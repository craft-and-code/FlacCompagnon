1//! Authenticity detections: three independent tests, each a plain yes/no.
2//!
3//! * **Upscaling**   — a low-resolution signal (≤16-bit) stored at a higher bit
4//!   depth. Detected from the effective vs. declared bit depth.
5//! * **Upsampling**  — a low-rate signal placed in a higher-sample-rate
6//!   container, inferred from limited bandwidth. This is a heuristic, not
7//!   proof that the sample rate was changed.
8//! * **Transcoding** — a lossy source re-wrapped as lossless. Decided by
9//!   [`crate::transcode`], which looks for the codec's own quantization
10//!   lattice; this module only reports the answer.
11//!
12//! Each test is independent: a file may trip none, one, or several. The
13//! reasoning is always surfaced in `detail`.
14//!
15//! ## Why there is no "suspected" any more
16//!
17//! Transcoding used to have three states, the middle one raised by a gentle
18//! spectral roll-off with no sharp cliff. It was the app's main source of
19//! false positives, and unavoidably so: acoustic, classical and analog-tape
20//! recordings are *naturally* dark, and no threshold separates "quiet above
21//! 16 kHz because a codec removed it" from "quiet above 16 kHz because nobody
22//! played anything up there". A verdict that fires on a whole genre is not a
23//! verdict.
24//!
25//! It is gone, along with the brick-wall and MDCT dead-zone rules that fed
26//! it. Transcoding is now decided solely by the re-quantization detectors,
27//! which provide statistical evidence of quantization, not a proof of origin.
28//! Missing checks are explained in detail; Clean means no finding, not that
29//! every check was available or that the recording history was verified.
30//! The cut-off frequency is still measured and still shown in its own
31//! column — it is useful information — but it no longer accuses anyone.

39/// The three independent detections plus a human summary.
40#[derive(Debug, Clone, Serialize, Deserialize)]
41pub struct Detections {
42    /// `true` when the content uses fewer bits than the container declares.
43    pub upscaling: bool,
44    /// `true` when a spectral heuristic suggests upsampling, not proof of origin.
45    pub upsampling: bool,
46    /// `true` when a lossy codec's quantization lattice was found in the
47    /// audio. Decided by [`crate::transcode`], not measured here.
48    pub transcoding: bool,
49    /// One-line explanation of the flagged issues (or why the file looks clean).
50    pub detail: String,
51    /// "Clean" for no findings, "Flagged" for any finding, or "Not analyzed"
52    /// for a placeholder. Missing checks remain explicit in `detail`.
53    pub summary: String,
54}
55
56impl Detections {
57    /// Build a binary summary from findings, independently of check availability.
58    pub fn from_findings(
59        upscaling: bool,
60        upsampling: bool,
61        transcoding: bool,
62        detail: String,
63    ) -> Self {
64        Self {
65            upscaling,
66            upsampling,
67            transcoding,
68            detail,
69            summary: if upscaling || upsampling || transcoding {
70                "Flagged"
71            } else {
72                "Clean"
73            }
74            .into(),
75        }
76    }
77
78    /// A neutral placeholder for a file that hasn't been (or couldn't be)
79    /// analyzed yet. A read failure is displayed separately as a file error.
80    pub fn not_analyzed() -> Self {
81        Detections {
82            upscaling: false,
83            upsampling: false,
84            transcoding: false,
85            detail: "Not yet analyzed.".to_string(),
86            summary: "Not analyzed".to_string(),
87        }
88    }
89}

91// --- Tunable thresholds -----------------------------------------------------
92/// Rates above this are "hi-res" containers subject to the upsampling check.
93const HIRES_RATE: u32 = 48_000;
94/// In a hi-res (> 48 kHz) container, content confined below this frequency
95/// (CD/DVD range, with margin) triggers the legacy bandwidth heuristic.
96/// This 30 kHz margin is not a validated boundary between native and resampled audio.
97const UPSAMPLE_MAX_HZ: f64 = 30_000.0;
98/// MDCT: fraction of frames that must show a dead zone. Calibrated on real
99/// transcodes: genuine music tops out around 0.30 (dark 1990s metal masters),
100/// while 128–192 kbps AAC sits at ~1.0, so 0.70 gives a safe margin both ways.
101const MDCT_DEAD_FRACTION: f32 = 0.70;
102/// MDCT: mean cut-off must be below this fraction of N to matter.
103const MDCT_CUTOFF_RATIO: f64 = 0.90;
104/// MDCT: the dead zone must be at least this flat/deep (dB rel. frame peak).
105const MDCT_DEAD_DB: f32 = -75.0;
106
107/// Whether the MDCT shows a consistent high-frequency dead zone.
108///
109/// Limited bandwidth can also come from native low-pass-filtered material.
110/// Even a spectrogram cannot distinguish identical samples with different
111/// histories; this measurement must never be described as proof of origin.
112fn mdct_signature(s: &AnalysisSummary) -> bool {
113    match (s.mdct_dead_fraction, s.mdct_cutoff_ratio, s.mdct_dead_db) {
114        (Some(frac), Some(cutoff), Some(dead)) => {
115            frac >= MDCT_DEAD_FRACTION && cutoff < MDCT_CUTOFF_RATIO && dead < MDCT_DEAD_DB
116        }
117        _ => false,
118    }
119}

121/// Run the three detections.
122///
123/// `transcode` is decided by the caller rather than measured here: it comes
124/// from [`crate::transcode`], which needs the whole decoded signal and a
125/// sweep of its own, neither of which belongs in a module whose job is to
126/// turn finished measurements into a verdict.
127///
128/// It arrives as the full [`crate::transcode::TranscodeEvidence`] rather than
129/// a bare `bool` so that the *score* reaches `detail`. That distinction
130/// matters in both directions: a user told only "Clean" cannot tell a file
131/// that scored 0.008 from one the detector never ran on, and a user told only
132/// "Transcoded" cannot tell a borderline finding from an overwhelming one.
133///
134/// [`crate::transcode::LatticeSkip`] is the third case — not tested — and it
135/// carries *why*, which ends up in `detail` verbatim. A dash in the Lattice
136/// column with no explanation is a dead end for the user and for whoever
137/// receives their bug report.
138pub fn classify(
139    summary: &AnalysisSummary,
140    sample_rate: u32,
141    declared_bits: Option<u32>,
142    real_bit_depth: Option<u32>,
143    transcode: LatticeResult,
144) -> Detections {
145    let mdct_dead = mdct_signature(summary);
146    let mdct_cutoff_hz = summary
147        .mdct_cutoff_ratio
148        .map(|r| r * (sample_rate as f64 / 2.0))
149        .unwrap_or(sample_rate as f64 / 2.0);
150    let mdct_khz = mdct_cutoff_hz / 1000.0;
151    let stft_khz = summary.cutoff_hz / 1000.0;
152
153    // 1. Upscaling: exact unused bits, or a persistent lower-depth grid with
154    //    a bounded residual. The latter is identified as an estimate in detail.
155    let upscaling = matches!(
156        (declared_bits, real_bit_depth),
157        (Some(d), Some(r)) if r < d
158    );
159
160    // 2. Upsampling candidate. In a hi-res container (> 48 kHz), a
161    //    consistent MDCT dead zone confined to the CD/DVD range suggests the extra
162    //    bandwidth is empty. (The STFT cut-off is unreliable on real music — it
163    //    usually reads full-band — so the MDCT dead zone is used here.)
164    let upsampling = sample_rate > HIRES_RATE && mdct_dead && mdct_cutoff_hz <= UPSAMPLE_MAX_HZ;
165
166    // 3. Transcoding — lossy source. Handed in by the caller; the only
167    //    decision left here is that it does not apply above 48 kHz, where the
168    //    lattice detectors have no tabulated bands.
169    let applies = sample_rate <= HIRES_RATE;
170    let transcoding = transcode.as_ref().is_ok_and(|e| e.detected) && applies;
171
172    // Build the human explanation.
173    let mut reasons: Vec<String> = Vec::new();
174    if upscaling {
175        let real = real_bit_depth.unwrap_or(0);
176        let declared = declared_bits.unwrap_or(0);
177        reasons.push(match summary.bit_depth_evidence {
178            Some(e) if e.method == BitDepthMethod::NarrowGrid => format!(
179                "Upscaling: estimated {real}-bit depth in {declared}-bit PCM from per-channel quantization grids with low-level residuals; stored samples occupy {} bits", e.stored_bits
180            ),
181            _ if summary.clipping.peak == 0.0 => format!(
182                "Upscaling: digital silence in {declared}-bit PCM (all samples are zero; reported as 1 bit by convention)"
183            ),
184            _ => format!("Upscaling: samples fit exactly in {real} bits within {declared}-bit PCM (zero low bits)"),
185        });
186    }
187    if upsampling {
188        reasons.push(format!(
189            "Possible upsampling: content stops at ~{:.1} kHz in a {:.1} kHz container; limited bandwidth is not proof of resampling",
190            mdct_khz,
191            sample_rate as f64 / 1000.0
192        ));
193    }
194    if transcoding {
195        reasons.push(format!(
196            "Transcoding: statistical evidence of a codec quantization lattice (score {:.4}); this score is not a probability or proof of origin",
197            transcode.as_ref().map(|e| e.likelihood).unwrap_or(0.0)
198        ));
199    }
200
201    let mut detections =
202        Detections::from_findings(upscaling, upsampling, transcoding, String::new());
203
204    // Always say what the lattice search found, flagged or not. A bare
205    // "Clean" hides the difference between "measured, and nothing there" and
206    // "never measured", and those deserve different amounts of trust.
207    let lattice = match (&transcode, applies) {
208        _ if transcoding => String::new(),
209        (Ok(e), true) => format!(
210            " No significant codec lattice evidence (score {:.4}).",
211            e.likelihood
212        ),
213        (_, false) => format!(
214            " Lattice search not applicable at {:.1} kHz; a lossy source cannot be ruled out.",
215            sample_rate as f64 / 1000.0
216        ),
217        // The *reason*, not just the fact. A row whose Lattice column is a
218        // dash is otherwise indistinguishable from one the search cleared,
219        // and the user has no way to tell which — nor to report it usefully.
220        (Err(skip), true) => format!(" Lattice search did not complete: {skip}."),
221    };
222
223    detections.detail = if reasons.is_empty() {
224        format!("{} — no anomaly detected by the completed checks; measured bandwidth to ~{stft_khz:.1} kHz.{lattice}", detections.summary)
225    } else {
226        format!("{}{lattice}", reasons.join(" · "))
227    };
228
229    detections
230}
