1//! Streaming analyzer.
2//!
3//! The decoder feeds audio one frame at a time (a frame = one sample per
4//! channel). Storing an entire album in memory would be wasteful, so every
5//! metric is accumulated incrementally:
6//!
7//! * spectrum      -> Hann-windowed FFT of a mono downmix, averaged over windows
8//! * clipping      -> full-scale sample counting with run detection
9//! * stereo        -> L-R difference, L/R correlation and mono cancellation
10//! * loudness      -> K-weighted integrated LUFS and loudness range
11//! * real bitdepth -> exact unused bits and persistent lower-depth integer grids
12//!
13//! Nothing here depends on a specific file format; [`decode`](crate::decode)
14//! adapts each codec to the [`StreamAnalyzer::push_frame`] interface.
15//!
16//! # Size
17//!
18//! Over CLAUDE.md's 300-line ceiling, deliberately. Each metric already lives
19//! in its own module ([`spectrum`], [`clipping`], [`stereo`](super::stereo),
20//! [`bitdepth`], [`loudness`](super::loudness), [`mdct`](super::mdct)); what is left here is the single hot
21//! loop that feeds them all from one pass over the samples,
22//! plus the state that loop carries. That single pass *is* the design — the
23//! whole reason this type exists rather than five independent analyzers is
24//! that an album must be read once, not five times. Splitting the loop would
25//! hide exactly the thing a reviewer needs to see whole, and any split would
26//! be by metric, which is the axis already covered by those modules.
27

42/// Full-scale detection threshold (normalized). Samples with |value| at or
43/// above this are treated as clipped.
44const CLIP_THRESHOLD: f32 = 0.9997;
45/// FFT window size. 8192 gives ~5.4 Hz resolution at 44.1 kHz.
46const FFT_SIZE: usize = 8192;
47
48// --- MDCT (AAC-SIN transcode detector) -------------------------------------
49/// Analyze one in every `MDCT_STRIDE` overlapping MDCT hops, so the transform
50/// samples the whole track without processing every frame (O(N²) per frame).
51const MDCT_STRIDE: u64 = 4;
52/// Cap on analyzed MDCT frames — keeps a long album fast.
53const MDCT_MAX_FRAMES: u32 = 240;
54/// A dead zone only "exists" if the per-frame cutoff is below this fraction of N.
55const MDCT_DEAD_TOP_RATIO: f32 = 0.92;
56
57// --- Dynamics (DR) ----------------------------------------------------------
58/// Frames per RMS block (~3 s at 44.1 kHz). The exact wall-clock length is not
59/// critical: the DR estimate averages the loudest blocks, so the granularity
60/// only needs to be long enough to smooth individual drum hits.
61const DYN_BLOCK_FRAMES: u64 = 131_072;
62/// Fraction of the loudest blocks that defines the "loud passages" RMS,
63/// mirroring the classic DR-meter approach (peak vs loudest-20% RMS).
64const DYN_TOP_FRACTION: f64 = 0.2;
65
66/// Aggregated results produced by [`StreamAnalyzer::finish`].
67#[derive(Debug, Clone)]
68pub struct AnalysisSummary {
69    /// Estimated frequency-response cutoff, in Hz.
70    pub cutoff_hz: f64,
71    /// [`cutoff_hz`](Self::cutoff_hz) as a fraction of Nyquist (0..1).
72    pub cutoff_ratio: f64,
73    /// How sharply the level drops at the cutoff (dB). Large == a brick wall.
74    pub cliff_db: f32,
75    /// Mean level just above the cutoff (dB rel. peak) — the dead-zone depth.
76    pub above_db: f32,
77    /// Averaged magnitude spectrum in dB, one entry per FFT bin (0..=N/2).
78    pub spectrum_db: Vec<f32>,
79    /// Clipping / true-peak-ish statistics accumulated over the stream.
80    pub clipping: ClippingInfo,
81    /// `true` when the left and right channels were identical (or near
82    /// enough) for long enough to suggest a mono source duplicated to stereo.
83    pub fake_stereo: bool,
84    /// Whole-stream correlation of the first two channels when both carry audio.
85    pub phase_correlation: Option<f32>,
86    /// True when the two channels are strongly opposed across the stream.
87    pub phase_inverted: bool,
88    /// Unweighted RMS balance, available only for two-channel streams.
89    pub stereo_balance: Option<super::stereo::StereoBalance>,
90    /// High-frequency Side/Mid measurement for qualifying stereo streams.
91    pub high_frequency_stereo: Option<HighFrequencyStereo>,
92    /// Whole-stream mean measured independently on each channel.
93    pub dc_offset: Option<DcOffset>,
94    /// Time-local correlation and frequency-band evidence for stereo streams.
95    pub local_phase: Option<LocalPhase>,
96    /// The bit depth actually used by the samples, when it could be
97    /// determined from an integer PCM source (`None` for float sources).
98    pub real_bit_depth: Option<u32>,
99    /// Exact occupied depth and whether the effective depth is a grid estimate.
100    pub bit_depth_evidence: Option<bitdepth::BitDepthEvidence>,
101
102    /// Dynamic-range estimate in dB: peak level vs the RMS of the loudest 20%
103    /// of ~3 s blocks (crest factor of the loud passages, DR-meter style).
104    /// High values (>= 12 dB) indicate a dynamic master; low values (< 8 dB)
105    /// a loudness-war master. `None` for silent or extremely short streams.
106    pub dr_db: Option<f32>,
107    /// EBU R 128 integrated loudness, in LUFS; absent when unmeasurable.
108    pub integrated_lufs: Option<f32>,
109    /// Maximum momentary and short-term loudness over real-audio windows.
110    pub loudness_peaks: Option<LoudnessPeaks>,
111    /// EBU Tech 3342 loudness range, in LU; absent when unmeasurable.
112    pub loudness_range_lu: Option<f32>,
113    /// Suspected short pulses and digital dropouts; absent if unmeasurable.
114    pub discontinuities: Option<DiscontinuityAnalysis>,
115
116    // --- MDCT (AAC-SIN) transcode evidence ---
117    /// Mean per-frame MDCT cutoff as a fraction of Nyquist (dead-zone frames).
118    pub mdct_cutoff_ratio: Option<f64>,
119    /// Mean level (dB rel. frame peak) of the MDCT dead zone above the cutoff.
120    pub mdct_dead_db: Option<f32>,
121    /// Fraction of analyzed MDCT frames that showed a high-frequency dead zone.
122    pub mdct_dead_fraction: Option<f32>,
123
124    /// Legacy detector result, retained for API compatibility. Always `None`;
125    /// transcoding is now evaluated by `crate::transcode` in the pipeline.
126    pub requant_rate: Option<f32>,
127}
128
129impl AnalysisSummary {
130    /// The FFT size [`spectrum_db`](Self::spectrum_db) was produced with.
131    ///
132    /// The spectrum holds bins `0..=N/2` (a real FFT's non-redundant half), so
133    /// `N` is recovered as `(len - 1) * 2`. Callers need it to turn a bin index
134    /// back into a frequency; it lives here rather than being re-derived at
135    /// each call site, where the off-by-one is easy to get wrong and silently
136    /// shifts every frequency band that depends on it.
137    pub fn fft_size(&self) -> usize {
138        self.spectrum_db.len().saturating_sub(1) * 2
139    }
140}

158    high_frequency_stereo: Option<HighFrequencyStereoMeter>,

163    diff_energy: f64,

165    r_energy: f64,
166    cross_energy: f64,
167    identical_frames: u64,
168    total_frames: u64,
169
170    // --- bit depth ---
171    bit_depth: bitdepth::BitDepthAnalyzer,
172
173    // --- MDCT (AAC-SIN) ---
174    mdct: &'static Mdct,
175    mdct_prev: Vec<f32>, // previous hop (N mono samples)
176    mdct_fill: Vec<f32>, // current hop being filled
177    mdct_have_prev: bool,

191impl StreamAnalyzer {
192    /// Start a fresh analysis for a stream with `channels` audio channels.
193    pub fn new(channels: usize, sample_rate: u32) -> Self {
194        let mut planner = FftPlanner::<f32>::new();
195        let fft = planner.plan_fft_forward(FFT_SIZE);
196        let hann: Vec<f32> = (0..FFT_SIZE)
197            .map(|n| {
198                let x = std::f32::consts::PI * n as f32 / (FFT_SIZE as f32 - 1.0);
199                x.sin().powi(2) // Hann window == sin^2
200            })
201            .collect();
202        Self {
203            channels: channels.max(1),
204            fft,
205            hann,
206            frame_buf: Vec::with_capacity(FFT_SIZE),
207            power_acc: vec![0.0; FFT_SIZE / 2 + 1],
208            window_count: 0,
209            clip_state: clipping::ClipState::new(CLIP_THRESHOLD),
210            true_peak: TruePeak::new(channels.max(1)),
211            loudness: LoudnessMeter::new(sample_rate, channels),
212            discontinuities: DiscontinuityDetector::new(sample_rate, channels),
213            high_frequency_stereo: HighFrequencyStereoMeter::new(sample_rate, channels),
214            dc_offset: DcOffsetMeter::new(channels),
215            local_phase: LocalPhaseMeter::new(sample_rate, channels),
216            diff_energy: 0.0,
217            l_energy: 0.0,
218            r_energy: 0.0,
219            cross_energy: 0.0,
220            identical_frames: 0,
221            total_frames: 0,
222            bit_depth: bitdepth::BitDepthAnalyzer::new(channels.max(1)),
223            mdct: Mdct::shared(),
224            mdct_prev: Vec::with_capacity(AAC_N),
225            mdct_fill: Vec::with_capacity(AAC_N),
226            mdct_have_prev: false,
227            mdct_hop: 0,
228            mdct_scratch: vec![0.0; AAC_N],
229            mdct_frames: 0,
230            mdct_dead_frames: 0,
231            mdct_cutoff_ratio_sum: 0.0,
232            mdct_dead_db_sum: 0.0,
233            dyn_block_sumsq: 0.0,
234            dyn_block_frames: 0,
235            dyn_blocks: Vec::new(),
236        }
237    }
238
239    /// Push one frame of normalized-float samples (`samples.len() == channels`),
240    /// optionally accompanied by the raw integer sample values for the same
241    /// frame (used for effective bit-depth estimation).
242    pub fn push_frame(&mut self, samples: &[f32], int_samples: Option<&[i32]>) {
243        if let Some(loudness) = &mut self.loudness {
244            loudness.push_frame(samples);
245        }
246        if let Some(phase) = &mut self.local_phase {
247            phase.push_frame(samples);
248        }
249        if let Some(dc_offset) = &mut self.dc_offset {
250            dc_offset.push_frame(samples);
251        }
252        if samples.is_empty() {
253            return;
254        }
255        self.total_frames += 1;
256
257        // Clipping + peak (per channel), and the frame's mean square for the
258        // dynamics blocks.
259        let mut frame_sumsq = 0.0f64;
260        for &s in samples {
261            self.clip_state.push(s);
262            frame_sumsq += (s as f64) * (s as f64);
263        }
264        self.dyn_block_sumsq += frame_sumsq / samples.len() as f64;
265        self.dyn_block_frames += 1;
266        self.true_peak.push_frame(samples);
267        if let Some(discontinuities) = &mut self.discontinuities {
268            discontinuities.push_frame(samples);
269        }
270        if let Some(high_frequency_stereo) = &mut self.high_frequency_stereo {
271            high_frequency_stereo.push_frame(samples);
272        }
273        if self.dyn_block_frames == DYN_BLOCK_FRAMES {
274            self.dyn_blocks
275                .push(self.dyn_block_sumsq / self.dyn_block_frames as f64);
276            self.dyn_block_sumsq = 0.0;
277            self.dyn_block_frames = 0;
278        }
279
280        // Stereo difference energy (first two channels).
281        if self.channels >= 2 && samples.len() >= 2 {
282            let l = samples[0] as f64;
283            let r = samples[1] as f64;
284            let d = l - r;
285            self.diff_energy += d * d;
286            self.l_energy += l * l;
287            self.r_energy += r * r;
288            self.cross_energy += l * r;
289            if (l - r).abs() < 1e-9 {
290                self.identical_frames += 1;
291            }
292        }
293
294        self.bit_depth.push_frame(int_samples);
295
296        // Mono downmix into the FFT buffer.
297        let mut mono = 0.0f32;
298        for &s in samples {
299            mono += s;
300        }
301        mono /= samples.len().max(1) as f32;
302        self.frame_buf.push(mono);
303        if self.frame_buf.len() == FFT_SIZE {
304            self.process_window();
305            self.frame_buf.clear();
306        }
307
308        // Feed the same mono sample to the MDCT pipeline (frame = 2N, hop = N).
309        self.mdct_fill.push(mono);
310        if self.mdct_fill.len() == AAC_N {
311            self.mdct_hop += 1;
312            if self.mdct_have_prev
313                && self.mdct_frames < MDCT_MAX_FRAMES
314                && self.mdct_hop.is_multiple_of(MDCT_STRIDE)
315            {
316                self.process_mdct();
317            }
318            std::mem::swap(&mut self.mdct_prev, &mut self.mdct_fill);
319            self.mdct_fill.clear();
320            self.mdct_have_prev = true;
321        }
322    }
323
324    /// Analyze one overlapping MDCT frame (previous hop + current hop) for the
325    /// AAC-SIN transcode signature: a flat, sharply-bounded high-frequency dead
326    /// zone in the sine-window MDCT domain.
327    fn process_mdct(&mut self) {
328        let n = AAC_N;
329        let mut frame = Vec::with_capacity(2 * n);
330        frame.extend_from_slice(&self.mdct_prev);
331        frame.extend_from_slice(&self.mdct_fill);
332
333        let m = self.mdct; // &'static, cheap to copy
334        m.forward(&frame, &mut self.mdct_scratch);
335
336        let mut peak = 0.0f32;
337        for &c in &self.mdct_scratch {
338            let a = c.abs();
339            if a > peak {
340                peak = a;
341            }
342        }
343        self.mdct_frames += 1;
344        if peak < 1e-7 {
345            return; // silent frame
346        }
347
348        let thr = peak * 1e-4; // -80 dB relative to the frame peak
349        let mut cutoff = 0usize;
350        for k in (0..n).rev() {
351            if self.mdct_scratch[k].abs() > thr {
352                cutoff = k;
353                break;
354            }
355        }
356
357        let top = (MDCT_DEAD_TOP_RATIO * n as f32) as usize;
358        if cutoff < top {
359            let start = (cutoff + 8).min(n);
360            if start < n {
361                let mut sum = 0.0f64;
362                let mut cnt = 0u32;
363                for k in start..n {
364                    let d = 20.0 * (self.mdct_scratch[k].abs() / peak).max(1e-12).log10();
365                    sum += d as f64;
366                    cnt += 1;
367                }
368                if cnt > 0 {
369                    self.mdct_dead_db_sum += sum / cnt as f64;
370                    self.mdct_cutoff_ratio_sum += cutoff as f64 / n as f64;
371                    self.mdct_dead_frames += 1;
372                }
373            }
374        }
375    }
376
377    fn process_window(&mut self) {
378        let mut buf: Vec<Complex<f32>> = self
379            .frame_buf
380            .iter()
381            .zip(self.hann.iter())
382            .map(|(&s, &w)| Complex { re: s * w, im: 0.0 })
383            .collect();
384        self.fft.process(&mut buf);
385        for (bin, c) in buf.iter().take(self.power_acc.len()).enumerate() {
386            self.power_acc[bin] += (c.re as f64).powi(2) + (c.im as f64).powi(2);
387        }
388        self.window_count += 1;
389    }
390
391    /// Finalize and compute all summary metrics for a stream at `sample_rate`.
392    /// `declared_bits` is the container's stated integer bit depth (if any) and
393    /// is used to bound the effective bit-depth estimate.
394    pub fn finish(mut self, sample_rate: u32, declared_bits: Option<u32>) -> AnalysisSummary {
395        // Flush a trailing partial window (zero padded) so short files still
396        // produce a spectrum.
397        if self.window_count == 0 && !self.frame_buf.is_empty() {
398            self.frame_buf.resize(FFT_SIZE, 0.0);
399            self.process_window();
400        }
401
402        let spectrum_db = spectrum::average_to_db(&self.power_acc, self.window_count);
403        let (cutoff_hz, cutoff_ratio) =
404            spectrum::detect_cutoff(&spectrum_db, sample_rate, FFT_SIZE);
405        let (cliff_db, above_db) =
406            spectrum::cutoff_context(&spectrum_db, sample_rate, FFT_SIZE, cutoff_hz);
407
408        // Dynamics: flush a meaningful trailing partial block (>= 1/4 length),
409        // then compare the peak with the RMS of the loudest 20% of blocks.
410        if self.dyn_block_frames >= DYN_BLOCK_FRAMES / 4 {
411            self.dyn_blocks
412                .push(self.dyn_block_sumsq / self.dyn_block_frames as f64);
413        }
414        let mut clipping = self.clip_state.finish(self.channels, self.total_frames);
415        // True peak from the 4x-oversampled stream. It can legitimately sit a
416        // hair above the sample peak on any material, and above 1.0 (positive
417        // dBTP) on loud masters — that's the inter-sample clipping signal.
418        clipping.true_peak = self.true_peak.peak();
419        clipping.true_peak_dbtp = self.true_peak.peak_dbtp();
420        let dr_db = {
421            let mut blocks = self.dyn_blocks.clone();
422            blocks.sort_by(|a, b| b.partial_cmp(a).unwrap_or(std::cmp::Ordering::Equal));
423            let top = ((blocks.len() as f64 * DYN_TOP_FRACTION).ceil() as usize).max(1);
424            let loud: Vec<f64> = blocks.into_iter().take(top).collect();
425            if loud.is_empty() {
426                None
427            } else {
428                let rms = (loud.iter().sum::<f64>() / loud.len() as f64).sqrt();
429                if rms > 1e-9 && clipping.peak > 0.0 {
430                    Some((20.0 * (clipping.peak as f64 / rms).log10()) as f32)
431                } else {
432                    None
433                }
434            }
435        };
436
437        let fake_stereo = if self.channels >= 2 {
438            super::stereo::is_fake(
439                self.diff_energy,
440                self.l_energy,
441                self.r_energy,
442                self.identical_frames,
443                self.total_frames,
444            )
445        } else {
446            false
447        };
448        let phase = super::stereo::analyze_phase(self.l_energy, self.r_energy, self.cross_energy);
449        let high_frequency_stereo = self
450            .high_frequency_stereo
451            .take()
452            .and_then(HighFrequencyStereoMeter::finish);
453
454        let measured_depth = self.bit_depth.finish(declared_bits);
455        let real_bit_depth = measured_depth.map(|(bits, _)| bits);
456        let bit_depth_evidence = measured_depth.map(|(_, evidence)| evidence);
457
458        let (mdct_cutoff_ratio, mdct_dead_db, mdct_dead_fraction) = if self.mdct_frames > 0 {
459            let frac = self.mdct_dead_frames as f32 / self.mdct_frames as f32;
460            if self.mdct_dead_frames > 0 {
461                (
462                    Some(self.mdct_cutoff_ratio_sum / self.mdct_dead_frames as f64),
463                    Some((self.mdct_dead_db_sum / self.mdct_dead_frames as f64) as f32),
464                    Some(frac),
465                )
466            } else {
467                (None, None, Some(frac))
468            }
469        } else {
470            (None, None, None)
471        };
472
473        // Capture programme loudness and M/S maxima before LRA appends 1.5 s of
474        // analysis-only silence for its centred 3 s tail windows.
475        let integrated_lufs = self
476            .loudness
477            .as_ref()
478            .and_then(LoudnessMeter::integrated_lufs);
479        let loudness_peaks = self.loudness.as_ref().and_then(LoudnessMeter::peaks);
480        let loudness_range_lu = self
481            .loudness
482            .take()
483            .and_then(LoudnessMeter::loudness_range_lu);
484
485        AnalysisSummary {
486            cutoff_hz,
487            cutoff_ratio,
488            cliff_db,
489            above_db,
490            spectrum_db,
491            clipping,
492            fake_stereo,
493            phase_correlation: phase.correlation,
494            phase_inverted: phase.likely_inverted,
495            stereo_balance: if self.channels == 2 {
496                super::stereo::analyze_balance(self.l_energy, self.r_energy)
497            } else {
498                None
499            },
500            high_frequency_stereo,
501            dc_offset: self.dc_offset.take().and_then(DcOffsetMeter::finish),
502            local_phase: self.local_phase.take().and_then(LocalPhaseMeter::finish),
503            real_bit_depth,
504            bit_depth_evidence,
505            dr_db,
506            integrated_lufs,
507            loudness_peaks,
508            loudness_range_lu,
509            discontinuities: self
510                .discontinuities
511                .take()
512                .and_then(DiscontinuityDetector::finish),
513            mdct_cutoff_ratio,
514            mdct_dead_db,
515            mdct_dead_fraction,
516            requant_rate: None,
517        }
518    }
