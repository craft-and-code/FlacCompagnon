1//! Streaming analyzer.
2//!
3//! The decoder feeds audio one frame at a time (a frame = one sample per
4//! channel). Storing an entire album in memory would be wasteful, so every
5//! metric is accumulated incrementally:
6//!
7//! * spectrum      -> Hann-windowed FFT of a mono downmix, averaged over windows
8//! * clipping      -> full-scale sample counting with run detection
9//! * stereo        -> L-R difference, L/R correlation and mono cancellation
10//! * loudness      -> K-weighted integrated LUFS and loudness range
11//! * real bitdepth -> exact unused bits and persistent lower-depth integer grids
12//!
13//! Nothing here depends on a specific file format; [`decode`](crate::decode)
14//! adapts each codec to the [`StreamAnalyzer::push_frame`] interface.
15//!
16//! # Size
17//!
18//! Over AGENTS.md's 300-line ceiling, deliberately. Each metric already lives
19//! in its own module ([`spectrum`], [`clipping`], [`stereo`](super::stereo),
20//! [`bitdepth`], [`loudness`](super::loudness), [`mdct`](super::mdct)); what is left here is the single hot
21//! loop that feeds them all from one pass over the samples,
22//! plus the state that loop carries. That single pass *is* the design — the
23//! whole reason this type exists rather than five independent analyzers is
24//! that an album must be read once, not five times. Splitting the loop would
25//! hide exactly the thing a reviewer needs to see whole, and any split would
26//! be by metric, which is the axis already covered by those modules.
27

42/// Full-scale detection threshold (normalized). Samples with |value| at or
43/// above this are treated as clipped.
44const CLIP_THRESHOLD: f32 = 0.9997;
45/// FFT window size. 8192 gives ~5.4 Hz resolution at 44.1 kHz.
46const FFT_SIZE: usize = 8192;
47
48// --- MDCT (AAC-SIN transcode detector) -------------------------------------
49/// Analyze one in every `MDCT_STRIDE` overlapping MDCT hops, so the transform
50/// samples the whole track without processing every frame (O(N²) per frame).
51const MDCT_STRIDE: u64 = 4;
52/// Cap on analyzed MDCT frames — keeps a long album fast.
53const MDCT_MAX_FRAMES: u32 = 240;
54/// A dead zone only "exists" if the per-frame cutoff is below this fraction of N.
55const MDCT_DEAD_TOP_RATIO: f32 = 0.92;
56
57// --- Dynamics (DR) ----------------------------------------------------------
58/// Frames per RMS block (~3 s at 44.1 kHz). The exact wall-clock length is not
59/// critical: the DR estimate averages the loudest blocks, so the granularity
60/// only needs to be long enough to smooth individual drum hits.
61const DYN_BLOCK_FRAMES: u64 = 131_072;
62/// Fraction of the loudest blocks that defines the "loud passages" RMS,
63/// mirroring the classic DR-meter approach (peak vs loudest-20% RMS).
64const DYN_TOP_FRACTION: f64 = 0.2;
65
66/// Aggregated results produced by [`StreamAnalyzer::finish`].
67#[derive(Debug, Clone)]
68pub struct AnalysisSummary {
69    /// Estimated frequency-response cutoff, in Hz.
70    pub cutoff_hz: f64,
71    /// [`cutoff_hz`](Self::cutoff_hz) as a fraction of Nyquist (0..1).
72    pub cutoff_ratio: f64,
73    /// How sharply the level drops at the cutoff (dB). Large == a brick wall.
74    pub cliff_db: f32,
75    /// Mean level just above the cutoff (dB rel. peak) — the dead-zone depth.
76    pub above_db: f32,
77    /// Averaged magnitude spectrum in dB, one entry per FFT bin (0..=N/2).
78    pub spectrum_db: Vec<f32>,
79    /// Clipping / true-peak-ish statistics accumulated over the stream.
80    pub clipping: ClippingInfo,
81    /// `true` when the left and right channels were identical (or near
82    /// enough) for long enough to suggest a mono source duplicated to stereo.
83    pub fake_stereo: bool,
84    /// Whole-stream correlation of the first two channels when both carry audio.
85    pub phase_correlation: Option<f32>,
86    /// True when the two channels are strongly opposed across the stream.
87    pub phase_inverted: bool,
88    /// Unweighted RMS balance, available only for two-channel streams.
89    pub stereo_balance: Option<super::stereo::StereoBalance>,
90    /// High-frequency Side/Mid measurement for qualifying stereo streams.
91    pub high_frequency_stereo: Option<HighFrequencyStereo>,
92    /// Whole-stream mean measured independently on each channel.
93    pub dc_offset: Option<DcOffset>,
94    /// Time-local correlation and frequency-band evidence for stereo streams.
95    pub local_phase: Option<LocalPhase>,
96    /// The bit depth actually used by the samples, when it could be
97    /// determined from an integer PCM source (`None` for float sources).
98    pub real_bit_depth: Option<u32>,
99    /// Exact occupied depth and whether the effective depth is a grid estimate.
100    pub bit_depth_evidence: Option<bitdepth::BitDepthEvidence>,
101
102    /// Dynamic-range estimate in dB: peak level vs the RMS of the loudest 20%
103    /// of ~3 s blocks (crest factor of the loud passages, DR-meter style).
104    /// High values (>= 12 dB) indicate a dynamic master; low values (< 8 dB)
105    /// a loudness-war master. `None` for silent or extremely short streams.
106    pub dr_db: Option<f32>,
107    /// EBU R 128 integrated loudness, in LUFS; absent when unmeasurable.
108    pub integrated_lufs: Option<f32>,
109    /// Maximum momentary and short-term loudness over real-audio windows.
110    pub loudness_peaks: Option<LoudnessPeaks>,
111    /// EBU Tech 3342 loudness range, in LU; absent when unmeasurable.
112    pub loudness_range_lu: Option<f32>,
113    /// Suspected short pulses and digital dropouts; absent if unmeasurable.
114    pub discontinuities: Option<DiscontinuityAnalysis>,
115
116    // --- MDCT (AAC-SIN) transcode evidence ---
117    /// Mean per-frame MDCT cutoff as a fraction of Nyquist (dead-zone frames).
118    pub mdct_cutoff_ratio: Option<f64>,
119    /// Mean level (dB rel. frame peak) of the MDCT dead zone above the cutoff.
120    pub mdct_dead_db: Option<f32>,
121    /// Fraction of analyzed MDCT frames that showed a high-frequency dead zone.
122    pub mdct_dead_fraction: Option<f32>,
123
124    /// Legacy detector result, retained for API compatibility. Always `None`;
125    /// transcoding is now evaluated by `crate::transcode` in the pipeline.
126    pub requant_rate: Option<f32>,
127}

129impl AnalysisSummary {
130    /// The FFT size [`spectrum_db`](Self::spectrum_db) was produced with.
131    ///
132    /// The spectrum holds bins `0..=N/2` (a real FFT's non-redundant half), so
133    /// `N` is recovered as `(len - 1) * 2`. Callers need it to turn a bin index
134    /// back into a frequency; it lives here rather than being re-derived at
135    /// each call site, where the off-by-one is easy to get wrong and silently
136    /// shifts every frequency band that depends on it.
137    pub fn fft_size(&self) -> usize {
138        self.spectrum_db.len().saturating_sub(1) * 2
139    }
140}

199    /// Build only the meters needed by the selection and its dependencies.
200    pub fn new_selected(channels: usize, sample_rate: u32, selection: AnalysisSelection) -> Self {
201        let mut planner = FftPlanner::<f32>::new();
202        let fft = selection
203            .spectrum()
204            .then(|| planner.plan_fft_forward(FFT_SIZE));
205        let hann: Vec<f32> = (0..if selection.spectrum() { FFT_SIZE } else { 0 })
206            .map(|n| {
207                let x = std::f32::consts::PI * n as f32 / (FFT_SIZE as f32 - 1.0);
208                x.sin().powi(2) // Hann window == sin^2
209            })
210            .collect();
211        Self {
212            channels: channels.max(1),
213            selection,
214            fft,
215            hann,
216            frame_buf: Vec::with_capacity(FFT_SIZE),
217            power_acc: if selection.spectrum() {
218                vec![0.0; FFT_SIZE / 2 + 1]
219            } else {
220                Vec::new()
221            },
222            window_count: 0,
223            clip_state: clipping::ClipState::new(CLIP_THRESHOLD),
224            true_peak: (selection.enabled(Kind::Clipping)).then(|| TruePeak::new(channels.max(1))),
225            loudness: (selection.enabled(Kind::Loudness))
226                .then(|| LoudnessMeter::new(sample_rate, channels))
227                .flatten(),
228            discontinuities: (selection.enabled(Kind::Clicks) || selection.enabled(Kind::Dropouts))
229                .then(|| {
230                    DiscontinuityDetector::new_selected(
231                        sample_rate,
232                        channels,
233                        selection.enabled(Kind::Clicks),
234                        selection.enabled(Kind::Dropouts),
235                    )
236                })
237                .flatten(),
238            high_frequency_stereo: (selection.enabled(Kind::HfStereo))
239                .then(|| HighFrequencyStereoMeter::new(sample_rate, channels))
240                .flatten(),
241            dc_offset: (selection.enabled(Kind::DcOffset))
242                .then(|| DcOffsetMeter::new(channels))
243                .flatten(),
244            local_phase: (selection.enabled(Kind::Phase))
245                .then(|| LocalPhaseMeter::new(sample_rate, channels))
246                .flatten(),
247            diff_energy: 0.0,
248            l_energy: 0.0,
249            r_energy: 0.0,
250            cross_energy: 0.0,
251            identical_frames: 0,
252            total_frames: 0,
253            stereo_valid: true,
254            bit_depth: selection
255                .bit_depth()
256                .then(|| bitdepth::BitDepthAnalyzer::new(channels.max(1))),
257            mdct: selection.enabled(Kind::Authenticity).then(Mdct::shared),
258            mdct_prev: Vec::with_capacity(AAC_N),
259            mdct_fill: Vec::with_capacity(AAC_N),
260            mdct_have_prev: false,
261            mdct_hop: 0,
262            mdct_scratch: if selection.enabled(Kind::Authenticity) {
263                vec![0.0; AAC_N]
264            } else {
265                Vec::new()
266            },
267            mdct_frames: 0,
268            mdct_dead_frames: 0,
269            mdct_cutoff_ratio_sum: 0.0,
270            mdct_dead_db_sum: 0.0,
271            dyn_block_sumsq: 0.0,
272            dyn_block_frames: 0,
273            dyn_blocks: Vec::new(),
274        }
275    }

277    /// Push one frame of normalized-float samples (`samples.len() == channels`),
278    /// optionally accompanied by the raw integer sample values for the same
279    /// frame (used for effective bit-depth estimation).
280    pub fn push_frame(&mut self, samples: &[f32], int_samples: Option<&[i32]>) {
281        // A malformed tail must not turn prefix energies into a whole-file
282        // channel relationship. Keep other measurement paths independent.
283        if self.selection.stereo()
284            && (samples.len() != self.channels || samples.iter().any(|s| !s.is_finite()))
285        {
286            self.stereo_valid = false;
287        }
288        if let Some(loudness) = &mut self.loudness {
289            loudness.push_frame(samples);
290        }
291        if let Some(phase) = &mut self.local_phase {
292            phase.push_frame(samples);
293        }
294        if let Some(dc_offset) = &mut self.dc_offset {
295            dc_offset.push_frame(samples);
296        }
297        if samples.is_empty() {
298            return;
299        }
300        self.total_frames += 1;
301
302        // Clipping + peak (per channel), and the frame's mean square for the
303        // dynamics blocks.
304        if self.selection.enabled(Kind::Clipping) {
305            self.clip_state.push_frame(samples);
306        }
307        let mut frame_sumsq = 0.0f64;
308        for &s in samples {
309            if !self.selection.enabled(Kind::Clipping) && self.selection.sample_peak() {
310                self.clip_state.push_peak(s);
311            }
312            if self.selection.enabled(Kind::Dynamics) {
313                frame_sumsq += (s as f64) * (s as f64);
314            }
315        }
316        if self.selection.enabled(Kind::Dynamics) {
317            self.dyn_block_sumsq += frame_sumsq / samples.len() as f64;
318            self.dyn_block_frames += 1;
319        }
320        if let Some(true_peak) = &mut self.true_peak {
321            true_peak.push_frame(samples);
322        }
323        if let Some(discontinuities) = &mut self.discontinuities {
324            discontinuities.push_frame(samples);
325        }
326        if let Some(high_frequency_stereo) = &mut self.high_frequency_stereo {
327            high_frequency_stereo.push_frame(samples);
328        }
329        if self.dyn_block_frames == DYN_BLOCK_FRAMES {
330            self.dyn_blocks
331                .push(self.dyn_block_sumsq / self.dyn_block_frames as f64);
332            self.dyn_block_sumsq = 0.0;
333            self.dyn_block_frames = 0;
334        }
335
336        // Stereo difference energy (first two channels).
337        if self.selection.stereo() && self.channels >= 2 && samples.len() >= 2 {
338            let l = samples[0] as f64;
339            let r = samples[1] as f64;
340            self.l_energy += l * l;
341            self.r_energy += r * r;
342            if self.selection.enabled(Kind::Stereo) {
343                let d = l - r;
344                self.diff_energy += d * d;
345                if l == r {
346                    self.identical_frames += 1;
347                }
348            }
349            if self.selection.enabled(Kind::Phase) {
350                self.cross_energy += l * r;
351            }
352        }
353
354        if let Some(depth) = &mut self.bit_depth {
355            depth.push_frame(int_samples);
356        }
357        if !self.selection.spectrum() && self.mdct.is_none() {
358            return;
359        }
360
361        // Mono downmix into the FFT buffer.
362        let mut mono = 0.0f32;
363        for &s in samples {
364            mono += s;
365        }
366        mono /= samples.len().max(1) as f32;
367        if self.selection.spectrum() {
368            self.frame_buf.push(mono);
369        }
370        if self.frame_buf.len() == FFT_SIZE {
371            self.process_window();
372            self.frame_buf.clear();
373        }
374
375        // Feed the same mono sample to the MDCT pipeline (frame = 2N, hop = N).
376        if self.mdct.is_none() {
377            return;
378        }
379        self.mdct_fill.push(mono);
380        if self.mdct_fill.len() == AAC_N {
381            self.mdct_hop += 1;
382            if self.mdct_have_prev
383                && self.mdct_frames < MDCT_MAX_FRAMES
384                && self.mdct_hop.is_multiple_of(MDCT_STRIDE)
385            {
386                self.process_mdct();
387            }
388            std::mem::swap(&mut self.mdct_prev, &mut self.mdct_fill);
389            self.mdct_fill.clear();
390            self.mdct_have_prev = true;
391        }
392    }

394    /// Analyze one overlapping MDCT frame (previous hop + current hop) for the
395    /// AAC-SIN transcode signature: a flat, sharply-bounded high-frequency dead
396    /// zone in the sine-window MDCT domain.
397    fn process_mdct(&mut self) {
398        let n = AAC_N;
399        let mut frame = Vec::with_capacity(2 * n);
400        frame.extend_from_slice(&self.mdct_prev);
401        frame.extend_from_slice(&self.mdct_fill);
402
403        let Some(m) = self.mdct else {
404            return;
405        };
406        m.forward(&frame, &mut self.mdct_scratch);
407
408        let mut peak = 0.0f32;
409        for &c in &self.mdct_scratch {
410            let a = c.abs();
411            if a > peak {
412                peak = a;
413            }
414        }
415        self.mdct_frames += 1;
416        if peak < 1e-7 {
417            return; // silent frame
418        }
419
420        let thr = peak * 1e-4; // -80 dB relative to the frame peak
421        let mut cutoff = 0usize;
422        for k in (0..n).rev() {
423            if self.mdct_scratch[k].abs() > thr {
424                cutoff = k;
425                break;
426            }
427        }
428
429        let top = (MDCT_DEAD_TOP_RATIO * n as f32) as usize;
430        if cutoff < top {
431            let start = (cutoff + 8).min(n);
432            if start < n {
433                let mut sum = 0.0f64;
434                let mut cnt = 0u32;
435                for k in start..n {
436                    let d = 20.0 * (self.mdct_scratch[k].abs() / peak).max(1e-12).log10();
437                    sum += d as f64;
438                    cnt += 1;
439                }
440                if cnt > 0 {
441                    self.mdct_dead_db_sum += sum / cnt as f64;
442                    self.mdct_cutoff_ratio_sum += cutoff as f64 / n as f64;
443                    self.mdct_dead_frames += 1;
444                }
445            }
446        }
447    }

449    fn process_window(&mut self) {
450        let mut buf: Vec<Complex<f32>> = self
451            .frame_buf
452            .iter()
453            .zip(self.hann.iter())
454            .map(|(&s, &w)| Complex { re: s * w, im: 0.0 })
455            .collect();
456        if let Some(fft) = &self.fft {
457            fft.process(&mut buf);
458        }
459        for (bin, c) in buf.iter().take(self.power_acc.len()).enumerate() {
460            self.power_acc[bin] += (c.re as f64).powi(2) + (c.im as f64).powi(2);
461        }
462        self.window_count += 1;
463    }

465    /// Finalize and compute all summary metrics for a stream at `sample_rate`.
466    /// `declared_bits` is the container's stated integer bit depth (if any) and
467    /// is used to bound the effective bit-depth estimate.
468    pub fn finish(mut self, sample_rate: u32, declared_bits: Option<u32>) -> AnalysisSummary {
469        // Flush a trailing partial window (zero padded) so short files still
470        // produce a spectrum.
471        if self.window_count == 0 && !self.frame_buf.is_empty() {
472            self.frame_buf.resize(FFT_SIZE, 0.0);
473            self.process_window();
474        }
475
476        let spectrum_db = if self.selection.spectrum() {
477            spectrum::average_to_db(&self.power_acc, self.window_count)
478        } else {
479            Vec::new()
480        };
481        let (cutoff_hz, cutoff_ratio, cliff_db, above_db) = if self.selection.spectrum() {
482            let (hz, ratio) = spectrum::detect_cutoff(&spectrum_db, sample_rate, FFT_SIZE);
483            let (cliff, above) = spectrum::cutoff_context(&spectrum_db, sample_rate, FFT_SIZE, hz);
484            (hz, ratio, cliff, above)
485        } else {
486            (0.0, 0.0, 0.0, 0.0)
487        };
488
489        // Dynamics: flush a meaningful trailing partial block (>= 1/4 length),
490        // then compare the peak with the RMS of the loudest 20% of blocks.
491        if self.dyn_block_frames >= DYN_BLOCK_FRAMES / 4 {
492            self.dyn_blocks
493                .push(self.dyn_block_sumsq / self.dyn_block_frames as f64);
494        }
495        let dynamics_peak = self.clip_state.unclamped_peak();
496        let mut clipping = if self.selection.sample_peak() {
497            self.clip_state.finish(self.channels, self.total_frames)
498        } else {
499            ClippingInfo::unmeasured()
500        };
501        // True peak from the 4x-oversampled stream. It can legitimately sit a
502        // hair above the sample peak on any material, and above 1.0 (positive
503        // dBTP) on loud masters — that's the inter-sample clipping signal.
504        if let Some(true_peak) = &mut self.true_peak {
505            true_peak.flush_tail();
506            clipping.true_peak = true_peak.peak();
507            clipping.true_peak_dbtp = true_peak.peak_dbtp();
508        }
509        let dr_db = if self.selection.enabled(Kind::Dynamics) {
510            self.dyn_blocks
511                .sort_by(|a, b| b.partial_cmp(a).unwrap_or(std::cmp::Ordering::Equal));
512            let top = ((self.dyn_blocks.len() as f64 * DYN_TOP_FRACTION).ceil() as usize).max(1);
513            let (sum, count) = self
514                .dyn_blocks
515                .iter()
516                .take(top)
517                .fold((0.0, 0usize), |(sum, count), power| {
518                    (sum + power, count + 1)
519                });
520            if count == 0 {
521                None
522            } else {
523                let rms = (sum / count as f64).sqrt();
524                if rms > 1e-9 && dynamics_peak > 0.0 {
525                    Some((20.0 * (dynamics_peak as f64 / rms).log10()) as f32)
526                } else {
527                    None
528                }
529            }
530        } else {
531            None
532        };
533
534        let fake_stereo =
535            if self.stereo_valid && self.selection.enabled(Kind::Stereo) && self.channels >= 2 {
536                super::stereo::is_fake(
537                    self.diff_energy,
538                    self.l_energy,
539                    self.r_energy,
540                    self.identical_frames,
541                    self.total_frames,
542                )
543            } else {
544                false
545            };
546        let phase = (self.stereo_valid && self.selection.enabled(Kind::Phase))
547            .then(|| super::stereo::analyze_phase(self.l_energy, self.r_energy, self.cross_energy));
548        let high_frequency_stereo = self
549            .high_frequency_stereo
550            .take()
551            .and_then(HighFrequencyStereoMeter::finish);
552
553        let measured_depth = self.bit_depth.and_then(|depth| depth.finish(declared_bits));
554        let real_bit_depth = measured_depth.map(|(bits, _)| bits);
555        let bit_depth_evidence = measured_depth.map(|(_, evidence)| evidence);
556
557        let (mdct_cutoff_ratio, mdct_dead_db, mdct_dead_fraction) = if self.mdct_frames > 0 {
558            let frac = self.mdct_dead_frames as f32 / self.mdct_frames as f32;
559            if self.mdct_dead_frames > 0 {
560                (
561                    Some(self.mdct_cutoff_ratio_sum / self.mdct_dead_frames as f64),
562                    Some((self.mdct_dead_db_sum / self.mdct_dead_frames as f64) as f32),
563                    Some(frac),
564                )
565            } else {
566                (None, None, Some(frac))
567            }
568        } else {
569            (None, None, None)
570        };
571
572        // Capture programme loudness and M/S maxima before LRA appends 1.5 s of
573        // analysis-only silence for its centred 3 s tail windows.
