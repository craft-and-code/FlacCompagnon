1//! Effective bit-depth estimation.
2//!
3//! Many "24-bit" files are really 16-bit content padded with zero low bits or
4//! exported with a tiny amount of dither. Exact padding is recovered from the
5//! bitwise OR of every sample. Dithered padding is recognized separately from
6//! the sample distribution around the 16-bit quantization grid.
7
8/// Maximum distance from the nearest 16-bit grid point in a 24-bit sample.
9/// Audacity's shaped dither on the regression file stays within ±10; ±16
10/// leaves headroom without accepting the 128-step spread of real 24-bit data.
11const DITHER_GRID_TOLERANCE: u32 = 16;
12/// A short or nearly silent signal cannot establish a quantization lattice.
13const DITHER_GRID_MIN_SAMPLES: u64 = 4_096;
14const DITHER_GRID_MIN_PEAK: u32 = 4_096;
15
16/// Streaming evidence for a 16-bit quantization grid inside 24-bit PCM.
17#[derive(Debug, Default)]
18pub struct Grid16Evidence {
19    samples: u64,
20    near_grid: u64,
21    peak: u32,
22}
23
24impl Grid16Evidence {
25    /// Add one integer sample in its declared native width.
26    pub fn push(&mut self, sample: i32) {
27        self.samples = self.samples.saturating_add(1);
28        self.peak = self.peak.max(sample.unsigned_abs());
29        let residue = sample.rem_euclid(256) as u32;
30        let distance = residue.min(256 - residue);
31        if distance <= DITHER_GRID_TOLERANCE {
32            self.near_grid = self.near_grid.saturating_add(1);
33        }
34    }
35
36    /// Whether a 24-bit stream follows a 16-bit grid hidden by low-level dither.
37    pub fn detected(&self, declared_bits: u32) -> bool {
38        declared_bits == 24
39            && self.samples >= DITHER_GRID_MIN_SAMPLES
40            && self.peak >= DITHER_GRID_MIN_PEAK
41            && self.near_grid == self.samples
42    }
43}
44
45/// Given the bitwise OR of every integer sample value and the container's
46/// declared bit width, return the effective (used) bit depth.
47///
48/// The OR mask is restricted to the low `declared_bits` bits so that
49/// sign-extension of negative two's-complement samples does not inflate the
50/// result. `effective = declared_bits - trailing_zero_bits`.
51///
52/// * All-zero input returns the arithmetic minimum, 1. This is not evidence
53///   of a one-bit recording: callers must treat silence as unmeasurable.
54pub fn effective_bits(or_mask: u32, declared_bits: u32) -> u32 {
55    let width = declared_bits.clamp(1, 32);
56    let mask = if width >= 32 {
57        u32::MAX
58    } else {
59        (1u32 << width) - 1
60    };
61    let low = or_mask & mask;
62    if low == 0 {
63        return 1;
64    }
65    let trailing = low.trailing_zeros();
66    width.saturating_sub(trailing).max(1)
67}
68
69/// Does non-silent integer content fit in 16 bits within a wider container?
70pub fn is_fake_hires(declared: u32, or_mask: u32) -> bool {
71    (24..=32).contains(&declared) && or_mask != 0 && effective_bits(or_mask, declared) <= 16
72}

74#[cfg(test)]
75mod tests {
76    use super::*;
77
78    // The underscore in each literal below sits at the padding boundary, not
79    // every four digits: `0x1234_00` reads as "the 16-bit sample 0x1234,
80    // shifted into a 24-bit field, low byte zero", which is the whole point of
81    // this test. Clippy's suggested regrouping (`0x0012_3400`) is uniform but
82    // says nothing, so the lint is turned off here rather than obeyed.
83    #[allow(clippy::unusual_byte_groupings)]
84    #[test]
85    fn sixteen_bit_padded_to_24_is_flagged() {
86        // 16-bit samples left-shifted into a 24-bit field => low 8 bits zero.
87        let mut mask = 0u32;
88        for v in [0x1234_00u32, 0x5600_00, 0x00AB_00, 0x7F00_00] {
89            mask |= v;
90        }
91        assert_eq!(effective_bits(mask, 24), 16);
92        assert!(is_fake_hires(24, mask));
93    }
94
95    #[test]
96    fn genuine_24bit_is_not_flagged() {
97        let mut mask = 0u32;
98        for v in [0x123456u32, 0x000001, 0xABCDEF, 0x000003] {
99            mask |= v;
100        }
101        assert!(effective_bits(mask, 24) > 16);
102        assert!(!is_fake_hires(24, mask));
103    }
104
105    #[test]
106    fn sign_extended_negatives_do_not_inflate() {
107        // -256 (0xFFFFFF00) has 8 trailing zero bits; within a 24-bit width the
108        // effective depth is 16, not 32.
109        let mask = (-256i32) as u32;
110        assert_eq!(effective_bits(mask, 24), 16);
111    }
112
113    #[test]
114    fn silence_is_not_evidence_of_fake_hires() {
115        assert!(!is_fake_hires(24, 0));
116        assert!(!is_fake_hires(32, 0));
117    }
118
119    #[test]
120    fn silence_reports_one_bit() {
121        assert_eq!(effective_bits(0, 24), 1);
122    }
123
124    #[test]
125    fn audacity_style_dither_still_reveals_the_16bit_grid() {
126        let mut evidence = Grid16Evidence::default();
127        for n in 0..DITHER_GRID_MIN_SAMPLES {
128            let base = (((n as i32 % 20_001) - 10_000) << 8).clamp(-8_388_608, 8_388_607);
129            let dither = (n as i32 % 21) - 10;
130            evidence.push(base + dither);
131        }
132        assert!(evidence.detected(24));
133    }
134
135    #[test]
136    fn genuine_24bit_residues_do_not_form_a_16bit_grid() {
137        let mut evidence = Grid16Evidence::default();
138        for n in 0..DITHER_GRID_MIN_SAMPLES {
139            evidence.push((n as i32 * 65_537) & 0x7f_ffff);
140        }
141        assert!(!evidence.detected(24));
142    }
143
144    #[test]
145    fn silence_with_low_level_noise_is_not_enough_evidence() {
146        let mut evidence = Grid16Evidence::default();
147        for n in 0..DITHER_GRID_MIN_SAMPLES {
148            evidence.push((n as i32 % 21) - 10);
149        }
150        assert!(!evidence.detected(24));
151    }
