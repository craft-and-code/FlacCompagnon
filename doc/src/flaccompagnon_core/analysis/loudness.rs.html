1//! Shared K-weighted power for integrated loudness, LRA and M/S maxima.
2//!
3//! The meter keeps 400 ms and 3 s of K-weighted power and sampled gate values.
4//! It does not retain decoded samples. Channel positions are required
5//! for a correct multichannel measurement, so unsupported layouts yield no
6//! value rather than incorrectly weighted loudness measurements.
7//! This is one streaming meter: filter history, overlapping gates and the
8//! shared M/S windows must advance together; splitting the state would obscure
9//! that invariant even though the complete algorithm exceeds 300 lines.
10
11use super::compensated_sum::CompensatedSum;
12use super::loudness_peaks::{LoudnessPeaks, LoudnessPeaksMeter};

14/// ITU-R BS.1770-5, Annex 1, Tables 1 and 2 (48 kHz reference filters).
15const SHELF_48K: ([f64; 3], [f64; 3]) = (
16    [
17        1.535_124_859_586_97,
18        -2.691_696_189_406_38,
19        1.198_392_810_852_85,
20    ],
21    [1.0, -1.690_659_293_182_41, 0.732_480_774_215_85],
22);
23const HIGH_PASS_48K: ([f64; 3], [f64; 3]) = (
24    [1.0, -2.0, 1.0],
25    [1.0, -1.990_047_454_833_98, 0.990_072_250_366_21],
26);
27
28/// ITU-R BS.1770-5, Annex 1, equations 2 and 6.
29const LOUDNESS_OFFSET: f64 = -0.691;
30const ABSOLUTE_GATE_LUFS: f64 = -70.0;
31const RELATIVE_GATE_LU: f64 = 10.0;
32/// EBU Tech 3342 (2023), section 3.1: LRA gating and percentile bounds.
33const LRA_RELATIVE_GATE_LU: f64 = 20.0;
34const LRA_LOW_PERCENTILE: f64 = 0.10;
35const LRA_HIGH_PERCENTILE: f64 = 0.95;
36
37#[derive(Clone, Copy)]
38struct Biquad {
39    b: [f64; 3],
40    a: [f64; 3],
41    x1: f64,
42    x2: f64,
43    y1: f64,
44    y2: f64,
45}
46
47impl Biquad {
48    fn new(reference: ([f64; 3], [f64; 3]), sample_rate: u32) -> Self {
49        // Recover the analogue polynomial from BS.1770-5 Tables 1/2.
50        // Prewarp its pole frequency when remapping to a different rate:
51        // a plain Fs/48000 ratio shifts the shelf at low sample rates.
52        fn retune(c: [f64; 3], ratio: f64) -> [f64; 3] {
53            let c0 = c[0] + c[1] + c[2];
54            let c1 = c[0] - c[2];
55            let c2 = c[0] - c[1] + c[2];
56            [
57                c0 + 2.0 * ratio * c1 + ratio * ratio * c2,
58                2.0 * (c0 - ratio * ratio * c2),
59                c0 - 2.0 * ratio * c1 + ratio * ratio * c2,
60            ]
61        }
62        let (b, a) = if sample_rate == 48_000 {
63            reference
64        } else {
65            let denominator = reference.1;
66            let pole = ((denominator[0] + denominator[1] + denominator[2])
67                / (denominator[0] - denominator[1] + denominator[2]))
68                .sqrt();
69            let ratio = pole / (pole.atan() * 48_000.0 / sample_rate as f64).tan();
70            let b = retune(reference.0, ratio);
71            let a = retune(reference.1, ratio);
72            let scale = a[0];
73            (b.map(|v| v / scale), [1.0, a[1] / scale, a[2] / scale])
74        };
75        Self {
76            b,
77            a,
78            x1: 0.0,
79            x2: 0.0,
80            y1: 0.0,
81            y2: 0.0,
82        }
83    }
84
85    fn push(&mut self, x: f64) -> f64 {
86        let y = self.b[0] * x + self.b[1] * self.x1 + self.b[2] * self.x2
87            - self.a[1] * self.y1
88            - self.a[2] * self.y2;
89        self.x2 = self.x1;
90        self.x1 = x;
91        self.y2 = self.y1;
92        self.y1 = y;
93        y
94    }
95}

97struct ChannelFilter {
98    shelf: Biquad,
99    high_pass: Biquad,
100}
101
102impl ChannelFilter {
103    fn new(sample_rate: u32) -> Self {
104        let mut high_pass = Biquad::new(HIGH_PASS_48K, sample_rate);
105        // Table 2's numerator is the unit second difference. Preserve it
106        // while retuning the poles; scaling it with the analogue prototype
107        // introduces a rate-dependent gain error (about 0.2 LU at 8 kHz).
108        high_pass.b = HIGH_PASS_48K.0;
109        Self {
110            shelf: Biquad::new(SHELF_48K, sample_rate),
111            high_pass,
112        }
113    }
114
115    fn push(&mut self, sample: f32) -> f64 {
116        let weighted = self.high_pass.push(self.shelf.push(sample as f64));
117        weighted * weighted
118    }
119}

121/// Streaming K-weighted integrated loudness, M/S maxima and Tech 3342 range.
122pub struct LoudnessMeter {
123    filters: Vec<ChannelFilter>,
124    power_ring: Vec<f64>,
125    ring_at: usize,
126    window_power: CompensatedSum,
127    frames: u64,
128    step_frames: u64,
129    absolute_gate_power: f64,
130    block_powers: Vec<f64>,
131    short_ring: Vec<f32>,
132    short_ring_at: usize,
133    short_window_power: CompensatedSum,
134    short_powers: Vec<f64>,
135    short_valid: bool,
136    valid: bool,
137    peaks: LoudnessPeaksMeter,
138}
139
140impl LoudnessMeter {
141    /// Create a meter for a stream. The channel count is limited to one or two
142    /// until decode paths can supply reliable speaker positions and LFE roles.
143    pub fn new(sample_rate: u32, channels: usize) -> Option<Self> {
144        // 768 kHz bounds both rings to ~12 MB even for a hostile header;
145        // below 8 kHz K-weighting loses useful bandwidth.
146        if !(1..=2).contains(&channels) || !(8_000..=768_000).contains(&sample_rate) {
147            return None;
148        }
149        // ITU-R BS.1770-5 Annex 1: 400 ms gates with 75% overlap. Round the
150        // hop down to whole samples to preserve EBU Tech 3342's >=10 Hz LRA
151        // update rate even at 11025 Hz. At common rates both lengths are exact.
152        let block_frames = (sample_rate as f64 * 0.4).round() as usize;
153        let step_frames = (sample_rate / 10) as u64;
154        let short_frames = sample_rate as usize * 3;
155        Some(Self {
156            filters: (0..channels)
157                .map(|_| ChannelFilter::new(sample_rate))
158                .collect(),
159            power_ring: vec![0.0; block_frames],
160            ring_at: 0,
161            window_power: CompensatedSum::default(),
162            frames: 0,
163            step_frames,
164            absolute_gate_power: 10f64.powf((ABSOLUTE_GATE_LUFS - LOUDNESS_OFFSET) / 10.0),
165            block_powers: Vec::new(),
166            short_ring: vec![0.0; short_frames],
167            short_ring_at: 0,
168            short_window_power: CompensatedSum::default(),
169            short_powers: Vec::new(),
170            short_valid: true,
171            valid: true,
172            peaks: LoudnessPeaksMeter::new(sample_rate, block_frames, short_frames),
173        })
174    }
175
176    /// Push a complete decoded frame, with one normalized sample per channel.
177    pub fn push_frame(&mut self, samples: &[f32]) {
178        // Never publish the valid prefix of a malformed stream as its
179        // whole-file loudness. NaN also poisons all later filter history.
180        if samples.len() != self.filters.len() || samples.iter().any(|s| !s.is_finite()) {
181            self.valid = false;
182        }
183        if !self.valid {
184            return;
185        }
186        let power: f64 = self
187            .filters
188            .iter_mut()
189            .zip(samples)
190            .map(|(filter, &sample)| filter.push(sample))
191            .sum();
192        // Add and remove separately: forming power-old first can erase quiet
193        // terms after a large float-PCM passage, even with compensated sums.
194        self.window_power.push(-self.power_ring[self.ring_at]);
195        self.window_power.push(power);
196        self.power_ring[self.ring_at] = power;
197        self.ring_at = (self.ring_at + 1) % self.power_ring.len();
198
199        // A f32 power ring limits the 3 s history to 9 MB at the highest
200        // supported rate; the running sum and final gates remain f64.
201        // Float PCM can contain enormous finite samples. If their squared
202        // power exceeds f32, the compact ring cannot represent the signal;
203        // withhold S/LRA instead of exporting an infinite or fabricated value.
204        if !power.is_finite() || power > f32::MAX as f64 {
205            self.short_valid = false;
206        }
207        let short_power = if self.short_valid { power as f32 } else { 0.0 };
208        self.short_window_power
209            .push(-f64::from(self.short_ring[self.short_ring_at]));
210        self.short_window_power.push(f64::from(short_power));
211        self.short_ring[self.short_ring_at] = short_power;
212        self.short_ring_at = (self.short_ring_at + 1) % self.short_ring.len();
213        self.frames += 1;
214        let window_power = self.window_power.total();
215        let short_window_power = self.short_window_power.total();
216        self.peaks
217            .push(window_power, short_window_power, self.frames);
218
219        let block_frames = self.power_ring.len() as u64;
220        if self.frames >= block_frames
221            && (self.frames - block_frames).is_multiple_of(self.step_frames)
222        {
223            let power = window_power / block_frames as f64;
224            // Below the fixed absolute gate a block can never contribute,
225            // regardless of the programme's eventual relative threshold.
226            if power > self.absolute_gate_power {
227                self.block_powers.push(power);
228            }
229        }
230        let short_frames = self.short_ring.len() as u64;
231        if self.frames >= short_frames
232            && (self.frames - short_frames).is_multiple_of(self.step_frames)
233        {
234            let power = short_window_power / short_frames as f64;
235            if power >= self.absolute_gate_power && self.short_valid {
236                self.short_powers.push(power);
237            }
238        }
239    }

241    /// Return the absolute- and relative-gated programme loudness in LUFS.
242    /// Silence and files shorter than one 400 ms gate have no reading.
243    pub fn integrated_lufs(&self) -> Option<f32> {
244        if !self.valid {
245            return None;
246        }
247        let absolute_power = self.absolute_gate_power;
248        let (sum, count) = self
249            .block_powers
250            .iter()
251            .filter(|&&power| power > absolute_power)
252            .fold((0.0, 0usize), |(sum, count), &power| {
253                (sum + power, count + 1)
254            });
255        if count == 0 {
256            return None;
257        }
258        let relative_power = (sum / count as f64) / 10f64.powf(RELATIVE_GATE_LU / 10.0);
259        let (sum, count) = self
260            .block_powers
261            .iter()
262            .filter(|&&power| power > absolute_power && power > relative_power)
263            .fold((0.0, 0usize), |(sum, count), &power| {
264                (sum + power, count + 1)
265            });
266        if count == 0 {
267            return None;
268        }
269        power_to_lufs(sum / count as f64)
270    }

272    /// Ungated M/S maxima over complete real-audio windows. Call before LRA
273    /// consumes the meter and adds analysis-only silence to its tail.
274    pub fn peaks(&self) -> Option<LoudnessPeaks> {
275        self.peaks.result(self.valid, self.short_valid)
276    }

278    /// EBU Tech 3342 loudness range in LU. Consumes the meter because the
279    /// file measurement extends the stream by at least 1.5 s of silence.
280    /// The final 3 s window's centre is within one update of the real end.
281    pub fn loudness_range_lu(mut self) -> Option<f32> {
282        if !self.valid || !self.short_valid || self.frames < self.short_ring.len() as u64 {
283            return None;
284        }
285        self.pad_lra_tail();
286
287        let absolute_power = self.absolute_gate_power;
288        let (sum, count) = self
289            .short_powers
290            .iter()
291            .filter(|&&power| power >= absolute_power)
292            .fold((0.0, 0usize), |(sum, count), &power| {
293                (sum + power, count + 1)
294            });
295        if count == 0 {
296            return None;
297        }
298        let relative_power = (sum / count as f64) / 10f64.powf(LRA_RELATIVE_GATE_LU / 10.0);
299        self.short_powers
300            .retain(|&power| power >= absolute_power && power >= relative_power);
301        if self.short_powers.is_empty() {
302            return None;
303        }
304        // Only two order statistics are needed. Reuse the recorded powers
305        // instead of allocating and sorting another programme-sized vector.
306        let last = self.short_powers.len() - 1;
307        let low_at = (last as f64 * LRA_LOW_PERCENTILE).round() as usize;
308        let high_at = (last as f64 * LRA_HIGH_PERCENTILE).round() as usize;
309        if low_at > last || high_at > last {
310            return None;
311        }
312        // Both indices have just been checked against the nonempty vector.
313        let low = *self
314            .short_powers
315            .select_nth_unstable_by(low_at, f64::total_cmp)
316            .1;
317        let high = *self
318            .short_powers
319            .select_nth_unstable_by(high_at, f64::total_cmp)
320            .1;
321        Some((10.0 * (high / low).log10()) as f32)
322    }
323
324    fn pad_lra_tail(&mut self) {
325        // Tech 3342's file measurement uses at least 1.5 s of trailing
326        // silence; round upward when 3*Fs is odd (for example 11025 Hz).
327        let channels = self.filters.len();
328        let silence = [0.0; 2];
329        for _ in 0..self.short_ring.len().div_ceil(2) {
330            self.push_frame(&silence[..channels]);
331        }
332    }
333}

335/// BS.1770-5 Annex 1, equation 2, after channel-weighted mean powers are summed.
336pub(super) fn power_to_lufs(power: f64) -> Option<f32> {
337    (power.is_finite() && power > 0.0).then(|| (LOUDNESS_OFFSET + 10.0 * power.log10()) as f32)
338}
