1//! Stereo relationship: dual-mono, opposing polarity and channel balance.
2//!
3//! A file can claim to be stereo while both channels carry an identical signal.
4//! Two independent conditions flag it:
5//! 1. Every frame had L == R (bit-exact dual mono), or
6//! 2. The L-R difference energy is >= 60 dB below the total channel energy.
7
8use serde::{Deserialize, Serialize};
9
10/// Full-stream, unweighted RMS balance. Silent channels are explicit states
11/// because an infinite dB difference cannot be represented in JSON.
12#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
13#[serde(tag = "state")]
14pub enum StereoBalance {
15    /// Both channels contain signal; zero means equal RMS, not identical audio.
16    Measured {
17        /// Right minus left RMS level in dB: positive means right is louder.
18        right_minus_left_db: f32,
19    },
20    /// Every left-channel sample is zero, with signal on the right.
21    LeftSilent,
22    /// Every right-channel sample is zero, with signal on the left.
23    RightSilent,
24}
25
26/// Compare channel energy over the same frames. Since RMS = sqrt(E / N),
27/// the level difference is 20 log10(RMS_R / RMS_L) = 10 log10(E_R / E_L).
28/// Separate logarithms avoid overflow for very unequal energies. This is a
29/// descriptive measurement: artistic panning can legitimately be asymmetric.
30/// Both channels silent, or invalid energies, yield no measurement.
31pub fn analyze_balance(l_energy: f64, r_energy: f64) -> Option<StereoBalance> {
32    if !l_energy.is_finite() || !r_energy.is_finite() || l_energy < 0.0 || r_energy < 0.0 {
33        return None;
34    }
35    match (l_energy > 0.0, r_energy > 0.0) {
36        (false, false) => None,
37        (false, true) => Some(StereoBalance::LeftSilent),
38        (true, false) => Some(StereoBalance::RightSilent),
39        (true, true) => Some(StereoBalance::Measured {
40            right_minus_left_db: (10.0 * (r_energy.log10() - l_energy.log10())) as f32,
41        }),
42    }
43}
44
45/// Threshold (in dB) below which the L-R difference is considered negligible.
46const DIFF_FLOOR_DB: f64 = -60.0;
47
48/// Project threshold for a likely channel-polarity inversion. A strongly
49/// negative whole-stream correlation detects R = -kL even when unequal channel
50/// gains prevent near-total cancellation in mono. Less negative values remain
51/// a mono-compatibility warning, not a polarity finding.
52const INVERTED_CORRELATION: f64 = -0.95;
53
54/// Correlation and a conservative polarity-inversion finding for two channels.
55#[derive(Debug, Clone, Copy)]
56pub struct PhaseAnalysis {
57    /// Whole-stream L/R correlation in -1..1; absent when either channel is silent.
58    pub correlation: Option<f32>,
59    /// True when the channels are strongly opposed across the whole stream.
60    pub likely_inverted: bool,
61}
62
63/// Measure phase correlation from accumulated channel energies. Absolute
64/// polarity cannot be known without a reference recording;
65/// this only compares the two channels to each other.
66pub fn analyze_phase(l_energy: f64, r_energy: f64, cross_energy: f64) -> PhaseAnalysis {
67    if !l_energy.is_finite()
68        || !r_energy.is_finite()
69        || !cross_energy.is_finite()
70        || l_energy <= f64::EPSILON
71        || r_energy <= f64::EPSILON
72    {
73        return PhaseAnalysis {
74            correlation: None,
75            likely_inverted: false,
76        };
77    }
78    let correlation = (cross_energy / (l_energy * r_energy).sqrt()).clamp(-1.0, 1.0);
79    PhaseAnalysis {
80        correlation: Some(correlation as f32),
81        likely_inverted: correlation <= INVERTED_CORRELATION,
82    }
83}
84
85/// Decide whether a >= 2 channel signal is really dual-mono, from accumulated
86/// energies and the count of bit-identical frames.
87pub fn is_fake(
88    diff_energy: f64,
89    l_energy: f64,
90    r_energy: f64,
91    identical_frames: u64,
92    total_frames: u64,
93) -> bool {
94    if total_frames == 0 {
95        return false;
96    }
97    if identical_frames == total_frames {
98        return true;
99    }
100    let sig_energy = l_energy + r_energy;
101    if sig_energy <= f64::EPSILON {
102        return false; // both channels silent
103    }
104    let ratio_db = 10.0 * (diff_energy / sig_energy).max(1e-30).log10();
105    ratio_db < DIFF_FLOOR_DB
106}
