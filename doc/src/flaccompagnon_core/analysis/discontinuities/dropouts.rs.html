1//! Abrupt digital silence bounded by signal; not a general silence detector.
2
3use super::{context_frames, DiscontinuityEvent, EventSummary};
4
5// Project thresholds: both boundaries must jump by >=0.02 full scale (~−34
6// dBFS). Context must exceed −60 dBFS RMS, and the resumed signal must stay
7// within 12 dB of the preceding RMS. These conservative checks exclude fades,
8// leading/trailing silence, and most unrelated starts/stops.
9const MIN_EDGE: f64 = 0.02;
10const MIN_POWER: f64 = 1e-6;
11const CONTEXT_POWER_RATIO: f64 = 16.0;
12
13struct Gap {
14    start: u64,
15    before_power: f64,
16}
17
18struct Resumption {
19    gap: Gap,
20    duration: u64,
21    power: f64,
22    nonzero: usize,
23    frames: usize,
24}
25
26pub(super) struct DropoutDetector {
27    history: Vec<f64>,
28    at: usize,
29    power: f64,
30    previous: f64,
31    frames: u64,
32    gap: Option<Gap>,
33    resumption: Option<Resumption>,
34    rate: u32,
35    channel: u16,
36}
37
38impl DropoutDetector {
39    pub(super) fn new(rate: u32, channel: u16) -> Self {
40        Self {
41            history: vec![0.0; context_frames(rate)],
42            at: 0,
43            power: 0.0,
44            previous: 0.0,
45            frames: 0,
46            gap: None,
47            resumption: None,
48            rate,
49            channel,
50        }
51    }
52
53    pub(super) fn push(&mut self, sample: f64, output: &mut EventSummary) {
54        let context = self.history.len();
55        if sample == 0.0 && self.previous != 0.0 {
56            let before_power = self.power.max(0.0) / context as f64;
57            if self.frames >= context as u64
58                && self.previous.abs() >= MIN_EDGE
59                && before_power >= MIN_POWER
60            {
61                self.gap = Some(Gap {
62                    start: self.frames,
63                    before_power,
64                });
65            }
66        } else if sample != 0.0 {
67            if let Some(gap) = self.gap.take() {
68                let duration = self.frames - gap.start;
69                // 2–250 ms excludes zero crossings and long musical pauses.
70                if duration >= (self.rate as u64).div_ceil(500)
71                    && duration <= self.rate as u64 / 4
72                    && sample.abs() >= MIN_EDGE
73                {
74                    self.resumption = Some(Resumption {
75                        gap,
76                        duration,
77                        power: 0.0,
78                        nonzero: 0,
79                        frames: 0,
80                    });
81                }
82            }
83        }
84        if let Some(after) = &mut self.resumption {
85            after.power += sample * sample;
86            after.nonzero += usize::from(sample != 0.0);
87            after.frames += 1;
88            if after.frames == context {
89                let power = after.power / context as f64;
90                if power >= MIN_POWER
91                    && after.nonzero * 10 >= context * 9
92                    && power >= after.gap.before_power / CONTEXT_POWER_RATIO
93                    && power <= after.gap.before_power * CONTEXT_POWER_RATIO
94                {
95                    output.record(DiscontinuityEvent {
96                        channel: self.channel,
97                        start_secs: after.gap.start as f64 / self.rate as f64,
98                        duration_secs: after.duration as f64 / self.rate as f64,
99                    });
100                }
101                self.resumption = None;
102            }
103        }
104        self.power += sample * sample - self.history[self.at];
105        self.history[self.at] = sample * sample;
106        self.at = (self.at + 1) % context;
107        self.previous = sample;
108        self.frames += 1;
109    }
110}
