1//! Playlist export (Simple or Extended M3U) — built entirely from data the
2//! frontend already has in memory (the table's current order, each file's
3//! analyzed duration, cached tags), never re-reading the tracks from disk on
4//! this side. Absolute paths are written (a selection can span several
5//! folders after a multi-drop), so the playlist opens correctly from
6//! anywhere, at the cost of breaking if the files are later moved.
7
8use serde::{Deserialize, Serialize};
9
10/// One playlist entry. `title`/`artist` are `None` when tags weren't
11/// available for that file — falls back to the file name, same as most
12/// players do for an untagged track.
13#[derive(Debug, Clone, Serialize, Deserialize)]
14pub struct PlaylistEntry {
15    /// Absolute path to the audio file.
16    pub path: String,
17    /// Track length in seconds, from the analysis already performed.
18    pub duration_secs: f64,
19    /// Track title, when a tag was available.
20    pub title: Option<String>,
21    /// Track artist, when a tag was available.
22    pub artist: Option<String>,
23}
24
25/// Which flavor of M3U to write — picked by the user in the export pop-in
26/// (Extended is the default). Externally-tagged as a bare string over the
27/// wire ("Simple" / "Extended"), same convention as the tag panel's
28/// `FieldEdit`/`CoverEdit` unit variants.
29#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
30pub enum PlaylistFormat {
31    /// One absolute path per line, nothing else — the original M3U format,
32    /// understood by literally everything.
33    Simple,
34    /// `#EXTM3U` header, then one `#EXTINF:duration,Artist - Title` line
35    /// before each path, so a compatible player can show a title without
36    /// opening the file itself.
37    Extended,
38}
39
40/// Build the playlist text in the requested format.
41pub fn build_playlist(entries: &[PlaylistEntry], format: PlaylistFormat) -> String {
42    match format {
43        PlaylistFormat::Simple => build_simple_m3u(entries),
44        PlaylistFormat::Extended => build_extended_m3u(entries),
45    }
46}
47
48/// Simple M3U: just the paths, one per line.
49pub fn build_simple_m3u(entries: &[PlaylistEntry]) -> String {
50    let mut out = String::new();
51    for e in entries {
52        out.push_str(&e.path);
53        out.push('\n');
54    }
55    out
56}
57
58/// Extended M3U: an `#EXTM3U` header, then one `#EXTINF` + path pair per
59/// entry, in the order given.
60pub fn build_extended_m3u(entries: &[PlaylistEntry]) -> String {
61    let mut out = String::from("#EXTM3U\n");
62    for e in entries {
63        let label = match (&e.artist, &e.title) {
64            (Some(a), Some(t)) if !a.is_empty() && !t.is_empty() => format!("{a} - {t}"),
65            (_, Some(t)) if !t.is_empty() => t.clone(),
66            (Some(a), _) if !a.is_empty() => format!("{a} - {}", file_stem(&e.path)),
67            _ => file_stem(&e.path),
68        };
69        out.push_str(&format!(
70            "#EXTINF:{},{}\n{}\n",
71            e.duration_secs.round().max(0.0) as i64,
72            label,
73            e.path,
74        ));
75    }
76    out
77}
78
79fn file_stem(path: &str) -> String {
80    let name = path.rsplit(['/', '\\']).next().unwrap_or(path);
81    match name.rfind('.') {
82        Some(i) if i > 0 => name[..i].to_string(),
83        _ => name.to_string(),
84    }
85}

87#[cfg(test)]
88mod tests {
89    use super::*;
90
91    #[test]
92    fn builds_extm3u_with_tags() {
93        let entries = vec![PlaylistEntry {
94            path: "/music/a.flac".into(),
95            duration_secs: 183.4,
96            title: Some("Song".into()),
97            artist: Some("Artist".into()),
98        }];
99        let m3u = build_extended_m3u(&entries);
100        assert!(m3u.starts_with("#EXTM3U\n"));
101        assert!(m3u.contains("#EXTINF:183,Artist - Song\n"));
102        assert!(m3u.contains("/music/a.flac\n"));
103    }
104
105    #[test]
106    fn falls_back_to_file_stem_without_tags() {
107        let entries = vec![PlaylistEntry {
108            path: "/music/b.flac".into(),
109            duration_secs: 10.0,
110            title: None,
111            artist: None,
112        }];
113        let m3u = build_extended_m3u(&entries);
114        assert!(m3u.contains("#EXTINF:10,b\n"));
115    }
116
117    #[test]
118    fn falls_back_to_artist_plus_file_stem_without_a_title() {
119        let entries = vec![PlaylistEntry {
120            path: "/music/c.flac".into(),
121            duration_secs: 5.0,
122            title: None,
123            artist: Some("Artist".into()),
124        }];
125        let m3u = build_extended_m3u(&entries);
126        assert!(m3u.contains("#EXTINF:5,Artist - c\n"));
127    }
128
129    #[test]
130    fn preserves_given_order() {
131        let entries = vec![
132            PlaylistEntry { path: "/m/2.flac".into(), duration_secs: 1.0, title: Some("Two".into()), artist: None },
133            PlaylistEntry { path: "/m/1.flac".into(), duration_secs: 1.0, title: Some("One".into()), artist: None },
134        ];
135        let m3u = build_extended_m3u(&entries);
136        assert!(m3u.find("Two").unwrap() < m3u.find("One").unwrap());
137    }
138
139    #[test]
140    fn simple_format_is_just_paths() {
141        let entries = vec![
142            PlaylistEntry { path: "/m/1.flac".into(), duration_secs: 1.0, title: Some("One".into()), artist: Some("Artist".into()) },
143            PlaylistEntry { path: "/m/2.flac".into(), duration_secs: 2.0, title: None, artist: None },
144        ];
145        let m3u = build_simple_m3u(&entries);
146        assert_eq!(m3u, "/m/1.flac\n/m/2.flac\n");
147        assert!(!m3u.contains("#EXTM3U"));
148        assert!(!m3u.contains("#EXTINF"));
149    }
150
151    #[test]
152    fn build_playlist_dispatches_on_format() {
153        let entries = vec![PlaylistEntry {
154            path: "/m/1.flac".into(),
155            duration_secs: 1.0,
156            title: None,
157            artist: None,
158        }];
159        assert_eq!(build_playlist(&entries, PlaylistFormat::Simple), "/m/1.flac\n");
160        assert!(build_playlist(&entries, PlaylistFormat::Extended).starts_with("#EXTM3U"));
161    }
