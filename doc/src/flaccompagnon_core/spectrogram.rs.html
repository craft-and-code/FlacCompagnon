1//! File-to-spectrogram adapter over the independent Rust renderer.
2//!
3//! Two bounded decode passes count real frames, then place every FFT window on
4//! that exact timeline. Untrusted duration hints cannot crop the end or stretch
5//! the axis. No complete track or generated image is written to disk here.
6
7use crate::{AnalysisError, BasicInfo};
8pub use flaccompagnon_spectrogram::*;
9use std::path::Path;

11/// Measure an audio file at its native decoded rate, without an external renderer.
12/// DSD64/128 mono/stereo uses the shared native wideband decoder.
13/// `ffmpeg` remains a compatibility option for other existing DSD rates.
14pub fn analyze_file(
15    path: &Path,
16    options: &AnalysisOptions,
17    ffmpeg: Option<&str>,
18    cancelled: &dyn Fn() -> bool,
19) -> Result<(Spectrogram, BasicInfo), AnalysisError> {
20    if cancelled() {
21        return Err(AnalysisError::Decode("cancelled".into()));
22    }
23    let before = std::fs::metadata(path)?;
24    if !before.is_file() {
25        return Err(AnalysisError::Decode(
26            "audio source is not a regular file".into(),
27        ));
28    }
29    let (info, frames) = visit(path, ffmpeg, &mut |_| Ok(()), cancelled)?;
30    let mut builder = SpectrogramBuilder::new(
31        SignalInfo {
32            sample_rate: info.sample_rate,
33            channels: info.channels,
34            frames,
35        },
36        options.clone(),
37    )
38    .map_err(convert_error)?;
39    let (second_info, second_frames) = visit(
40        path,
41        ffmpeg,
42        &mut |samples| {
43            builder
44                .push_interleaved_cancellable(samples, cancelled)
45                .map_err(convert_error)
46        },
47        cancelled,
48    )?;
49    let after = std::fs::metadata(path)?;
50    if frames != second_frames
51        || info.sample_rate != second_info.sample_rate
52        || info.channels != second_info.channels
53        || before.len() != after.len()
54        || before.modified().ok() != after.modified().ok()
55    {
56        return Err(AnalysisError::Decode(
57            "audio source changed during spectrogram generation".into(),
58        ));
59    }
60    let data = builder
61        .finish_cancellable(cancelled)
62        .map_err(convert_error)?;
63    Ok((data, info))
64}

66/// Format actual decoded properties for a standalone image; unknown depth stays unknown.
67pub fn audio_description(info: &BasicInfo) -> String {
68    let rate = f64::from(info.sample_rate) / 1000.0;
69    let depth = info
70        .bits
71        .map(|bits| format!("{bits}-bit"))
72        .unwrap_or_else(|| info.format.clone());
73    let channels = match info.channels {
74        1 => "Mono".into(),
75        2 => "Stereo".into(),
76        count => format!("{count} channels"),
77    };
78    format!("{rate} kHz · {depth} · {channels}")
79}

81/// Whether a source uses the optional external DSD decoder rather than native PCM decoding.
82pub fn requires_dsd_decoder(path: &Path) -> bool {
83    path.extension()
84        .and_then(|value| value.to_str())
85        .is_some_and(|extension| {
86            extension.eq_ignore_ascii_case("dsf") || extension.eq_ignore_ascii_case("dff")
87        })
88}

90fn visit(
91    path: &Path,
92    ffmpeg: Option<&str>,
93    sink: &mut dyn FnMut(&[f32]) -> Result<(), AnalysisError>,
94    cancelled: &dyn Fn() -> bool,
95) -> Result<(BasicInfo, u64), AnalysisError> {
96    if requires_dsd_decoder(path) {
97        let dsd = crate::dsd::parse(path)?;
98        let program = ffmpeg.ok_or_else(|| {
99            AnalysisError::Decode(
100                "DSD content decoding requires FFmpeg; PCM spectrograms do not".into(),
101            )
102        })?;
103        let rate = dsd.sample_rate / 8;
104        let frames =
105            crate::decode::dsd::visit_dsd_pcm(program, path, dsd.channels, rate, sink, cancelled)?;
106        Ok((
107            BasicInfo {
108                sample_rate: rate,
109                channels: dsd.channels,
110                bits: None,
111                format: format!("{} (decoded PCM)", dsd.label()),
112            },
113            frames,
114        ))
115    } else {
116        crate::decode::visit_pcm(path, sink, cancelled)
117    }
118}

120fn convert_error(error: flaccompagnon_spectrogram::Error) -> AnalysisError {
121    AnalysisError::Decode(error.to_string())
122}
