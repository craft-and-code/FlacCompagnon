1//! Assembling the `FileAnalysis` record itself — the parts that describe the
2//! file rather than the audio inside it.
3//!
4//! Split out of `pipeline` because they answer a different question and change
5//! for different reasons: `pipeline` decides *which decode path a file takes*
6//! and what the measurements mean, while everything here is filesystem
7//! metadata and arithmetic over it — size, modification time, fingerprints,
8//! the bitrate that falls out of size and duration. None of it needs a decoder,
9//! and all of it must work on a file that fails to decode at all.
10
11use std::path::Path;
12
13use crate::analysis::detections::Detections;
14use crate::types::{ClippingInfo, FileAnalysis};
15
16/// The record we can already fill in before decoding anything, so an
17/// unreadable file still produces a useful row.
18pub(super) fn skeleton(path: &Path) -> FileAnalysis {
19    // Read once, up front, from the filesystem: available even for a file
20    // whose audio fails to decode, so an unreadable track still reports an
21    // honest size (and modification time) in the table. One `metadata` call
22    // rather than two separate ones for size and mtime.
23    let meta = std::fs::metadata(path).ok();
24    let digest = crate::hash::file_digest(path).ok();
25    let size_bytes = meta.as_ref().map(|m| m.len()).unwrap_or(0);
26    let modified_unix = meta.as_ref().and_then(|m| m.modified().ok()).and_then(|t| {
27        t.duration_since(std::time::UNIX_EPOCH)
28            .ok()
29            .and_then(|d| i64::try_from(d.as_secs()).ok())
30    });
31
32    FileAnalysis {
33        path: path.to_string_lossy().to_string(),
34        file_name: path
35            .file_name()
36            .and_then(|n| n.to_str())
37            .unwrap_or("")
38            .to_string(),
39        format: String::new(),
40        codec: None,
41        ext_mismatch: false,
42        sample_rate: 0,
43        channels: 0,
44        declared_bits: None,
45        duration_secs: 0.0,
46        size_bytes,
47        bitrate_kbps: None, // filled in once `duration_secs` is known — see `analyze_file`
48        modified_unix,
49        detections: Detections::unknown(),
50        cutoff_hz: None,
51        cutoff_ratio: None,
52        real_bit_depth: None,
53        lattice_score: None,
54        fake_stereo: None,
55        badge: None,
56        clipping: ClippingInfo::unmeasured(),
57        dr_db: None,
58        flac_md5: None,
59        // Computed here, beside `size_bytes` and `modified_unix`, and for the
60        // same reason: it describes the file as an object rather than the
61        // audio inside it, so a track that fails to decode still gets an
62        // honest fingerprint. One extra read of the file — negligible next to
63        // the spectral analysis and the lattice sweep that follow.
64        file_md5: digest.as_ref().map(|d| d.md5.clone()),
65        file_crc32: digest.as_ref().map(|d| d.crc32.clone()),
66        error: None,
67    }
68}

70/// Average bitrate in kbps from the file's own size and duration — see
71/// [`FileAnalysis::bitrate_kbps`] for why this (not a codec-reported figure)
72/// is what's shown.
73pub(super) fn bitrate_kbps(size_bytes: u64, duration_secs: f64) -> Option<u32> {
74    if duration_secs <= 0.0 {
75        return None;
76    }
77    let kbps = (size_bytes as f64 * 8.0) / duration_secs / 1000.0;
78    if kbps.is_finite() && kbps >= 0.0 {
79        Some(kbps.round() as u32)
80    } else {
81        None
82    }
83}

86#[cfg(test)]
87mod tests {
88    use super::*;
89
90    /// The skeleton must be usable for a file the decoder will later reject —
91    /// that is the whole reason it exists before decoding.
92    #[test]
93    fn a_file_that_cannot_be_decoded_still_gets_size_and_fingerprints() {
94        let dir = tempfile::tempdir().expect("tempdir");
95        let path = dir.path().join("not-audio.flac");
96        std::fs::write(&path, b"this is not a FLAC file").expect("write");
97
98        let r = skeleton(&path);
99        assert_eq!(r.file_name, "not-audio.flac");
100        assert_eq!(r.size_bytes, 23);
101        assert!(r.modified_unix.is_some());
102        // The fingerprints are of the bytes on disk, so they exist here even
103        // though nothing about this file is audio.
104        assert_eq!(r.file_md5.as_deref().map(str::len), Some(32));
105        assert_eq!(r.file_crc32.as_deref().map(str::len), Some(8));
106        // Nothing has been measured yet, so nothing is claimed.
107        assert!(!r.detections.upscaling && !r.detections.upsampling && !r.detections.transcoding);
108        assert_eq!(r.sample_rate, 0);
109        assert!(r.error.is_none());
110    }
111
112    /// A path that does not exist must not panic, and must not invent values.
113    #[test]
114    fn a_missing_file_leaves_the_optional_fields_empty() {
115        let dir = tempfile::tempdir().expect("tempdir");
116        let r = skeleton(&dir.path().join("absent.flac"));
117        assert_eq!(r.size_bytes, 0);
118        assert!(r.modified_unix.is_none());
119        assert!(r.file_md5.is_none());
120        assert!(r.file_crc32.is_none());
121    }
122
123    #[test]
124    fn bitrate_is_size_over_duration_and_refuses_the_degenerate_cases() {
125        // 1 MB over 8 seconds = 1000 kbps exactly.
126        assert_eq!(bitrate_kbps(1_000_000, 8.0), Some(1000));
127        assert_eq!(bitrate_kbps(1_000_000, 0.0), None, "zero duration");
128        assert_eq!(bitrate_kbps(0, 300.0), Some(0), "an empty file has no bitrate, not no answer");
129        assert_eq!(bitrate_kbps(1_000_000, f64::NAN), None, "NaN duration");
130    }
131}
