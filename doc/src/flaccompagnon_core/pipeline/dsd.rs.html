1//! The DSD branch of the analysis pipeline.
2//!
3//! Separate from the PCM path in [`super`] because almost nothing is shared:
4//! the container is parsed by hand rather than probed, decoding needs an
5//! external ffmpeg, only one of the three detections can even apply, and the
6//! badge is granted on different grounds. The two only meet at
7//! [`FileAnalysis`], which both fill in.
8//!
9//! `crate::dsd` (container parsing + spectral heuristics) is aliased to
10//! `dsd_format` here so it is never confused with this module.
11
12use std::path::Path;
13
14use crate::decode;
15use crate::detections::{Detections, TranscodeState};
16use crate::dsd as dsd_format;
17use crate::types::{FileAnalysis, ScanOptions};
18
19/// ffmpeg decodes a 1-bit DSD stream to PCM at one eighth of its rate
20/// (352.8 kHz for DSD64). The spectral analysis must use *that* rate, not the
21/// 1-bit rate from the header.
22const DSD_TO_PCM_DIVISOR: u32 = 8;
23
24/// Below this boundary the brick wall came from a 44.1 kHz source, above it
25/// from a 48 kHz one — the two Nyquist frequencies are 22.05 and 24 kHz, so
26/// anything under 23 kHz is the former.
27const CD_NYQUIST_CEILING_HZ: f64 = 23_000.0;
28
29/// Exact DSF/DFF header verification plus, when ffmpeg is available, a content
30/// check on the decoded PCM (the PCM-source brick wall).
31pub(super) fn analyze(path: &Path, opts: &ScanOptions, result: &mut FileAnalysis) {
32    let info = match dsd_format::parse(path) {
33        Ok(i) => i,
34        Err(e) => {
35            result.error = Some(e.to_string());
36            return;
37        }
38    };
39    result.format = info.label();
40    result.sample_rate = info.sample_rate;
41    result.channels = info.channels;
42    result.duration_secs = info.duration_secs();
43
44    let mut flagged: Option<dsd_format::PcmSourceCheck> = None;
45    let mut analyzed = false;
46
47    if info.dst_compressed {
48        result.detections = verdict(
49            false,
50            "DST-compressed DFF: header verified; content analysis is not supported for DST streams.",
51            "Unknown",
52        );
53    } else if let Some(ffmpeg) = &opts.ffmpeg {
54        let decoded_rate = (info.sample_rate / DSD_TO_PCM_DIVISOR).max(1);
55        match decode::decode_and_analyze_dsd(ffmpeg, path, info.channels, decoded_rate) {
56            Ok(outcome) => {
57                analyzed = true;
58                // The DFF header carries a byte size, not a sample count, so
59                // its duration is often unknown until the audio is decoded.
60                if result.duration_secs == 0.0 {
61                    result.duration_secs = outcome.duration_secs;
62                }
63                let summary = outcome.analyzer.finish(decoded_rate, None);
64                result.cutoff_hz = Some(summary.cutoff_hz);
65                result.cutoff_ratio = Some(summary.cutoff_ratio);
66                result.clipping = summary.clipping.clone();
67                result.dr_db = summary.dr_db;
68                if info.channels >= 2 {
69                    result.fake_stereo = Some(summary.fake_stereo);
70                }
71                flagged = dsd_format::pcm_source_check(
72                    &summary.spectrum_db,
73                    decoded_rate,
74                    summary.fft_size(),
75                );
76                result.detections = match flagged {
77                    Some(hit) => verdict(true, &pcm_source_detail(hit), "Flagged"),
78                    None => verdict(
79                        false,
80                        "Content blends into the sigma-delta noise shaping with no PCM brick wall — consistent with native DSD.",
81                        "Clean",
82                    ),
83                };
84            }
85            Err(e) => result.error = Some(e.to_string()),
86        }
87    } else {
88        result.detections = verdict(
89            false,
90            "DSD header verified. Install ffmpeg to enable the content authenticity check.",
91            "Unknown",
92        );
93    }
94
95    // DSD badge: header authentic and content not flagged as PCM-sourced.
96    // "(unverified)" when only the header could be checked, so the badge never
97    // claims more than was actually measured.
98    if result.error.is_none() && flagged.is_none() {
99        result.badge = Some(if analyzed {
100            info.label()
101        } else {
102            format!("{} (unverified)", info.label())
103        });
104    }
105}

107fn pcm_source_detail(hit: dsd_format::PcmSourceCheck) -> String {
108    let source = if hit.boundary_hz < CD_NYQUIST_CEILING_HZ {
109        "44.1 kHz"
110    } else {
111        "48 kHz"
112    };
113    format!(
114        "Upsampling: PCM-sourced DSD — digital brick wall at ~{:.2} kHz ({:.0} dB drop). \
115         The 1-bit stream was converted from a {source} PCM source.",
116        hit.boundary_hz / 1000.0,
117        hit.drop_db,
118    )
119}
120
121/// A DSD verdict. Only `upsampling` can ever fire on this path: the other two
122/// detections need a PCM bit depth and an MDCT grid, neither of which a 1-bit
123/// stream has — so hardcoding them false here is a statement, not a shortcut.
124fn verdict(upsampling: bool, detail: &str, summary: &str) -> Detections {
125    Detections {
126        upscaling: false,
127        upsampling,
128        transcoding: TranscodeState::None,
129        detail: detail.to_string(),
130        summary: summary.to_string(),
131    }
132}

134#[cfg(test)]
135mod tests {
136    use super::*;
137
138    #[test]
139    fn attributes_the_brick_wall_to_the_right_source_rate() {
140        let cd = pcm_source_detail(dsd_format::PcmSourceCheck {
141            boundary_hz: 22_050.0,
142            drop_db: 51.0,
143        });
144        assert!(cd.contains("44.1 kHz"), "{cd}");
145        assert!(cd.contains("22.05 kHz"), "{cd}");
146
147        let dvd = pcm_source_detail(dsd_format::PcmSourceCheck {
148            boundary_hz: 24_000.0,
149            drop_db: 48.0,
150        });
151        assert!(dvd.contains("48 kHz"), "{dvd}");
152    }
153
154    /// The three non-flagged verdicts must never claim a detection fired.
155    #[test]
156    fn informational_verdicts_flag_nothing() {
157        for v in [
158            verdict(false, "header only", "Unknown"),
159            verdict(false, "native", "Clean"),
160        ] {
161            assert!(!v.upscaling);
162            assert!(!v.upsampling);
163            assert_eq!(v.transcoding, TranscodeState::None);
164        }
165    }
166}
