1//! The bridge between a file on disk and the lattice detectors.
2//!
3//! [`crate::transcode`] is deliberately pure: it takes samples and returns a
4//! verdict, touches no filesystem, and can therefore be tested on synthetic
5//! signals with no fixtures. Getting the samples to it is this file's job, and
6//! it lives in the pipeline because that is where decoding and orchestration
7//! already belong.
8//!
9//! # Why this decodes a second time
10//!
11//! The main analysis pass streams: it consumes packets, folds each into
12//! running statistics, and drops them. That is what lets a 200-track folder be
13//! analyzed in parallel without filling memory.
14//!
15//! The detectors cannot work that way. They re-measure the same frames at
16//! 1024 different sample alignments, so they need the whole signal resident,
17//! in `f64`, for the duration. A five-minute stereo track at 44.1 kHz is about
18//! 210 MB in that form — acceptable for one file at a time, ruinous for a
19//! dozen decoded in parallel.
20//!
21//! The file is decoded again here and the buffer is released after its
22//! sweeps. Concurrent callers each hold their own buffer; this function
23//! does not serialize them or impose a global memory ceiling.
24

33/// How many channels the detectors look at.
34///
35/// Two: a stereo master's content is in the first pair, and the detectors
36/// already test that pair four ways (L, R, M, S). Surround channels would
37/// multiply the cost of the most expensive step in the app for content that
38/// carries the same lattice as the front pair.
39const MAX_CHANNELS: usize = 2;
40
41/// Decode `path` and ask **every** codec detector whether the audio sits on
42/// its lattice, returning the strongest evidence found.
43///
44/// Both detectors run, and both must: they look in different transforms, so a
45/// file the AAC sweep calls clean can still be an MP3 transcode, and vice
46/// versa. Measured on a real 128 kbps MP3 re-wrapped as FLAC, the MP3 sweep
47/// scored 0.184 against its threshold while the AAC sweep scored 0.011 against
48/// its own 0.0125 — the AAC detector would have declared that file genuine on
49/// its own.
50///
51/// Doubling the sweep doubles the cost, which is the most expensive step in
52/// the app. It is not optional all the same: half a detector is a detector
53/// that reports "clean" for the wrong reason.
54///
55/// `Err` when the file could not be decoded, its sample rate has no
56/// tabulated bands, it is too short, or `cancelled` fired. An `Err` is not
57/// "clean" — it is "no answer" — and it says which of those happened, because
58/// a bare "no answer" is exactly as unactionable in a bug report as it is in
59/// the results table.
60pub fn detect(
61    path: &Path,
62    aac_params: &AacParams,
63    mp3_params: &Mp3Params,
64    cancelled: &(dyn Fn() -> bool + Sync),
65) -> LatticeResult {
66    if cancelled() {
67        return Err(LatticeSkip::Cancelled);
68    }
69    // Check the decoded rate before either sweep so the reason is
70    // the *specific* one: the detectors would refuse an untabulated rate too,
71    // but by then it is indistinguishable from every other refusal.
72    let pcm = decode_to_pcm(path).map_err(|e| LatticeSkip::Undecodable(e.to_string()))?;
73    if supported_rate_khz(pcm.sample_rate).is_none() {
74        return Err(LatticeSkip::UnsupportedRate(pcm.sample_rate));
75    }
76    let channels = deinterleave(&pcm.samples, pcm.channels).ok_or(LatticeSkip::NoSamples)?;
77    // Dropped before the sweep starts: the interleaved `f32` copy is dead
78    // weight from here on, and this is the one place in the app where holding
79    // an extra copy of a whole track actually matters.
80    drop(pcm.samples);
81
82    let found_aac = aac::detect(&channels, pcm.sample_rate, aac_params, cancelled);
83    if cancelled() {
84        return Err(LatticeSkip::Cancelled);
85    }
86    let found_mp3 = mp3::detect(&channels, pcm.sample_rate, mp3_params, cancelled);
87    if cancelled() {
88        return Err(LatticeSkip::Cancelled);
89    }
90    combine(
91        found_aac,
92        found_mp3,
93        aac_params.significance,
94        mp3_params.significance,
95    )
96}

98fn combine(
99    found_aac: Option<TranscodeEvidence>,
100    found_mp3: Option<TranscodeEvidence>,
101    aac_threshold: f64,
102    mp3_threshold: f64,
103) -> LatticeResult {
104    // A positive from either codec wins outright — the likelihoods are *not*
105    // comparable across detectors, since each is a fraction measured against
106    // its own threshold, and the two thresholds differ. Comparing the raw
107    // numbers would let a near-miss on one codec outrank a clear hit on the
108    // other.
109    match (found_aac, found_mp3) {
110        (Some(a), Some(m)) => Ok(match (a.detected, m.detected) {
111            (true, false) => a,
112            (false, true) => m,
113            // Both agree: report whichever is further past its own threshold.
114            _ => {
115                let margin = |e: &TranscodeEvidence, lambda: f64| e.likelihood / lambda;
116                if margin(&m, mp3_threshold) > margin(&a, aac_threshold) {
117                    m
118                } else {
119                    a
120                }
121            }
122        }),
123        // One positive is useful evidence, but one negative cannot clear a
124        // codec that lacked enough frames to run its calibrated search.
125        (Some(e), None) | (None, Some(e)) if e.detected => Ok(e),
126        _ => Err(LatticeSkip::TooShort),
127    }
128}

130/// Split interleaved samples into one `f64` buffer per channel, keeping at
131/// most [`MAX_CHANNELS`].
132///
133/// `None` for a zero-channel or empty decode — both mean there is nothing to
134/// analyze, and neither should reach the detector as an empty channel list it
135/// would have to guess about.
136fn deinterleave(samples: &[f32], channels: usize) -> Option<Vec<Vec<f64>>> {
137    if channels == 0 || samples.is_empty() {
138        return None;
139    }
140    let kept = channels.min(MAX_CHANNELS);
141    // Integer division: a truncated final frame (a decoder returning half a
142    // frame at EOF) is dropped rather than padded with silence, which would
143    // put a discontinuity into the very last analysis window.
144    let frames = samples.len() / channels;
145    if frames == 0 {
146        return None;
147    }
148
149    let mut out = vec![Vec::with_capacity(frames); kept];
150    for f in 0..frames {
151        for (c, buf) in out.iter_mut().enumerate() {
152            // `get` rather than indexing: `samples` comes from a decoded file,
153            // and this crate does not panic on anything a file produced.
154            if let Some(s) = samples.get(f * channels + c) {
155                buf.push(*s as f64);
156            }
157        }
158    }
159    Some(out)
160}

166    #[test]
167    fn a_missing_codec_cannot_be_cleared_by_the_other_codecs_negative() {
168        let negative = || TranscodeEvidence {
169            likelihood: 0.01,
170            offset: 0,
171            detected: false,
172        };
173        assert!(matches!(
174            combine(None, Some(negative()), 0.0125, 0.031),
175            Err(LatticeSkip::TooShort)
176        ));
177        assert!(matches!(
178            combine(Some(negative()), None, 0.0125, 0.031),
179            Err(LatticeSkip::TooShort)
180        ));
181        assert!(
182            !combine(Some(negative()), Some(negative()), 0.0125, 0.031)
183                .unwrap()
184                .detected
185        );
186    }

207    #[test]
208    fn deinterleave_splits_channels_in_order() {
209        // L R L R L R
210        let inter = [1.0f32, -1.0, 2.0, -2.0, 3.0, -3.0];
211        let got = deinterleave(&inter, 2).expect("two channels");
212        assert_eq!(got.len(), 2);
213        assert_eq!(got[0], vec![1.0, 2.0, 3.0]);
214        assert_eq!(got[1], vec![-1.0, -2.0, -3.0]);
215    }

217    #[test]
218    fn mono_stays_one_channel() {
219        let got = deinterleave(&[1.0f32, 2.0, 3.0], 1).expect("mono");
220        assert_eq!(got.len(), 1);
221        assert_eq!(got[0], vec![1.0, 2.0, 3.0]);
222    }

224    /// Surround is truncated to the front pair, and — the part worth
225    /// asserting — the front pair must still be the *right* samples, not
226    /// whatever the first two slots of a 6-channel frame happen to hold after
227    /// a stride mistake.
228    #[test]
229    fn surround_keeps_the_front_pair_correctly_strided() {
230        let mut inter = Vec::new();
231        for f in 0..4 {
232            for c in 0..6 {
233                inter.push((f * 10 + c) as f32);
234            }
235        }
236        let got = deinterleave(&inter, 6).expect("surround");
237        assert_eq!(got.len(), MAX_CHANNELS);
238        assert_eq!(got[0], vec![0.0, 10.0, 20.0, 30.0]);
239        assert_eq!(got[1], vec![1.0, 11.0, 21.0, 31.0]);
240    }

242    /// A decoder returning a partial final frame must not produce channels of
243    /// unequal length: the detector zips them against each other.
244    #[test]
245    fn a_truncated_final_frame_is_dropped_not_padded() {
246        // Five samples across two channels: two whole frames plus a stray.
247        let got = deinterleave(&[1.0f32, 2.0, 3.0, 4.0, 5.0], 2).expect("ragged");
248        assert_eq!(got[0].len(), got[1].len(), "channels must match in length");
249        assert_eq!(got[0], vec![1.0, 3.0]);
250        assert_eq!(got[1], vec![2.0, 4.0]);
251    }

299    /// Every reason must read as a sentence a user can act on — an empty or
300    /// duplicated one would leave the table saying nothing again.
301    #[test]
302    fn every_skip_reason_is_distinct_and_non_empty() {
303        let all = [
304            LatticeSkip::Undecodable("boom".into()),
305            LatticeSkip::NoSamples,
306            LatticeSkip::UnsupportedRate(96_000),
307            LatticeSkip::TooShort,
308            LatticeSkip::Cancelled,
309        ];
310        let texts: Vec<String> = all.iter().map(|s| s.to_string()).collect();
311        for t in &texts {
312            assert!(!t.is_empty());
313        }
314        for (i, a) in texts.iter().enumerate() {
315            for b in texts.iter().skip(i + 1) {
316                assert_ne!(a, b);
317            }
318        }
319        assert!(texts[0].contains("boom"), "{}", texts[0]);
320        assert!(texts[2].contains("96.0 kHz"), "{}", texts[2]);
321    }
