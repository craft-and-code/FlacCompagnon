1//! Map decoded measurements and requested authenticity evidence into a record.
2use crate::analysis::detections;
3use crate::decode;
4use crate::dsd as dsd_format;
5use crate::types::FileAnalysis;
6
7/// Fold a successful decode into `result`, and return the DSD-heritage rise
8/// (in dB) if the ultrasonic content shows one.
9pub(super) fn apply_outcome(
10    outcome: decode::DecodeOutcome,
11    result: &mut FileAnalysis,
12    transcoded: Option<crate::transcode::LatticeResult>,
13) -> Option<f32> {
14    result.format = outcome.format;
15    result.codec = outcome.codec;
16    result.sample_rate = outcome.sample_rate;
17    result.channels = outcome.channels;
18    result.declared_bits = outcome.declared_bits;
19    result.duration_secs = outcome.duration_secs;
20
21    let summary = outcome
22        .analyzer
23        .finish(outcome.sample_rate, outcome.declared_bits);
24
25    result.cutoff_hz = Some(summary.cutoff_hz);
26    result.cutoff_ratio = Some(summary.cutoff_ratio);
27    // The *old* detector's `summary.requant_rate` is no longer surfaced: it
28    // came from the spectral machinery this rewrite replaced. What the column
29    // shows now is the lattice search's own score.
30    // Borrowed, not consumed: `classify` needs the same value below, and
31    // `LatticeResult` carries a `String` in its error case so it is not `Copy`.
32    result.lattice_score = transcoded
33        .as_ref()
34        .and_then(|r| r.as_ref().ok())
35        .map(|e| e.likelihood as f32);
36    result.clipping = summary.clipping.clone();
37    result.dr_db = summary.dr_db;
38    result.integrated_lufs = summary.integrated_lufs;
39    result.loudness_peaks = summary.loudness_peaks;
40    result.loudness_range_lu = summary.loudness_range_lu;
41    // Bounded event lists are cheap to copy; classification still borrows
42    // the complete summary below.
43    result.discontinuities = summary.discontinuities.clone();
44    result.stereo_balance = summary.stereo_balance;
45    result.high_frequency_stereo = summary.high_frequency_stereo;
46    // At most 32 means; retain the summary for classification below.
47    result.dc_offset = summary.dc_offset.clone();
48    result.local_phase = summary.local_phase;
49    result.bit_depth_evidence = summary.bit_depth_evidence;
50
51    if outcome.channels >= 2 {
52        result.fake_stereo = Some(summary.fake_stereo);
53    }
54    if outcome.channels == 2 {
55        result.phase_correlation = summary.phase_correlation;
56        result.phase_inverted = summary.phase_correlation.map(|_| summary.phase_inverted);
57    }
58    let real_bits = match (outcome.declared_bits, summary.real_bit_depth) {
59        (Some(_), Some(real)) => {
60            result.real_bit_depth = Some(real);
61            Some(real)
62        }
63        _ => None,
64    };
65    // `transcoded` is handed in — see the call site for why. An `Err` there
66    // means "no answer" (unsupported rate, too short, undecodable), never
67    // "clean": an un-run test must leave the flag down and say why, which is
68    // what carrying the reason all the way here buys.
69    let transcoded = transcoded?;
70    result.detections = detections::classify(
71        &summary,
72        outcome.sample_rate,
73        outcome.declared_bits,
74        real_bits,
75        transcoded,
76    );
77
78    dsd_format::dsd_heritage_check(
79        &summary.spectrum_db,
80        outcome.sample_rate,
81        summary.fft_size(),
82    )
83}
