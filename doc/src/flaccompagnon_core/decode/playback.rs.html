1//! Decoding for the preview player rather than for analysis: no analyzer, no
2//! integer reconstruction, just interleaved `f32` out.
3//!
4//! Two shapes, because the player needs both:
5//!
6//! * [`PcmStreamDecoder`] — progressive. Exposes the sample rate and channel
7//!   count from the header alone, then yields one packet at a time, so the
8//!   audio device can open and start sounding before the file has finished
9//!   decoding. This is the normal path.
10//! * [`decode_to_pcm`] — whole file in memory. Only the resampling fallback
11//!   needs this (it has to know the whole signal before it can resample it);
12//!   a typical track is well under 100 MB decoded.
13//!
14//! DSD (`.dsf`/`.dff`) is not handled here — Symphonia cannot decode it, and
15//! callers check for it before getting this far.
16
17use std::path::Path;
18
19use symphonia::core::codecs::Decoder;
20use symphonia::core::errors::Error as SymError;
21use symphonia::core::formats::FormatReader;
22
23use super::probe::{probe, InterleavedBuf};
24use crate::AnalysisError;
25
26/// Fully-decoded PCM audio: interleaved `f32` samples in `[-1.0, 1.0]`, at the
27/// file's own sample rate and channel count.
28pub struct PcmAudio {
29    /// Interleaved `f32` samples, `[-1.0, 1.0]`.
30    pub samples: Vec<f32>,
31    /// Sample rate the file was decoded at, in Hz.
32    pub sample_rate: u32,
33    /// Channel count the samples are interleaved across.
34    pub channels: usize,
35}
36
37/// Decode `path` into an in-memory interleaved PCM buffer.
38///
39/// Built on [`PcmStreamDecoder`] rather than repeating the decode loop: that
40/// costs one `Vec` per packet over a hand-rolled version, which is invisible
41/// next to decoding the file at all, and it means a fix to the packet loop
42/// cannot land in one of the two paths and miss the other.
43pub fn decode_to_pcm(path: &Path) -> Result<PcmAudio, AnalysisError> {
44    let mut decoder = PcmStreamDecoder::open(path)?;
45    let sample_rate = decoder.sample_rate;
46    let channels = decoder.channels;
47
48    let mut samples: Vec<f32> = Vec::new();
49    while let Some(chunk) = decoder.next_chunk()? {
50        samples.extend_from_slice(&chunk);
51    }
52
53    if samples.is_empty() {
54        return Err(AnalysisError::Decode("no audio data decoded".into()));
55    }
56
57    Ok(PcmAudio {
58        samples,
59        sample_rate,
60        channels,
61    })
62}

64/// Progressive decoder: opens the file and exposes its `sample_rate` /
65/// `channels` immediately (from the container header, before decoding any
66/// audio), then hands out one packet's worth of interleaved `f32` samples at
67/// a time via [`next_chunk`](Self::next_chunk).
68pub struct PcmStreamDecoder {
69    format: Box<dyn FormatReader>,
70    decoder: Box<dyn Decoder>,
71    track_id: u32,
72    buf: InterleavedBuf,
73    /// Sample rate read from the container header, in Hz.
74    pub sample_rate: u32,
75    /// Channel count read from the container header.
76    pub channels: usize,
77}
78
79impl PcmStreamDecoder {
80    /// Open `path` and probe its header, without decoding any packets yet.
81    pub fn open(path: &Path) -> Result<Self, AnalysisError> {
82        let probed = probe(path, true)?;
83        let sample_rate = probed.sample_rate()?;
84        let channels = probed.channels()?;
85        let decoder = probed.make_decoder()?;
86        Ok(Self {
87            format: probed.format,
88            decoder,
89            track_id: probed.track_id,
90            buf: InterleavedBuf::default(),
91            sample_rate,
92            channels,
93        })
94    }
95
96    /// Decode and return the next packet's interleaved `f32` samples, or
97    /// `Ok(None)` at end of stream. A corrupt packet is skipped rather than
98    /// ending the stream early — one bad packet in the middle of a file
99    /// should cost a few milliseconds of audio, not the rest of the track.
100    pub fn next_chunk(&mut self) -> Result<Option<Vec<f32>>, AnalysisError> {
101        loop {
102            let packet = match self.format.next_packet() {
103                Ok(p) => p,
104                Err(SymError::IoError(e)) if e.kind() == std::io::ErrorKind::UnexpectedEof => {
105                    return Ok(None)
106                }
107                Err(SymError::ResetRequired) => return Ok(None),
108                Err(e) => return Err(AnalysisError::Decode(format!("packet error: {e}"))),
109            };
110            if packet.track_id() != self.track_id {
111                continue;
112            }
113
114            match self.decoder.decode(&packet) {
115                Ok(decoded) => return Ok(Some(self.buf.fill(decoded).to_vec())),
116                Err(SymError::DecodeError(_)) => continue, // skip a corrupt packet
117                Err(SymError::IoError(e)) if e.kind() == std::io::ErrorKind::UnexpectedEof => {
118                    return Ok(None)
119                }
120                Err(e) => return Err(AnalysisError::Decode(format!("decode error: {e}"))),
121            }
122        }
123    }
124}
