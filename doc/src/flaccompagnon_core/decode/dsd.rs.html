1//! DSD decoding, delegated to ffmpeg.
2//!
3//! Symphonia has no DSD decoder, and demodulating a 1-bit sigma-delta stream
4//! to PCM properly is a filter design problem in its own right — so this shells
5//! out to the ffmpeg the user already has (the same one the spectrogram
6//! feature needs). No shell is involved: arguments go through `Command::arg`,
7//! never a command line.
8
9use std::io::Read;
10use std::path::Path;
11use std::process::{Command, Stdio};
12
13use super::DecodeOutcome;
14use crate::analyzer::StreamAnalyzer;
15use crate::AnalysisError;
16
17/// Read size for ffmpeg's stdout. Large enough that the pipe is rarely the
18/// bottleneck, small enough to stay a rounding error against the decoded
19/// audio itself.
20const PIPE_CHUNK: usize = 1 << 16;
21
22/// Decode a DSD file (DSF/DFF) through `ffmpeg` and stream the PCM into the
23/// analyzer.
24///
25/// ffmpeg converts DSD to float PCM at `dsd_rate / 8` (352.8 kHz for DSD64);
26/// `decoded_rate` is that *decoded* rate, and it is what the spectral analysis
27/// must use — not the 1-bit rate from the container header.
28pub fn decode_and_analyze_dsd(
29    ffmpeg: &str,
30    path: &Path,
31    channels: usize,
32    decoded_rate: u32,
33) -> Result<DecodeOutcome, AnalysisError> {
34    if channels == 0 {
35        return Err(AnalysisError::Decode("DSD: zero channels".into()));
36    }
37
38    let mut child = Command::new(ffmpeg)
39        .arg("-v")
40        .arg("error")
41        .arg("-i")
42        .arg(path)
43        .args(["-f", "f32le", "-"])
44        .stdout(Stdio::piped())
45        .stderr(Stdio::null())
46        .stdin(Stdio::null())
47        .spawn()
48        .map_err(|e| AnalysisError::Decode(format!("ffmpeg spawn failed: {e}")))?;
49
50    let mut stdout = child
51        .stdout
52        .take()
53        .ok_or_else(|| AnalysisError::Decode("ffmpeg produced no output pipe".into()))?;
54
55    let mut analyzer = StreamAnalyzer::new(channels);
56    let mut carry: Vec<u8> = Vec::new();
57    let mut buf = vec![0u8; PIPE_CHUNK];
58    let mut frame = vec![0f32; channels];
59    let frame_bytes = channels * 4;
60    let mut frame_count: u64 = 0;
61
62    loop {
63        let n = stdout
64            .read(&mut buf)
65            .map_err(|e| AnalysisError::Decode(format!("ffmpeg read failed: {e}")))?;
66        if n == 0 {
67            break;
68        }
69        carry.extend_from_slice(&buf[..n]);
70        // Only whole frames can be pushed; a partial one at the end of this
71        // read stays in `carry` and is completed by the next one.
72        let whole = carry.len() / frame_bytes * frame_bytes;
73        for chunk in carry[..whole].chunks_exact(frame_bytes) {
74            for (c, s) in frame.iter_mut().enumerate() {
75                let o = c * 4;
76                *s = f32::from_le_bytes([chunk[o], chunk[o + 1], chunk[o + 2], chunk[o + 3]]);
77            }
78            analyzer.push_frame(&frame, None);
79            frame_count += 1;
80        }
81        carry.drain(..whole);
82    }
83
84    let status = child
85        .wait()
86        .map_err(|e| AnalysisError::Decode(format!("ffmpeg wait failed: {e}")))?;
87    if !status.success() || frame_count == 0 {
88        return Err(AnalysisError::Decode(
89            "ffmpeg could not decode this DSD file".into(),
90        ));
91    }
92
93    Ok(DecodeOutcome {
94        format: "DSD".to_string(),
95        sample_rate: decoded_rate,
96        channels,
97        declared_bits: None, // 1-bit stream; PCM bit depth does not apply
98        duration_secs: frame_count as f64 / decoded_rate.max(1) as f64,
99        analyzer,
100    })
101}
