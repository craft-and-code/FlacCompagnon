1//! DSD decoding, delegated to ffmpeg.
2//!
3//! Symphonia has no DSD decoder, and demodulating a 1-bit sigma-delta stream
4//! to PCM properly is a filter design problem in its own right — so this shells
5//! out to the ffmpeg the user already has (the same one the spectrogram
6//! feature needs). No shell is involved: arguments go through `Command::arg`,
7//! never a command line.
8

18/// Read size for ffmpeg's stdout. Large enough that the pipe is rarely the
19/// bottleneck, small enough to stay a rounding error against the decoded
20/// audio itself.
21const PIPE_CHUNK: usize = 1 << 16;
22
23/// Decode a DSD file (DSF/DFF) through `ffmpeg` and stream the PCM into the
24/// analyzer.
25///
26/// ffmpeg converts DSD to float PCM at `dsd_rate / 8` (352.8 kHz for DSD64);
27/// `decoded_rate` is that *decoded* rate, and it is what the spectral analysis
28/// must use — not the 1-bit rate from the container header.
29pub fn decode_and_analyze_dsd(
30    ffmpeg: &str,
31    path: &Path,
32    channels: usize,
33    decoded_rate: u32,
34) -> Result<DecodeOutcome, AnalysisError> {
35    decode_and_analyze_dsd_selected(
36        ffmpeg,
37        path,
38        channels,
39        decoded_rate,
40        crate::AnalysisSelection::all(),
41    )
42}
43
44pub(crate) fn decode_and_analyze_dsd_selected(
45    ffmpeg: &str,
46    path: &Path,
47    channels: usize,
48    decoded_rate: u32,
49    selection: crate::AnalysisSelection,
50) -> Result<DecodeOutcome, AnalysisError> {
51    if !(1..=8).contains(&channels) || decoded_rate == 0 {
52        return Err(AnalysisError::Decode(
53            "DSD: invalid stream parameters".into(),
54        ));
55    }
56
57    let mut child = DecodeProcess(
58        local_ffmpeg_command(ffmpeg)
59            .arg("-v")
60            .arg("error")
61            .arg("-i")
62            .arg(path)
63            .args(["-f", "f32le", "-"])
64            .stdout(Stdio::piped())
65            .stderr(Stdio::null())
66            .stdin(Stdio::null())
67            .spawn()
68            .map_err(|e| AnalysisError::Decode(format!("ffmpeg spawn failed: {e}")))?,
69    );
70
71    let mut stdout = child
72        .0
73        .stdout
74        .take()
75        .ok_or_else(|| AnalysisError::Decode("ffmpeg produced no output pipe".into()))?;
76
77    let (analyzer, frame_count) =
78        analyze_pcm_output(&mut stdout, channels, decoded_rate, selection)?;
79
80    let status = child
81        .0
82        .wait()
83        .map_err(|e| AnalysisError::Decode(format!("ffmpeg wait failed: {e}")))?;
84    if !status.success() || frame_count == 0 {
85        return Err(AnalysisError::Decode(
86            "ffmpeg could not decode this DSD file".into(),
87        ));
88    }
89
90    Ok(DecodeOutcome {
91        format: "DSD".to_string(),
92        codec: None, // container already says everything this field would
93        sample_rate: decoded_rate,
94        channels,
95        declared_bits: None, // 1-bit stream; PCM bit depth does not apply
96        duration_secs: frame_count as f64 / decoded_rate.max(1) as f64,
97        analyzer,
98    })
99}

101// Dropping std::process::Child does not stop or reap it. Read/validation errors
102// must not leave FFmpeg running against a closed pipe or retain a zombie.
103struct DecodeProcess(Child);
104
105impl Drop for DecodeProcess {
106    fn drop(&mut self) {
107        let _ = self.0.kill();
108        let _ = self.0.wait();
109    }
110}

112fn analyze_pcm_output(
113    stdout: &mut impl Read,
114    channels: usize,
115    decoded_rate: u32,
116    selection: crate::AnalysisSelection,
117) -> Result<(StreamAnalyzer, u64), AnalysisError> {
118    let mut analyzer = StreamAnalyzer::new_selected(channels, decoded_rate, selection);
119    let mut carry: Vec<u8> = Vec::new();
120    let mut buf = vec![0u8; PIPE_CHUNK];
121    let mut frame = vec![0f32; channels];
122    let frame_bytes = channels * 4;
123    let mut frame_count: u64 = 0;
124
125    loop {
126        let n = stdout
127            .read(&mut buf)
128            .map_err(|e| AnalysisError::Decode(format!("ffmpeg read failed: {e}")))?;
129        if n == 0 {
130            break;
131        }
132        carry.extend_from_slice(&buf[..n]);
133        // Only whole frames can be pushed; a partial one at the end of this
134        // read stays in `carry` and is completed by the next one.
135        let whole = carry.len() / frame_bytes * frame_bytes;
136        for chunk in carry[..whole].chunks_exact(frame_bytes) {
137            for (c, s) in frame.iter_mut().enumerate() {
138                let o = c * 4;
139                *s = f32::from_le_bytes([chunk[o], chunk[o + 1], chunk[o + 2], chunk[o + 3]]);
140            }
141            validate_pcm_samples(&frame)?;
142            analyzer.push_frame(&frame, None);
143            frame_count += 1;
144        }
145        carry.drain(..whole);
146    }
147
148    if !carry.is_empty() {
149        return Err(AnalysisError::Decode("incomplete FFmpeg PCM frame".into()));
150    }
151    Ok((analyzer, frame_count))
152}
