1//! FLAC's fused decode path: one pass that feeds the streaming analyzer *and*
2//! computes the STREAMINFO MD5 over the exact original integer samples.
3//!
4//! Fused because both need every sample of the file: doing them separately
5//! would decode the whole track twice for no gain. claxon is used here rather
6//! than Symphonia because it hands back the raw decoded integers, which is
7//! what the MD5 and bit-depth measurement must use. Spectral analysis can
8//! use normalized floats without discarding the raw integer evidence.
9

19/// Decode a FLAC into an interleaved `f32` buffer using claxon.
20///
21/// # Why this exists next to the Symphonia path
22///
23/// [`super::decode_to_pcm`] normally goes through Symphonia, which handles
24/// every format the app accepts. Symphonia is also the stricter of the two
25/// FLAC readers, and it refuses streams that claxon and ffmpeg accept — a
26/// STREAMINFO that reports a variable block size while the frame headers say
27/// fixed, for one, which is a file this app used to produce itself (see
28/// `convert::flac::declare_fixed_block_size`). That bug is fixed at the
29/// source now, but files written before the fix are on users' disks, and the
30/// transcoding detector reported them as un-analysable.
31///
32/// So FLAC — the format this app is named after and the one it is most often
33/// pointed at — gets decoded by the same library that already decodes it for
34/// analysis and MD5 verification everywhere else. Symphonia stays as the
35/// fallback, which keeps a claxon-specific failure from being fatal either.
36///
37/// Deliberately *not* merged with [`decode_and_analyze_flac`]: that function
38/// exists to hash and analyse in one pass and would have to grow a mode flag
39/// and a discarded output to serve this caller. The two share a library, not
40/// a reason to change.
41pub fn decode_flac_to_pcm(path: &Path) -> Result<crate::decode::PcmAudio, AnalysisError> {
42    let mut reader = claxon::FlacReader::open(path)
43        .map_err(|e| AnalysisError::Decode(format!("flac open failed: {e}")))?;
44    let info = reader.streaminfo();
45    let (sample_rate, channels, bits) = (
46        info.sample_rate,
47        info.channels as usize,
48        info.bits_per_sample,
49    );
50    if sample_rate == 0 || channels == 0 || bits == 0 || bits > 32 {
51        return Err(AnalysisError::Decode("invalid FLAC stream info".into()));
52    }
53    let scale = 1.0f32 / (1u64 << (bits - 1)) as f32;
54
55    let mut samples: Vec<f32> = Vec::new();
56    // `info.samples` is a hint from a header this crate does not trust, so it
57    // sizes the allocation but never bounds the loop.
58    if let Some(total) = info.samples {
59        if let Ok(n) = usize::try_from(total) {
60            samples.reserve(n.saturating_mul(channels).min(1 << 28));
61        }
62    }
63
64    let mut blocks = reader.blocks();
65    let mut buffer: Vec<i32> = Vec::new();
66    let mut frame_count = 0;
67    loop {
68        let block = match blocks.read_next_or_eof(buffer) {
69            Ok(Some(b)) => b,
70            Ok(None) => break,
71            Err(e) => return Err(AnalysisError::Decode(format!("flac decode error: {e}"))),
72        };
73        validate_block_channels(&block, channels)?;
74        for t in 0..block.duration() as usize {
75            for c in 0..channels {
76                samples.push(block_sample(&block, c, t)? as f32 * scale);
77            }
78        }
79        frame_count += u64::from(block.duration());
80        buffer = block.into_buffer();
81    }
82    validate_decoded_frames(frame_count, info.samples)?;
83    Ok(crate::decode::PcmAudio {
84        samples,
85        sample_rate,
86        channels,
87    })
88}

90/// Fused single-pass FLAC decode.
91///
92/// Correctness guarantees:
93/// * The MD5 is hashed from claxon's raw decoded integers — no float
94///   round-trip — using the exact layout mandated by the FLAC spec:
95///   interleaved samples, each written as `ceil(bits/8)` little-endian
96///   two's-complement bytes. Bit-identical to what `flac -t` verifies.
97/// * The analyzer receives `s / 2^(bits-1)` as `f32`, which is exact for
98///   FLAC's ≤ 24-bit integers (every such integer fits in the f32 mantissa),
99///   plus the raw integers for effective-bit-depth analysis.
100///
101/// When `verify_md5` is false the hash comparison is skipped (the decode cost
102/// is the same either way since analysis needs every sample).
103pub fn decode_and_analyze_flac(
104    path: &Path,
105    verify_md5: bool,
106) -> Result<(DecodeOutcome, FlacMd5Status), AnalysisError> {
107    let mut reader = claxon::FlacReader::open(path)
108        .map_err(|e| AnalysisError::Decode(format!("flac open failed: {e}")))?;
109    let info = reader.streaminfo();
110    let sample_rate = info.sample_rate;
111    let channels = info.channels as usize;
112    let bits = info.bits_per_sample;
113    // Guards the shifts and the per-sample byte width below; a STREAMINFO
114    // claiming 0 or > 32 bits is either corrupt or hostile.
115    if sample_rate == 0 || channels == 0 || bits == 0 || bits > 32 {
116        return Err(AnalysisError::Decode("invalid FLAC stream info".into()));
117    }
118    let total_frames_hint = info.samples;
119    let stored = info.md5sum;
120    let has_signature = stored != [0u8; 16];
121
122    let scale = 1.0f32 / (1u64 << (bits - 1)) as f32;
123    let bytes_per_sample = bits.div_ceil(8) as usize;
124    let mut hasher = (has_signature && verify_md5).then(Md5::new);
125
126    let mut analyzer = StreamAnalyzer::new(channels, sample_rate);
127    let mut frame_f32 = vec![0.0f32; channels];
128    let mut frame_i32 = vec![0i32; channels];
129    let mut byte_row: Vec<u8> = Vec::new();
130    let mut frame_count: u64 = 0;
131
132    let mut blocks = reader.blocks();
133    let mut buffer: Vec<i32> = Vec::new();
134    loop {
135        let block = match blocks.read_next_or_eof(buffer) {
136            Ok(Some(b)) => b,
137            Ok(None) => break,
138            Err(e) => return Err(AnalysisError::Decode(format!("flac decode error: {e}"))),
139        };
140        validate_block_channels(&block, channels)?;
141        let n = block.duration() as usize;
142        if hasher.is_some() {
143            byte_row.clear();
144            byte_row.reserve(n * channels * bytes_per_sample);
145        }
146        for t in 0..n {
147            for c in 0..channels {
148                let s = block_sample(&block, c, t)?;
149                frame_i32[c] = s;
150                frame_f32[c] = s as f32 * scale;
151                if hasher.is_some() {
152                    let le = (s as u32).to_le_bytes();
153                    byte_row.extend_from_slice(&le[..bytes_per_sample]);
154                }
155            }
156            analyzer.push_frame(&frame_f32, Some(&frame_i32));
157            frame_count += 1;
158        }
159        if let Some(h) = &mut hasher {
160            h.update(&byte_row);
161        }
162        buffer = block.into_buffer();
163    }
164    validate_decoded_frames(frame_count, total_frames_hint)?;
165
166    let md5_status = if !has_signature {
167        FlacMd5Status::NoSignature
168    } else if let Some(h) = hasher {
169        // `finalize` consumes the hasher, so this cannot be a match guard.
170        if h.finalize().as_slice() == stored.as_slice() {
171            FlacMd5Status::Match
172        } else {
173            FlacMd5Status::Mismatch
174        }
175    } else {
176        // A signature is present but verification was not asked for.
177        FlacMd5Status::Present
178    };
179
180    let duration_secs = total_frames_hint
181        .map(|n| n as f64 / sample_rate as f64)
182        .unwrap_or(frame_count as f64 / sample_rate as f64);
183
184    Ok((
185        DecodeOutcome {
186            format: "FLAC".to_string(),
187            codec: None, // container already says everything this field would
188            sample_rate,
189            channels,
190            declared_bits: Some(bits),
191            duration_secs,
192            analyzer,
193        },
194        md5_status,
195    ))
196}

198fn validate_block_channels(
199    block: &claxon::frame::Block,
200    channels: usize,
201) -> Result<(), AnalysisError> {
202    if block.channels() as usize != channels {
203        return Err(AnalysisError::Decode(
204            "FLAC channel count changed during decoding".into(),
205        ));
206    }
207    Ok(())
208}
209
210fn block_sample(
211    block: &claxon::frame::Block,
212    channel: usize,
213    frame: usize,
214) -> Result<i32, AnalysisError> {
215    // `channel()` can panic, so validate against this block, not STREAMINFO.
216    if channel >= block.channels() as usize {
217        return Err(AnalysisError::Decode("invalid FLAC channel index".into()));
218    }
219    block
220        .channel(channel as u32)
221        .get(frame)
222        .copied()
223        .ok_or_else(|| AnalysisError::Decode("incomplete FLAC block".into()))
224}
