1//! FLAC's fused decode path: one pass that feeds the streaming analyzer *and*
2//! computes the STREAMINFO MD5 over the exact original integer samples.
3//!
4//! Fused because both need every sample of the file: doing them separately
5//! would decode the whole track twice for no gain. claxon is used here rather
6//! than Symphonia because it hands back the raw decoded integers, which is
7//! what the MD5 must be computed over — a float round-trip would not be
8//! bit-exact for the hash even though it is exact for the analysis.
9
10use std::path::Path;
11
12use md5::{Digest, Md5};
13
14use super::DecodeOutcome;
15use crate::analyzer::StreamAnalyzer;
16use crate::flac_md5::FlacMd5Status;
17use crate::AnalysisError;

19/// Fused single-pass FLAC decode.
20///
21/// Correctness guarantees:
22/// * The MD5 is hashed from claxon's raw decoded integers — no float
23///   round-trip — using the exact layout mandated by the FLAC spec:
24///   interleaved samples, each written as `ceil(bits/8)` little-endian
25///   two's-complement bytes. Bit-identical to what `flac -t` verifies.
26/// * The analyzer receives `s / 2^(bits-1)` as `f32`, which is exact for
27///   FLAC's ≤ 24-bit integers (every such integer fits in the f32 mantissa),
28///   plus the raw integers for effective-bit-depth analysis.
29///
30/// When `verify_md5` is false the hash comparison is skipped (the decode cost
31/// is the same either way since analysis needs every sample).
32pub fn decode_and_analyze_flac(
33    path: &Path,
34    verify_md5: bool,
35) -> Result<(DecodeOutcome, FlacMd5Status), AnalysisError> {
36    let mut reader = claxon::FlacReader::open(path)
37        .map_err(|e| AnalysisError::Decode(format!("flac open failed: {e}")))?;
38    let info = reader.streaminfo();
39    let sample_rate = info.sample_rate;
40    let channels = info.channels as usize;
41    let bits = info.bits_per_sample;
42    // Guards the shifts and the per-sample byte width below; a STREAMINFO
43    // claiming 0 or > 32 bits is either corrupt or hostile.
44    if sample_rate == 0 || channels == 0 || bits == 0 || bits > 32 {
45        return Err(AnalysisError::Decode("invalid FLAC stream info".into()));
46    }
47    let total_frames_hint = info.samples;
48    let stored = info.md5sum;
49    let has_signature = stored != [0u8; 16];
50
51    let scale = 1.0f32 / (1u64 << (bits - 1)) as f32;
52    let bytes_per_sample = bits.div_ceil(8) as usize;
53    let mut hasher = (has_signature && verify_md5).then(Md5::new);
54
55    let mut analyzer = StreamAnalyzer::new(channels);
56    let mut frame_f32 = vec![0.0f32; channels];
57    let mut frame_i32 = vec![0i32; channels];
58    let mut byte_row: Vec<u8> = Vec::new();
59    let mut frame_count: u64 = 0;
60
61    let mut blocks = reader.blocks();
62    let mut buffer: Vec<i32> = Vec::new();
63    loop {
64        let block = match blocks.read_next_or_eof(buffer) {
65            Ok(Some(b)) => b,
66            Ok(None) => break,
67            Err(e) => return Err(AnalysisError::Decode(format!("flac decode error: {e}"))),
68        };
69        let n = block.duration() as usize;
70        if hasher.is_some() {
71            byte_row.clear();
72            byte_row.reserve(n * channels * bytes_per_sample);
73        }
74        for t in 0..n {
75            for c in 0..channels {
76                let s = block.channel(c as u32)[t];
77                frame_i32[c] = s;
78                frame_f32[c] = s as f32 * scale;
79                if hasher.is_some() {
80                    let le = (s as u32).to_le_bytes();
81                    byte_row.extend_from_slice(&le[..bytes_per_sample]);
82                }
83            }
84            analyzer.push_frame(&frame_f32, Some(&frame_i32));
85            frame_count += 1;
86        }
87        if let Some(h) = &mut hasher {
88            h.update(&byte_row);
89        }
90        buffer = block.into_buffer();
91    }
92
93    let md5_status = if !has_signature {
94        FlacMd5Status::NoSignature
95    } else if let Some(h) = hasher {
96        // `finalize` consumes the hasher, so this cannot be a match guard.
97        if h.finalize().as_slice() == stored.as_slice() {
98            FlacMd5Status::Match
99        } else {
100            FlacMd5Status::Mismatch
101        }
102    } else {
103        // A signature is present but verification was not asked for.
104        FlacMd5Status::Present
105    };
106
107    let duration_secs = total_frames_hint
108        .map(|n| n as f64 / sample_rate as f64)
109        .unwrap_or(frame_count as f64 / sample_rate as f64);
110
111    Ok((
112        DecodeOutcome {
113            format: "FLAC".to_string(),
114            sample_rate,
115            channels,
116            declared_bits: Some(bits),
117            duration_secs,
118            analyzer,
119        },
120        md5_status,
121    ))
122}

124#[cfg(test)]
125mod tests {
126    /// The fused FLAC path feeds the analyzer with `s * (1 / 2^(bits-1))` as
127    /// f32. This must be a *lossless* round-trip for every integer the format
128    /// can produce at ≤ 24 bits — otherwise analysis results could drift from
129    /// the exact samples the MD5 is computed over.
130    #[test]
131    fn f32_normalization_roundtrips_exactly_up_to_24_bits() {
132        for bits in [8u32, 12, 16, 20, 24] {
133            let scale = 1.0f32 / (1u64 << (bits - 1)) as f32;
134            let max = (1i64 << (bits - 1)) - 1;
135            let probes = [0i64, 1, -1, 2, -2, max, -max - 1, max / 3, -(max / 7), max - 1];
136            for &s in &probes {
137                let f = s as f32 * scale;
138                let back = (f / scale).round() as i64;
139                assert_eq!(back, s, "bits={bits} sample={s}");
140            }
141        }
142    }
143
144    /// The byte width the MD5 is fed must match the FLAC spec's
145    /// `ceil(bits/8)` for every depth the format allows.
146    #[test]
147    fn bytes_per_sample_matches_the_spec() {
148        for (bits, expected) in [
149            (8u32, 1usize),
150            (12, 2),
151            (16, 2),
152            (20, 3),
153            (24, 3),
154            (32, 4),
155        ] {
156            assert_eq!(bits.div_ceil(8) as usize, expected, "bits={bits}");
157        }
158    }
