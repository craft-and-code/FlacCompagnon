1//! Opening a file and finding its default audio track.
2//!
3//! This exists because four call sites needed the same twenty lines — probe
4//! the container, take the default track, pull the sample rate and channel
5//! count off its codec parameters — and had four copies of them, which is how
6//! they had already started to drift apart (only one of them disabled gapless
7//! playback, for no stated reason). [`ProbedTrack`] is that step, once.
8

22/// A probed file: its reader, positioned before the first packet, and the
23/// codec parameters of the track worth decoding.
24pub(super) struct ProbedTrack {
25    pub format: Box<dyn FormatReader>,
26    pub track_id: u32,
27    pub params: CodecParameters,
28}
29
30/// Probe `path` and locate its default track.
31///
32/// `gapless` mirrors Symphonia's `enable_gapless`: it trims the encoder delay
33/// and padding declared by formats that record them (MP3/AAC). Analysis and
34/// playback want it on, so what we measure and what a player would output are
35/// the same samples. [`probe_info`] leaves it off because it never reads a
36/// packet, so the setting cannot affect its answer either way.
37pub(super) fn probe(path: &Path, gapless: bool) -> Result<ProbedTrack, AnalysisError> {
38    let file = File::open(path)?;
39    let mss = MediaSourceStream::new(Box::new(file), Default::default());
40
41    let mut hint = Hint::new();
42    if let Some(ext) = path.extension().and_then(|e| e.to_str()) {
43        hint.with_extension(ext);
44    }
45
46    let probed = symphonia::default::get_probe()
47        .format(
48            &hint,
49            mss,
50            &FormatOptions {
51                enable_gapless: gapless,
52                ..Default::default()
53            },
54            &MetadataOptions::default(),
55        )
56        .map_err(|e| AnalysisError::Decode(format!("probe failed: {e}")))?;
57
58    let format = probed.format;
59    let track = format
60        .default_track()
61        .ok_or_else(|| AnalysisError::Decode("no default track".into()))?;
62    let track_id = track.id;
63    let params = track.codec_params.clone();
64
65    Ok(ProbedTrack {
66        format,
67        track_id,
68        params,
69    })
70}
71
72impl ProbedTrack {
73    pub(super) fn sample_rate(&self) -> Result<u32, AnalysisError> {
74        self.params
75            .sample_rate
76            .ok_or_else(|| AnalysisError::Decode("unknown sample rate".into()))
77    }
78
79    pub(super) fn channels(&self) -> Result<usize, AnalysisError> {
80        self.params
81            .channels
82            .map(|c| c.count())
83            .ok_or_else(|| AnalysisError::Decode("unknown channel layout".into()))
84    }
85
86    pub(super) fn make_decoder(&self) -> Result<Box<dyn Decoder>, AnalysisError> {
87        symphonia::default::get_codecs()
88            .make(&self.params, &DecoderOptions::default())
89            .map_err(|e| AnalysisError::Decode(format!("no decoder: {e}")))
90    }
91}

93/// A reusable interleaved-`f32` scratch buffer for decoded packets.
94///
95/// Symphonia hands back a typed `AudioBufferRef` per packet; every caller
96/// wants the same flat `&[f32]`. Allocating one buffer per packet would churn
97/// the allocator once per ~4096 frames for the whole file, so the buffer is
98/// kept and only reallocated when a later packet needs a bigger one (block
99/// sizes can vary within a stream).
100#[derive(Default)]
101pub(super) struct InterleavedBuf {
102    buf: Option<SampleBuffer<f32>>,
103    frames: u64,
104}
105
106impl InterleavedBuf {
107    pub(super) fn fill(&mut self, decoded: AudioBufferRef<'_>) -> &[f32] {
108        let needed = decoded.capacity() as u64;
109        if self.buf.is_none() || needed > self.frames {
110            let spec = *decoded.spec();
111            self.buf = Some(SampleBuffer::<f32>::new(needed, spec));
112            self.frames = needed;
113        }
114        // Set on the line above if it was missing, so this cannot be None.
115        let buf = self.buf.as_mut().expect("buffer just ensured present");
116        buf.copy_interleaved_ref(decoded);
117        buf.samples()
118    }
119}

121/// Lightweight header information obtained without decoding the whole file.
122#[derive(Debug, Clone)]
123pub struct BasicInfo {
124    /// Sample rate declared by the header, in Hz.
125    pub sample_rate: u32,
126    /// Channel count declared by the header.
127    pub channels: usize,
128    /// Bit depth declared by the header, when the format has one.
129    pub bits: Option<u32>,
130    /// Human-readable format label (e.g. "FLAC", "DSF").
131    pub format: String,
132}
133
134/// Read basic stream parameters from a file's header only (fast — no full
135/// decode). Used e.g. to caption spectrogram images.
136pub fn probe_info(path: &Path) -> Result<BasicInfo, AnalysisError> {
137    let probed = probe(path, false)?;
138    Ok(BasicInfo {
139        // Unlike the decode paths, a missing rate or channel count is not
140        // fatal here: this only captions an image, and "0" is a better answer
141        // than refusing to render the spectrogram at all.
142        sample_rate: probed.params.sample_rate.unwrap_or(0),
143        channels: probed.params.channels.map(|c| c.count()).unwrap_or(0),
144        bits: probed.params.bits_per_sample,
145        format: format_label(path),
146    })
147}
