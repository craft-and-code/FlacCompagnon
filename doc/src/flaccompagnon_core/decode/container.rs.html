1//! What a file *is* versus what it claims to be.
2//!
3//! [`detect_container`] reads magic bytes, [`ext_canonical`] reads the
4//! extension, and the analyzer compares them — a `.flac` that is really an MP4
5//! is worth saying out loud, and it is exactly the kind of thing a re-wrapped
6//! lossy file looks like.
7

18/// Detect the *real* container from the file's magic bytes, independent of its
19/// extension. Returns a canonical short name, or `None` if unrecognized.
20pub fn detect_container(path: &Path) -> Option<&'static str> {
21    use std::io::Read;
22    let mut f = std::fs::File::open(path).ok()?;
23    let mut b = [0u8; 16];
24    let n = f.read(&mut b).ok()?;
25    let head = b.get(..n)?;
26    if head.len() < 4 {
27        return None;
28    }
29    // Helper: does the header have `tag` at `at`? False when it is too short,
30    // so every check below is bounds-safe on a file of any length.
31    let at = |pos: usize, tag: &[u8]| head.get(pos..pos + tag.len()) == Some(tag);
32
33    if at(0, b"fLaC") {
34        return Some("FLAC");
35    }
36    if at(0, b"DSD ") {
37        return Some("DSF");
38    }
39    if at(0, b"FRM8") && at(12, b"DSD ") {
40        return Some("DFF");
41    }
42    if (at(0, b"RIFF") || at(0, b"RF64")) && at(8, b"WAVE") {
43        return Some("WAV");
44    }
45    if at(0, b"FORM") && (at(8, b"AIFF") || at(8, b"AIFC")) {
46        return Some("AIFF");
47    }
48    if at(0, b"OggS") {
49        return Some("OGG");
50    }
51    if at(0, b"caff") {
52        return Some("CAF");
53    }
54    if at(4, b"ftyp") {
55        return Some("MP4");
56    }
57    if at(0, b"ID3") {
58        return Some("MP3");
59    }
60    // Frame-sync patterns, checked last: they are only two bytes, so a longer
61    // magic that happens to start the same way must win first.
62    if let (Some(&0xFF), Some(&b1)) = (head.first(), head.get(1)) {
63        if b1 & 0xF6 == 0xF0 {
64            return Some("AAC"); // ADTS
65        }
66        if b1 & 0xE0 == 0xE0 {
67            return Some("MP3"); // MPEG-1/2 audio frame sync
68        }
69    }
70    None
71}

73/// Canonical container name expected from a file's extension.
74pub fn ext_canonical(path: &Path) -> Option<&'static str> {
75    match lower_ext(path).as_deref() {
76        Some("flac") => Some("FLAC"),
77        Some("wav" | "wave") => Some("WAV"),
78        Some("aif" | "aiff" | "aifc") => Some("AIFF"),
79        Some("m4a" | "mp4" | "alac") => Some("MP4"),
80        Some("caf") => Some("CAF"),
81        // `.opus` is an Ogg container like the other two — the extension only
82        // narrows the codec, not the wrapper. It has to be listed here even
83        // though `format_label` shows it as "Opus": an extension this function
84        // doesn't know returns `None`, and `flag_container_mismatch` treats
85        // that as "unrecognized" and overwrites the label with the bare
86        // detected container, which would throw the codec name away again.
87        Some("ogg" | "oga" | "opus") => Some("OGG"),
88        Some("mp3") => Some("MP3"),
89        Some("aac") => Some("AAC"),
90        Some("dsf") => Some("DSF"),
91        Some("dff") => Some("DFF"),
92        _ => None,
93    }
94}

96/// Human-readable container/codec label from the file extension, and — for
97/// containers that can hold more than one codec — the codec Symphonia
98/// actually identified.
99///
100/// Deliberately not the same mapping as [`ext_canonical`]: this one is shown
101/// to the user, and falls back to the uppercased extension for anything
102/// unrecognized, while `ext_canonical` returns a strict canonical name used to
103/// compare against the detected container.
104///
105/// `codec` only changes anything for `.m4a`/`.mp4`/`.alac`: an MP4 container
106/// can hold ALAC (lossless) or AAC (lossy), and always assuming ALAC — as an
107/// earlier version of this function did — meant an AAC-in-MP4 file (lossy)
108/// displayed as "ALAC/MP4", reading as lossless when it plainly is not, in
109/// an app whose whole point is catching exactly that kind of
110/// mislabeling. `None` (codec unresolved, e.g. a probe failure) falls back
111/// to the previous "ALAC/MP4" default, since that's still the more common
112/// case for this extension.
113pub(super) fn format_label(path: &Path, codec: Option<&str>) -> String {
114    match lower_ext(path).as_deref() {
115        Some("flac") => "FLAC".to_string(),
116        Some("wav" | "wave") => "WAV".to_string(),
117        Some("aif" | "aiff" | "aifc") => "AIFF".to_string(),
118        Some("alac" | "m4a" | "mp4") => match codec {
119            Some("ALAC") | None => "ALAC/MP4".to_string(),
120            Some(other) => format!("{other}/MP4"),
121        },
122        Some("caf") => "CAF".to_string(),
123        Some("ogg" | "oga") => "OGG".to_string(),
124        // An `.opus` file is an Ogg stream too, but unlike a generic `.ogg` it
125        // can only carry Opus — so the label says the codec outright, where
126        // "OGG" would leave the one useful fact to the codec column.
127        Some("opus") => "Opus".to_string(),
128        Some("mp3") => "MP3".to_string(),
129        Some("aac") => "AAC".to_string(),
130        Some(other) => other.to_uppercase(),
131        None => "?".to_string(),
132    }
133}

135/// Human-readable codec name for containers that can hold more than one —
136/// an M4A/MP4 might be ALAC or AAC, an OGG might be Vorbis or Opus, and the
137/// container name alone can't tell those apart (see
138/// [`crate::types::FileAnalysis::codec`] and [`format_label`], above, which
139/// both use this). `None` for anything not in this list, which just means
140/// the caller falls back to a container-only label; it is never wrong, only
141/// sometimes less specific.
142pub(super) fn codec_label(codec: CodecType) -> Option<&'static str> {
143    Some(match codec {
144        CODEC_TYPE_AAC => "AAC",
145        CODEC_TYPE_ALAC => "ALAC",
146        CODEC_TYPE_FLAC => "FLAC",
147        CODEC_TYPE_MP1 => "MP1",
148        CODEC_TYPE_MP2 => "MP2",
149        CODEC_TYPE_MP3 => "MP3",
150        CODEC_TYPE_OPUS => "Opus",
151        CODEC_TYPE_VORBIS => "Vorbis",
152        CODEC_TYPE_PCM_S8 | CODEC_TYPE_PCM_U8 | CODEC_TYPE_PCM_S16LE | CODEC_TYPE_PCM_S16BE
153        | CODEC_TYPE_PCM_S24LE | CODEC_TYPE_PCM_S24BE | CODEC_TYPE_PCM_S32LE
154        | CODEC_TYPE_PCM_S32BE | CODEC_TYPE_PCM_F32LE | CODEC_TYPE_PCM_F32BE
155        | CODEC_TYPE_PCM_F64LE | CODEC_TYPE_PCM_F64BE => "PCM",
156        _ => return None,
157    })
158}

160/// Why no decoder exists for `codec`, when the reason is a known gap in
161/// Symphonia rather than an unrecognized stream.
162///
163/// Symphonia probes far more codecs than it can decode: it will happily
164/// identify an Opus track in an Ogg stream and then have nothing to hand it
165/// to, because `symphonia-codec-opus` is a placeholder that isn't even pulled
166/// in by the `all` feature. The raw failure that comes back from
167/// `get_codecs().make()` is a bare "unsupported codec", which reads like a
168/// corrupt file rather than a missing feature. This turns the handful of cases
169/// we know about into something a user can act on.
170///
171/// Returns `None` for anything else, so a genuinely unreadable stream still
172/// reports Symphonia's own error rather than a guess.
173pub(super) fn missing_decoder_reason(codec: CodecType) -> Option<&'static str> {
174    Some(match codec {
175        // Encoding *to* Opus works (that path uses libopus directly), which is
176        // why this one is worth naming: the app can write a format it cannot
177        // read back, and that asymmetry is otherwise baffling.
178        CODEC_TYPE_OPUS => {
179            "Opus decoding is not supported yet — Symphonia has no Opus decoder \
180             (conversion to Opus does work, it uses libopus directly)"
181        }
182        _ => return None,
183    })
184}

186fn lower_ext(path: &Path) -> Option<String> {
187    path.extension()
188        .and_then(|e| e.to_str())
189        .map(|e| e.to_ascii_lowercase())
190}
