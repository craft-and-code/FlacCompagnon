1//! The generic decode-and-analyze path: any format Symphonia supports, fed
2//! frame by frame into a [`StreamAnalyzer`].
3//!
4//! FLAC does not come through here — it has a fused path ([`super::flac`])
5//! that also computes the STREAMINFO MD5 in the same pass — but it is the
6//! fallback when that path fails on a malformed FLAC.
7

19/// Decode `path` and run streaming analysis over its samples.
20pub fn decode_and_analyze(path: &Path) -> Result<DecodeOutcome, AnalysisError> {
21    let mut probed = probe(path, true)?;
22    let sample_rate = probed.sample_rate()?;
23    let channels = probed.channels()?;
24    let codec = codec_label(probed.params.codec).map(str::to_string);
25    let declared_bits = probed.params.bits_per_sample;
26    let declared_duration = probed
27        .params
28        .n_frames
29        .map(|n| n as f64 / sample_rate as f64)
30        .unwrap_or(0.0);
31    let track_id = probed.track_id;
32    let mut decoder = probed.make_decoder()?;
33
34    // Integer conversion aligns samples to 32 bits. Restore the declared
35    // width without a float round-trip, which would erase low bits above 24.
36    let int_shift = declared_bits
37        .filter(|b| (1..=32).contains(b))
38        .map(|b| 32 - b);
39
40    let mut analyzer = StreamAnalyzer::new(channels);
41    let mut buf = InterleavedBuf::<f32>::default();
42    let mut int_buf = InterleavedBuf::<i32>::default();
43    let mut int_packet: Vec<i32> = Vec::new();
44    let mut frame_count: u64 = 0;
45    let mut saw_integer = false;
46
47    loop {
48        let packet = match probed.format.next_packet() {
49            Ok(p) => p,
50            Err(SymError::IoError(e)) if e.kind() == std::io::ErrorKind::UnexpectedEof => break,
51            Err(SymError::ResetRequired) => {
52                return Err(AnalysisError::Decode(
53                    "stream changed during analysis".into(),
54                ));
55            }
56            Err(e) => return Err(AnalysisError::Decode(format!("packet error: {e}"))),
57        };
58        if packet.track_id() != track_id {
59            continue;
60        }
61
62        match decoder.decode(&packet) {
63            Ok(decoded) => {
64                // Float sources carry no meaningful integer bit depth.
65                let is_int = !matches!(&decoded, AudioBufferRef::F32(_) | AudioBufferRef::F64(_));
66
67                // Reconstruct native integers for the whole packet, once, into
68                // a buffer reused across packets rather than a fresh Vec each
69                // time (this runs per packet for the length of the file).
70                int_packet.clear();
71                if let (true, Some(shift)) = (is_int, int_shift) {
72                    int_packet.extend(int_buf.fill(decoded.clone()).iter().map(|&s| s >> shift));
73                }
74                let f32_samples = buf.fill(decoded);
75                let ints_ready = !int_packet.is_empty();
76                saw_integer |= ints_ready;
77
78                let ch = channels.max(1);
79                let n_frames = f32_samples.len() / ch;
80                for f in 0..n_frames {
81                    let base = f * ch;
82                    let frame = &f32_samples[base..base + ch];
83                    let ints = ints_ready.then(|| &int_packet[base..base + ch]);
84                    analyzer.push_frame(frame, ints);
85                }
86                frame_count += n_frames as u64;
87            }
88            // Missing samples could contain the only non-zero low bit. An
89            // incomplete decode cannot establish whole-file zero padding.
90            Err(SymError::DecodeError(e)) => {
91                return Err(AnalysisError::Decode(format!("corrupt audio packet: {e}")));
92            }
93            Err(e) => return Err(AnalysisError::Decode(format!("decode error: {e}"))),
94        }
95    }
96
97    // A normal end-of-packets also occurs on truncated PCM. Do not turn
98    // analysis of a prefix into an exact claim about the entire file.
99    if saw_integer && probed.params.n_frames.is_some_and(|n| frame_count < n) {
100        return Err(AnalysisError::Decode(
101            "incomplete integer audio stream".into(),
102        ));
103    }
104
105    // Prefer the container's declared length; fall back to what we counted
106    // when it doesn't declare one (common for streamed/edited files).
107    let duration_secs = if declared_duration > 0.0 {
108        declared_duration
109    } else {
110        frame_count as f64 / sample_rate as f64
111    };
112
113    Ok(DecodeOutcome {
114        format: format_label(path, codec.as_deref()),
115        codec,
116        sample_rate,
117        channels,
118        declared_bits,
119        duration_secs,
120        analyzer,
121    })
122}

124#[cfg(test)]
125mod tests {
126    use super::*;
127    use crate::analysis::detections::classify;

130    fn measure(bits: u16, channels: u16, samples: &[i32]) -> Option<u32> {
131        let dir = tempfile::tempdir().expect("temporary directory");
132        let path = dir.path().join("integer.wav");
133        let mut writer = hound::WavWriter::create(
134            &path,
135            hound::WavSpec {
136                channels,
137                sample_rate: 96_000,
138                bits_per_sample: bits,
139                sample_format: hound::SampleFormat::Int,
140            },
141        )
142        .expect("WAV writer");
143        for &sample in samples {
144            writer.write_sample(sample).expect("valid integer sample");
145        }
146        writer.finalize().expect("finish WAV");
147        let decoded = decode_and_analyze(&path).expect("decode WAV");
148        let summary = decoded
149            .analyzer
150            .finish(decoded.sample_rate, decoded.declared_bits);
151        let verdict = classify(
152            &summary,
153            decoded.sample_rate,
154            decoded.declared_bits,
155            summary.real_bit_depth,
156            Err(LatticeSkip::TooShort),
157        );
158        if samples.iter().all(|&s| s == 0) {
159            assert_eq!(verdict.summary, "Unknown");
160            assert!(verdict.detail.contains("Digital silence"));
161        }
162        assert_eq!(
163            verdict.upscaling,
164            summary.real_bit_depth.is_some_and(|r| r < u32::from(bits))
165        );
166        summary.real_bit_depth
167    }

169    #[test]
170    fn digital_silence_does_not_claim_one_bit_upscaling() {
171        for bits in [8, 16, 24, 32] {
172            assert_eq!(measure(bits, 1, &[0; 64]), None);
173        }
174    }

176    #[test]
177    fn full_resolution_32bit_samples_keep_their_low_bits() {
178        // All are odd, but f32 rounds every one to an even integer.
179        let samples = [0x4000_0001, -0x4000_0001, 0x2000_0001, -0x2000_0001];
180        assert_eq!(measure(32, 1, &samples), Some(32));
181    }

183    #[test]
184    fn exact_padding_is_detected_across_integer_widths() {
185        for (bits, source) in [(16, 8), (24, 16), (24, 20), (32, 16), (32, 24)] {
186            let shift = bits - source;
187            let samples = [123 << shift, -125 << shift, 0, 127 << shift];
188            assert_eq!(measure(bits, 1, &samples), Some(u32::from(source)));
189        }
190    }

210    #[test]
211    fn audacity_style_dithered_16bit_export_is_detected_in_24bit() {
212        let samples: Vec<i32> = (0..20_000)
213            .map(|n| {
214                let source_16bit = ((n % 60_001) - 30_000) << 8;
215                let dither = (n % 21) - 10;
216                source_16bit + dither
217            })
218            .collect();
219        assert_eq!(measure(24, 1, &samples), Some(16));
220    }

222    #[test]
223    fn truncated_pcm_cannot_receive_a_whole_file_padding_verdict() {
224        let dir = tempfile::tempdir().expect("temporary directory");
225        let path = dir.path().join("truncated.wav");
226        let mut writer = hound::WavWriter::create(
227            &path,
228            hound::WavSpec {
229                channels: 1,
230                sample_rate: 96_000,
231                bits_per_sample: 24,
232                sample_format: hound::SampleFormat::Int,
233            },
234        )
235        .expect("WAV writer");
236        for _ in 0..20_000 {
237            writer.write_sample(123i32 << 8).expect("padded sample");
238        }
239        writer.write_sample(1i32).expect("only nonzero low bit");
240        writer.finalize().expect("finish WAV");
241        assert!(decode_and_analyze(&path).is_ok());
242        let file = std::fs::OpenOptions::new()
243            .write(true)
244            .open(&path)
245            .expect("open WAV");
246        let len = file.metadata().expect("WAV length").len();
247        file.set_len(len - 3).expect("remove last sample");
248        assert!(decode_and_analyze(&path).is_err());
249    }
