1//! The generic decode-and-analyze path: any format Symphonia supports, fed
2//! frame by frame into a [`StreamAnalyzer`].
3//!
4//! FLAC does not come through here — it has a fused path ([`super::flac`])
5//! that also computes the STREAMINFO MD5 in the same pass — but it is the
6//! fallback when that path fails on a malformed FLAC.
7
8use std::path::Path;
9
10use symphonia::core::audio::AudioBufferRef;
11use symphonia::core::errors::Error as SymError;
12
13use super::container::format_label;
14use super::probe::{probe, InterleavedBuf};
15use super::DecodeOutcome;
16use crate::analyzer::StreamAnalyzer;
17use crate::AnalysisError;
18
19/// Decode `path` and run streaming analysis over its samples.
20pub fn decode_and_analyze(path: &Path) -> Result<DecodeOutcome, AnalysisError> {
21    let mut probed = probe(path, true)?;
22    let sample_rate = probed.sample_rate()?;
23    let channels = probed.channels()?;
24    let declared_bits = probed.params.bits_per_sample;
25    let declared_duration = probed
26        .params
27        .n_frames
28        .map(|n| n as f64 / sample_rate as f64)
29        .unwrap_or(0.0);
30    let track_id = probed.track_id;
31    let mut decoder = probed.make_decoder()?;
32
33    // Integer reconstruction scale (full-scale = 2^(bits-1)).
34    let int_scale: Option<f32> = declared_bits.map(|b| 2f32.powi(b as i32 - 1));
35
36    let mut analyzer = StreamAnalyzer::new(channels);
37    let mut buf = InterleavedBuf::default();
38    let mut int_packet: Vec<i32> = Vec::new();
39    let mut frame_count: u64 = 0;
40
41    loop {
42        let packet = match probed.format.next_packet() {
43            Ok(p) => p,
44            Err(SymError::IoError(e)) if e.kind() == std::io::ErrorKind::UnexpectedEof => break,
45            Err(SymError::ResetRequired) => break,
46            Err(e) => return Err(AnalysisError::Decode(format!("packet error: {e}"))),
47        };
48        if packet.track_id() != track_id {
49            continue;
50        }
51
52        match decoder.decode(&packet) {
53            Ok(decoded) => {
54                // Float sources carry no meaningful integer bit depth.
55                let is_int = !matches!(&decoded, AudioBufferRef::F32(_) | AudioBufferRef::F64(_));
56                let f32_samples = buf.fill(decoded);
57
58                // Reconstruct native integers for the whole packet, once, into
59                // a buffer reused across packets rather than a fresh Vec each
60                // time (this runs per packet for the length of the file).
61                int_packet.clear();
62                if let (true, Some(scale)) = (is_int, int_scale) {
63                    int_packet.extend(f32_samples.iter().map(|&s| (s * scale).round() as i32));
64                }
65                let ints_ready = !int_packet.is_empty();
66
67                let ch = channels.max(1);
68                let n_frames = f32_samples.len() / ch;
69                for f in 0..n_frames {
70                    let base = f * ch;
71                    let frame = &f32_samples[base..base + ch];
72                    let ints = ints_ready.then(|| &int_packet[base..base + ch]);
73                    analyzer.push_frame(frame, ints);
74                }
75                frame_count += n_frames as u64;
76            }
77            Err(SymError::DecodeError(_)) => continue, // skip a corrupt packet
78            Err(SymError::IoError(e)) if e.kind() == std::io::ErrorKind::UnexpectedEof => break,
79            Err(e) => return Err(AnalysisError::Decode(format!("decode error: {e}"))),
80        }
81    }
82
83    // Prefer the container's declared length; fall back to what we counted
84    // when it doesn't declare one (common for streamed/edited files).
85    let duration_secs = if declared_duration > 0.0 {
86        declared_duration
87    } else {
88        frame_count as f64 / sample_rate as f64
89    };
90
91    Ok(DecodeOutcome {
92        format: format_label(path),
93        sample_rate,
94        channels,
95        declared_bits,
96        duration_secs,
97        analyzer,
98    })
99}
