1//! WAV encoding via `hound` — the same crate this crate's own tests already
2//! use to synthesize fixtures (see e.g. `tags::mod::tests::synth_wav`), so
3//! the call shape here follows an already-proven pattern rather than a fresh
4//! guess at the API.
5//!
6//! Fixed at 16-bit PCM: unlike FLAC, WAV has no lossless-at-any-depth
7//! encoder API to lean on here, and 16-bit keeps `hound`'s typed
8//! `write_sample::<i16>` call unambiguous. This isn't a hi-res archival
9//! format in this app's conversion panel — it exists as a guaranteed-honest
10//! PCM copy of a file this app may have flagged as fake-lossless, and 16-bit
11//! already exceeds what a genuinely lossy source ever contained.
12
13use std::path::Path;
14
15use super::ConvertError;
16use crate::decode::PcmAudio;
17
18const BIT_DEPTH: u32 = 16;
19
20/// Encode `pcm` as 16-bit PCM WAV to `dest`.
21pub(super) fn encode(pcm: &PcmAudio, dest: &Path) -> Result<(), ConvertError> {
22    let name = || dest.display().to_string();
23    let spec = hound::WavSpec {
24        channels: pcm.channels as u16,
25        sample_rate: pcm.sample_rate,
26        bits_per_sample: BIT_DEPTH as u16,
27        sample_format: hound::SampleFormat::Int,
28    };
29    let mut writer =
30        hound::WavWriter::create(dest, spec).map_err(|e| ConvertError::Encode(name(), e.to_string()))?;
31
32    let ints = super::f32_to_ints(&pcm.samples, BIT_DEPTH);
33    for s in ints {
34        writer
35            .write_sample(s as i16)
36            .map_err(|e| ConvertError::Encode(name(), e.to_string()))?;
37    }
38    writer
39        .finalize()
40        .map_err(|e| ConvertError::Encode(name(), e.to_string()))
41}

43#[cfg(test)]
44mod tests {
45    use super::*;
46
47    /// PCM WAV is trivially lossless at a fixed depth: what goes in as an
48    /// integer must come back as the exact same integer. `hound` reads its
49    /// own files back here — a second, independent decode isn't needed the
50    /// way it is for FLAC's compressed bitstream, since there is no
51    /// bitstream format to get subtly wrong, only a fixed-width sample
52    /// layout `hound` itself defines on both ends.
53    #[test]
54    fn round_trips_exactly_at_16_bit() {
55        let samples = vec![-1.0f32, -0.5, 0.0, 0.25, 0.5, 1.0, -1.0, 0.5];
56        let pcm = PcmAudio {
57            samples: samples.clone(),
58            sample_rate: 44_100,
59            channels: 2,
60        };
61        let dir = tempfile::tempdir().expect("tempdir");
62        let dest = dir.path().join("tone.wav");
63        encode(&pcm, &dest).expect("encode");
64
65        let mut reader = hound::WavReader::open(&dest).expect("reopen");
66        let decoded: Vec<i32> = reader
67            .samples::<i16>()
68            .map(|s| s.expect("decode") as i32)
69            .collect();
70        assert_eq!(decoded, super::super::f32_to_ints(&samples, BIT_DEPTH));
71    }
72
73    /// A malformed destination (a parent folder that doesn't exist) must
74    /// come back as a named [`ConvertError`], not a panic — `convert_file`
75    /// creates the parent first in the normal path, but this module's own
76    /// `encode` doesn't assume it always will.
77    #[test]
78    fn missing_parent_folder_is_a_named_error_not_a_panic() {
79        let pcm = PcmAudio {
80            samples: vec![0.0, 0.0],
81            sample_rate: 44_100,
82            channels: 2,
83        };
84        let dest = Path::new("/definitely/not/a/real/folder/track.wav");
85        assert!(encode(&pcm, dest).is_err());
86    }
87}
