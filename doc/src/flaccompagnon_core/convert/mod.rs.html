1//! Converting audio files to another format: FLAC, Opus, MP3 or WAV.
2//!
3//! Like [`crate::tags`], this module **does** write to disk beyond the source
4//! file it reads — one converted file per source, plus (optionally, see
5//! [`passthrough_files`]) plain copies of whatever else shares its folder.
6//! That is a deliberate reading of "no I/O beyond the audio files it is
7//! given" (see CLAUDE.md): the files it is *given* are exactly what it
8//! produces output for, the same way `tags::write_tags` reads that rule as
9//! covering writes to the files it is asked to tag, not just reads.
10//!
11//! DSD (`.dsf`/`.dff`) is out of scope for now: [`crate::decode::decode_to_pcm`]
12//! doesn't handle it (Symphonia cannot decode DSD; only the ffmpeg-backed
13//! analysis path can), so a DSD source fails fast with
14//! [`ConvertError::Unsupported`] rather than silently producing nothing or
15//! panicking partway through.
16//!
17//! ## Format choice
18//!
19//! All four targets are free to depend on: FLAC via a pure-Rust encoder
20//! ([`flac`], crate `flacenc`, no C toolchain at all), WAV via plain PCM
21//! muxing ([`wav`], crate `hound`), and Opus/MP3 ([`opus`], [`mp3`]) via
22//! crates that vendor their respective C codec (`libopus`, LAME) from source
23//! rather than linking a system library — so a built binary has no runtime
24//! dependency to install. *Building* is a different matter, and this comment
25//! used to get it wrong: `audiopus_sys` 0.1.x (what `audiopus` 0.2 pins)
26//! configures its vendored libopus with the autotools, so `autoreconf` has to
27//! be on PATH. That is fine on Linux, where the toolchain is usually already
28//! there, and not on a bare macOS box — see the README's prerequisites and the
29//! `brew install autoconf automake libtool` step in `.github/workflows/`.
30//! Opus and MP3's *patents* are what mattered here, not their build story:
31//! Opus was designed royalty-free from the start, and MP3's patents have all
32//! expired (the last, in the US, in 2017) — this app holds no codec license
33//! either way, so both are as free to ship as FLAC.
34//!
35//! ## Module layout
36//!
37//! One file per encoder ([`flac`], [`wav`], [`opus`], [`mp3`]), because each
38//! wraps a different codec crate with essentially nothing in common. What's
39//! shared — deciding *where* converted files go, not *how* they're encoded —
40//! lives here: [`convert_file`] dispatches to the right encoder for one file,
41//! [`plan_batch`] works out every destination path up front (mirroring the
42//! source folder structure under a new root), and [`passthrough_files`]
43//! copies everything else in that source folder verbatim. Two neighbours sit
44//! beside them for reasons of their own: [`pcm`] (sample reshaping shared by
45//! the encoders) and [`cleanup`] (removing what a cancelled batch already
46//! wrote).

65/// A conversion target format.
66#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
67#[serde(rename_all = "lowercase")]
68pub enum ConvertFormat {
69    /// Lossless — the default, and the only lossless target here besides WAV.
70    Flac,
71    /// Lossy, royalty-free — the modern choice when size matters.
72    Opus,
73    /// Lossy — kept for compatibility with players/hardware that don't speak
74    /// Opus.
75    Mp3,
76    /// Uncompressed PCM. Not really a "conversion" so much as a guaranteed
77    /// real copy — useful for a file this app has flagged as fake-lossless,
78    /// where the point is a known-honest baseline rather than smaller files.
79    Wav,
80}
81
82impl ConvertFormat {
83    /// The file extension a converted file gets, without a leading dot.
84    pub fn extension(self) -> &'static str {
85        match self {
86            ConvertFormat::Flac => "flac",
87            ConvertFormat::Opus => "opus",
88            ConvertFormat::Mp3 => "mp3",
89            ConvertFormat::Wav => "wav",
90        }
91    }
92}

94/// Default Opus bitrate: comfortably transparent for music at this codec's
95/// efficiency (see Opus's own listening-test results), well below where
96/// diminishing returns set in.
97pub const DEFAULT_OPUS_KBPS: u32 = 160;
98/// Default MP3 bitrate — LAME's own commonly-recommended "high quality" rate.
99pub const DEFAULT_MP3_KBPS: u32 = 256;

101/// What to convert to, and how. `bitrate_kbps` only matters for the two lossy
102/// formats; it is ignored for FLAC and WAV, which have no such setting.
103#[derive(Debug, Clone, Serialize, Deserialize)]
104pub struct ConvertSettings {
105    /// The output format.
106    pub format: ConvertFormat,
107    /// Target bitrate in kbps, for [`ConvertFormat::Opus`]/[`ConvertFormat::Mp3`].
108    /// `None` falls back to [`DEFAULT_OPUS_KBPS`]/[`DEFAULT_MP3_KBPS`].
109    pub bitrate_kbps: Option<u32>,
110}

112/// Errors that can occur while converting one file. Every variant carries the
113/// path it happened to, the same shape as [`crate::tags::TagError`], since a
114/// batch conversion reports failures per file rather than aborting the rest.
115#[derive(Debug, thiserror::Error)]
116pub enum ConvertError {
117    /// Could not create the destination folder, or write the output file
118    /// (path, reason).
119    #[error("{0}: {1}")]
120    Io(String, String),
121    /// The source file could not be decoded (path, reason).
122    #[error("{0}: could not decode the source file ({1})")]
123    Decode(String, String),
124    /// The encoder rejected the audio or failed mid-encode (path, reason).
125    #[error("{0}: could not encode the output file ({1})")]
126    Encode(String, String),
127    /// The source format, or something about it, isn't convertible yet
128    /// (path, reason).
129    #[error("{0}: {1}")]
130    Unsupported(String, String),
131    /// The caller asked to stop partway through this file (path). Distinct
132    /// from the other variants because it isn't a failure to report to the
133    /// user — a cancelled batch is discarded whole (see [`undo_batch`]), and
134    /// counting this file as "failed" in the summary would be misleading.
135    #[error("{0}: cancelled")]
136    Cancelled(String),
137}

139/// Convert `src` to `dest` per `settings`. `dest`'s parent folder is created
140/// if it doesn't exist yet (see [`plan_batch`], which is what typically
141/// produces `dest` in the first place).
142///
143/// `is_cancelled` is polled while decoding — once per packet, and again just
144/// before the encoder starts — and returns [`ConvertError::Cancelled`]
145/// without writing anything as soon as it answers `true`. Without it, a
146/// cancelled batch still has to wait for every in-flight file to decode *and*
147/// re-encode in full before it can stop, which on a long hi-res track is
148/// several seconds of a UI that looks hung; the encode itself is one opaque
149/// call into a codec crate and stays uninterruptible, so the decode half is
150/// what there is to give back.
151pub fn convert_file(
152    src: &Path,
153    dest: &Path,
154    settings: &ConvertSettings,
155    is_cancelled: &dyn Fn() -> bool,
156) -> Result<(), ConvertError> {
157    let name = || src.display().to_string();
158
159    if matches!(lower_ext(src).as_deref(), Some("dsf" | "dff")) {
160        return Err(ConvertError::Unsupported(
161            name(),
162            "DSD conversion isn't supported yet".to_string(),
163        ));
164    }
165    // Checked before the source is even opened, not only inside the decode
166    // loop: a worker that picks up its next file just as Cancel lands should
167    // not pay for opening and probing it first.
168    if is_cancelled() {
169        return Err(ConvertError::Cancelled(name()));
170    }
171
172    let pcm = pcm::decode_cancellable(src, is_cancelled)?;
173    if is_cancelled() {
174        return Err(ConvertError::Cancelled(name()));
175    }
176    ensure_parent_dir(dest)?;
177
178    match settings.format {
179        ConvertFormat::Flac => flac::encode(&pcm, dest, source_bit_depth(src)),
180        ConvertFormat::Wav => wav::encode(&pcm, dest),
181        ConvertFormat::Opus => {
182            opus::encode(&pcm, dest, settings.bitrate_kbps.unwrap_or(DEFAULT_OPUS_KBPS))
183        }
184        ConvertFormat::Mp3 => {
185            mp3::encode(&pcm, dest, settings.bitrate_kbps.unwrap_or(DEFAULT_MP3_KBPS))
186        }
187    }
188}

190/// The bit depth to encode FLAC output at: the source's own declared depth
191/// when there is one, clamped to 16–24 (FLAC's practical range, and the
192/// range every encoder here is expected to see), 16 otherwise — a header
193/// read, not a decode, so this costs nothing next to the decode
194/// [`convert_file`] already did.
195fn source_bit_depth(src: &Path) -> u32 {
196    decode::probe_info(src)
197        .ok()
198        .and_then(|info| info.bits)
199        .map(|b| b.clamp(16, 24))
200        .unwrap_or(16)
201}
202
203fn lower_ext(path: &Path) -> Option<String> {
204    path.extension()
205        .and_then(|e| e.to_str())
206        .map(|e| e.to_ascii_lowercase())
207}
208
209fn ensure_parent_dir(dest: &Path) -> Result<(), ConvertError> {
210    if let Some(parent) = dest.parent() {
211        std::fs::create_dir_all(parent)
212            .map_err(|e| ConvertError::Io(dest.display().to_string(), e.to_string()))?;
213    }
214    Ok(())
215}

217/// Work out one destination path per source in `sources`, mirroring each
218/// one's position relative to `input_root` under `output_root` instead —
219/// same folder, same subfolders, only the root and the extension change.
220///
221/// A source that (unexpectedly) doesn't live under `input_root` falls back
222/// to just its own file name directly under `output_root`, rather than
223/// failing the whole batch over one path oddity — the common root is
224/// computed from the same file list on the frontend (`commonDir`), so this
225/// should not normally happen, but a destination that's merely flatter than
226/// intended is a far smaller problem than a batch that refuses to run.
227pub fn plan_batch(
228    sources: &[PathBuf],
229    input_root: &Path,
230    output_root: &Path,
231    format: ConvertFormat,
232) -> Vec<PathBuf> {
233    sources
234        .iter()
235        .map(|src| {
236            let rel: &Path = src.strip_prefix(input_root).unwrap_or_else(|_| {
237                src.file_name().map(Path::new).unwrap_or(src.as_path())
238            });
239            output_root.join(rel).with_extension(format.extension())
240        })
241        .collect()
242}

244/// Copy every file under `input_root` that isn't in `exclude` (the tracks
245/// just converted) to the same relative position under `output_root` —
246/// covers, `.m3u` playlists, generated spectrograms, anything else that
247/// shares the folder, verbatim. "Tout ou rien": there is no per-file choice,
248/// by design (see the module's callers) — a file the caller wants left out
249/// belongs in `exclude`, not a filter this function grows.
250pub fn passthrough_files(
251    input_root: &Path,
252    output_root: &Path,
253    exclude: &HashSet<PathBuf>,
254) -> std::io::Result<Vec<PathBuf>> {
255    let mut written = Vec::new();
256    for entry in walkdir::WalkDir::new(input_root)
257        .into_iter()
258        .filter_map(Result::ok)
259    {
260        let path = entry.into_path();
261        if !path.is_file() || exclude.contains(&path) {
262            continue;
263        }
264        let Ok(rel) = path.strip_prefix(input_root) else {
265            continue;
266        };
267        let dest = output_root.join(rel);
268        if let Some(parent) = dest.parent() {
269            std::fs::create_dir_all(parent)?;
270        }
271        std::fs::copy(&path, &dest)?;
272        written.push(dest);
273    }
274    Ok(written)
275}

281    #[test]
282    fn plan_batch_mirrors_the_source_structure_under_the_new_root() {
283        let input_root = Path::new("/music/Album");
284        let output_root = Path::new("/export/Album (FLAC)");
285        let sources = vec![
286            PathBuf::from("/music/Album/01 track.mp3"),
287            PathBuf::from("/music/Album/Disc 2/02 track.mp3"),
288        ];
289        let got = plan_batch(&sources, input_root, output_root, ConvertFormat::Flac);
290        assert_eq!(
291            got,
292            vec![
293                PathBuf::from("/export/Album (FLAC)/01 track.flac"),
294                PathBuf::from("/export/Album (FLAC)/Disc 2/02 track.flac"),
295            ]
296        );
297    }

299    /// A source outside `input_root` (shouldn't normally happen — see the
300    /// doc comment) still gets a destination, just a flatter one, rather
301    /// than panicking or silently dropping the file from the batch.
302    #[test]
303    fn plan_batch_falls_back_to_the_file_name_for_a_path_outside_the_root() {
304        let input_root = Path::new("/music/Album");
305        let output_root = Path::new("/export");
306        let sources = vec![PathBuf::from("/elsewhere/loose.wav")];
307        let got = plan_batch(&sources, input_root, output_root, ConvertFormat::Wav);
308        assert_eq!(got, vec![PathBuf::from("/export/loose.wav")]);
309    }

311    #[test]
312    fn passthrough_copies_everything_except_the_excluded_paths() {
313        let dir = tempfile::tempdir().expect("tempdir");
314        let input_root = dir.path().join("in");
315        let output_root = dir.path().join("out");
316        std::fs::create_dir_all(input_root.join("Disc 1")).expect("mkdir");
317        std::fs::write(input_root.join("track.mp3"), b"audio").expect("write");
318        std::fs::write(input_root.join("cover.jpg"), b"jpeg").expect("write");
319        std::fs::write(input_root.join("Disc 1/playlist.m3u"), b"m3u").expect("write");
320
321        let mut exclude = HashSet::new();
322        exclude.insert(input_root.join("track.mp3"));
323
324        let written = passthrough_files(&input_root, &output_root, &exclude).expect("copy");
325        assert_eq!(written.len(), 2, "{written:?}");
326        assert!(output_root.join("cover.jpg").exists());
327        assert!(output_root.join("Disc 1/playlist.m3u").exists());
328        assert!(!output_root.join("track.mp3").exists());
329    }

331    /// Cancellation is checked before the first packet is decoded, so an
332    /// already-cancelled batch never reads or writes anything at all — and
333    /// in particular never creates `dest`, which is what [`undo_batch`]
334    /// would otherwise have to clean up.
335    #[test]
336    fn an_already_cancelled_convert_writes_nothing() {
337        let dir = tempfile::tempdir().expect("tempdir");
338        let src = dir.path().join("in.wav");
339        let dest = dir.path().join("out/track.flac");
340        // Contents don't matter: cancellation is checked before any decode.
341        std::fs::write(&src, b"not really a wav").expect("write");
342
343        let settings = ConvertSettings {
344            format: ConvertFormat::Flac,
345            bitrate_kbps: None,
346        };
347        let err = convert_file(&src, &dest, &settings, &|| true).expect_err("cancelled");
348        assert!(matches!(err, ConvertError::Cancelled(_)), "{err:?}");
349        assert!(!dest.exists());
350    }
