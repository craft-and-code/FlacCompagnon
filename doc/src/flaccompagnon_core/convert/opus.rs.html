1//! Opus encoding: `audiopus` (vendors `libopus` via `audiopus_sys`, no
2//! system library needed) produces raw Opus packets; this module also mixes
3//! them into an Ogg container by hand (RFC 7845, "Ogg Opus"), using the
4//! generic `ogg` crate's page writer — `audiopus` only speaks the codec, not
5//! the container a `.opus` file actually needs in order to be playable at
6//! all.
7//!
8//! Opus only accepts four fixed sample rates (8/12/16/24/48 kHz); a source at
9//! any other rate (44.1 kHz being the common case for music) is resampled to
10//! 48 kHz first with [`super::resample_linear`] — the same plain linear
11//! interpolator [`super::mp3`] also uses, since neither format's own lossy
12//! coding needs a mastering-grade resampler ahead of it (see that function's
13//! own doc comment for the full reasoning).
14

17use audiopus::coder::Encoder;
18use audiopus::{Application, Bitrate, Channels, SampleRate};
19use ogg::writing::{PacketWriteEndInfo, PacketWriter};

21use super::ConvertError;
22use crate::decode::PcmAudio;

24const OPUS_SAMPLE_RATE: u32 = 48_000;
25/// 20ms frames — the size most Opus encoders default to, and a reasonable
26/// balance of overhead vs. flexibility for a file being written to disk
27/// rather than streamed (there's no real-time latency constraint either
28/// way).
29const FRAME_MS: u32 = 20;
30const FRAME_SAMPLES_PER_CHANNEL: usize = (OPUS_SAMPLE_RATE * FRAME_MS / 1000) as usize;
31/// Opus's own documented worst case for a packet at this frame size.
32const MAX_PACKET_BYTES: usize = 4000;
33/// Every page in one logical Ogg stream shares one serial number; this file
34/// never holds more than one stream, so a fixed constant is enough.
35const OGG_SERIAL: u32 = 1;
36
37/// Encode `pcm` as Ogg Opus to `dest`, at `bitrate_kbps`.
38pub(super) fn encode(pcm: &PcmAudio, dest: &Path, bitrate_kbps: u32) -> Result<(), ConvertError> {
39    let name = || dest.display().to_string();
40
41    let channels_enum = match pcm.channels {
42        1 => Channels::Mono,
43        2 => Channels::Stereo,
44        n => {
45            return Err(ConvertError::Unsupported(
46                name(),
47                format!("Opus only supports mono or stereo here (got {n} channels)"),
48            ))
49        }
50    };
51
52    let resampled = if pcm.sample_rate == OPUS_SAMPLE_RATE {
53        pcm.samples.clone()
54    } else {
55        super::resample_linear(&pcm.samples, pcm.channels, pcm.sample_rate, OPUS_SAMPLE_RATE)
56    };
57
58    let mut encoder = Encoder::new(SampleRate::Hz48000, channels_enum, Application::Audio)
59        .map_err(|e| ConvertError::Encode(name(), e.to_string()))?;
60    encoder
61        .set_bitrate(Bitrate::BitsPerSecond(
62            bitrate_kbps.saturating_mul(1000) as i32,
63        ))
64        .map_err(|e| ConvertError::Encode(name(), e.to_string()))?;
65    // Opus's own algorithmic delay — folded into the granule position so a
66    // decoder can compensate for it, per RFC 7845 §4.
67    let pre_skip = encoder.lookahead().unwrap_or(0).min(u16::MAX as u32) as u16;
68
69    let file = std::fs::File::create(dest).map_err(|e| ConvertError::Io(name(), e.to_string()))?;
70    let mut writer = PacketWriter::new(file);
71    write_header_pages(&mut writer, pcm.channels as u8, pre_skip, pcm.sample_rate)
72        .map_err(|e| ConvertError::Io(name(), e.to_string()))?;
73
74    let frame_len = FRAME_SAMPLES_PER_CHANNEL * pcm.channels;
75    let mut out_buf = vec![0u8; MAX_PACKET_BYTES];
76    let mut samples_encoded: u64 = pre_skip as u64;
77    // At least one frame even for a near-empty source, so the stream always
78    // gets its `EndStream` packet.
79    let total_frames = resampled.len().div_ceil(frame_len.max(1)).max(1);
80
81    for frame_index in 0..total_frames {
82        let start = frame_index * frame_len;
83        let end = (start + frame_len).min(resampled.len());
84        let mut frame = resampled.get(start..end).unwrap_or(&[]).to_vec();
85        // Pads only the final, partial frame with silence — Opus frames must
86        // all be the same fixed length; a player trims this back out using
87        // the stream's final granule position, which stays at the *true*
88        // sample count below rather than following this padding.
89        frame.resize(frame_len, 0.0);
90
91        let n = encoder
92            .encode_float(&frame, &mut out_buf)
93            .map_err(|e| ConvertError::Encode(name(), e.to_string()))?;
94
95        let real_samples_this_frame = end.saturating_sub(start) / pcm.channels.max(1);
96        samples_encoded += real_samples_this_frame as u64;
97
98        let is_last = frame_index + 1 == total_frames;
99        let end_info = if is_last {
100            PacketWriteEndInfo::EndStream
101        } else {
102            PacketWriteEndInfo::NormalPacket
103        };
104        writer
105            .write_packet(out_buf[..n].to_vec(), OGG_SERIAL, end_info, samples_encoded)
106            .map_err(|e| ConvertError::Io(name(), e.to_string()))?;
107    }
108
109    Ok(())
110}

112/// Writes the two mandatory header pages an Ogg Opus stream opens with (RFC
113/// 7845 §5.1/§5.2): `OpusHead` (channel count, pre-skip, the source's own
114/// sample rate as an informational field only — decoders always run at
115/// 48kHz internally) and `OpusTags` (just a vendor string; no comments —
116/// track tags live in the file's own tag system, not duplicated into the
117/// Opus stream itself). Each gets its own page per spec
118/// (`PacketWriteEndInfo::EndPage`).
119fn write_header_pages(
120    writer: &mut PacketWriter<std::fs::File>,
121    channels: u8,
122    pre_skip: u16,
123    input_sample_rate: u32,
124) -> std::io::Result<()> {
125    let mut head = Vec::with_capacity(19);
126    head.extend_from_slice(b"OpusHead");
127    head.push(1); // version
128    head.push(channels);
129    head.extend_from_slice(&pre_skip.to_le_bytes());
130    head.extend_from_slice(&input_sample_rate.to_le_bytes());
131    head.extend_from_slice(&0i16.to_le_bytes()); // output gain
132    head.push(0); // channel mapping family 0: mono/stereo, no mapping table
133    writer.write_packet(head, OGG_SERIAL, PacketWriteEndInfo::EndPage, 0)?;
134
135    let vendor = b"FlacCompagnon";
136    let mut tags = Vec::with_capacity(8 + 4 + vendor.len() + 4);
137    tags.extend_from_slice(b"OpusTags");
138    tags.extend_from_slice(&(vendor.len() as u32).to_le_bytes());
139    tags.extend_from_slice(vendor);
140    tags.extend_from_slice(&0u32.to_le_bytes()); // zero user comments
141    writer.write_packet(tags, OGG_SERIAL, PacketWriteEndInfo::EndPage, 0)
142}

144#[cfg(test)]
145mod tests {
146    use super::*;
147
148    /// A full round trip through the real encoder: encode a short stereo
149    /// tone, then read the file back with `ogg`'s own `PacketReader` and
150    /// confirm the container shape a real Opus decoder would also check —
151    /// `OpusHead`/`OpusTags` magic first, then at least one audio packet with
152    /// a positive granule position. This doesn't decode the Opus audio
153    /// itself (lossy, so there is no bit-exact ground truth the way FLAC/WAV
154    /// have) — it verifies the *container* this module hand-builds is the
155    /// shape RFC 7845 requires, independent of `audiopus`'s own correctness.
156    #[test]
157    fn produces_a_well_formed_ogg_opus_stream() {
158        let sample_rate = 44_100u32;
159        let channels = 2usize;
160        let frames = 4000;
161        let mut samples = Vec::with_capacity(frames * channels);
162        for t in 0..frames {
163            let phase = t as f32 / sample_rate as f32;
164            let s = (phase * 440.0 * std::f32::consts::TAU).sin() * 0.4;
165            samples.push(s);
166            samples.push(s);
167        }
168        let pcm = PcmAudio {
169            samples,
170            sample_rate,
171            channels,
172        };
173        let dir = tempfile::tempdir().expect("tempdir");
174        let dest = dir.path().join("tone.opus");
175        encode(&pcm, &dest, 96).expect("encode");
176
177        let file = std::fs::File::open(&dest).expect("reopen");
178        let mut reader = ogg::reading::PacketReader::new(file);
179        let head = reader
180            .read_packet()
181            .expect("read head packet")
182            .expect("head packet present");
183        assert!(head.data.starts_with(b"OpusHead"));
184        let tags = reader
185            .read_packet()
186            .expect("read tags packet")
187            .expect("tags packet present");
188        assert!(tags.data.starts_with(b"OpusTags"));
189        let first_audio = reader
190            .read_packet()
191            .expect("read first audio packet")
192            .expect("audio packet present");
193        assert!(!first_audio.data.is_empty());
194        assert!(first_audio.absgp_page() > 0);
195    }
