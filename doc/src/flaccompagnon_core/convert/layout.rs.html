1//! Where the output goes: one destination per converted file, and the
2//! non-audio files that travelled with them.
3//!
4//! Split out of [`super`] because it answers a different question from the
5//! rest of that module. The encoders decide *how* a file is written; this
6//! decides *where*, and it is the part users notice when it is wrong — a
7//! flattened album is immediately visible, a slightly different LPC order is
8//! not.
9//!
10//! ## Why the base folder is remembered, not computed
11//!
12//! An earlier version derived one shared root from the file list, as the
13//! deepest common ancestor of every source's parent. That is subtly not what
14//! the user asked for, and the gap showed up in the most ordinary case there
15//! is: drop a band folder holding a single album, and every file shares the
16//! parent `Band/Album`, so the "common root" *is* `Band/Album` and stripping
17//! it drops both folders — the tracks land flat in the destination, the band
18//! and album folders gone. The user dropped a folder; the folder is what they
19//! expect to find.
20//!
21//! So each source carries the base its layout is expressed against, set from
22//! what was actually dropped rather than recovered afterwards. Per file, not
23//! per batch, which also fixes a second-order oddity: two unrelated folders
24//! no longer get forced under whatever distant ancestor they happen to share.
25
26use std::collections::HashSet;
27use std::path::{Path, PathBuf};
28
29use super::ConvertFormat;
30
31/// One source file and the folder its layout should be mirrored from.
32#[derive(Debug, Clone)]
33pub struct ConvertSource {
34    /// The audio file to convert.
35    pub path: PathBuf,
36    /// The folder `path`'s position is expressed relative to — the *parent*
37    /// of what was dropped, so a dropped folder is itself recreated at the
38    /// destination. For a loose file this is simply its own parent, so it
39    /// lands directly in the output folder.
40    pub base: PathBuf,
41}
42
43/// Work out one destination path per source, reproducing each file's position
44/// relative to its own [`ConvertSource::base`] under `output_root` — same
45/// folders, same nesting, only the root and the extension change.
46///
47/// A source that (unexpectedly) doesn't live under its base falls back to
48/// just its own file name directly under `output_root`, rather than failing
49/// the whole batch over one path oddity: a destination that is merely flatter
50/// than intended is a far smaller problem than a batch that refuses to run.
51pub fn plan_batch(
52    sources: &[ConvertSource],
53    output_root: &Path,
54    format: ConvertFormat,
55) -> Vec<PathBuf> {
56    sources
57        .iter()
58        .map(|s| {
59            let rel: &Path = s.path.strip_prefix(&s.base).unwrap_or_else(|_| {
60                s.path.file_name().map(Path::new).unwrap_or(s.path.as_path())
61            });
62            output_root.join(rel).with_extension(format.extension())
63        })
64        .collect()
65}
66
67/// The folders to sweep for non-audio files, paired with the base each one's
68/// contents are mirrored against.
69///
70/// Recovered from the sources rather than passed in: a source's first path
71/// component under its base *is* what was dropped. `Band/Album/01.flac`
72/// relative to `/music` means `/music/Band` was dropped, so that is the
73/// folder to sweep — sweeping `/music` itself would drag in every other band
74/// sitting beside it.
75fn sweep_roots(sources: &[ConvertSource]) -> HashSet<(PathBuf, PathBuf)> {
76    sources
77        .iter()
78        .filter_map(|s| {
79            let rel = s.path.strip_prefix(&s.base).ok()?;
80            let mut comps = rel.components();
81            let first = comps.next()?;
82            // A single component means the file sits directly in `base` — it
83            // was dropped as a loose file, so the folder to sweep is `base`.
84            let root = if comps.next().is_none() {
85                s.base.clone()
86            } else {
87                s.base.join(first)
88            };
89            Some((root, s.base.clone()))
90        })
91        .collect()
92}

94/// Copy every file that isn't one of the tracks being converted — covers,
95/// `.m3u` playlists, generated spectrograms, anything else sharing the
96/// folder — to the same relative position under `output_root`.
97///
98/// "Tout ou rien": there is no per-file choice, by design. The audio files in
99/// `sources` are the exclusion list, so a caller cannot get it out of step
100/// with what was actually converted.
101///
102/// ## Overlapping drops
103///
104/// Dropping a folder *and* something inside it means one neighbouring file is
105/// reachable from two sweeps, under two different bases — and those bases give
106/// it two *different* destinations (`out/Band/Album/cover.jpg` from `/music`,
107/// `out/Album/cover.jpg` from `/music/Band`). So the thing that has to be
108/// unique is the **source**, not the destination: deduplicating on the
109/// destination lets the same file through twice, which is exactly what it did.
110///
111/// Which of the two destinations wins then has to be decided rather than left
112/// to chance. The sweeps run outermost base first, so the deepest surviving
113/// structure wins — consistent with the rest of this module: the user dropped
114/// `Band`, so `Band/` is what they expect to find. Sorting also makes the
115/// returned list stable, which a `HashSet` iteration order alone was not.
116pub fn passthrough_files(
117    sources: &[ConvertSource],
118    output_root: &Path,
119) -> std::io::Result<Vec<PathBuf>> {
120    let audio: HashSet<&Path> = sources.iter().map(|s| s.path.as_path()).collect();
121    let mut written = Vec::new();
122    let mut copied: HashSet<PathBuf> = HashSet::new();
123
124    // A path sorts after every prefix of itself, so ordering by base puts the
125    // outermost (shortest) one first; the root breaks ties between two sweeps
126    // sharing a base.
127    let mut roots: Vec<(PathBuf, PathBuf)> = sweep_roots(sources).into_iter().collect();
128    roots.sort_by(|(a_root, a_base), (b_root, b_base)| {
129        a_base.cmp(b_base).then_with(|| a_root.cmp(b_root))
130    });
131
132    for (root, base) in roots {
133        let walk = walkdir::WalkDir::new(&root).sort_by_file_name();
134        for entry in walk.into_iter().filter_map(Result::ok) {
135            let path = entry.into_path();
136            if !path.is_file() || audio.contains(path.as_path()) {
137                continue;
138            }
139            let Ok(rel) = path.strip_prefix(&base) else {
140                continue;
141            };
142            // Only after `strip_prefix` succeeded: a file this sweep can't
143            // place must stay eligible for a later one that can.
144            if !copied.insert(path.clone()) {
145                continue;
146            }
147            let dest = output_root.join(rel);
148            if let Some(parent) = dest.parent() {
149                std::fs::create_dir_all(parent)?;
150            }
151            std::fs::copy(&path, &dest)?;
152            written.push(dest);
153        }
154    }
155    Ok(written)
156}
