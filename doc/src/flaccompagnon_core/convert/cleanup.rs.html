1//! Undoing a cancelled conversion batch.
2//!
3//! Cancellation here is not the same problem it is for analysis. Analysis
4//! only *reads*, so abandoning it mid-run leaves nothing behind — dropping
5//! the partial results is the whole cleanup. A conversion batch writes real
6//! files, and the workers that were already mid-file when Cancel was pressed
7//! go on to finish and write theirs (see `parallel_map_ordered`'s own
8//! comment: in-flight items are allowed to complete rather than being killed
9//! mid-write, which would leave a truncated file instead of no file). So
10//! "cancelled" without this module means the user gets told the run was
11//! cancelled while a folder of half a batch's output sits on disk — which is
12//! exactly the behaviour this exists to fix.
13//!
14//! The one thing it must not do is delete something it didn't create. A user
15//! converting into a folder that already holds a previous run's output would
16//! otherwise have those earlier files removed by cancelling this one, which
17//! is worse than the problem being solved. Hence `preexisting`: the caller
18//! records which destination paths were already on disk *before* the batch
19//! started, and only the rest are ever removed.
20
21use std::collections::HashSet;
22use std::path::{Path, PathBuf};
23
24/// Delete the output files a cancelled batch created under `output_root`,
25/// then remove any folders that creating them had left empty. `planned` is
26/// every destination the batch was going to write (see
27/// [`super::plan_batch`]); `preexisting` is the subset of those that already
28/// existed before it started, which are left exactly as they were.
29///
30/// Returns how many files were actually removed. Errors are deliberately not
31/// propagated: this runs while unwinding a cancelled batch, where a file that
32/// can't be deleted (permissions, a lock, something removed it already) is
33/// worth neither failing the cancellation over nor reporting on top of it.
34pub fn undo_batch(planned: &[PathBuf], preexisting: &HashSet<PathBuf>, output_root: &Path) -> usize {
35    let mut removed = 0usize;
36    let mut parents: HashSet<PathBuf> = HashSet::new();
37
38    for dest in planned {
39        if preexisting.contains(dest) || !dest.starts_with(output_root) || !dest.is_file() {
40            continue;
41        }
42        if std::fs::remove_file(dest).is_ok() {
43            removed += 1;
44            if let Some(parent) = dest.parent() {
45                parents.insert(parent.to_path_buf());
46            }
47        }
48    }
49
50    // Deepest first, so a nested folder is gone before its parent is tried —
51    // otherwise the parent still looks non-empty and survives a run that
52    // emptied it.
53    let mut parents: Vec<PathBuf> = parents.into_iter().collect();
54    parents.sort_by_key(|p| std::cmp::Reverse(p.components().count()));
55    for parent in parents {
56        prune_empty_dirs(&parent, output_root);
57    }
58
59    removed
60}
61
62/// Remove `dir` and each of its ancestors up to (but never including)
63/// `output_root`, stopping at the first one that isn't empty.
64/// `std::fs::remove_dir` refuses to remove a non-empty folder, so its failure
65/// *is* the emptiness test — no separate `read_dir` race between checking and
66/// removing.
67fn prune_empty_dirs(mut dir: &Path, output_root: &Path) {
68    while dir.starts_with(output_root) && dir != output_root {
69        if std::fs::remove_dir(dir).is_err() {
70            return;
71        }
72        match dir.parent() {
73            Some(parent) => dir = parent,
74            None => return,
75        }
76    }
77}

79#[cfg(test)]
80mod tests {
81    use super::*;
82
83    fn touch(path: &Path) {
84        if let Some(parent) = path.parent() {
85            std::fs::create_dir_all(parent).expect("mkdir");
86        }
87        std::fs::write(path, b"data").expect("write");
88    }
89
90    #[test]
91    fn removes_the_files_the_batch_wrote() {
92        let dir = tempfile::tempdir().expect("tempdir");
93        let root = dir.path();
94        let a = root.join("a.flac");
95        let b = root.join("Disc 2/b.flac");
96        touch(&a);
97        touch(&b);
98
99        let planned = vec![a.clone(), b.clone()];
100        assert_eq!(undo_batch(&planned, &HashSet::new(), root), 2);
101        assert!(!a.exists());
102        assert!(!b.exists());
103    }
104
105    /// The case this module's `preexisting` argument exists for: converting
106    /// into a folder that already holds an earlier run's output, then
107    /// cancelling, must not take that earlier output down with it.
108    #[test]
109    fn leaves_files_that_were_already_there_before_the_batch() {
110        let dir = tempfile::tempdir().expect("tempdir");
111        let root = dir.path();
112        let old = root.join("old.flac");
113        let fresh = root.join("fresh.flac");
114        touch(&old);
115        touch(&fresh);
116
117        let planned = vec![old.clone(), fresh.clone()];
118        let preexisting: HashSet<PathBuf> = [old.clone()].into_iter().collect();
119        assert_eq!(undo_batch(&planned, &preexisting, root), 1);
120        assert!(old.exists(), "a file that predates the batch must survive it");
121        assert!(!fresh.exists());
122    }
123
124    /// A planned destination that never got written (the batch was cancelled
125    /// before reaching it) is simply not there — that's the normal case for
126    /// most of the list, not an error.
127    #[test]
128    fn a_destination_that_was_never_written_is_not_an_error() {
129        let dir = tempfile::tempdir().expect("tempdir");
130        let root = dir.path();
131        let never = root.join("never.flac");
132
133        assert_eq!(undo_batch(&[never], &HashSet::new(), root), 0);
134    }
135
136    /// Subfolders the batch created to mirror the source layout go too, so a
137    /// cancelled run doesn't leave an empty skeleton of the album's folder
138    /// structure behind.
139    #[test]
140    fn prunes_folders_the_batch_created_and_then_emptied() {
141        let dir = tempfile::tempdir().expect("tempdir");
142        let root = dir.path();
143        let nested = root.join("Album/Disc 2/b.flac");
144        touch(&nested);
145
146        undo_batch(&[nested], &HashSet::new(), root);
147        assert!(!root.join("Album/Disc 2").exists());
148        assert!(!root.join("Album").exists());
149        assert!(root.exists(), "the output root itself is never removed");
150    }
151
152    /// A folder that still holds something else — a cover the user had put
153    /// there, a file from an earlier run — must survive, along with every
154    /// ancestor above it.
155    #[test]
156    fn keeps_a_folder_that_still_has_something_else_in_it() {
157        let dir = tempfile::tempdir().expect("tempdir");
158        let root = dir.path();
159        let converted = root.join("Album/b.flac");
160        let cover = root.join("Album/cover.jpg");
161        touch(&converted);
162        touch(&cover);
163
164        undo_batch(&[converted], &HashSet::new(), root);
165        assert!(cover.exists());
166        assert!(root.join("Album").exists());
167    }
168
169    /// Defence in depth against a bad `output_root`/`planned` pairing: this
170    /// deletes files, so a path outside the folder the user picked is never
171    /// touched, whatever the caller passed.
172    #[test]
173    fn never_touches_anything_outside_the_output_root() {
174        let dir = tempfile::tempdir().expect("tempdir");
175        let root = dir.path().join("out");
176        std::fs::create_dir_all(&root).expect("mkdir");
177        let outside = dir.path().join("elsewhere/keep.flac");
178        touch(&outside);
179
180        assert_eq!(undo_batch(&[outside.clone()], &HashSet::new(), &root), 0);
181        assert!(outside.exists());
182    }
