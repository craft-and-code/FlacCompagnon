1//! The PCM pipeline between a decoder and an encoder: reading a source into
2//! samples, resampling them, and quantizing them to integers.
3//!
4//! Split out of [`super`] rather than left beside the batch-planning code it
5//! has nothing to do with — this is the whole of the conversion path that
6//! touches sample values themselves, and none of it is specific to one
7//! output format ([`super::opus`]/[`super::mp3`] resample, all four
8//! quantize, every one of them decodes first). The two helpers stay
9//! re-exported from [`super`] so the encoders keep calling
10//! `super::f32_to_ints` / `super::resample_linear` unchanged.
11
12use std::path::Path;

14use super::ConvertError;

17/// [`decode::decode_to_pcm`], but polling `is_cancelled` between packets.
18/// Deliberately not a flag added to `decode_to_pcm` itself: that one is also
19/// the playback engine's path, where there is nothing to cancel, and it would
20/// pay for a callback it never uses on every file the app plays.
21pub(super) fn decode_cancellable(
22    src: &Path,
23    is_cancelled: &dyn Fn() -> bool,
24) -> Result<decode::PcmAudio, ConvertError> {
25    let name = || src.display().to_string();
26    let mut decoder = decode::PcmStreamDecoder::open(src)
27        .map_err(|e| ConvertError::Decode(name(), e.to_string()))?;
28    let sample_rate = decoder.sample_rate;
29    let channels = decoder.channels;
30
31    let mut samples: Vec<f32> = Vec::new();
32    loop {
33        if is_cancelled() {
34            return Err(ConvertError::Cancelled(name()));
35        }
36        let chunk = decoder
37            .next_chunk()
38            .map_err(|e| ConvertError::Decode(name(), e.to_string()))?;
39        match chunk {
40            Some(chunk) => samples.extend_from_slice(&chunk),
41            None => break,
42        }
43    }
44
45    if samples.is_empty() {
46        return Err(ConvertError::Decode(
47            name(),
48            "no audio data decoded".to_string(),
49        ));
50    }
51    Ok(decode::PcmAudio {
52        samples,
53        sample_rate,
54        channels,
55    })
56}

58/// Simple linear-interpolation resampler, shared by [`super::opus`] and
59/// [`super::mp3`] — both formats only accept a short, fixed list of sample
60/// rates (Opus: 8/12/16/24/48kHz; MP3/LAME: the MPEG1/2/2.5 rates), so a
61/// source at any other rate (44.1kHz being the common case for music, and any
62/// of this app's own hi-res detections being another) needs resampling before
63/// either encoder will accept it. Not a mastering-grade resampler with proper
64/// anti-aliasing filtering — a windowed-sinc/polyphase resampler would do
65/// better, at the cost of a filter design neither format's own lossy coding
66/// otherwise needs; both already discard far more information than this
67/// step adds at any ordinary music sample rate, which is why the simpler
68/// approach was judged good enough here — worth revisiting if it turns out
69/// to matter in practice.
70pub(crate) fn resample_linear(
71    samples: &[f32],
72    channels: usize,
73    from_hz: u32,
74    to_hz: u32,
75) -> Vec<f32> {
76    if from_hz == to_hz || channels == 0 {
77        return samples.to_vec();
78    }
79    let frames_in = samples.len() / channels;
80    if frames_in == 0 {
81        return Vec::new();
82    }
83    let ratio = to_hz as f64 / from_hz as f64;
84    let frames_out = ((frames_in as f64) * ratio).round().max(1.0) as usize;
85    let mut out = Vec::with_capacity(frames_out * channels);
86    for i in 0..frames_out {
87        let src_pos = i as f64 / ratio;
88        let i0 = (src_pos.floor() as usize).min(frames_in - 1);
89        let i1 = (i0 + 1).min(frames_in - 1);
90        let frac = (src_pos - i0 as f64) as f32;
91        for c in 0..channels {
92            let a = samples[i0 * channels + c];
93            let b = samples[i1 * channels + c];
94            out.push(a + (b - a) * frac);
95        }
96    }
97    out
98}

100/// Converts normalized `[-1.0, 1.0]` samples to signed integers at
101/// `bit_depth`, clamped to that depth's representable range — the reverse of
102/// the reconstruction `decode::stream` does for analysis (see its
103/// `int_scale` comment there).
104pub(crate) fn f32_to_ints(samples: &[f32], bit_depth: u32) -> Vec<i32> {
105    let scale = 2f32.powi(bit_depth as i32 - 1);
106    let max = scale - 1.0;
107    let min = -scale;
108    samples
109        .iter()
110        .map(|&s| (s * scale).round().clamp(min, max) as i32)
111        .collect()
112}

114#[cfg(test)]
115mod tests {
116    use super::*;
117
118    #[test]
119    fn f32_to_ints_clamps_at_the_target_bit_depth() {
120        let samples = [-2.0f32, -1.0, 0.0, 0.5, 1.0, 2.0];
121        let got = f32_to_ints(&samples, 16);
122        assert_eq!(got, vec![-32768, -32768, 0, 16384, 32767, 32767]);
123    }
124
125    /// Halving the rate must halve the frame count (not the sample count —
126    /// that stays interleaved by `channels`), and the resampled frames must
127    /// still be interleaved in the same channel order. A resampler that
128    /// treated the buffer as mono would pass a frame-count check and still
129    /// scramble left and right, which is why this asserts on values too.
130    #[test]
131    fn halving_the_rate_halves_the_frames_and_keeps_the_channels_apart() {
132        // Two channels held at distinct constant values: any interpolation
133        // between them stays at that value, so a channel swap is visible.
134        let channels = 2;
135        let samples: Vec<f32> = (0..8).flat_map(|_| [1.0f32, -1.0]).collect();
136        let got = resample_linear(&samples, channels, 48_000, 24_000);
137        assert_eq!(got.len(), 4 * channels);
138        for frame in got.chunks(channels) {
139            assert_eq!(frame[0], 1.0);
140            assert_eq!(frame[1], -1.0);
141        }
142    }
143
144    #[test]
145    fn an_unchanged_rate_is_returned_verbatim() {
146        let samples = vec![0.25f32, -0.25, 0.5, -0.5];
147        assert_eq!(resample_linear(&samples, 2, 44_100, 44_100), samples);
148    }
149
150    /// Degenerate inputs must return rather than panicking on a
151    /// divide-by-zero or an out-of-range index — the callers hand this
152    /// whatever `decode_to_pcm` produced, and a zero-length or zero-channel
153    /// decode is exactly the malformed-file case this app exists to run
154    /// into. A zero channel count has no frame layout to resample *to*, so
155    /// the buffer comes back untouched rather than emptied: dropping the
156    /// samples would turn a header oddity into silent data loss further down
157    /// the encoder chain.
158    #[test]
159    fn empty_or_zero_channel_input_does_not_panic() {
160        assert!(resample_linear(&[], 2, 44_100, 48_000).is_empty());
161        assert_eq!(resample_linear(&[0.1, 0.2], 0, 44_100, 48_000), vec![0.1, 0.2]);
162    }
163}
