1//! MP3 encoding via `mp3lame-encoder` (vendors LAME through `mp3lame-sys`, no
2//! system library needed).
3//!
4//! LAME only accepts a short fixed list of input sample rates — 32/44.1/48kHz
5//! for MPEG1, 16/22.05/24kHz for MPEG2, 8/11.025/12kHz for MPEG2.5, and
6//! nothing above 48kHz at all. That last part matters here specifically:
7//! hi-res sources (88.2/96/176.4/192kHz) are exactly what this app spends
8//! most of its analysis code detecting, so they are a routine input to this
9//! encoder, not an edge case. A source outside the supported set is
10//! resampled first with [`super::resample_linear`] (the same helper
11//! [`super::opus`] uses) down to the nearest rate LAME accepts, preferring an
12//! even division (88.2k -> 44.1k, 96k -> 48k, ...) over the closest rate by
13//! raw distance, since that keeps the common hi-res cases landing on the
14//! rate a listener would expect.
15//!
16//! LAME's own encoded output is a bare MP3 frame stream: no ID3 tag is
17//! written here (this app's own tag writer handles that separately, the same
18//! way it already does for the other three formats), and no VBR — output is
19//! constant bitrate at the requested [`Bitrate`] preset, the closest match to
20//! the `bitrate_kbps` setting the caller asked for.
21
22use std::mem::MaybeUninit;
23use std::path::Path;
24
25use mp3lame_encoder::{Bitrate, Builder, FlushNoGap, InterleavedPcm};
26
27use super::ConvertError;
28use crate::decode::PcmAudio;

30/// Sample rates LAME accepts as encoder input, across all three MPEG
31/// versions it can produce.
32const LAME_SUPPORTED_RATES: &[u32] = &[
33    8_000, 11_025, 12_000, 16_000, 22_050, 24_000, 32_000, 44_100, 48_000,
34];
35
36/// Encode `pcm` as MP3 to `dest`, at the bitrate preset closest to
37/// `bitrate_kbps`.
38pub(super) fn encode(pcm: &PcmAudio, dest: &Path, bitrate_kbps: u32) -> Result<(), ConvertError> {
39    let name = || dest.display().to_string();
40
41    if pcm.channels == 0 || pcm.channels > 2 {
42        return Err(ConvertError::Unsupported(
43            name(),
44            format!("MP3 only supports mono or stereo here (got {} channels)", pcm.channels),
45        ));
46    }
47
48    let target_rate = nearest_supported_rate(pcm.sample_rate);
49    let resampled = if target_rate == pcm.sample_rate {
50        pcm.samples.clone()
51    } else {
52        super::resample_linear(&pcm.samples, pcm.channels, pcm.sample_rate, target_rate)
53    };
54
55    let mut encoder = Builder::new()
56        .ok_or_else(|| ConvertError::Encode(name(), "could not allocate the LAME encoder".to_string()))?
57        .with_num_channels(pcm.channels as u8)
58        .map_err(|e| ConvertError::Encode(name(), format!("channel count: {e:?}")))?
59        .with_sample_rate(target_rate)
60        .map_err(|e| ConvertError::Encode(name(), format!("sample rate: {e:?}")))?
61        .with_brate(nearest_bitrate_preset(bitrate_kbps))
62        .map_err(|e| ConvertError::Encode(name(), format!("bitrate: {e:?}")))?
63        .with_quality(mp3lame_encoder::Quality::Best)
64        .map_err(|e| ConvertError::Encode(name(), format!("quality: {e:?}")))?
65        .build()
66        .map_err(|e| ConvertError::Encode(name(), format!("{e:?}")))?;
67
68    // `InterleavedPcm<f32>` is fed our normalized `[-1.0, 1.0]` samples
69    // as-is, on the assumption that LAME's `ieee_float` input path expects
70    // that range (matching every other float-PCM API this app already
71    // touches) rather than a 16-bit-full-scale float — unverified without
72    // compiling and actually listening to the output, since the crate's own
73    // docs don't state the convention explicitly. Worth a listening check
74    // before shipping: a scale mismatch here would produce either near-silent
75    // or heavily clipped MP3s, not a build failure.
76    let frames = resampled.len() / pcm.channels.max(1);
77    let mut out = Vec::with_capacity(mp3lame_encoder::max_required_buffer_size(frames));
78
79    // SAFETY: `encode`/`flush` only ever write into the spare capacity they
80    // are given and report back exactly how many bytes they initialized;
81    // `set_len` below only ever grows the vec by that reported count, never
82    // past what was just written — the same contract the crate's own
83    // documented example relies on.
84    let written = encoder
85        .encode(InterleavedPcm(&resampled), spare_capacity(&mut out))
86        .map_err(|e| ConvertError::Encode(name(), format!("{e:?}")))?;
87    unsafe {
88        out.set_len(out.len() + written);
89    }
90
91    let flushed = encoder
92        .flush::<FlushNoGap>(spare_capacity(&mut out))
93        .map_err(|e| ConvertError::Encode(name(), format!("flush: {e:?}")))?;
94    unsafe {
95        out.set_len(out.len() + flushed);
96    }
97
98    std::fs::write(dest, &out).map_err(|e| ConvertError::Io(name(), e.to_string()))
99}
100
101/// `Vec::spare_capacity_mut`, reserving room for at least one more flush call
102/// first — `encode`/`flush` both refuse to write past the slice they are
103/// given rather than growing it themselves.
104fn spare_capacity(buf: &mut Vec<u8>) -> &mut [MaybeUninit<u8>] {
105    if buf.spare_capacity_mut().len() < 7200 {
106        // LAME's own documented minimum for a final flush call.
107        buf.reserve(7200);
108    }
109    buf.spare_capacity_mut()
110}

112/// The closest rate LAME actually supports. Prefers an integer-division
113/// match against the common hi-res multiples this app already detects
114/// (88.2k/176.4k -> 44.1k, 96k/192k -> 48k) before falling back to nearest by
115/// raw distance for anything else.
116fn nearest_supported_rate(rate: u32) -> u32 {
117    let mut r = rate;
118    while r > 48_000 && r.is_multiple_of(2) {
119        r /= 2;
120    }
121    if LAME_SUPPORTED_RATES.contains(&r) {
122        return r;
123    }
124    LAME_SUPPORTED_RATES
125        .iter()
126        .min_by_key(|&&s| (s as i64 - rate as i64).abs())
127        .copied()
128        .unwrap_or(44_100)
129}
130
131/// The [`Bitrate`] preset closest to `kbps` — LAME only accepts one of 16
132/// fixed steps, not an arbitrary integer.
133fn nearest_bitrate_preset(kbps: u32) -> Bitrate {
134    const PRESETS: &[(u32, Bitrate)] = &[
135        (8, Bitrate::Kbps8),
136        (16, Bitrate::Kbps16),
137        (24, Bitrate::Kbps24),
138        (32, Bitrate::Kbps32),
139        (40, Bitrate::Kbps40),
140        (48, Bitrate::Kbps48),
141        (64, Bitrate::Kbps64),
142        (80, Bitrate::Kbps80),
143        (96, Bitrate::Kbps96),
144        (112, Bitrate::Kbps112),
145        (128, Bitrate::Kbps128),
146        (160, Bitrate::Kbps160),
147        (192, Bitrate::Kbps192),
148        (224, Bitrate::Kbps224),
149        (256, Bitrate::Kbps256),
150        (320, Bitrate::Kbps320),
151    ];
152    PRESETS
153        .iter()
154        .min_by_key(|&&(k, _)| (k as i64 - kbps as i64).abs())
155        .map(|&(_, b)| b)
156        .unwrap_or(Bitrate::Kbps256)
157}

159#[cfg(test)]
160mod tests {
161    use super::*;
162
163    #[test]
164    fn nearest_supported_rate_prefers_integer_division_for_hi_res_multiples() {
165        assert_eq!(nearest_supported_rate(88_200), 44_100);
166        assert_eq!(nearest_supported_rate(96_000), 48_000);
167        assert_eq!(nearest_supported_rate(176_400), 44_100);
168        assert_eq!(nearest_supported_rate(192_000), 48_000);
169        assert_eq!(nearest_supported_rate(44_100), 44_100);
170    }
171
172    /// MP3 is lossy, so there is no bit-exact ground truth to check the way
173    /// there is for FLAC/WAV — this only verifies the encoder actually
174    /// produces a plausible MP3 bitstream (non-empty, starts with a frame
175    /// sync or an ID3 marker) for both a directly-supported rate and a
176    /// hi-res rate that must be resampled first.
177    #[test]
178    fn produces_a_non_empty_mp3_stream_at_a_hi_res_rate() {
179        let sample_rate = 96_000u32;
180        let channels = 2usize;
181        let frames = 8000;
182        let mut samples = Vec::with_capacity(frames * channels);
183        for t in 0..frames {
184            let phase = t as f32 / sample_rate as f32;
185            let s = (phase * 440.0 * std::f32::consts::TAU).sin() * 0.4;
186            samples.push(s);
187            samples.push(s);
188        }
189        let pcm = PcmAudio {
190            samples,
191            sample_rate,
192            channels,
193        };
194        let dir = tempfile::tempdir().expect("tempdir");
195        let dest = dir.path().join("tone.mp3");
196        encode(&pcm, &dest, 256).expect("encode");
197
198        let data = std::fs::read(&dest).expect("reopen");
199        assert!(!data.is_empty());
200        // A raw MPEG frame sync is 11 set bits: 0xFF followed by the top
201        // three bits of the next byte also set.
202        assert!(data[0] == 0xFF && (data[1] & 0xE0) == 0xE0);
203    }
204
205    #[test]
206    fn rejects_more_than_two_channels_without_panicking() {
207        let pcm = PcmAudio {
208            samples: vec![0.0; 300],
209            sample_rate: 44_100,
210            channels: 3,
211        };
212        let dir = tempfile::tempdir().expect("tempdir");
213        let dest = dir.path().join("tone.mp3");
214        assert!(encode(&pcm, &dest, 256).is_err());
215    }
