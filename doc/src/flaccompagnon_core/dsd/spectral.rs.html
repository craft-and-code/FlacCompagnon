1//! Calibrated spectral heuristics around DSD, in both directions.
2//!
3//! * [`pcm_source_check`] — a *DSD file* whose content came from PCM. Real DSD
4//!   blends smoothly into the sigma-delta noise shaping that rises above
5//!   ~25 kHz; PCM-sourced DSD instead shows a digital brick wall at the
6//!   source's Nyquist (22.05 or 24 kHz). Measured on synthetic ground truth (a
7//!   2nd-order delta-sigma modulator fed native-band vs 44.1k-band-limited
8//!   content): the drop across the boundary is ≈ 3 dB for native DSD and
9//!   ≈ 50 dB for PCM-sourced.
10//! * [`dsd_heritage_check`] — the mirror case: *hi-res PCM* (FLAC/WAV ≥ 96 kHz)
11//!   converted from a DSD master, which carries the sigma-delta noise ramp
12//!   into the PCM file.
13//!
14//! Both work from the averaged spectrum the analyzer already produced, so
15//! neither reads a file or knows anything about container layout.
16
17/// Result of the PCM-source check on decoded DSD content.
18#[derive(Debug, Clone, Copy)]
19pub struct PcmSourceCheck {
20    /// Boundary where the brick wall sits (22 050 or 24 000 Hz).
21    pub boundary_hz: f64,
22    /// Level drop across the boundary in dB.
23    pub drop_db: f32,
24}
25
26/// Minimum drop across a PCM-Nyquist boundary to call it a brick wall.
27/// Calibrated: native ≈ 3 dB, PCM-sourced ≈ 50 dB.
28pub const PCM_CLIFF_DB: f32 = 30.0;
29
30/// Minimum ultrasonic rise for [`dsd_heritage_check`]. Measured on a real
31/// DSD-sourced 192 kHz FLAC: valley ≈ −85 dB, 36–70 kHz ≈ −55 dB → +30 dB.
32const DSD_RAMP_DB: f32 = 15.0;
33
34/// Mean level over `[lo, hi]` Hz of an averaged dB spectrum, or `None` when
35/// the band falls outside the spectrum, spans `min_span` bins or fewer, or
36/// holds no finite value.
37///
38/// `min_span` is why this takes a parameter instead of being one shared
39/// constant: the brick-wall check reads narrow bands right next to each other
40/// and only needs them non-empty, while the ramp check averages wide bands
41/// and would be noise-driven if it accepted a two-bin one.
42fn mean_band(spectrum_db: &[f32], bin_hz: f64, lo: f64, hi: f64, min_span: usize) -> Option<f32> {
43    if bin_hz <= 0.0 || spectrum_db.is_empty() {
44        return None;
45    }
46    let a = (lo / bin_hz) as usize;
47    let b = ((hi / bin_hz) as usize).min(spectrum_db.len() - 1);
48    if b <= a + min_span {
49        return None;
50    }
51    let (sum, n) = spectrum_db[a..=b]
52        .iter()
53        .filter(|v| v.is_finite())
54        .fold((0.0f32, 0usize), |(s, n), v| (s + v, n + 1));
55    (n > 0).then(|| sum / n as f32)
56}

58/// Inspect the averaged spectrum (from the analyzer, computed on the decoded
59/// PCM at `decoded_rate`) for a digital brick wall at a PCM source's Nyquist.
60pub fn pcm_source_check(
61    spectrum_db: &[f32],
62    decoded_rate: u32,
63    fft_size: usize,
64) -> Option<PcmSourceCheck> {
65    if spectrum_db.len() < 16 || decoded_rate == 0 || fft_size == 0 {
66        return None;
67    }
68    let bin_hz = decoded_rate as f64 / fft_size as f64;
69
70    let mut best: Option<PcmSourceCheck> = None;
71    for boundary in [22_050.0f64, 24_000.0] {
72        let below = mean_band(spectrum_db, bin_hz, boundary - 2_000.0, boundary - 200.0, 0);
73        let above = mean_band(spectrum_db, bin_hz, boundary + 300.0, boundary + 2_000.0, 0);
74        let (Some(lo), Some(hi)) = (below, above) else {
75            continue;
76        };
77        // Require actual content below the boundary (not silence).
78        if lo <= -80.0 {
79            continue;
80        }
81        let drop = lo - hi;
82        if drop >= PCM_CLIFF_DB && best.is_none_or(|b| drop > b.drop_db) {
83            best = Some(PcmSourceCheck {
84                boundary_hz: boundary,
85                drop_db: drop,
86            });
87        }
88    }
89    best
90}

92/// Detect the sigma-delta heritage of a DSD master inside hi-res *PCM*
93/// (FLAC/WAV at ≥ 96 kHz): genuine PCM recordings decay monotonically into the
94/// ultrasonic range, while DSD-converted PCM shows a valley around 22–30 kHz
95/// followed by a strong **rising** noise ramp. Returns the rise in dB.
96pub fn dsd_heritage_check(spectrum_db: &[f32], sample_rate: u32, fft_size: usize) -> Option<f32> {
97    if sample_rate < 96_000 || spectrum_db.len() < 16 || fft_size == 0 {
98        return None;
99    }
100    let nyq = sample_rate as f64 / 2.0;
101    let bin_hz = sample_rate as f64 / fft_size as f64;
102
103    let valley = mean_band(spectrum_db, bin_hz, 22_000.0, 30_000.0, 3)?;
104    let ramp = mean_band(
105        spectrum_db,
106        bin_hz,
107        36_000.0,
108        (0.92 * nyq).min(75_000.0),
109        3,
110    )?;
111    let rise = ramp - valley;
112    (rise >= DSD_RAMP_DB && ramp > -75.0).then_some(rise)
113}
