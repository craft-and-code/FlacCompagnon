1//! Exact DSF / DFF container parsing.
2//!
3//! DSF (Sony) is little-endian with a fixed header layout; DFF (Philips
4//! DSDIFF) is a big-endian IFF chunk tree. Both are parsed directly from the
5//! first bytes of the file — this authenticates the *container* and says
6//! nothing about whether its content is genuine DSD (that's [`super::spectral`]).
7//!
8//! Every read is bounds-checked against the buffer actually read from disk:
9//! these headers come from files the app exists to be suspicious of, so a
10//! truncated or hostile header must produce an error, never a panic.
11
12use std::io::Read;
13use std::path::Path;
14
15use serde::{Deserialize, Serialize};
16
17use crate::AnalysisError;

19/// Base DSD64 bit rate (64 × 44.1 kHz).
20pub const DSD64_RATE: u32 = 2_822_400;
21
22/// How much of the file to read looking for the header. Both formats put
23/// everything this parser needs near the start; DFF's chunk walk stops as soon
24/// as it has the rate and channel count.
25const HEADER_SCAN_BYTES: usize = 16384;
26
27/// Exact information read from a DSF/DFF header.
28#[derive(Debug, Clone, Serialize, Deserialize)]
29pub struct DsdInfo {
30    /// Container: "DSF" or "DFF".
31    pub container: &'static str,
32    /// 1-bit sample rate (e.g. 2 822 400 for DSD64).
33    pub sample_rate: u32,
34    /// Channel count declared by the header.
35    pub channels: usize,
36    /// Total 1-bit samples per channel, when the header declares it.
37    pub sample_count: Option<u64>,
38    /// DSD speed grade: 64, 128, 256… (sample_rate / 44100 rounded).
39    pub multiple: u32,
40    /// `true` for DST-compressed DFF (content analysis unavailable).
41    pub dst_compressed: bool,
42}
43
44impl DsdInfo {
45    /// Display label, e.g. "DSD64".
46    pub fn label(&self) -> String {
47        format!("DSD{}", self.multiple)
48    }
49
50    /// Track length in seconds, derived from the declared sample count and
51    /// rate; `0.0` when the header didn't declare a sample count.
52    pub fn duration_secs(&self) -> f64 {
53        self.sample_count
54            .map(|n| n as f64 / self.sample_rate as f64)
55            .unwrap_or(0.0)
56    }
57}
58
59/// `N` bytes at `at`, or `None` if they aren't all there.
60///
61/// The callers below walk offsets derived from the file's own declared chunk
62/// sizes — exactly the input that cannot be trusted to stay inside the buffer —
63/// so every read has to answer `None` rather than panic. Note `checked_add`
64/// and not `at + N`: such an offset can be enormous, and `at + N` would
65/// overflow (a panic in debug) *before* `get` ever got to answer.
66fn rd<const N: usize>(b: &[u8], at: usize) -> Option<[u8; N]> {
67    b.get(at..at.checked_add(N)?)?.try_into().ok()
68}
69
70// Fixed-width integer reads, all built on `rd` so none of them can be the one
71// that forgets a bounds check.
72
73fn rd_u32_le(b: &[u8], at: usize) -> Option<u32> {
74    rd(b, at).map(u32::from_le_bytes)
75}
76fn rd_u64_le(b: &[u8], at: usize) -> Option<u64> {
77    rd(b, at).map(u64::from_le_bytes)
78}
79fn rd_u32_be(b: &[u8], at: usize) -> Option<u32> {
80    rd(b, at).map(u32::from_be_bytes)
81}
82fn rd_u64_be(b: &[u8], at: usize) -> Option<u64> {
83    rd(b, at).map(u64::from_be_bytes)
84}
85fn rd_u16_be(b: &[u8], at: usize) -> Option<u16> {
86    rd(b, at).map(u16::from_be_bytes)
87}
88
89fn truncated(what: &str) -> AnalysisError {
90    AnalysisError::Decode(format!("truncated {what} header"))
91}

93/// Parse a DSF or DFF header (first bytes of the file decide which).
94pub fn parse(path: &Path) -> Result<DsdInfo, AnalysisError> {
95    let mut f = std::fs::File::open(path)?;
96    let mut head = vec![0u8; HEADER_SCAN_BYTES];
97    let n = f.read(&mut head)?;
98    head.truncate(n);
99    if head.len() >= 4 && &head[0..4] == b"DSD " {
100        parse_dsf(&head)
101    } else if head.len() >= 16 && &head[0..4] == b"FRM8" && &head[12..16] == b"DSD " {
102        parse_dff(&head)
103    } else {
104        Err(AnalysisError::Decode("not a DSF/DFF file".into()))
105    }
106}
107
108/// DSF (Sony): little-endian. Layout: "DSD " chunk (28 bytes), then "fmt "
109/// chunk (52 bytes) holding version, format id, channel type, channel count,
110/// sampling frequency, bits per sample, sample count, block size.
111fn parse_dsf(head: &[u8]) -> Result<DsdInfo, AnalysisError> {
112    const FMT: usize = 28; // "fmt " chunk starts right after the 28-byte DSD chunk
113    const FMT_LEN: usize = 52; // the spec fixes the fmt chunk's size
114
115    // The *whole* fmt chunk has to be there. Checking only the fields we read
116    // would accept a file that stops at byte 72, since the last of them
117    // (sample_count) ends exactly there — but such a file is truncated, and
118    // the block size that follows is part of what makes it a valid DSF.
119    if head.len() < FMT + FMT_LEN {
120        return Err(truncated("DSF"));
121    }
122    if head.get(FMT..FMT + 4) != Some(b"fmt ") {
123        return Err(AnalysisError::Decode("DSF: missing fmt chunk".into()));
124    }
125    // The per-field reads below are still bounds-checked rather than trusting
126    // the length test above: one upfront check that every later offset
127    // silently depends on is exactly the pattern that breaks when a field is
128    // added.
129    let channels = rd_u32_le(head, FMT + 24).ok_or_else(|| truncated("DSF"))? as usize;
130    let sample_rate = rd_u32_le(head, FMT + 28).ok_or_else(|| truncated("DSF"))?;
131    let bits = rd_u32_le(head, FMT + 32).ok_or_else(|| truncated("DSF"))?;
132    let sample_count = rd_u64_le(head, FMT + 36).ok_or_else(|| truncated("DSF"))?;
133    if bits != 1 && bits != 8 {
134        return Err(AnalysisError::Decode(format!(
135            "DSF: unexpected bits per sample {bits}"
136        )));
137    }
138    validate(sample_rate, channels)?;
139    Ok(DsdInfo {
140        container: "DSF",
141        sample_rate,
142        channels,
143        sample_count: Some(sample_count),
144        multiple: multiple_of(sample_rate),
145        dst_compressed: false,
146    })
147}

149/// DFF (Philips DSDIFF): big-endian IFF. "FRM8" + size + "DSD ", then chunks;
150/// "PROP" (type "SND ") contains "FS  " (rate), "CHNL" (count), "CMPR" (codec).
151///
152/// Chunk sizes come from the file, so every step forward is a checked add: a
153/// header declaring a huge or overflowing size must end the walk, not wrap
154/// around into a re-scan of the same bytes.
155fn parse_dff(head: &[u8]) -> Result<DsdInfo, AnalysisError> {
156    let mut sample_rate = 0u32;
157    let mut channels = 0usize;
158    let mut dst = false;
159    // Annotated because `saturating_add` is called on it below: a bare
160    // `12 + 4` stays an ambiguous `{integer}` until something pins it, and a
161    // method call is not something that can.
162    let mut pos: usize = 12 + 4; // after FRM8 header + form type
163
164    // `saturating_add` in the condition, not just at the assignment below:
165    // `pos` is derived from a size the file declared, so it can legitimately
166    // reach `usize::MAX` after the saturating step — and `pos + 12` would then
167    // overflow here, panicking in debug before the loop ever got a chance to
168    // end. Same reasoning for `p` in the inner walk.
169    while pos.saturating_add(12) <= head.len() {
170        let id = &head[pos..pos + 4];
171        let size = rd_u64_be(head, pos + 4).ok_or_else(|| truncated("DFF"))? as usize;
172        let body = pos + 12;
173
174        if id == b"PROP" && head.get(body..body + 4) == Some(b"SND ") {
175            // Walk the local chunks inside PROP.
176            let mut p = body + 4;
177            let end = body.saturating_add(size).min(head.len());
178            while p.saturating_add(12) <= end {
179                let lid = &head[p..p + 4];
180                let lsize = rd_u64_be(head, p + 4).ok_or_else(|| truncated("DFF"))? as usize;
181                let lbody = p + 12;
182                match lid {
183                    b"FS  " => sample_rate = rd_u32_be(head, lbody).unwrap_or(sample_rate),
184                    b"CHNL" => {
185                        channels = rd_u16_be(head, lbody).map_or(channels, |c| c as usize)
186                    }
187                    b"CMPR" => dst = head.get(lbody..lbody + 4) == Some(b"DST "),
188                    _ => {}
189                }
190                // IFF chunks are 2-byte aligned. A zero-size chunk would leave
191                // `p` unchanged and spin forever, so the step is forced to be
192                // strictly positive (12 bytes of header at minimum).
193                let next = lbody.saturating_add(lsize).saturating_add(lsize & 1);
194                if next <= p {
195                    break;
196                }
197                p = next;
198            }
199        }
200
201        let next = body.saturating_add(size).saturating_add(size & 1);
202        if next <= pos {
203            break; // same guard at the outer level
204        }
205        pos = next;
206        if sample_rate != 0 && channels != 0 {
207            break;
208        }
209    }
210
211    validate(sample_rate, channels)?;
212    Ok(DsdInfo {
213        container: "DFF",
214        sample_rate,
215        channels,
216        sample_count: None, // DFF stores a data-chunk byte size, not a count
217        multiple: multiple_of(sample_rate),
218        dst_compressed: dst,
219    })
220}
221
222fn validate(sample_rate: u32, channels: usize) -> Result<(), AnalysisError> {
223    // Accept DSD64..DSD512 at 44.1k- and 48k-based grids.
224    let ok_rate = (2_000_000..=25_000_000).contains(&sample_rate);
225    if !ok_rate || channels == 0 || channels > 8 {
226        return Err(AnalysisError::Decode(format!(
227            "implausible DSD parameters: {sample_rate} Hz, {channels} ch"
228        )));
229    }
230    Ok(())
231}
232
233fn multiple_of(sample_rate: u32) -> u32 {
234    ((sample_rate as f64 / 44_100.0) as u32).max(1)
235}

241    /// Build a minimal valid DSF header (the exact layout our parser reads).
242    fn dsf_header(rate: u32, channels: u32, samples: u64) -> Vec<u8> {
243        let mut v = Vec::new();
244        v.extend(b"DSD ");
245        v.extend(28u64.to_le_bytes());
246        v.extend(0u64.to_le_bytes()); // total size (unused by parser)
247        v.extend(0u64.to_le_bytes()); // metadata ptr
248        v.extend(b"fmt ");
249        v.extend(52u64.to_le_bytes());
250        v.extend(1u32.to_le_bytes()); // version
251        v.extend(0u32.to_le_bytes()); // format id
252        v.extend(2u32.to_le_bytes()); // channel type
253        v.extend(channels.to_le_bytes());
254        v.extend(rate.to_le_bytes());
255        v.extend(1u32.to_le_bytes()); // bits per sample
256        v.extend(samples.to_le_bytes());
257        v.extend(4096u32.to_le_bytes());
258        v.extend(0u32.to_le_bytes());
259        v
260    }

262    #[test]
263    fn parses_dsf_header() {
264        let h = dsf_header(DSD64_RATE, 2, 2_822_400 * 60);
265        let info = parse_dsf(&h).expect("valid");
266        assert_eq!(info.container, "DSF");
267        assert_eq!(info.sample_rate, DSD64_RATE);
268        assert_eq!(info.channels, 2);
269        assert_eq!(info.multiple, 64);
270        assert_eq!(info.label(), "DSD64");
271        assert!((info.duration_secs() - 60.0).abs() < 1e-9);
272    }
