1//! Exact DSF / DFF container parsing.
2//!
3//! DSF (Sony) is little-endian with a fixed header layout; DFF (Philips
4//! DSDIFF) is a big-endian IFF chunk tree. Both are parsed directly from the
5//! first bytes of the file — this authenticates the *container* and says
6//! nothing about whether its content is genuine DSD (that's [`super::spectral`]).
7//!
8//! DSF fields are checked against its fixed header; DFF offsets are bounded
9//! by each chunk, its parent and the actual file size. A truncated or hostile
10//! header must produce an error, never a panic.
11
12use std::io::{Read, Seek, SeekFrom};
13use std::path::Path;

19/// Base DSD64 bit rate (64 × 44.1 kHz).
20pub const DSD64_RATE: u32 = 2_822_400;
21
22/// Exact information read from a DSF/DFF header.
23#[derive(Debug, Clone, Serialize, Deserialize)]
24pub struct DsdInfo {
25    /// Container: "DSF" or "DFF".
26    pub container: &'static str,
27    /// 1-bit sample rate (e.g. 2 822 400 for DSD64).
28    pub sample_rate: u32,
29    /// Channel count declared by the header.
30    pub channels: usize,
31    /// Total 1-bit samples per channel, when the header declares it.
32    pub sample_count: Option<u64>,
33    /// DSD speed grade: 64, 128, 256… (sample_rate / 44100 rounded).
34    pub multiple: u32,
35    /// `true` for DST-compressed DFF (content analysis unavailable).
36    pub dst_compressed: bool,
37}
38
39impl DsdInfo {
40    /// Display label, e.g. "DSD64".
41    pub fn label(&self) -> String {
42        format!("DSD{}", self.multiple)
43    }
44
45    /// Track length in seconds, derived from the declared sample count and
46    /// rate; `0.0` when the header didn't declare a sample count.
47    pub fn duration_secs(&self) -> f64 {
48        self.sample_count
49            .map(|n| n as f64 / self.sample_rate as f64)
50            .unwrap_or(0.0)
51    }
52}
53
54/// `N` bytes at `at`, or `None` if they aren't all there.
55///
56/// The callers below walk offsets derived from the file's own declared chunk
57/// sizes — exactly the input that cannot be trusted to stay inside the buffer —
58/// so every read has to answer `None` rather than panic. Note `checked_add`
59/// and not `at + N`: such an offset can be enormous, and `at + N` would
60/// overflow (a panic in debug) *before* `get` ever got to answer.
61fn rd<const N: usize>(b: &[u8], at: usize) -> Option<[u8; N]> {
62    b.get(at..at.checked_add(N)?)?.try_into().ok()
63}
64
65// Fixed-width integer reads, all built on `rd` so none of them can be the one
66// that forgets a bounds check.
67
68fn rd_u32_le(b: &[u8], at: usize) -> Option<u32> {
69    rd(b, at).map(u32::from_le_bytes)
70}
71fn rd_u64_le(b: &[u8], at: usize) -> Option<u64> {
72    rd(b, at).map(u64::from_le_bytes)
73}
74fn rd_u64_be(b: &[u8], at: usize) -> Option<u64> {
75    rd(b, at).map(u64::from_be_bytes)
76}

78fn truncated(what: &str) -> AnalysisError {
79    AnalysisError::Decode(format!("truncated {what} header"))
80}

82/// Parse a DSF or DFF header (first bytes of the file decide which).
83pub fn parse(path: &Path) -> Result<DsdInfo, AnalysisError> {
84    let mut file = std::fs::File::open(path)?;
85    let magic = read_array::<4>(&mut file)?;
86    file.seek(SeekFrom::Start(0))?;
87    match &magic {
88        b"DSD " => parse_dsf(&read_array::<80>(&mut file)?),
89        b"FRM8" => {
90            let size = file.metadata()?.len();
91            parse_dff_reader(&mut file, size)
92        }
93        _ => Err(AnalysisError::Decode("not a DSF/DFF file".into())),
94    }
95}
96
97fn read_array<const N: usize>(reader: &mut impl Read) -> Result<[u8; N], AnalysisError> {
98    let mut bytes = [0; N];
99    reader.read_exact(&mut bytes)?;
100    Ok(bytes)
101}

103/// DSF (Sony): little-endian. Layout: "DSD " chunk (28 bytes), then "fmt "
104/// chunk (52 bytes) holding version, format id, channel type, channel count,
105/// sampling frequency, bits per sample, sample count, block size.
106fn parse_dsf(head: &[u8]) -> Result<DsdInfo, AnalysisError> {
107    const FMT: usize = 28; // "fmt " chunk starts right after the 28-byte DSD chunk
108    const FMT_LEN: usize = 52; // the spec fixes the fmt chunk's size
109
110    // The *whole* fmt chunk has to be there. Checking only the fields we read
111    // would accept a file that stops at byte 72, since the last of them
112    // (sample_count) ends exactly there — but such a file is truncated, and
113    // the block size that follows is part of what makes it a valid DSF.
114    if head.len() < FMT + FMT_LEN {
115        return Err(truncated("DSF"));
116    }
117    if head.get(FMT..FMT + 4) != Some(b"fmt ") {
118        return Err(AnalysisError::Decode("DSF: missing fmt chunk".into()));
119    }
120    if rd_u64_le(head, 4) != Some(28) || rd_u64_le(head, FMT + 4) != Some(FMT_LEN as u64) {
121        return Err(AnalysisError::Decode(
122            "DSF: invalid header chunk size".into(),
123        ));
124    }
125    // The per-field reads below are still bounds-checked rather than trusting
126    // the length test above: one upfront check that every later offset
127    // silently depends on is exactly the pattern that breaks when a field is
128    // added.
129    let channels = rd_u32_le(head, FMT + 24).ok_or_else(|| truncated("DSF"))? as usize;
130    let sample_rate = rd_u32_le(head, FMT + 28).ok_or_else(|| truncated("DSF"))?;
131    let bits = rd_u32_le(head, FMT + 32).ok_or_else(|| truncated("DSF"))?;
132    let sample_count = rd_u64_le(head, FMT + 36).ok_or_else(|| truncated("DSF"))?;
133    if bits != 1 && bits != 8 {
134        return Err(AnalysisError::Decode(format!(
135            "DSF: unexpected bits per sample {bits}"
136        )));
137    }
138    validate(sample_rate, channels)?;
139    Ok(DsdInfo {
140        container: "DSF",
141        sample_rate,
142        channels,
143        sample_count: Some(sample_count),
144        multiple: multiple_of(sample_rate),
145        dst_compressed: false,
146    })
147}

149struct DffChunk {
150    id: [u8; 4],
151    size: u64,
152    end: u64,
153    next: u64,
154}
155
156/// DSDIFF 1.5 §2.3: sizes exclude the 12-byte header and optional even pad.
157/// Bounds use u64 offsets so skipping a large extension never allocates it.
158fn dff_chunk(
159    reader: &mut (impl Read + Seek),
160    at: u64,
161    limit: u64,
162) -> Result<DffChunk, AnalysisError> {
163    let body = at
164        .checked_add(12)
165        .filter(|end| *end <= limit)
166        .ok_or_else(|| truncated("DFF chunk"))?;
167    reader.seek(SeekFrom::Start(at))?;
168    let header = read_array::<12>(reader)?;
169    let id = rd(&header, 0).ok_or_else(|| truncated("DFF chunk"))?;
170    let size = rd_u64_be(&header, 4).ok_or_else(|| truncated("DFF chunk"))?;
171    let end = body
172        .checked_add(size)
173        .ok_or_else(|| truncated("DFF chunk"))?;
174    let next = end
175        .checked_add(size & 1)
176        .filter(|end| *end <= limit)
177        .ok_or_else(|| truncated("DFF chunk"))?;
178    Ok(DffChunk {
179        id,
180        size,
181        end,
182        next,
183    })
184}

186/// Walk DFF headers without a fixed scan cutoff. Unrecognized extensions are
187/// skipped on disk, while known fields remain bounded by their own payloads.
188fn parse_dff_reader(
189    reader: &mut (impl Read + Seek),
190    file_size: u64,
191) -> Result<DsdInfo, AnalysisError> {
192    let header = read_array::<16>(reader)?;
193    if header.get(..4) != Some(b"FRM8") || header.get(12..16) != Some(b"DSD ") {
194        return Err(AnalysisError::Decode("not a DFF file".into()));
195    }
196    let form_end = rd_u64_be(&header, 4)
197        .and_then(|size| size.checked_add(12))
198        .filter(|end| (16..=file_size).contains(end))
199        .ok_or_else(|| truncated("DFF"))?;
200    let mut pos = 16;
201    while pos < form_end {
202        let chunk = dff_chunk(reader, pos, form_end)?;
203        if &chunk.id == b"PROP" && chunk.size >= 4 && read_array::<4>(reader)? == *b"SND " {
204            return parse_dff_properties(reader, chunk.end);
205        }
206        pos = chunk.next;
207    }
208    Err(AnalysisError::Decode(
209        "DFF: missing sound properties".into(),
210    ))
211}

213fn parse_dff_properties(
214    reader: &mut (impl Read + Seek),
215    end: u64,
216) -> Result<DsdInfo, AnalysisError> {
217    let mut sample_rate = None;
218    let mut channels = None;
219    let mut dst = None;
220    let mut pos = reader.stream_position()?;
221    while pos < end {
222        let chunk = dff_chunk(reader, pos, end)?;
223        // DSDIFF 1.5 §§3.2.1–3.2.3: mandatory properties are unique.
224        match &chunk.id {
225            b"FS  " if sample_rate.is_none() && chunk.size == 4 => {
226                sample_rate = Some(u32::from_be_bytes(read_array(reader)?));
227            }
228            b"CHNL" if channels.is_none() && chunk.size >= 2 => {
229                let count = u16::from_be_bytes(read_array(reader)?);
230                if chunk.size != 2 + u64::from(count) * 4 {
231                    return Err(truncated("DFF channel IDs"));
232                }
233                channels = Some(usize::from(count));
234            }
235            b"CMPR" if dst.is_none() && chunk.size >= 5 => {
236                let codec = read_array::<5>(reader)?;
237                let name_len = codec
238                    .get(4)
239                    .copied()
240                    .ok_or_else(|| truncated("DFF compression"))?;
241                if chunk.size != 5 + u64::from(name_len) {
242                    return Err(truncated("DFF compression name"));
243                }
244                dst = Some(match codec.get(..4) {
245                    Some(b"DSD ") => false,
246                    Some(b"DST ") => true,
247                    _ => return Err(AnalysisError::Decode("DFF: unsupported compression".into())),
248                });
249            }
250            b"FS  " | b"CHNL" | b"CMPR" => {
251                return Err(AnalysisError::Decode(
252                    "DFF: invalid or duplicate sound property".into(),
253                ));
254            }
255            _ => {}
256        }
257        pos = chunk.next;
258    }
259    let sample_rate = sample_rate.ok_or_else(|| truncated("DFF sample rate"))?;
260    let channels = channels.ok_or_else(|| truncated("DFF channels"))?;
261    let dst_compressed = dst.ok_or_else(|| truncated("DFF compression"))?;
262    validate(sample_rate, channels)?;
263    Ok(DsdInfo {
264        container: "DFF",
265        sample_rate,
266        channels,
267        sample_count: None,
268        multiple: multiple_of(sample_rate),
269        dst_compressed,
270    })
271}
272
273fn validate(sample_rate: u32, channels: usize) -> Result<(), AnalysisError> {
274    // Accept DSD64..DSD512 at 44.1k- and 48k-based grids.
275    let ok_rate = (2_000_000..=25_000_000).contains(&sample_rate);
276    if !ok_rate || channels == 0 || channels > 8 {
277        return Err(AnalysisError::Decode(format!(
278            "implausible DSD parameters: {sample_rate} Hz, {channels} ch"
279        )));
280    }
281    Ok(())
282}
283
284fn multiple_of(sample_rate: u32) -> u32 {
285    ((sample_rate as f64 / 44_100.0) as u32).max(1)
286}
