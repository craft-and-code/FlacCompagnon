1//! MPEG-1 Layer III constants: the polyphase analysis window, the scalefactor
2//! band layouts, and the aliasing-reduction coefficients.
3//!
4//! All three are **from the standard**, ISO/IEC 11172-3. The analysis window
5//! is Table C.1 (the 512 coefficients `Ci` of the polyphase filterbank); the
6//! band layouts are the Layer III scalefactor band tables; the butterfly
7//! coefficients are the eight `ci` of the alias-reduction stage. They are
8//! reproduced as published — nobody's invention, and no authorship attaches
9//! to them.
10//!
11//! # Why MP3 needs all this and AAC needed almost none of it
12//!
13//! AAC transforms the signal once. MP3 transforms it three times over: a
14//! 32-band polyphase filter, an 18-point MDCT inside each of those bands, then
15//! a butterfly stage that partially undoes the aliasing the first two
16//! introduced. The quantization lattice this detector looks for only exists at
17//! the *end* of that chain, so every stage has to be reproduced exactly —
18//! including the alias reduction, which is not optional cosmetics but part of
19//! what defines where the coefficients land.
20//!
21//! This file is one long table by design; see the module ceiling exemption in
22//! CLAUDE.md.
23

215/// Precomputed alias-reduction factors `cs[i]` and `ca[i]`.
216///
217/// `cs = 1/sqrt(1 + ci^2)`, `ca = ci/sqrt(1 + ci^2)` — derived rather than
218/// tabulated, so the two can never drift apart from the `ci` above.
219pub fn butterfly_coefficients() -> ([f64; 8], [f64; 8]) {
220    let mut cs = [0.0; 8];
221    let mut ca = [0.0; 8];
222    for (i, &c) in CI.iter().enumerate() {
223        let d = (1.0 + c * c).sqrt();
224        cs[i] = 1.0 / d;
225        ca[i] = c / d;
226    }
227    (cs, ca)
228}

230/// The four window shapes Layer III applies to a subband's 36 samples.
231///
232/// Same idea as AAC's, at a twelfth of the size and one stage further down
233/// the chain: these are applied inside each of the 32 subbands, not to the
234/// PCM. `start` and `stop` are the transition shapes bridging long and short
235/// blocks; both must be tested, because a frame the encoder coded as a
236/// transition carries its lattice under that shape and no other.
237pub struct Windows {
238    /// Plain long window, 36 samples.
239    pub long: Vec<f64>,
240    /// Long rising edge, short falling edge.
241    pub start: Vec<f64>,
242    /// Short rising edge, long falling edge.
243    pub stop: Vec<f64>,
244    /// Applied to each of the three short transforms, 12 samples.
245    pub short: Vec<f64>,
246}
247
248impl Windows {
249    /// Build the four shapes. Cheap; built once per worker.
250    pub fn new() -> Self {
251        let sine = |len: usize, from: usize, count: usize| -> Vec<f64> {
252            (from..from + count)
253                .map(|n| (std::f64::consts::PI * (n as f64 + 0.5) / len as f64).sin())
254                .collect()
255        };
256        const FILL: usize = 6;
257        let half_long = N_LONG / 2;
258        let half_short = N_SHORT / 2;
259
260        let mut start = sine(N_LONG, 0, half_long);
261        start.extend(std::iter::repeat_n(1.0, FILL));
262        start.extend(sine(N_SHORT, FILL, half_short));
263        start.extend(std::iter::repeat_n(0.0, FILL));
264
265        let mut stop = vec![0.0; FILL];
266        stop.extend(sine(N_SHORT, 0, half_short));
267        stop.extend(std::iter::repeat_n(1.0, FILL));
268        stop.extend(sine(N_LONG, half_long, half_long));
269
270        Self {
271            long: sine(N_LONG, 0, N_LONG),
272            start,
273            stop,
274            short: sine(N_SHORT, 0, N_SHORT),
275        }
276    }
277}
278
279
280impl Default for Windows {
281    fn default() -> Self {
282        Self::new()
283    }
284}

286/// Long-transform scalefactor bands for `rate_khz`.
287///
288/// Unlike AAC's, these do **not** tile the whole spectrum: the topmost band
289/// stops at 418 of 576 coefficients at 44.1 kHz (384 at 48, 550 at 32). Above
290/// that is the region a Layer III encoder discards outright, so there is no
291/// lattice up there to find and the standard does not assign it a scalefactor.
292pub fn long_bands(rate_khz: u32) -> Option<Vec<Band>> {
293    let (starts, end): (&[usize], usize) = match rate_khz {
294        32 => (&LONG_32, LONG_END_32),
295        44 => (&LONG_44, LONG_END_44),
296        48 => (&LONG_48, LONG_END_48),
297        _ => return None,
298    };
299    Some(bands(starts, end))
300}
301
302/// Short-transform scalefactor bands for `rate_khz`.
303pub fn short_bands(rate_khz: u32) -> Option<Vec<Band>> {
304    let (starts, end): (&[usize], usize) = match rate_khz {
305        32 => (&SHORT_32, SHORT_END_32),
306        44 => (&SHORT_44, SHORT_END_44),
307        48 => (&SHORT_48, SHORT_END_48),
308        _ => return None,
309    };
310    Some(bands(starts, end))
311}
312
313fn bands(starts: &[usize], end: usize) -> Vec<Band> {
314    starts
315        .iter()
316        .enumerate()
317        .map(|(i, &start)| Band {
318            start,
319            end: starts.get(i + 1).copied().unwrap_or(end),
320        })
321        .collect()
322}
