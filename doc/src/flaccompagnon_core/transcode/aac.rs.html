1//! The MPEG-AAC detector: does this signal sit on an AAC quantization
2//! lattice?
3//!
4//! Implements §1.4 of the paper. The shape of the search is a stack of loops,
5//! and every level exists because the detector does not know something the
6//! encoder did:
7//!
8//! | Parameter to infer | Search strategy |
9//! |---|---|
10//! | where the encoder's frames began | try all 1024 sample alignments |
11//! | which window shape each frame used | try all four, keep the best |
12//! | what scalefactor each band carried | try a grid of them |
13//! | which stereo mode was coded | try L/R and M/S, keep the best |
14//!
15//! The best shape's counts are pooled across frames; the final score is
16//! maximized over alignments and the selected stereo signals. This follows
17//! the MATLAB aggregation, whose relation to the paper's pseudocode still
18//! needs author confirmation. A maximum does not rule out false positives.
19//!
20//! This file is one algorithm read top to bottom — the sweep, the per-frame
21//! measurement it drives, and the buffers both need. Splitting it would put
22//! the loop in one file and the thing it loops over in another, which is a
23//! split by layer, not by responsibility.
24

35/// Search settings.
36///
37/// The defaults are the paper's own: `δ ∈ [0.3, 0.7]` from §2, and 64 frames
38/// by 64 scalefactors, the most thorough setting it evaluates.
39#[derive(Debug, Clone, Copy)]
40pub struct AacParams {
41    /// `Nf` — how many high-energy frames to measure.
42    pub frames: usize,
43    /// `Nsf` — how many scalefactors to try per band.
44    pub scalefactors: usize,
45    /// Lower relative scalefactor bound `δmin`.
46    pub delta_min: f64,
47    /// Upper relative scalefactor bound `δmax`.
48    pub delta_max: f64,
49    /// `λ` — the likelihood above which the verdict is "transcoded".
50    ///
51    /// This pairs with `frames`/`scalefactors` and cannot be moved
52    /// independently of them. These thresholds were calibrated empirically
53    /// for each complete configuration. The author's
54    /// reference lists 0.031 for 8×8, 0.019 for 16×16, 0.0145 for 32×32 and
55    /// 0.0125 for 64×64.
56    pub significance: f64,
57    /// How many sample alignments to try, from 0 upwards.
58    ///
59    /// Always [`LONG_LEN`] in production — the encoder's frame boundary can
60    /// sit anywhere in a transform's span, and testing fewer alignments means
61    /// simply missing the files whose boundary falls outside the range. It is
62    /// a field only so the unit tests can exercise the plumbing without
63    /// running the full sweep, which takes minutes by design.
64    pub alignments: usize,
65}
66
67impl Default for AacParams {
68    fn default() -> Self {
69        Self {
70            frames: 64,
71            scalefactors: 64,
72            delta_min: 0.3,
73            delta_max: 0.7,
74            significance: 0.0125,
75            alignments: LONG_LEN,
76        }
77    }
78}

80/// Run the detector over one file's decoded channels.
81///
82/// `channels` holds one slice per channel. Mono is analyzed directly; stereo
83/// is analyzed as L/R and as M/S and the higher score wins, because an
84/// encoder chooses between those two modes per frame and nothing in the
85/// decoded signal records which it picked. More than two channels: the first
86/// two are used, which is where a stereo master's content lives.
87///
88/// `cancelled` is polled once per alignment so a long sweep can be stopped.
89/// Returns `None` when it fires, when the file cannot supply all requested frames,
90/// or when its sample rate has no tabulated bands.
91pub fn detect(
92    channels: &[Vec<f64>],
93    sample_rate: u32,
94    params: &AacParams,
95    cancelled: &(dyn Fn() -> bool + Sync),
96) -> Option<TranscodeEvidence> {
97    let rate_khz = supported_rate_khz(sample_rate)?;
98    let bands = BandSet::new(rate_khz, params)?;
99    let first = channels.first()?;
100
101    let mut best = TranscodeEvidence {
102        likelihood: 0.0,
103        offset: 0,
104        detected: false,
105    };
106
107    if channels.len() == 1 {
108        let picked = select_frames(first, LONG_LEN, 2, params.frames)?;
109        best = sweep(first, &picked, &bands, params, cancelled)?;
110    } else {
111        let left = first.as_slice();
112        let right = channels[1].as_slice();
113        // AAC's stereo matrix is unscaled: M = L + R, S = L − R.
114        let (mid, side) = mid_side(left, right, 1.0);
115
116        // Frames are chosen on the mid channel and then reused to compare the
117        // four signals, so all four are judged over the same stretches of
118        // music rather than each over its own loudest moments.
119        let picked = select_frames(&mid, LONG_LEN, 2, params.frames)?;
120        let mean_of = |x: &[f64]| mean_energy_db(&frame_energies_db(x, LONG_LEN), &picked);
121
122        // Only the louder of each pair is swept: the quieter one carries the
123        // same lattice more faintly, so it costs a full sweep to learn
124        // nothing new.
125        let lr: &[f64] = if mean_of(left) >= mean_of(right) {
126            left
127        } else {
128            right
129        };
130        let ms: &[f64] = if mean_of(&mid) >= mean_of(&side) {
131            &mid
132        } else {
133            &side
134        };
135
136        for signal in [lr, ms] {
137            let evidence = sweep(signal, &picked, &bands, params, cancelled)?;
138            if evidence.likelihood > best.likelihood {
139                best = evidence;
140            }
141        }
142    }
143
144    best.detected = best.likelihood > params.significance;
145    Some(best)
146}

148/// Band layouts and everything derived from them, built once per file.
149struct BandSet {
150    long: Vec<Band>,
151    short: Vec<Band>,
152    tau_long: Vec<f64>,
153    tau_short: Vec<f64>,
154    deltas: Vec<f64>,
155}
156
157impl BandSet {
158    fn new(rate_khz: u32, params: &AacParams) -> Option<Self> {
159        let long = long_bands(rate_khz)?;
160        let short = short_bands(rate_khz)?;
161        let widths_long: Vec<usize> = long.iter().map(|b| b.width()).collect();
162        // A short band is measured across all eight short transforms at once,
163        // so its statistic averages over eight times as many coefficients and
164        // its threshold has to be computed for that width, not the band's.
165        let widths_short: Vec<usize> = short.iter().map(|b| b.width() * SHORT_COUNT).collect();
166        Some(Self {
167            tau_long: subband_thresholds(&widths_long, TARGET_PROBABILITY),
168            tau_short: subband_thresholds(&widths_short, TARGET_PROBABILITY),
169            long,
170            short,
171            deltas: scalefactor_grid(params.delta_min, params.delta_max, params.scalefactors),
172        })
173    }
174}

176/// Running best across the four window shapes of one frame.
177///
178/// A plain struct rather than a closure over local variables: the comparison
179/// has to read and write the same state the caller keeps afterwards, and a
180/// closure capturing it mutably would lock the caller out of it.
181struct BestShape {
182    hits: usize,
183    trials: usize,
184    ratio: f64,
185}
186
187impl BestShape {
188    fn new() -> Self {
189        // −1 so the first offer always wins, even one that scored zero.
190        Self {
191            hits: 0,
192            trials: 0,
193            ratio: -1.0,
194        }
195    }
196
197    fn offer(&mut self, hits: usize, trials: usize) {
198        let ratio = if trials == 0 {
199            0.0
200        } else {
201            hits as f64 / trials as f64
202        };
203        if ratio > self.ratio {
204            self.ratio = ratio;
205            self.hits = hits;
206            self.trials = trials;
207        }
208    }
209}

211/// Count on-grid trials across a spectrum's bands.
212///
213/// A free function taking `sub` separately rather than a method: the caller
214/// holds both the scratch quantizer and the spectrum as fields of one struct,
215/// and a method would borrow the whole struct twice over.
216///
217/// The topmost band is skipped, as in the reference. It is the wide remainder
218/// up to Nyquist, and in a transcoded file it is precisely the region the
219/// encoder threw away — near-silent, no lattice left in it, and its emptiness
220/// would dilute the ratio identically for every file.
221///
222/// `trials` counts every scalefactor offered, including those in bands too
223/// silent to test. That is deliberate and matches the reference: a file whose
224/// bands are mostly empty should score *low*, and crediting it by shrinking
225/// the denominator would do the opposite.
226fn count_bands(
227    sub: &mut PreparedSubband,
228    spectrum: &[f64],
229    bands: &[Band],
230    taus: &[f64],
231    deltas: &[f64],
232) -> (usize, usize) {
233    let mut hits = 0;
234    let mut trials = 0;
235    let n = bands.len().saturating_sub(1);
236    for (b, &tau) in bands.iter().zip(taus).take(n) {
237        let Some(slice) = spectrum.get(b.start..b.end) else {
238            continue;
239        };
240        trials += deltas.len();
241        if sub.prepare(slice, SF_BIAS) {
242            hits += sub.on_grid_count(deltas, SF_BIAS, tau);
243        }
244    }
245    (hits, trials)
246}

248/// Everything one thread needs, allocated once and reused for every alignment
249/// it handles.
250///
251/// A sweep runs on the order of a million transforms; allocating per
252/// transform would cost more than the transforms do. The long and short
253/// windowing buffers are separate fields rather than one shared buffer so
254/// that borrowing one never conflicts with borrowing the other.
255struct Worker {
256    mdct_long: Mdct,
257    mdct_short: Mdct,
258    windowed_long: Vec<f64>,
259    windowed_short: Vec<f64>,
260    spectrum: Vec<f64>,
261    short_spectra: Vec<f64>,
262    gather: Vec<f64>,
263    sub: PreparedSubband,
264}
265
266impl Worker {
267    fn new() -> Self {
268        Self {
269            mdct_long: Mdct::new(LONG_LEN),
270            mdct_short: Mdct::new(SHORT_LEN),
271            windowed_long: vec![0.0; 2 * LONG_LEN],
272            windowed_short: vec![0.0; 2 * SHORT_LEN],
273            spectrum: vec![0.0; LONG_LEN],
274            short_spectra: vec![0.0; SHORT_COUNT * SHORT_LEN],
275            gather: vec![0.0; SHORT_COUNT * SHORT_LEN],
276            sub: PreparedSubband::new(),
277        }
278    }
279
280    /// Measure one 2048-sample frame under all four window shapes and return
281    /// the best as a `(hits, trials)` pair.
282    fn measure_frame(&mut self, win: &[f64], bands: &BandSet, w: &Windows) -> (usize, usize) {
283        let mut best = BestShape::new();
284
285        for shape in [&w.long, &w.start, &w.stop] {
286            for (dst, (s, sh)) in self
287                .windowed_long
288                .iter_mut()
289                .zip(win.iter().zip(shape.iter()))
290            {
291                *dst = s * sh;
292            }
293            if !self
294                .mdct_long
295                .transform(&self.windowed_long, &mut self.spectrum)
296            {
297                continue;
298            }
299            let (h, t) = count_bands(
300                &mut self.sub,
301                &self.spectrum,
302                &bands.long,
303                &bands.tau_long,
304                &bands.deltas,
305            );
306            best.offer(h, t);
307        }
308
309        if self.transform_short_block(win, w) {
310            let (h, t) = self.count_short_bands(bands);
311            best.offer(h, t);
312        }
313
314        (best.hits, best.trials)
315    }
316
317    /// The eight short transforms of one frame, written into
318    /// [`Worker::short_spectra`]. Returns `false` if the frame is too short
319    /// to hold them, which leaves the short shape simply unmeasured.
320    fn transform_short_block(&mut self, win: &[f64], w: &Windows) -> bool {
321        for k in 0..SHORT_COUNT {
322            let from = k * SHORT_LEN + SHORT_OFFSET;
323            let Some(src) = win.get(from..from + 2 * SHORT_LEN) else {
324                return false;
325            };
326            for (dst, (s, sh)) in self
327                .windowed_short
328                .iter_mut()
329                .zip(src.iter().zip(w.short.iter()))
330            {
331                *dst = s * sh;
332            }
333            let out = &mut self.short_spectra[k * SHORT_LEN..(k + 1) * SHORT_LEN];
334            if !self.mdct_short.transform(&self.windowed_short, out) {
335                return false;
336            }
337        }
338        true
339    }
340
341    /// One short band spans the same frequency range in all eight transforms.
342    /// The reference measures those eight stretches as a single population
343    /// rather than as eight small ones — which is why they are gathered into
344    /// a contiguous buffer here before being handed to the quantizer.
345    fn count_short_bands(&mut self, bands: &BandSet) -> (usize, usize) {
346        let mut hits = 0;
347        let mut trials = 0;
348        let n = bands.short.len().saturating_sub(1);
349
350        for (b, &tau) in bands.short.iter().zip(&bands.tau_short).take(n) {
351            let width = b.width();
352            let mut filled = 0;
353            for k in 0..SHORT_COUNT {
354                let base = k * SHORT_LEN;
355                if base + b.end > self.short_spectra.len() || filled + width > self.gather.len() {
356                    break;
357                }
358                self.gather[filled..filled + width]
359                    .copy_from_slice(&self.short_spectra[base + b.start..base + b.end]);
360                filled += width;
361            }
362            trials += bands.deltas.len();
363            if self.sub.prepare(&self.gather[..filled], SF_BIAS) {
364                hits += self.sub.on_grid_count(&bands.deltas, SF_BIAS, tau);
365            }
366        }
367        (hits, trials)
368    }
369}

371/// Sweep every alignment of one signal, returning the best likelihood found.
372///
373/// Alignments are split across threads because they are perfectly
374/// independent — each one re-measures the same frames from a different
375/// starting sample and shares nothing with its neighbours. `std::thread`
376/// rather than a work-stealing pool: the work is uniform, so a fixed split is
377/// as good as any, and it keeps the crate free of a parallelism dependency.
378fn sweep(
379    signal: &[f64],
380    picked: &[usize],
381    bands: &BandSet,
382    params: &AacParams,
383    cancelled: &(dyn Fn() -> bool + Sync),
384) -> Option<TranscodeEvidence> {
385    if picked.len() != params.frames || picked.is_empty() || bands.deltas.is_empty() {
386        return None;
387    }
388
389    let alignments = params.alignments.clamp(1, LONG_LEN);
390    let threads = std::thread::available_parallelism()
391        .map(|n| n.get())
392        .unwrap_or(1)
393        .min(alignments);
394    let per = alignments.div_ceil(threads);
395
396    let results: Vec<Option<(f64, usize)>> = std::thread::scope(|scope| {
397        let handles: Vec<_> = (0..threads)
398            .map(|t| {
399                let lo = t * per;
400                let hi = ((t + 1) * per).min(alignments);
401                scope.spawn(move || {
402                    let windows = Windows::new();
403                    let mut worker = Worker::new();
404                    let mut best = (0.0f64, lo);
405                    for offset in lo..hi {
406                        if cancelled() {
407                            return None;
408                        }
409                        let mut hits = 0usize;
410                        let mut trials = 0usize;
411                        for &f in picked {
412                            let start = f * LONG_LEN + offset;
413                            let win = signal.get(start..start + 2 * LONG_LEN)?;
414                            let (h, t) = worker.measure_frame(win, bands, &windows);
415                            hits += h;
416                            trials += t;
417                        }
418                        let ratio = if trials == 0 {
419                            0.0
420                        } else {
421                            hits as f64 / trials as f64
422                        };
423                        if ratio > best.0 {
424                            best = (ratio, offset);
425                        }
426                    }
427                    Some(best)
428                })
429            })
430            .collect();
431        handles
432            .into_iter()
433            .map(|h| match h.join() {
434                Ok(v) => v,
435                // A worker panicked. `.ok()` used to turn that into `None`,
436                // which the caller reads as "no answer" and the UI as
437                // **clean** — a crash silently rebranded as a verdict, with
438                // nothing printed anywhere. Re-raising it makes the bug
439                // impossible to miss instead of impossible to see. This is a
440                // programming fault by construction: every index in the sweep
441                // is derived from constants, never from the file.
442                Err(payload) => std::panic::resume_unwind(payload),
443            })
444            .collect()
445    });
446
447    let mut best = (0.0f64, 0usize);
448    for result in results {
449        let (ratio, offset) = result?;
450        if ratio > best.0 {
451            best = (ratio, offset);
452        }
453    }
454    Some(TranscodeEvidence {
455        likelihood: best.0,
456        offset: best.1,
457        detected: best.0 > params.significance,
458    })
459}
