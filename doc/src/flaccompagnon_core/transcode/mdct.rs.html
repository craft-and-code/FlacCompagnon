1//! The Modified Discrete Cosine Transform, computed through an FFT.
2//!
3//! Separate from [`crate::analysis::mdct`], which serves the spectral
4//! heuristics and works in `f32` on its own window sizes. This one exists for
5//! the transcoding detectors and differs in the two ways that matter to them:
6//! it is `f64` throughout, and it reuses one FFT plan across the millions of
7//! transforms an offset sweep performs.
8//!
9//! `f64` is not caution for its own sake. The whole detector rests on
10//! measuring how far a coefficient sits from the nearest integer *after* it
11//! has been scaled up by a large power of two. In `f32` the scaling itself
12//! costs enough significant digits to blur exactly the quantity being
13//! measured, and the collapse towards zero that signals a transcode would be
14//! partly hidden under the transform's own rounding.
15
16use std::f64::consts::PI;
17use std::sync::Arc;
18
19use rustfft::num_complex::Complex;
20use rustfft::{Fft, FftPlanner};
21
22/// An MDCT of one fixed size, with its FFT plan and scratch buffers kept
23/// between calls.
24///
25/// Built once per window size and reused: a single offset sweep runs on the
26/// order of a million transforms, and planning an FFT allocates and computes
27/// twiddle factors, so doing it per call would dominate the runtime.
28pub struct Mdct {
29    /// Number of output coefficients — half the input length.
30    n: usize,
31    fft: Arc<dyn Fft<f64>>,
32    /// Pre-rotation `exp(-i·π·t / 2N)` for `t` in `0..2N`.
33    pre: Vec<Complex<f64>>,
34    /// Post-rotation `exp(-i·π·(f + ½)(N + 1) / 2N)` for `f` in `0..N`.
35    post: Vec<Complex<f64>>,
36    buf: Vec<Complex<f64>>,
37    scratch: Vec<Complex<f64>>,
38}
39
40impl Mdct {
41    /// Prepare an MDCT producing `n` coefficients from `2 * n` input samples.
42    pub fn new(n: usize) -> Self {
43        let len = 2 * n;
44        let mut planner = FftPlanner::new();
45        let fft = planner.plan_fft_forward(len);
46        let scratch = vec![Complex::new(0.0, 0.0); fft.get_inplace_scratch_len()];
47
48        let pre = (0..len)
49            .map(|t| Complex::from_polar(1.0, -PI * t as f64 / (2.0 * n as f64)))
50            .collect();
51        let post = (0..n)
52            .map(|f| {
53                let angle = -PI * (f as f64 + 0.5) * (n as f64 + 1.0) / (2.0 * n as f64);
54                Complex::from_polar(1.0, angle)
55            })
56            .collect();
57
58        Self {
59            n,
60            fft,
61            pre,
62            post,
63            buf: vec![Complex::new(0.0, 0.0); len],
64            scratch,
65        }
66    }
67
68    /// Number of coefficients this instance produces.
69    pub fn len(&self) -> usize {
70        self.n
71    }
72
73    /// `true` when this MDCT produces no coefficients — impossible for any
74    /// instance [`Mdct::new`] is called with a non-zero size, but `clippy`
75    /// asks for it wherever `len` exists and a caller may genuinely want it.
76    pub fn is_empty(&self) -> bool {
77        self.n == 0
78    }
79
80    /// Transform `input` (already windowed, `2 * n` samples) into `out`
81    /// (`n` coefficients).
82    ///
83    /// Both lengths are checked rather than asserted: these buffers are sized
84    /// from a file's own sample rate and channel count, and this crate does
85    /// not panic on anything derived from a file. A wrong size returns
86    /// `false` and leaves `out` untouched.
87    #[must_use]
88    pub fn transform(&mut self, input: &[f64], out: &mut [f64]) -> bool {
89        if input.len() != 2 * self.n || out.len() != self.n {
90            return false;
91        }
92
93        // Pre-twiddle, FFT, keep the first half, post-twiddle, take the real
94        // part. This is the standard FFT factorisation of the MDCT; the
95        // rotations are what turn a complex DFT into the real cosine
96        // transform with the right phase convention.
97        for (b, (x, p)) in self.buf.iter_mut().zip(input.iter().zip(&self.pre)) {
98            *b = p * *x;
99        }
100        self.fft.process_with_scratch(&mut self.buf, &mut self.scratch);
101        for (o, (b, p)) in out.iter_mut().zip(self.buf.iter().zip(&self.post)) {
102            *o = (b * p).re;
103        }
104        true
105    }
106}

108#[cfg(test)]
109mod tests {
110    use super::*;

112    /// The MDCT computed straight from its definition, with no FFT anywhere
113    /// near it:
114    ///
115    /// `X(f) = Σ x(t) · cos( π/N · (t + ½ + N/2)(f + ½) )`
116    ///
117    /// Deliberately the slowest possible implementation. Its whole value is
118    /// that it shares no code, no library and no factorisation with
119    /// [`Mdct::transform`] — if both agree, the fast path's twiddle factors
120    /// and phase convention are right, which is precisely what a test written
121    /// against the fast path's own output could never tell us.
122    fn naive_mdct(input: &[f64]) -> Vec<f64> {
123        let n = input.len() / 2;
124        (0..n)
125            .map(|f| {
126                (0..2 * n)
127                    .map(|t| {
128                        let phase = PI / n as f64
129                            * (t as f64 + 0.5 + n as f64 / 2.0)
130                            * (f as f64 + 0.5);
131                        input[t] * phase.cos()
132                    })
133                    .sum()
134            })
135            .collect()
136    }
137
138    fn assert_close(got: &[f64], want: &[f64], tol: f64, what: &str) {
139        assert_eq!(got.len(), want.len(), "{what}: length");
140        for (i, (g, w)) in got.iter().zip(want).enumerate() {
141            assert!(
142                (g - w).abs() < tol,
143                "{what}: bin {i} got {g} want {w} (diff {})",
144                (g - w).abs()
145            );
146        }
147    }
148
149    #[test]
150    fn matches_an_independently_derived_transform() {
151        // Every size the two detectors actually use, and a couple more.
152        //
153        // 6 and 18 are MP3's, and they were missing from this list for a
154        // while — which mattered more than it looks: they are the only sizes
155        // here that are **not powers of two**, so they are the only ones that
156        // exercise rustfft's mixed-radix path rather than its radix-2 one.
157        // A transform that is correct at 128 and wrong at 18 would leave every
158        // test in this file green while the MP3 detector quietly reported
159        // every file clean.
160        for n in [6usize, 8, 9, 18, 128, 1024] {
161            let input: Vec<f64> = (0..2 * n)
162                .map(|t| {
163                    let x = t as f64 / (2 * n) as f64;
164                    (7.0 * PI * x).sin() * 0.6 + (31.0 * PI * x).cos() * 0.3
165                })
166                .collect();
167
168            let mut mdct = Mdct::new(n);
169            let mut got = vec![0.0; n];
170            assert!(mdct.transform(&input, &mut got));
171            // Tolerance scales with the transform size: an N-point sum
172            // accumulates rounding, and the naive version sums in a different
173            // order from the FFT.
174            assert_close(&got, &naive_mdct(&input), 1e-9 * n as f64, &format!("n={n}"));
175        }
176    }
177
178    /// A plan is reused across calls, so a stale buffer would show up as the
179    /// second call returning the first call's answer.
180    #[test]
181    fn reusing_one_instance_gives_independent_results() {
182        let n = 64;
183        let mut mdct = Mdct::new(n);
184
185        let a: Vec<f64> = (0..2 * n).map(|t| (t as f64 * 0.11).sin()).collect();
186        let b: Vec<f64> = (0..2 * n).map(|t| (t as f64 * 0.37).cos()).collect();
187
188        let (mut got_a, mut got_b) = (vec![0.0; n], vec![0.0; n]);
189        assert!(mdct.transform(&a, &mut got_a));
190        assert!(mdct.transform(&b, &mut got_b));
191
192        assert_close(&got_a, &naive_mdct(&a), 1e-7, "first");
193        assert_close(&got_b, &naive_mdct(&b), 1e-7, "second");
194    }

196    #[test]
197    fn silence_transforms_to_silence() {
198        let n = 128;
199        let mut mdct = Mdct::new(n);
200        let mut out = vec![f64::NAN; n];
201        assert!(mdct.transform(&vec![0.0; 2 * n], &mut out));
202        assert!(out.iter().all(|c| c.abs() < 1e-12), "{out:?}");
203    }

205    /// Buffer sizes come from a file's own header, so a mismatch is a
206    /// malformed-input case, not a programming error: it must be reported,
207    /// not panicked on.
208    #[test]
209    fn wrong_buffer_sizes_are_refused_without_panicking() {
210        let mut mdct = Mdct::new(32);
211        let mut out = vec![0.0; 32];
212        assert!(!mdct.transform(&[0.0; 63], &mut out), "input too short");
213        assert!(!mdct.transform(&[0.0; 65], &mut out), "input too long");
214        assert!(!mdct.transform(&[], &mut out), "input empty");
215
216        let input = vec![0.0; 64];
217        assert!(!mdct.transform(&input, &mut [0.0; 31]), "output too short");
218        assert!(!mdct.transform(&input, &mut []), "output empty");
219
220        // The refused calls must not have written anything.
221        let mut fresh = vec![f64::NAN; 32];
222        assert!(!mdct.transform(&[0.0; 10], &mut fresh));
223        assert!(fresh.iter().all(|v| v.is_nan()), "output was touched");
224    }
