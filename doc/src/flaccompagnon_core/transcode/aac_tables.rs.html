1//! MPEG-AAC constants: scalefactor band layouts and window shapes.
2//!
3//! The band tables are **from the standard**, ISO/IEC 13818-7 (MPEG-2 AAC),
4//! §"Scalefactor bands and grouping" — they define where one scalefactor's
5//! reach ends and the next begins, and a detector that got them wrong would
6//! be averaging its measurement across boundaries the encoder never crossed.
7//! They are reproduced here as published; they are not anyone's invention and
8//! carry no authorship.
9//!
10//! Only the three sample rates a lossy transcode realistically involves are
11//! tabulated. 44.1 and 48 kHz share a layout in the standard; 32 kHz gets two
12//! extra bands at the top because its Nyquist sits lower, so the same
13//! coefficient index covers less bandwidth and the upper bands can be
14//! narrower.
15//!
16//! # Window shapes
17//!
18//! AAC switches between one long transform and eight short ones, with two
19//! transition shapes bridging them. All four are built from the same sine
20//! window, and only sine windows are implemented: the paper checked that
21//! Kaiser-Bessel-derived windows "does not alter the detection performance in
22//! a significant way" (§1.2), and sine is what common AAC encoders emit.
23
24/// Coefficients in one long transform.
25pub const LONG_LEN: usize = 1024;
26/// Coefficients in one short transform.
27pub const SHORT_LEN: usize = 128;
28/// Short transforms packed into the span of one long transform.
29pub const SHORT_COUNT: usize = LONG_LEN / SHORT_LEN;
30/// Where the first short transform starts inside the long window.
31pub const SHORT_OFFSET: usize = (LONG_LEN - SHORT_LEN) / 2;
32
33/// Scalefactor coding bias for AAC.
34///
35/// See [`super::criterion::PreparedSubband::prepare`] for what this is and
36/// why the search runs in biased units.
37pub const SF_BIAS: f64 = 60.0;

41/// Band start offsets for the long transform, 44.1 and 48 kHz.
42const LONG_44_48: [usize; 49] = [
43    0, 4, 8, 12, 16, 20, 24, 28, 32, 36, 40, 48, 56, 64, 72, 80, 88, 96, 108, 120, 132, 144, 160,
44    176, 196, 216, 240, 264, 292, 320, 352, 384, 416, 448, 480, 512, 544, 576, 608, 640, 672, 704,
45    736, 768, 800, 832, 864, 896, 928,
46];
47
48/// Band start offsets for the long transform, 32 kHz.
49const LONG_32: [usize; 51] = [
50    0, 4, 8, 12, 16, 20, 24, 28, 32, 36, 40, 48, 56, 64, 72, 80, 88, 96, 108, 120, 132, 144, 160,
51    176, 196, 216, 240, 264, 292, 320, 352, 384, 416, 448, 480, 512, 544, 576, 608, 640, 672, 704,
52    736, 768, 800, 832, 864, 896, 928, 960, 992,
53];
54
55/// Band start offsets for the short transform. The same for all three rates.
56const SHORT_ALL: [usize; 14] = [0, 4, 8, 12, 16, 20, 28, 36, 44, 56, 68, 80, 96, 112];
57
58fn bands_from_starts(starts: &[usize], total: usize) -> Vec<Band> {
59    starts
60        .iter()
61        .enumerate()
62        .map(|(i, &start)| Band {
63            start,
64            end: starts.get(i + 1).copied().unwrap_or(total),
65        })
66        .collect()
67}
68
69/// The long-transform bands for `rate_khz`, or `None` for an untabulated
70/// rate. Use [`super::supported_rate_khz`] to obtain `rate_khz`.
71pub fn long_bands(rate_khz: u32) -> Option<Vec<Band>> {
72    let starts: &[usize] = match rate_khz {
73        44 | 48 => &LONG_44_48,
74        32 => &LONG_32,
75        _ => return None,
76    };
77    Some(bands_from_starts(starts, LONG_LEN))
78}
79
80/// The short-transform bands for `rate_khz`.
81pub fn short_bands(rate_khz: u32) -> Option<Vec<Band>> {
82    match rate_khz {
83        32 | 44 | 48 => Some(bands_from_starts(&SHORT_ALL, SHORT_LEN)),
84        _ => None,
85    }
86}

88/// A symmetric sine window of `len` samples: `w[n] = sin(π(n + ½)/len)`.
89///
90/// This is the window MPEG defines for both transform sizes; the four shapes
91/// below are all assembled from two of these.
92fn sine_window(len: usize) -> Vec<f64> {
93    (0..len)
94        .map(|n| (std::f64::consts::PI * (n as f64 + 0.5) / len as f64).sin())
95        .collect()
96}

98/// The four window shapes AAC can apply to one long frame.
99///
100/// `start` and `stop` are the transition shapes: an encoder switching from
101/// long to short transforms cannot do it abruptly without breaking the
102/// transform's perfect-reconstruction property, so it inserts a window that
103/// is long on one side and short on the other. Both must be tested, because
104/// a frame the encoder used as a transition carries its lattice under that
105/// shape and under no other.
106pub struct Windows {
107    /// Plain long window.
108    pub long: Vec<f64>,
109    /// Long rising edge, short falling edge.
110    pub start: Vec<f64>,
111    /// Short rising edge, long falling edge.
112    pub stop: Vec<f64>,
113    /// The window applied to each of the eight short transforms.
114    pub short: Vec<f64>,
115}
116
117impl Windows {
118    /// Build the four shapes. Deterministic and cheap; built once per
119    /// analysis.
120    pub fn new() -> Self {
121        let long_full = sine_window(2 * LONG_LEN);
122        let short_full = sine_window(2 * SHORT_LEN);
123        let (long_rise, long_fall) = long_full.split_at(LONG_LEN);
124        let (short_rise, short_fall) = short_full.split_at(SHORT_LEN);
125
126        // start: long rise, flat top, short fall, then silence. The flat and
127        // zero runs are what let a short transform sit inside a long frame
128        // without the two overlapping incorrectly.
129        let mut start = Vec::with_capacity(2 * LONG_LEN);
130        start.extend_from_slice(long_rise);
131        start.extend(std::iter::repeat_n(1.0, SHORT_OFFSET));
132        start.extend_from_slice(short_fall);
133        start.extend(std::iter::repeat_n(0.0, SHORT_OFFSET));
134
135        // stop: the mirror image.
136        let mut stop = Vec::with_capacity(2 * LONG_LEN);
137        stop.extend(std::iter::repeat_n(0.0, SHORT_OFFSET));
138        stop.extend_from_slice(short_rise);
139        stop.extend(std::iter::repeat_n(1.0, SHORT_OFFSET));
140        stop.extend_from_slice(long_fall);
141
142        Self {
143            long: long_full.clone(),
144            start,
145            stop,
146            short: short_full,
147        }
148    }
149}
150
151impl Default for Windows {
152    fn default() -> Self {
153        Self::new()
154    }
155}
