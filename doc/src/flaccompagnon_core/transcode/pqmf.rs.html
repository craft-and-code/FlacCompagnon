1//! MP3's polyphase analysis filterbank — the first of its three transform
2//! stages.
3//!
4//! Where AAC takes one MDCT straight off the PCM, Layer III first splits the
5//! signal into 32 equal-width subbands with a 512-tap polyphase filter, and
6//! only then applies a small MDCT inside each band. This file is that first
7//! stage; [`super::mp3`] drives the rest.
8//!
9//! # The shape of one step
10//!
11//! Each output step consumes 32 fresh samples and looks back over 512 in
12//! total, so consecutive steps overlap by 480. The standard's own formulation
13//! — reproduced here because it is what makes the arithmetic cheap — is:
14//!
15//! 1. take the last 512 samples, **time-reversed**;
16//! 2. multiply by the analysis window `C`;
17//! 3. fold the 512 products down to 64 by summing every 64th;
18//! 4. multiply that 64-vector by a fixed 32×64 cosine matrix.
19//!
20//! The fold is what makes it polyphase: a naive implementation would need
21//! 32 × 512 multiplies per step, this needs 512 + 32 × 64.
22//!
23//! # Why one granule needs the previous one
24//!
25//! A granule is 576 samples — 18 steps of 32. The first of those steps still
26//! reaches 512 samples into the past, most of which belong to the *previous*
27//! granule. [`Pqmf::granule`] therefore takes both, and the caller must feed
28//! them in order.
29
30use super::mp3_tables::{GRANULE_LEN, GRANULE_SIZE, PQMF_BANDS, PQMF_WINDOW, PQMF_WINDOW_LEN};
31
32/// Blocks of 32 samples the window looks back over, beyond the current step.
33const LOOKBACK_BLOCKS: usize = 16;
34/// Where in the previous granule the carried tail begins: `(18 − 16 + 1) × 32
35/// = 96` samples in.
36const CARRY_START: usize = (GRANULE_SIZE - LOOKBACK_BLOCKS + 1) * PQMF_BANDS;
37
38/// Samples of the previous granule the first step still needs: 480.
39///
40/// Plus a full 576-sample granule, that is exactly the 1056 the eighteenth
41/// step's window ends on. Off by one block in either direction and the last
42/// step reads past the buffer or the first one starts in the wrong place —
43/// and the tell is subtle, because a steady tone still lands in the right
44/// band either way. (This constant was briefly the *offset* rather than the
45/// *count*, 96 instead of 480; the test below is what it exists for.)
46const CARRY: usize = GRANULE_LEN - CARRY_START;
47
48/// The analysis filterbank, with its cosine matrix built once.
49pub struct Pqmf {
50    /// `cos((2i + 1)(k − 16)π / 64)` for `i` in `0..32`, `k` in `0..64`,
51    /// stored row-major.
52    modulation: Vec<f64>,
53    /// Scratch: the 512 windowed products, then folded to 64.
54    window: Vec<f64>,
55    folded: Vec<f64>,
56    /// Scratch: `prev`'s tail followed by the current granule.
57    buffer: Vec<f64>,
58}
59
60impl Pqmf {
61    /// Build the filterbank. Cheap, but done once per worker rather than per
62    /// granule — a sweep runs this thousands of times.
63    pub fn new() -> Self {
64        let mut modulation = Vec::with_capacity(PQMF_BANDS * 64);
65        for i in 0..PQMF_BANDS {
66            for k in 0..64 {
67                let angle = (2.0 * i as f64 + 1.0) * (k as f64 - 16.0) * std::f64::consts::PI / 64.0;
68                modulation.push(angle.cos());
69            }
70        }
71        Self {
72            modulation,
73            window: vec![0.0; PQMF_WINDOW_LEN],
74            folded: vec![0.0; 64],
75            buffer: vec![0.0; CARRY + GRANULE_LEN],
76        }
77    }
78
79    /// Analyze one granule into `out`, a `32 × 18` array stored **band-major**
80    /// (`out[band * 18 + step]`).
81    ///
82    /// `prev` is the granule immediately before `cur`; both must be
83    /// [`GRANULE_LEN`] samples and `out` must hold `32 × 18`. Wrong sizes
84    /// return `false` rather than panicking — these come from a decoded file.
85    #[must_use]
86    pub fn granule(&mut self, prev: &[f64], cur: &[f64], out: &mut [f64]) -> bool {
87        if prev.len() != GRANULE_LEN
88            || cur.len() != GRANULE_LEN
89            || out.len() != PQMF_BANDS * GRANULE_SIZE
90        {
91            return false;
92        }
93
94        self.buffer[..CARRY].copy_from_slice(&prev[GRANULE_LEN - CARRY..]);
95        self.buffer[CARRY..].copy_from_slice(cur);
96
97        for step in 0..GRANULE_SIZE {
98            let from = step * PQMF_BANDS;
99            let slice = &self.buffer[from..from + PQMF_WINDOW_LEN];
100
101            // Time-reversed and windowed. The reversal is part of the
102            // standard's definition, not an implementation choice: it is what
103            // makes the fold below equivalent to the direct convolution.
104            for (n, w) in self.window.iter_mut().enumerate() {
105                *w = slice[PQMF_WINDOW_LEN - 1 - n] * PQMF_WINDOW[n];
106            }
107
108            // Fold 512 down to 64 by summing every 64th.
109            for (k, f) in self.folded.iter_mut().enumerate() {
110                let mut acc = 0.0;
111                for block in 0..8 {
112                    acc += self.window[block * 64 + k];
113                }
114                *f = acc;
115            }
116
117            for band in 0..PQMF_BANDS {
118                let row = &self.modulation[band * 64..(band + 1) * 64];
119                let mut acc = 0.0;
120                for (m, f) in row.iter().zip(&self.folded) {
121                    acc += m * f;
122                }
123                out[band * GRANULE_SIZE + step] = acc;
124            }
125        }
126        true
127    }
128}
129
130impl Default for Pqmf {
131    fn default() -> Self {
132        Self::new()
133    }
134}

136/// Flip the sign of odd subbands at odd time steps.
137///
138/// Layer III's filterbank alternates the phase of every second subband, and
139/// the decoder undoes it before the inverse MDCT. An analysis that skipped
140/// this would produce coefficients that are individually the right magnitude
141/// but collectively on the wrong lattice — the detector would find nothing,
142/// and nothing about the output would look wrong.
143///
144/// `coeffs` is `32 × steps`, band-major.
145pub fn flip_odd_subbands(coeffs: &mut [f64], steps: usize) {
146    if steps == 0 {
147        return;
148    }
149    let mut band = 1;
150    while band < PQMF_BANDS {
151        let row = band * steps;
152        let mut step = 1;
153        while step < steps {
154            if let Some(v) = coeffs.get_mut(row + step) {
155                *v = -*v;
156            }
157            step += 2;
158        }
159        band += 2;
160    }
161}

163#[cfg(test)]
164mod tests {
165    use super::*;
166
167    /// The buffer arithmetic, asserted rather than trusted: the eighteenth
168    /// step's 512-sample window must end exactly on the last sample of the
169    /// granule, with nothing left over and nothing missing.
170    #[test]
171    fn the_lookback_buffer_is_exactly_long_enough() {
172        assert_eq!(CARRY_START, 96, "the tail starts 96 samples into the granule");
173        assert_eq!(CARRY, 480, "480 samples carried from the previous granule");
174        assert_eq!(CARRY + GRANULE_LEN, 1056);
175        let last_step_end = (GRANULE_SIZE - 1) * PQMF_BANDS + PQMF_WINDOW_LEN;
176        assert_eq!(last_step_end, CARRY + GRANULE_LEN, "must end flush");
177    }

179    fn tone(hz: f64, sr: f64, n: usize) -> Vec<f64> {
180        (0..n)
181            .map(|i| (2.0 * std::f64::consts::PI * hz * i as f64 / sr).sin() * 0.5)
182            .collect()
183    }
184
185    /// A pure tone must land in the subband that covers its frequency, and
186    /// almost nowhere else. Each of the 32 bands spans Nyquist/32 — 689 Hz at
187    /// 44.1 kHz.
188    ///
189    /// This is the check that the modulation matrix, the window and the fold
190    /// agree with each other: get any one of them wrong and the energy lands
191    /// in the wrong band, or smears across all of them.
192    ///
193    /// The tones are placed at band *centres* rather than at round
194    /// frequencies. A round number can fall near a band edge — 9 kHz sits
195    /// 42 Hz above the 13/14 boundary — where energy legitimately splits
196    /// between neighbours (72/28 in that case) and a concentration assertion
197    /// fails for a reason that has nothing to do with correctness.
198    #[test]
199    fn a_tone_lands_in_the_band_that_covers_it() {
200        let sr = 44_100.0;
201        let band_hz = sr / 2.0 / PQMF_BANDS as f64;
202        let mut pqmf = Pqmf::new();
203        let mut out = vec![0.0; PQMF_BANDS * GRANULE_SIZE];
204
205        for band in [1usize, 4, 13, 21, 28] {
206            let hz = (band as f64 + 0.5) * band_hz;
207            let sig = tone(hz, sr, GRANULE_LEN * 6);
208            // Granules 2 and 3, well past the filterbank's start-up.
209            let prev = &sig[2 * GRANULE_LEN..3 * GRANULE_LEN];
210            let cur = &sig[3 * GRANULE_LEN..4 * GRANULE_LEN];
211            assert!(pqmf.granule(prev, cur, &mut out));
212
213            let energy: Vec<f64> = (0..PQMF_BANDS)
214                .map(|b| out[b * GRANULE_SIZE..(b + 1) * GRANULE_SIZE].iter().map(|v| v * v).sum())
215                .collect();
216            let loudest = energy
217                .iter()
218                .enumerate()
219                .max_by(|a, b| a.1.partial_cmp(b.1).unwrap_or(std::cmp::Ordering::Equal))
220                .map(|(i, _)| i)
221                .expect("32 bands");
222            assert_eq!(loudest, band, "{hz:.0} Hz should sit in band {band}");
223
224            // And it must dominate: a filterbank that smeared energy across
225            // bands could still pick the right maximum by a hair.
226            let total: f64 = energy.iter().sum();
227            assert!(
228                energy[loudest] / total > 0.99,
229                "{hz:.0} Hz: only {:.1}% of energy in its own band",
230                100.0 * energy[loudest] / total
231            );
232        }
233    }

235    #[test]
236    fn silence_analyzes_to_silence() {
237        let mut pqmf = Pqmf::new();
238        let mut out = vec![f64::NAN; PQMF_BANDS * GRANULE_SIZE];
239        let zeros = vec![0.0; GRANULE_LEN];
240        assert!(pqmf.granule(&zeros, &zeros, &mut out));
241        assert!(out.iter().all(|v| v.abs() < 1e-12), "{:?}", &out[..8]);
242    }

244    #[test]
245    fn wrong_buffer_sizes_are_refused_without_panicking() {
246        let mut pqmf = Pqmf::new();
247        let g = vec![0.0; GRANULE_LEN];
248        let short = vec![0.0; GRANULE_LEN - 1];
249        let mut out = vec![0.0; PQMF_BANDS * GRANULE_SIZE];
250        assert!(!pqmf.granule(&short, &g, &mut out));
251        assert!(!pqmf.granule(&g, &short, &mut out));
252        assert!(!pqmf.granule(&g, &g, &mut [0.0; 8]));
253        assert!(!pqmf.granule(&[], &[], &mut out));
254    }

256    /// Odd bands at odd steps, and nothing else. Written as an explicit map
257    /// because an off-by-one in either index is invisible in the output.
258    #[test]
259    fn the_sign_flip_touches_exactly_the_odd_odd_cells() {
260        let steps = 4;
261        let mut coeffs = vec![1.0; PQMF_BANDS * steps];
262        flip_odd_subbands(&mut coeffs, steps);
263        for band in 0..PQMF_BANDS {
264            for step in 0..steps {
265                let want = if band % 2 == 1 && step % 2 == 1 { -1.0 } else { 1.0 };
266                assert_eq!(
267                    coeffs[band * steps + step], want,
268                    "band {band} step {step}"
269                );
270            }
271        }
272        // Applying it twice restores the original — it is an involution.
273        flip_odd_subbands(&mut coeffs, steps);
274        assert!(coeffs.iter().all(|v| *v == 1.0));
275
276        // Degenerate input must not panic.
277        flip_odd_subbands(&mut [], 0);
278        flip_odd_subbands(&mut [1.0], 1);
279    }
