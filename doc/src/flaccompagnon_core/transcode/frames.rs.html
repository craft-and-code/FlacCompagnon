1//! Choosing *where* in the file to look, and *which* channel signal to look
2//! at.
3//!
4//! Both decisions are codec-independent, both come from the paper's §1.4, and
5//! both exist for the same reason: the detector is expensive, so it is pointed
6//! at the parts of the file most likely to answer.
7//!
8//! # Why the loudest frames
9//!
10//! §1.3 puts it plainly — "detection can be improved by selecting only
11//! high-energy frames". A quiet passage gives the encoder little to quantize:
12//! most coefficients land near zero, the lattice is coarse relative to the
13//! signal, and the rounding error carries almost no information either way.
14//! Loud frames exercise the quantizer across its range, so that is where the
15//! grid is visible.
16
17/// Energy of every analysis frame, in dB.
18///
19/// Frames overlap by half: frame `f` spans `f·frame_len` to
20/// `(f+2)·frame_len`, matching the codec's own 50%-overlapped transform. The
21/// last two frames' worth of samples are dropped rather than zero-padded —
22/// a padded frame's energy is not comparable to a full one's, and it would
23/// compete for a place in the top list on false terms.
24///
25/// Returns an empty vector for a signal shorter than three frames, which is
26/// the caller's signal that the file is too short to analyze.
27pub fn frame_energies_db(signal: &[f64], frame_len: usize) -> Vec<f64> {
28    if frame_len == 0 {
29        return Vec::new();
30    }
31    let count = (signal.len() / frame_len).saturating_sub(2);
32    (0..count)
33        .map(|f| {
34            let start = f * frame_len;
35            let energy: f64 = signal
36                .get(start..start + 2 * frame_len)
37                .map(|w| w.iter().map(|s| s * s).sum())
38                .unwrap_or(0.0);
39            // dB purely so the values stay comparable when printed; the
40            // ordering is the same either way since log is monotonic. Silence
41            // gives -inf, which sorts last — correct, it is the least useful
42            // frame there is.
43            10.0 * energy.log10()
44        })
45        .collect()
46}

48/// Indices of the `wanted` highest-energy frames.
49///
50/// Ties are broken by frame index so the choice is reproducible: a file
51/// containing a long stretch of identical frames (digital silence, a held
52/// tone) would otherwise give a different answer per run depending on the
53/// sort's internal ordering, and a detector whose verdict moves between runs
54/// on the same file is worse than a slightly less optimal one.
55///
56/// Fewer than `wanted` frames are returned when the file is short; the caller
57/// decides whether that is still enough to judge.
58pub fn top_frames(energies_db: &[f64], wanted: usize) -> Vec<usize> {
59    let mut order: Vec<usize> = (0..energies_db.len()).collect();
60    order.sort_by(|&a, &b| {
61        energies_db[b]
62            .partial_cmp(&energies_db[a])
63            // NaN cannot arise from a sum of squares, but `partial_cmp` has
64            // to be handled and falling back on the index keeps the order
65            // total rather than letting `sort_by` see an inconsistent
66            // comparator.
67            .unwrap_or(std::cmp::Ordering::Equal)
68            .then(a.cmp(&b))
69    });
70    order.truncate(wanted);
71    order
72}

74/// Select exactly the calibrated number of frames, with enough trailing
75/// samples for the transform at every alignment. `context_frames` is 2 for
76/// AAC and 3 for MP3 (which also needs the preceding PQMF granule).
77///
78/// Returning a shorter selection would silently apply a threshold calibrated
79/// for a different number of observations. Select once on M in stereo and
80/// pass the same indices to every channel sweep, as in the MATLAB reference.
81pub fn select_frames(
82    signal: &[f64],
83    frame_len: usize,
84    context_frames: usize,
85    wanted: usize,
86) -> Option<Vec<usize>> {
87    if frame_len == 0 || wanted == 0 {
88        return None;
89    }
90    let mut energies = frame_energies_db(signal, frame_len);
91    energies.truncate((signal.len() / frame_len).saturating_sub(context_frames));
92    if energies.len() < wanted {
93        return None;
94    }
95    Some(top_frames(&energies, wanted))
96}

98/// Mean energy, in dB, over the given frames.
99///
100/// The mean of the decibel values rather than the decibels of the mean
101/// energy: those differ, and this one is what the reference uses to decide
102/// which of two channels to test. Returns `-inf` for an empty selection, so
103/// a channel with nothing to measure always loses the comparison.
104pub fn mean_energy_db(energies_db: &[f64], frames: &[usize]) -> f64 {
105    if frames.is_empty() {
106        return f64::NEG_INFINITY;
107    }
108    let sum: f64 = frames.iter().filter_map(|&f| energies_db.get(f)).sum();
109    sum / frames.len() as f64
110}

112/// Mid and side signals, `(l + r)·k` and `(l − r)·k`.
113///
114/// `k` is 1 for AAC and ½ for MP3 — the reference implementations use different matrices, and it is not a detail that can be normalised away here: the
115/// scale feeds straight into `φ^dz` (eq. 2), which is a logarithm of the
116/// coefficient peak, so a factor of two shifts every scalefactor the search
117/// tries by exactly 4 steps.
118///
119/// Channels of unequal length are truncated to the shorter, rather than
120/// refused: a decoder that returns a partial last frame on one channel should
121/// not sink the whole analysis.
122pub fn mid_side(left: &[f64], right: &[f64], k: f64) -> (Vec<f64>, Vec<f64>) {
123    let n = left.len().min(right.len());
124    let mut mid = Vec::with_capacity(n);
125    let mut side = Vec::with_capacity(n);
126    for i in 0..n {
127        mid.push((left[i] + right[i]) * k);
128        side.push((left[i] - right[i]) * k);
129    }
130    (mid, side)
131}

137    #[test]
138    fn short_files_cannot_shrink_the_calibrated_population() {
139        for (context, wanted) in [(2, 64), (3, 8)] {
140            let hop = 16;
141            let minimum = (wanted + context) * hop;
142            assert!(select_frames(&vec![1.0; minimum - 1], hop, context, wanted).is_none());
143            let selected = select_frames(&vec![1.0; minimum], hop, context, wanted).unwrap();
144            assert_eq!(selected.len(), wanted);
145            for frame in selected {
146                // Last alignment plus complete AAC window / MP3 context.
147                assert!(frame * hop + (hop - 1) + context * hop <= minimum);
148            }
149        }
150    }

152    #[test]
153    fn frames_overlap_by_half_and_drop_the_incomplete_tail() {
154        // 10 frames' worth of samples gives 8 analysis frames: the last two
155        // starts would run past the end of the signal.
156        let signal = vec![1.0; 10 * 64];
157        assert_eq!(frame_energies_db(&signal, 64).len(), 8);
158
159        // Too short to analyze at all, at every boundary.
160        for n in [0usize, 63, 64, 127, 128, 191] {
161            assert!(
162                frame_energies_db(&vec![1.0; n], 64).len() <= 1,
163                "n={n} produced too many frames"
164            );
165        }
166        assert!(
167            frame_energies_db(&[1.0; 100], 0).is_empty(),
168            "zero-length frame"
169        );
170    }

172    /// The energy of a frame must cover *two* frame lengths — using one would
173    /// silently measure the wrong window and pick the wrong frames.
174    #[test]
175    fn energy_covers_two_frame_lengths() {
176        let mut signal = vec![0.0; 4 * 8];
177        // Put energy only in the second half of frame 0's span.
178        for s in signal.iter_mut().take(16).skip(8) {
179            *s = 1.0;
180        }
181        let db = frame_energies_db(&signal, 8);
182        // Frame 0 spans samples 0..16, so it must see those eight ones.
183        assert!((db[0] - 10.0 * 8.0f64.log10()).abs() < 1e-12, "{db:?}");
184    }

186    #[test]
187    fn top_frames_picks_the_loudest_and_breaks_ties_by_index() {
188        let db = [1.0, 9.0, 5.0, 9.0, 2.0];
189        assert_eq!(top_frames(&db, 3), vec![1, 3, 2]);
190        // Asking for more than exist yields what exists, in order.
191        assert_eq!(top_frames(&db, 99).len(), 5);
192        assert_eq!(top_frames(&[], 4), Vec::<usize>::new());
193        assert_eq!(top_frames(&db, 0), Vec::<usize>::new());
194
195        // Identical frames must give the same answer every time.
196        let flat = [3.0; 6];
197        assert_eq!(top_frames(&flat, 3), vec![0, 1, 2]);
198        assert_eq!(top_frames(&flat, 3), top_frames(&flat, 3));
199    }

201    #[test]
202    fn mean_energy_ignores_missing_frames_and_empty_selections() {
203        let db = [10.0, 20.0, 30.0];
204        assert!((mean_energy_db(&db, &[0, 2]) - 20.0).abs() < 1e-12);
205        assert_eq!(mean_energy_db(&db, &[]), f64::NEG_INFINITY);
206        // An out-of-range index must not panic — it is dropped.
207        assert!((mean_energy_db(&db, &[1, 99]) - 10.0).abs() < 1e-12);
208    }

210    #[test]
211    fn mid_side_applies_its_scale_and_tolerates_ragged_channels() {
212        let (m, s) = mid_side(&[1.0, 2.0], &[3.0, 4.0], 1.0);
213        assert_eq!(m, vec![4.0, 6.0]);
214        assert_eq!(s, vec![-2.0, -2.0]);
215
216        let (m_half, s_half) = mid_side(&[1.0, 2.0], &[3.0, 4.0], 0.5);
217        assert_eq!(m_half, vec![2.0, 3.0]);
218        assert_eq!(s_half, vec![-1.0, -1.0]);
219
220        let (m2, s2) = mid_side(&[1.0, 2.0, 3.0], &[1.0], 1.0);
221        assert_eq!(m2.len(), 1);
222        assert_eq!(s2.len(), 1);
223        let (m3, s3) = mid_side(&[], &[1.0], 1.0);
224        assert!(m3.is_empty() && s3.is_empty());
225    }
