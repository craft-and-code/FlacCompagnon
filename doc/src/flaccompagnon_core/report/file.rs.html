11/// Resource budget for an imported JSON report: 64 MiB, ample for large libraries.
12/// Both the metadata and the bytes actually read are checked, including files
13/// that grow while being read.
14pub const MAX_JSON_BYTES: u64 = 64 * 1024 * 1024;
15
16/// Reject links and special files before replacing an export destination.
17/// A missing destination is allowed; its parent must exist when writing.
18pub fn validate_destination(dest: &Path) -> io::Result<()> {
19    match dest.symlink_metadata() {
20        Ok(metadata) if !metadata.file_type().is_file() => Err(io::Error::new(
21            io::ErrorKind::InvalidInput,
22            "Report destination must be a regular file, not a symbolic link or directory.",
23        )),
24        Ok(_) => Ok(()),
25        Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(()),
26        Err(error) => Err(error),
27    }
28}

30/// Read a regular JSON report with a bounded allocation.
31/// The opened handle is checked, and Unix opens are nonblocking without
32/// following links, so a concurrent path swap cannot stall cache discovery.
33pub fn read_json(path: &Path) -> Result<FolderReport, String> {
34    let file = open_regular_file(path).map_err(|error| error.to_string())?;
35    let metadata = file.metadata().map_err(|error| error.to_string())?;
36    if metadata.len() > MAX_JSON_BYTES {
37        return Err("JSON report exceeds the 64 MiB import limit.".into());
38    }
39    let mut text = String::new();
40    file.take(MAX_JSON_BYTES + 1)
41        .read_to_string(&mut text)
42        .map_err(|error| error.to_string())?;
43    if text.len() as u64 > MAX_JSON_BYTES {
44        return Err("JSON report exceeds the 64 MiB import limit.".into());
45    }
46    parse_json(&text)
47}

49/// Open a regular file without following links or accepting special files.
50/// The opened handle is validated because a concurrent rename could replace
51/// a previously checked file with a link or pipe. Unix opens are nonblocking.
52pub fn open_regular_file(path: &Path) -> io::Result<File> {
53    let mut options = OpenOptions::new();
54    options.read(true);
55    #[cfg(unix)]
56    {
57        use std::os::unix::fs::OpenOptionsExt;
58        options.custom_flags(libc::O_NOFOLLOW | libc::O_NONBLOCK);
59    }
60    #[cfg(windows)]
61    {
62        use std::os::windows::fs::OpenOptionsExt;
63        // FILE_FLAG_OPEN_REPARSE_POINT opens the link itself, not its target.
64        options.custom_flags(0x0020_0000);
65    }
66    let file = options.open(path)?;
67    if !file.metadata()?.file_type().is_file() {
68        return Err(io::Error::new(
69            io::ErrorKind::InvalidInput,
70            "Input must be a regular file, not a symbolic link or directory.",
71        ));
72    }
73    Ok(file)
74}

76/// Atomically replace an exported file with `bytes`, rejecting links and special files.
77/// Renaming a sibling temporary file avoids truncating an existing export on
78/// an interrupted write, and never follows a destination link to an audio file.
79pub fn write_atomic_bytes(dest: &Path, bytes: &[u8]) -> io::Result<()> {
80    validate_destination(dest)?;
81    let parent = dest
82        .parent()
83        .filter(|p| !p.as_os_str().is_empty())
84        .unwrap_or(Path::new("."));
85    let mut temporary = tempfile::NamedTempFile::new_in(parent)?;
86    temporary.write_all(bytes)?;
87    temporary.as_file().sync_all()?;
88    temporary.persist(dest).map_err(|error| error.error)?;
89    Ok(())
90}

92/// Atomically write the CSV report to `dest`, rejecting links and special files.
93pub fn write_csv(dest: &Path, report: &FolderReport) -> io::Result<()> {
94    write_atomic_bytes(dest, build_csv(report).as_bytes())
95}

97/// Atomically write the JSON report to `dest`, rejecting links and special files.
98pub fn write_json(dest: &Path, report: &FolderReport) -> io::Result<()> {
99    let text =
100        build_json(report).map_err(|error| io::Error::new(io::ErrorKind::InvalidData, error))?;
101    write_atomic_bytes(dest, text.as_bytes())
102}
