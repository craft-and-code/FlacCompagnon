9/// Build the CSV text for a folder report.
10///
11/// Column order mirrors the results table left-to-right (`ResultsTable.tsx`'s
12/// `headers`), not the order fields were originally added to this struct —
13/// so a column's position here means the same thing it means on screen.
14/// Fields the table folds into a single cell (Detections: `status` +
15/// `upscaling`/`upsampling`/`transcoding`/`lattice_score`; Clipping: `clipped` +
16/// `clip_events`/`peak_dbfs`) are grouped together at that cell's position
17/// rather than split across the row. No column is dropped — every field the
18/// old order exported is still here, just reordered.
19///
20/// `file_md5` and `file_crc32` are the exception to that mirroring: they are
21/// appended at the end because their columns are hidden by default in the
22/// table, so there is no on-screen position to mirror.
23///
24/// `codec` and `bitrate_kbps` are deliberately slotted mid-row — right after
25/// `format` and right before `sample_rate` respectively — rather than at the
26/// end, on the maintainer's explicit request, even though the table's own
27/// column order is otherwise just a display preference (`useColumnPrefs.ts`)
28/// this file doesn't follow. This *is* a breaking change for any script or
29/// spreadsheet already reading the CSV by position: those two columns moved.
30/// `modified_unix` stays appended at the end — nobody asked to move that one,
31/// and there's no natural mid-row spot for it the way there is for the other
32/// two.
33pub fn build_csv(report: &FolderReport) -> String {
34    let mut out = String::new();
35    out.push_str(
36        "file,format,codec,badge,bitrate_kbps,sample_rate,declared_bits,real_bit_depth,\
37         duration_s,size_bytes,status,upscaling,upsampling,transcoding,lattice_score,cutoff_hz,\
38         cutoff_ratio,channels,fake_stereo,phase_correlation,phase_inverted,clipped,clip_events,peak_dbfs,true_peak_dbtp,integrated_lufs,max_momentary_lufs,momentary_max_start_s,max_short_term_lufs,short_term_max_start_s,loudness_range_lu,dr_db,\
39         md5,bit_depth_method,stored_bits,balance_right_minus_left_db,balance_silent_channel,hf_side_to_mid_db,hf_reference_side_to_mid_db,hf_narrowed_block_fraction,hf_stereo_narrowed,suspected_clicks,suspected_dropouts,click_locations,dropout_locations,modified_unix,dc_offset_max_abs,dc_offset_channel_means,",
40    );
41    out.push_str(&phase::headers().join(","));
42    out.push_str(",file_md5,file_crc32\n");
43    for f in &report.files {
44        let md5 = f
45            .flac_md5
46            .as_ref()
47            .map(|m| match m {
48                FlacMd5Status::NoSignature => "none",
49                FlacMd5Status::Present => "present",
50                FlacMd5Status::Match => "ok",
51                FlacMd5Status::Mismatch => "mismatch",
52                FlacMd5Status::Error(_) => "error",
53            })
54            .unwrap_or("");
55        use crate::analysis::stereo::StereoBalance;
56        let (balance_db, silent_channel) = match f.stereo_balance {
57            Some(StereoBalance::Measured {
58                right_minus_left_db,
59            }) => (format!("{right_minus_left_db:.2}"), ""),
60            Some(StereoBalance::LeftSilent) => (String::new(), "left"),
61            Some(StereoBalance::RightSilent) => (String::new(), "right"),
62            None => (String::new(), ""),
63        };
64        let (
65            hf_side_to_mid_db,
66            hf_reference_side_to_mid_db,
67            hf_narrowed_block_fraction,
68            hf_stereo_narrowed,
69        ) = match f.high_frequency_stereo {
70            Some(measurement) => (
71                format!("{:.2}", measurement.side_to_mid_db),
72                format!("{:.2}", measurement.reference_side_to_mid_db),
73                format!("{:.3}", measurement.narrowed_block_fraction),
74                measurement.narrowed.to_string(),
75            ),
76            None => (String::new(), String::new(), String::new(), String::new()),
77        };
78        let mut row = vec![
79            csv_text(&f.file_name),
80            csv_text(&f.format),
81            csv_text(f.codec.as_deref().unwrap_or_default()),
82            csv_text(f.badge.as_deref().unwrap_or_default()),
83            opt(f.bitrate_kbps),
84            f.sample_rate.to_string(),
85            opt(f.declared_bits),
86            opt(f.real_bit_depth),
87            format!("{:.3}", f.duration_secs),
88            f.size_bytes.to_string(),
89            csv_text(&f.detections.summary),
90            f.detections.upscaling.to_string(),
91            f.detections.upsampling.to_string(),
92            f.detections.transcoding.to_string(),
93            f.lattice_score
94                .map(|v| format!("{v:.4}"))
95                .unwrap_or_default(),
96            f.cutoff_hz.map(|v| format!("{v:.0}")).unwrap_or_default(),
97            f.cutoff_ratio
98                .map(|v| format!("{v:.3}"))
99                .unwrap_or_default(),
100            f.channels.to_string(),
101            opt_bool(f.fake_stereo),
102            f.phase_correlation
103                .map(|v| format!("{v:.3}"))
104                .unwrap_or_default(),
105            opt_bool(f.phase_inverted),
106            f.clipping.clipped.to_string(),
107            f.clipping.clip_events.to_string(),
108            format!("{:.2}", f.clipping.peak_dbfs),
109            format!("{:.2}", f.clipping.true_peak_dbtp),
110            f.integrated_lufs
111                .map(|v| format!("{v:.1}"))
112                .unwrap_or_default(),
113            opt(f.loudness_peaks.and_then(|p| p.momentary).map(|p| p.lufs)),
114            opt(f
115                .loudness_peaks
116                .and_then(|p| p.momentary)
117                .map(|p| p.start_secs)),
118            opt(f.loudness_peaks.and_then(|p| p.short_term).map(|p| p.lufs)),
119            opt(f
120                .loudness_peaks
121                .and_then(|p| p.short_term)
122                .map(|p| p.start_secs)),
123            f.loudness_range_lu
124                .map(|v| format!("{v:.1}"))
125                .unwrap_or_default(),
126            f.dr_db.map(|v| format!("{v:.1}")).unwrap_or_default(),
127            md5.to_string(),
128            f.bit_depth_evidence
129                .map(|e| match e.method {
130                    crate::analysis::bitdepth::BitDepthMethod::Stored => "Stored",
131                    crate::analysis::bitdepth::BitDepthMethod::NarrowGrid => "NarrowGrid",
132                })
133                .unwrap_or("")
134                .to_string(),
135            opt(f.bit_depth_evidence.map(|e| e.stored_bits)),
136            balance_db,
137            silent_channel.to_string(),
138            hf_side_to_mid_db,
139            hf_reference_side_to_mid_db,
140            hf_narrowed_block_fraction,
141            hf_stereo_narrowed,
142            opt(f.discontinuities.as_ref().map(|d| d.clicks.count)),
143            opt(f.discontinuities.as_ref().map(|d| d.dropouts.count)),
144            discontinuity_locations(f.discontinuities.as_ref().map(|d| &d.clicks)),
145            discontinuity_locations(f.discontinuities.as_ref().map(|d| &d.dropouts)),
146            opt(f.modified_unix),
147            f.dc_offset
148                .as_ref()
149                .map(|dc| dc.max_abs.to_string())
150                .unwrap_or_default(),
151            f.dc_offset
152                .as_ref()
153                .map(|dc| {
154                    dc.channel_means
155                        .iter()
156                        .map(ToString::to_string)
157                        .collect::<Vec<_>>()
158                        .join(";")
159                })
160                .unwrap_or_default(),
161        ];
162        row.extend(phase::values(f.local_phase.as_ref()));
163        row.push(csv_text(f.file_md5.as_deref().unwrap_or_default()));
164        row.push(csv_text(f.file_crc32.as_deref().unwrap_or_default()));
165        out.push_str(&row.join(","));
166        out.push('\n');
167    }
168    out
169}

171fn discontinuity_locations(
172    summary: Option<&crate::analysis::discontinuities::EventSummary>,
173) -> String {
174    summary
175        .map(|s| {
176            s.events
177                .iter()
178                .map(|event| {
179                    format!(
180                        "ch{}@{:.6}s/{:.6}s",
181                        event.channel, event.start_secs, event.duration_secs
182                    )
183                })
184                .collect::<Vec<_>>()
185                .join(";")
186        })
187        .unwrap_or_default()
188}

190// Imported reports and file names are untrusted text. Spreadsheet programs
191// interpret leading operators even inside quoted CSV fields; an apostrophe
192// keeps those cells textual while numeric measurement columns remain numeric.
193fn csv_text(s: &str) -> String {
194    let dangerous = matches!(s.chars().next(), Some('\t' | '\r' | '\n'))
195        || matches!(s.trim_start().chars().next(), Some('=' | '+' | '-' | '@'));
196    if dangerous {
197        csv_escape(&format!("'{s}"))
198    } else {
199        csv_escape(s)
200    }
201}
202
203fn csv_escape(s: &str) -> String {
204    if s.contains(',') || s.contains('"') || s.contains('\n') || s.contains('\r') {
205        format!("\"{}\"", s.replace('"', "\"\""))
206    } else {
207        s.to_string()
208    }
209}

211fn opt<T: std::fmt::Display>(v: Option<T>) -> String {
212    v.map(|x| x.to_string()).unwrap_or_default()
213}
214
215fn opt_bool(v: Option<bool>) -> String {
216    v.map(|x| x.to_string()).unwrap_or_default()
217}
