2//! round-trips the full [`FolderReport`] — every field the app computed,
3//! nested detections included — so a saved analysis can be dropped back onto
4//! the window later and rendered without re-decoding a single audio file.

19/// Same stem, JSON sibling — `save` always writes both from one dialog pick.
20pub const JSON_FILE_NAME: &str = "FlacCompagnon.json";
21
22/// Marker written into every JSON report so a dropped file can be recognized
23/// (and rejected with a clear message) before attempting to parse it as one.
24const JSON_FORMAT_MARKER: &str = "flaccompagnon-report";
25/// Bumped if the JSON shape ever changes incompatibly.
26const JSON_FORMAT_VERSION: u32 = 1;
27
28// A report is exchanged between app builds, the CLI and other Rust software.
29// Their optimized floating-point code may differ by one final IEEE-754 bit
30// while yielding the same measurement, which used to make two analyses of the
31// same file look different in a JSON diff. Nine decimal places are far beyond
32// the display precision and preserve sub-microsecond event positions.
33const JSON_FLOAT_SCALE: f64 = 1_000_000_000.0;
34
35/// On-disk shape of the JSON report: the marker/version let a dropped file be
36/// identified and versioned independently of [`FolderReport`]'s own shape.
37#[derive(Debug, Clone, Serialize, Deserialize)]
38struct JsonReport {
39    format: String,
40    version: u32,
41    report: FolderReport,
42}
43
44/// Build the JSON text for a folder report (pretty-printed, wrapped with a
45/// format marker and version — see `JsonReport`).
46///
47/// # Complete, always
48///
49/// Unlike the CSV, this is a full serialization of every [`crate::FileAnalysis`]
50/// field, in the order the files are given (which is the table's display
51/// order — the frontend hands them over already sorted). **What the user has
52/// chosen to show or hide in the table has no effect here**: a hidden column's
53/// value is still in the JSON, because this file is what a saved analysis is
54/// reloaded from, and a reload that lost whatever happened to be hidden at
55/// save time would be a data-loss bug rather than a formatting choice.
56pub fn build_json(report: &FolderReport) -> serde_json::Result<String> {
57    let mut report = report.clone();
58    stabilize_report_floats(&mut report);
59    let wrapped = JsonReport {
60        format: JSON_FORMAT_MARKER.to_string(),
61        version: JSON_FORMAT_VERSION,
62        report,
63    };
64    serde_json::to_string_pretty(&wrapped)
65}
66
67/// Round the `f64` fields that cross the persisted report boundary.
68///
69/// The analysis keeps its full in-memory precision, while JSON becomes stable
70/// across debug and release builds and across clients that use the same core
71/// version. This works on the typed fields rather than generic JSON numbers:
72/// serializing an `f32` through `serde_json::Value` would expand it to a longer
73/// `f64` spelling despite its value already being stable.
74fn stabilize_report_floats(report: &mut FolderReport) {
75    for file in &mut report.files {
76        file.duration_secs = stable_float(file.duration_secs);
77        file.cutoff_hz = file.cutoff_hz.map(stable_float);
78        file.cutoff_ratio = file.cutoff_ratio.map(stable_float);
79
80        if let Some(offset) = &mut file.dc_offset {
81            offset.channel_means.iter_mut().for_each(|mean| {
82                *mean = stable_float(*mean);
83            });
84            offset.max_abs = stable_float(offset.max_abs);
85        }
86
87        if let Some(phase) = &mut file.local_phase {
88            phase.window_secs = stable_float(phase.window_secs);
89            phase.hop_secs = stable_float(phase.hop_secs);
90            stabilize_phase_summary(phase.broadband.as_mut());
91            for band in &mut phase.bands {
92                band.low_hz = stable_float(band.low_hz);
93                band.high_hz = stable_float(band.high_hz);
94                stabilize_phase_summary(band.summary.as_mut());
95            }
96        }
97
98        if let Some(peaks) = &mut file.loudness_peaks {
99            if let Some(peak) = &mut peaks.momentary {
100                peak.start_secs = stable_float(peak.start_secs);
101            }
102            if let Some(peak) = &mut peaks.short_term {
103                peak.start_secs = stable_float(peak.start_secs);
104            }
105        }
106
107        if let Some(discontinuities) = &mut file.discontinuities {
108            for event in discontinuities
109                .clicks
110                .events
111                .iter_mut()
112                .chain(discontinuities.dropouts.events.iter_mut())
113            {
114                event.start_secs = stable_float(event.start_secs);
115                event.duration_secs = stable_float(event.duration_secs);
116            }
117        }
118    }
119}
120
121fn stabilize_phase_summary(summary: Option<&mut crate::analysis::local_phase::PhaseSummary>) {
122    if let Some(summary) = summary {
123        summary.minimum_start_secs = stable_float(summary.minimum_start_secs);
124        summary.opposed_fraction = stable_float(summary.opposed_fraction);
125    }
126}
127
128fn stable_float(value: f64) -> f64 {
129    let rounded = (value * JSON_FLOAT_SCALE).round() / JSON_FLOAT_SCALE;
130    if rounded.is_finite() {
131        rounded
132    } else {
133        value
134    }
135}

137/// Parse a previously-saved JSON report back into a [`FolderReport`], so it
138/// can be rendered without re-analyzing any audio. Rejects JSON that doesn't
139/// carry FlacCompagnon's format marker, with a message meant for end users
140/// (someone dropped an unrelated `.json` file).
141///
142/// ```
143/// use flaccompagnon_core::{report, FolderReport};
144///
145/// let empty = FolderReport { root: "/music".into(), files: vec![], has_flac: false };
146///
147/// // build_json / parse_json round-trip the whole report.
148/// let json = report::build_json(&empty).unwrap();
149/// let back = report::parse_json(&json).unwrap();
150/// assert_eq!(back.root, "/music");
151///
152/// // Unrelated JSON is refused with a message meant for the user.
153/// assert!(report::parse_json(r#"{"hello":"world"}"#).is_err());
154/// ```
155pub fn parse_json(text: &str) -> Result<FolderReport, String> {
156    let wrapped: JsonReport = serde_json::from_str(text)
157        .map_err(|e| format!("This doesn't look like a FlacCompagnon JSON report ({e})."))?;
158    if wrapped.format != JSON_FORMAT_MARKER {
159        return Err("This JSON file wasn't exported by FlacCompagnon.".to_string());
160    }
161    if wrapped.version > JSON_FORMAT_VERSION {
162        return Err(
163            "This report was saved by a newer version of FlacCompagnon — please update the app."
164                .to_string(),
165        );
166    }
167    if wrapped.report.files.iter().any(|file| {
168        !matches!(
169            file.detections.summary.as_str(),
170            "Clean" | "Flagged" | "Not analyzed"
171        )
172    }) {
173        return Err(
174            "This report uses an unsupported detection status. Reanalyze the audio files.".into(),
175        );
176    }
177    Ok(wrapped.report)
178}
