1//! Report generation: a spreadsheet-friendly CSV, and a re-importable JSON that
2//! round-trips the full [`FolderReport`] — every field the app computed,
3//! nested detections included — so a saved analysis can be dropped back onto
4//! the window later and rendered without re-decoding a single audio file.

15/// Default file name suggested when saving a report.
16pub const CSV_FILE_NAME: &str = "FlacCompagnon.csv";
17/// Same stem, JSON sibling — `save` always writes both from one dialog pick.
18pub const JSON_FILE_NAME: &str = "FlacCompagnon.json";
19
20/// Marker written into every JSON report so a dropped file can be recognized
21/// (and rejected with a clear message) before attempting to parse it as one.
22const JSON_FORMAT_MARKER: &str = "flaccompagnon-report";
23/// Bumped if the JSON shape ever changes incompatibly.
24const JSON_FORMAT_VERSION: u32 = 1;
25
26// A report is exchanged between app builds, the CLI and other Rust software.
27// Their optimized floating-point code may differ by one final IEEE-754 bit
28// while yielding the same measurement, which used to make two analyses of the
29// same file look different in a JSON diff. Nine decimal places are far beyond
30// the display precision and preserve sub-microsecond event positions.
31const JSON_FLOAT_SCALE: f64 = 1_000_000_000.0;
32
33/// On-disk shape of the JSON report: the marker/version let a dropped file be
34/// identified and versioned independently of [`FolderReport`]'s own shape.
35#[derive(Debug, Clone, Serialize, Deserialize)]
36struct JsonReport {
37    format: String,
38    version: u32,
39    report: FolderReport,
40}

42/// Build the CSV text for a folder report.
43///
44/// Column order mirrors the results table left-to-right (`ResultsTable.tsx`'s
45/// `headers`), not the order fields were originally added to this struct —
46/// so a column's position here means the same thing it means on screen.
47/// Fields the table folds into a single cell (Detections: `status` +
48/// `upscaling`/`upsampling`/`transcoding`/`lattice_score`; Clipping: `clipped` +
49/// `clip_events`/`peak_dbfs`) are grouped together at that cell's position
50/// rather than split across the row. No column is dropped — every field the
51/// old order exported is still here, just reordered.
52///
53/// `file_md5` and `file_crc32` are the exception to that mirroring: they are
54/// appended at the end because their columns are hidden by default in the
55/// table, so there is no on-screen position to mirror.
56///
57/// `codec` and `bitrate_kbps` are deliberately slotted mid-row — right after
58/// `format` and right before `sample_rate` respectively — rather than at the
59/// end, on the maintainer's explicit request, even though the table's own
60/// column order is otherwise just a display preference (`useColumnPrefs.ts`)
61/// this file doesn't follow. This *is* a breaking change for any script or
62/// spreadsheet already reading the CSV by position: those two columns moved.
63/// `modified_unix` stays appended at the end — nobody asked to move that one,
64/// and there's no natural mid-row spot for it the way there is for the other
65/// two.
66pub fn build_csv(report: &FolderReport) -> String {
67    let mut out = String::new();
68    out.push_str(
69        "file,format,codec,badge,bitrate_kbps,sample_rate,declared_bits,real_bit_depth,\
70         duration_s,size_bytes,status,upscaling,upsampling,transcoding,lattice_score,cutoff_hz,\
71         cutoff_ratio,channels,fake_stereo,phase_correlation,phase_inverted,clipped,clip_events,peak_dbfs,true_peak_dbtp,integrated_lufs,max_momentary_lufs,momentary_max_start_s,max_short_term_lufs,short_term_max_start_s,loudness_range_lu,dr_db,\
72         md5,bit_depth_method,stored_bits,balance_right_minus_left_db,balance_silent_channel,hf_side_to_mid_db,hf_reference_side_to_mid_db,hf_narrowed_block_fraction,hf_stereo_narrowed,suspected_clicks,suspected_dropouts,click_locations,dropout_locations,modified_unix,dc_offset_max_abs,dc_offset_channel_means,",
73    );
74    out.push_str(&phase::headers().join(","));
75    out.push_str(",file_md5,file_crc32\n");
76    for f in &report.files {
77        let md5 = f
78            .flac_md5
79            .as_ref()
80            .map(|m| match m {
81                FlacMd5Status::NoSignature => "none",
82                FlacMd5Status::Present => "present",
83                FlacMd5Status::Match => "ok",
84                FlacMd5Status::Mismatch => "mismatch",
85                FlacMd5Status::Error(_) => "error",
86            })
87            .unwrap_or("");
88        use crate::analysis::stereo::StereoBalance;
89        let (balance_db, silent_channel) = match f.stereo_balance {
90            Some(StereoBalance::Measured {
91                right_minus_left_db,
92            }) => (format!("{right_minus_left_db:.2}"), ""),
93            Some(StereoBalance::LeftSilent) => (String::new(), "left"),
94            Some(StereoBalance::RightSilent) => (String::new(), "right"),
95            None => (String::new(), ""),
96        };
97        let (
98            hf_side_to_mid_db,
99            hf_reference_side_to_mid_db,
100            hf_narrowed_block_fraction,
101            hf_stereo_narrowed,
102        ) = match f.high_frequency_stereo {
103            Some(measurement) => (
104                format!("{:.2}", measurement.side_to_mid_db),
105                format!("{:.2}", measurement.reference_side_to_mid_db),
106                format!("{:.3}", measurement.narrowed_block_fraction),
107                measurement.narrowed.to_string(),
108            ),
109            None => (String::new(), String::new(), String::new(), String::new()),
110        };
111        let mut row = vec![
112            csv_escape(&f.file_name),
113            f.format.clone(),
114            f.codec.clone().unwrap_or_default(),
115            f.badge.clone().unwrap_or_default(),
116            opt(f.bitrate_kbps),
117            f.sample_rate.to_string(),
118            opt(f.declared_bits),
119            opt(f.real_bit_depth),
120            format!("{:.3}", f.duration_secs),
121            f.size_bytes.to_string(),
122            f.detections.summary.clone(),
123            f.detections.upscaling.to_string(),
124            f.detections.upsampling.to_string(),
125            f.detections.transcoding.to_string(),
126            f.lattice_score
127                .map(|v| format!("{v:.4}"))
128                .unwrap_or_default(),
129            f.cutoff_hz.map(|v| format!("{v:.0}")).unwrap_or_default(),
130            f.cutoff_ratio
131                .map(|v| format!("{v:.3}"))
132                .unwrap_or_default(),
133            f.channels.to_string(),
134            opt_bool(f.fake_stereo),
135            f.phase_correlation
136                .map(|v| format!("{v:.3}"))
137                .unwrap_or_default(),
138            opt_bool(f.phase_inverted),
139            f.clipping.clipped.to_string(),
140            f.clipping.clip_events.to_string(),
141            format!("{:.2}", f.clipping.peak_dbfs),
142            format!("{:.2}", f.clipping.true_peak_dbtp),
143            f.integrated_lufs
144                .map(|v| format!("{v:.1}"))
145                .unwrap_or_default(),
146            opt(f.loudness_peaks.and_then(|p| p.momentary).map(|p| p.lufs)),
147            opt(f
148                .loudness_peaks
149                .and_then(|p| p.momentary)
150                .map(|p| p.start_secs)),
151            opt(f.loudness_peaks.and_then(|p| p.short_term).map(|p| p.lufs)),
152            opt(f
153                .loudness_peaks
154                .and_then(|p| p.short_term)
155                .map(|p| p.start_secs)),
156            f.loudness_range_lu
157                .map(|v| format!("{v:.1}"))
158                .unwrap_or_default(),
159            f.dr_db.map(|v| format!("{v:.1}")).unwrap_or_default(),
160            md5.to_string(),
161            f.bit_depth_evidence
162                .map(|e| match e.method {
163                    crate::analysis::bitdepth::BitDepthMethod::Stored => "Stored",
164                    crate::analysis::bitdepth::BitDepthMethod::NarrowGrid => "NarrowGrid",
165                })
166                .unwrap_or("")
167                .to_string(),
168            opt(f.bit_depth_evidence.map(|e| e.stored_bits)),
169            balance_db,
170            silent_channel.to_string(),
171            hf_side_to_mid_db,
172            hf_reference_side_to_mid_db,
173            hf_narrowed_block_fraction,
174            hf_stereo_narrowed,
175            opt(f.discontinuities.as_ref().map(|d| d.clicks.count)),
176            opt(f.discontinuities.as_ref().map(|d| d.dropouts.count)),
177            discontinuity_locations(f.discontinuities.as_ref().map(|d| &d.clicks)),
178            discontinuity_locations(f.discontinuities.as_ref().map(|d| &d.dropouts)),
179            opt(f.modified_unix),
180            f.dc_offset
181                .as_ref()
182                .map(|dc| dc.max_abs.to_string())
183                .unwrap_or_default(),
184            f.dc_offset
185                .as_ref()
186                .map(|dc| {
187                    dc.channel_means
188                        .iter()
189                        .map(ToString::to_string)
190                        .collect::<Vec<_>>()
191                        .join(";")
192                })
193                .unwrap_or_default(),
194        ];
195        row.extend(phase::values(f.local_phase.as_ref()));
196        row.push(f.file_md5.clone().unwrap_or_default());
197        row.push(f.file_crc32.clone().unwrap_or_default());
198        out.push_str(&row.join(","));
199        out.push('\n');
200    }
201    out
202}

204fn discontinuity_locations(
205    summary: Option<&crate::analysis::discontinuities::EventSummary>,
206) -> String {
207    summary
208        .map(|s| {
209            s.events
210                .iter()
211                .map(|event| {
212                    format!(
213                        "ch{}@{:.6}s/{:.6}s",
214                        event.channel, event.start_secs, event.duration_secs
215                    )
216                })
217                .collect::<Vec<_>>()
218                .join(";")
219        })
220        .unwrap_or_default()
221}

229/// Build the JSON text for a folder report (pretty-printed, wrapped with a
230/// format marker and version — see `JsonReport`).
231///
232/// # Complete, always
233///
234/// Unlike the CSV, this is a full serialization of every [`crate::FileAnalysis`]
235/// field, in the order the files are given (which is the table's display
236/// order — the frontend hands them over already sorted). **What the user has
237/// chosen to show or hide in the table has no effect here**: a hidden column's
238/// value is still in the JSON, because this file is what a saved analysis is
239/// reloaded from, and a reload that lost whatever happened to be hidden at
240/// save time would be a data-loss bug rather than a formatting choice.
241pub fn build_json(report: &FolderReport) -> serde_json::Result<String> {
242    let mut report = report.clone();
243    stabilize_report_floats(&mut report);
244    let wrapped = JsonReport {
245        format: JSON_FORMAT_MARKER.to_string(),
246        version: JSON_FORMAT_VERSION,
247        report,
248    };
249    serde_json::to_string_pretty(&wrapped)
250}

252/// Round the `f64` fields that cross the persisted report boundary.
253///
254/// The analysis keeps its full in-memory precision, while JSON becomes stable
255/// across debug and release builds and across clients that use the same core
256/// version. This works on the typed fields rather than generic JSON numbers:
257/// serializing an `f32` through `serde_json::Value` would expand it to a longer
258/// `f64` spelling despite its value already being stable.
259fn stabilize_report_floats(report: &mut FolderReport) {
260    for file in &mut report.files {
261        file.duration_secs = stable_float(file.duration_secs);
262        file.cutoff_hz = file.cutoff_hz.map(stable_float);
263        file.cutoff_ratio = file.cutoff_ratio.map(stable_float);
264
265        if let Some(offset) = &mut file.dc_offset {
266            offset.channel_means.iter_mut().for_each(|mean| {
267                *mean = stable_float(*mean);
268            });
269            offset.max_abs = stable_float(offset.max_abs);
270        }
271
272        if let Some(phase) = &mut file.local_phase {
273            phase.window_secs = stable_float(phase.window_secs);
274            phase.hop_secs = stable_float(phase.hop_secs);
275            stabilize_phase_summary(phase.broadband.as_mut());
276            for band in &mut phase.bands {
277                band.low_hz = stable_float(band.low_hz);
278                band.high_hz = stable_float(band.high_hz);
279                stabilize_phase_summary(band.summary.as_mut());
280            }
281        }
282
283        if let Some(peaks) = &mut file.loudness_peaks {
284            if let Some(peak) = &mut peaks.momentary {
285                peak.start_secs = stable_float(peak.start_secs);
286            }
287            if let Some(peak) = &mut peaks.short_term {
288                peak.start_secs = stable_float(peak.start_secs);
289            }
290        }
291
292        if let Some(discontinuities) = &mut file.discontinuities {
293            for event in discontinuities
294                .clicks
295                .events
296                .iter_mut()
297                .chain(discontinuities.dropouts.events.iter_mut())
298            {
299                event.start_secs = stable_float(event.start_secs);
300                event.duration_secs = stable_float(event.duration_secs);
301            }
302        }
303    }
304}

306fn stabilize_phase_summary(summary: Option<&mut crate::analysis::local_phase::PhaseSummary>) {
307    if let Some(summary) = summary {
308        summary.minimum_start_secs = stable_float(summary.minimum_start_secs);
309        summary.opposed_fraction = stable_float(summary.opposed_fraction);
310    }
311}

313fn stable_float(value: f64) -> f64 {
314    let rounded = (value * JSON_FLOAT_SCALE).round() / JSON_FLOAT_SCALE;
315    if rounded.is_finite() {
316        rounded
317    } else {
318        value
319    }
320}

322/// Write the JSON report to `dest`.
323pub fn write_json(dest: &Path, report: &FolderReport) -> std::io::Result<()> {
324    let text =
325        build_json(report).map_err(|e| std::io::Error::new(std::io::ErrorKind::InvalidData, e))?;
326    let mut file = std::fs::File::create(dest)?;
327    file.write_all(text.as_bytes())
328}

330/// Parse a previously-saved JSON report back into a [`FolderReport`], so it
331/// can be rendered without re-analyzing any audio. Rejects JSON that doesn't
332/// carry FlacCompagnon's format marker, with a message meant for end users
333/// (someone dropped an unrelated `.json` file).
334///
335/// ```
336/// use flaccompagnon_core::{report, FolderReport};
337///
338/// let empty = FolderReport { root: "/music".into(), files: vec![], has_flac: false };
339///
340/// // build_json / parse_json round-trip the whole report.
341/// let json = report::build_json(&empty).unwrap();
342/// let back = report::parse_json(&json).unwrap();
343/// assert_eq!(back.root, "/music");
344///
345/// // Unrelated JSON is refused with a message meant for the user.
346/// assert!(report::parse_json(r#"{"hello":"world"}"#).is_err());
347/// ```
348pub fn parse_json(text: &str) -> Result<FolderReport, String> {
349    let wrapped: JsonReport = serde_json::from_str(text)
350        .map_err(|e| format!("This doesn't look like a FlacCompagnon JSON report ({e})."))?;
351    if wrapped.format != JSON_FORMAT_MARKER {
352        return Err("This JSON file wasn't exported by FlacCompagnon.".to_string());
353    }
354    if wrapped.version > JSON_FORMAT_VERSION {
355        return Err(
356            "This report was saved by a newer version of FlacCompagnon — please update the app."
357                .to_string(),
358        );
359    }
360    if wrapped.report.files.iter().any(|file| {
361        !matches!(
362            file.detections.summary.as_str(),
363            "Clean" | "Flagged" | "Not analyzed"
364        )
365    }) {
366        return Err(
367            "This report uses an unsupported detection status. Reanalyze the audio files.".into(),
368        );
369    }
370    Ok(wrapped.report)
371}

373fn csv_escape(s: &str) -> String {
374    if s.contains(',') || s.contains('"') || s.contains('\n') {
375        format!("\"{}\"", s.replace('"', "\"\""))
376    } else {
377        s.to_string()
378    }
379}

381fn opt<T: std::fmt::Display>(v: Option<T>) -> String {
382    v.map(|x| x.to_string()).unwrap_or_default()
