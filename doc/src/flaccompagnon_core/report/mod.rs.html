1//! Report generation: a spreadsheet-friendly CSV, and a re-importable JSON that
2//! round-trips the full [`FolderReport`] — every field the app computed,
3//! nested detections included — so a saved analysis can be dropped back onto
4//! the window later and rendered without re-decoding a single audio file.

15/// Default file name suggested when saving a report.
16pub const CSV_FILE_NAME: &str = "FlacCompagnon.csv";
17/// Same stem, JSON sibling — `save` always writes both from one dialog pick.
18pub const JSON_FILE_NAME: &str = "FlacCompagnon.json";

20/// Marker written into every JSON report so a dropped file can be recognized
21/// (and rejected with a clear message) before attempting to parse it as one.
22const JSON_FORMAT_MARKER: &str = "flaccompagnon-report";
23/// Bumped if the JSON shape ever changes incompatibly.
24const JSON_FORMAT_VERSION: u32 = 1;
25
26/// On-disk shape of the JSON report: the marker/version let a dropped file be
27/// identified and versioned independently of [`FolderReport`]'s own shape.
28#[derive(Debug, Clone, Serialize, Deserialize)]
29struct JsonReport {
30    format: String,
31    version: u32,
32    report: FolderReport,
33}

35/// Build the CSV text for a folder report.
36///
37/// Column order mirrors the results table left-to-right (`ResultsTable.tsx`'s
38/// `headers`), not the order fields were originally added to this struct —
39/// so a column's position here means the same thing it means on screen.
40/// Fields the table folds into a single cell (Detections: `status` +
41/// `upscaling`/`upsampling`/`transcoding`/`lattice_score`; Clipping: `clipped` +
42/// `clip_events`/`peak_dbfs`) are grouped together at that cell's position
43/// rather than split across the row. No column is dropped — every field the
44/// old order exported is still here, just reordered.
45///
46/// `file_md5` and `file_crc32` are the exception to that mirroring: they are
47/// appended at the end because their columns are hidden by default in the
48/// table, so there is no on-screen position to mirror.
49///
50/// `codec` and `bitrate_kbps` are deliberately slotted mid-row — right after
51/// `format` and right before `sample_rate` respectively — rather than at the
52/// end, on the maintainer's explicit request, even though the table's own
53/// column order is otherwise just a display preference (`useColumnPrefs.ts`)
54/// this file doesn't follow. This *is* a breaking change for any script or
55/// spreadsheet already reading the CSV by position: those two columns moved.
56/// `modified_unix` stays appended at the end — nobody asked to move that one,
57/// and there's no natural mid-row spot for it the way there is for the other
58/// two.
59pub fn build_csv(report: &FolderReport) -> String {
60    let mut out = String::new();
61    out.push_str(
62        "file,format,codec,badge,bitrate_kbps,sample_rate,declared_bits,real_bit_depth,\
63         duration_s,size_bytes,status,upscaling,upsampling,transcoding,lattice_score,cutoff_hz,\
64         cutoff_ratio,channels,fake_stereo,phase_correlation,phase_inverted,clipped,clip_events,peak_dbfs,true_peak_dbtp,integrated_lufs,max_momentary_lufs,momentary_max_start_s,max_short_term_lufs,short_term_max_start_s,loudness_range_lu,dr_db,\
65         md5,bit_depth_method,stored_bits,balance_right_minus_left_db,balance_silent_channel,hf_side_to_mid_db,hf_reference_side_to_mid_db,hf_narrowed_block_fraction,hf_stereo_narrowed,suspected_clicks,suspected_dropouts,click_locations,dropout_locations,modified_unix,dc_offset_max_abs,dc_offset_channel_means,",
66    );
67    out.push_str(&phase::headers().join(","));
68    out.push_str(",file_md5,file_crc32\n");
69    for f in &report.files {
70        let md5 = f
71            .flac_md5
72            .as_ref()
73            .map(|m| match m {
74                FlacMd5Status::NoSignature => "none",
75                FlacMd5Status::Present => "present",
76                FlacMd5Status::Match => "ok",
77                FlacMd5Status::Mismatch => "mismatch",
78                FlacMd5Status::Error(_) => "error",
79            })
80            .unwrap_or("");
81        use crate::analysis::stereo::StereoBalance;
82        let (balance_db, silent_channel) = match f.stereo_balance {
83            Some(StereoBalance::Measured {
84                right_minus_left_db,
85            }) => (format!("{right_minus_left_db:.2}"), ""),
86            Some(StereoBalance::LeftSilent) => (String::new(), "left"),
87            Some(StereoBalance::RightSilent) => (String::new(), "right"),
88            None => (String::new(), ""),
89        };
90        let (
91            hf_side_to_mid_db,
92            hf_reference_side_to_mid_db,
93            hf_narrowed_block_fraction,
94            hf_stereo_narrowed,
95        ) = match f.high_frequency_stereo {
96            Some(measurement) => (
97                format!("{:.2}", measurement.side_to_mid_db),
98                format!("{:.2}", measurement.reference_side_to_mid_db),
99                format!("{:.3}", measurement.narrowed_block_fraction),
100                measurement.narrowed.to_string(),
101            ),
102            None => (String::new(), String::new(), String::new(), String::new()),
103        };
104        let mut row = vec![
105            csv_escape(&f.file_name),
106            f.format.clone(),
107            f.codec.clone().unwrap_or_default(),
108            f.badge.clone().unwrap_or_default(),
109            opt(f.bitrate_kbps),
110            f.sample_rate.to_string(),
111            opt(f.declared_bits),
112            opt(f.real_bit_depth),
113            format!("{:.3}", f.duration_secs),
114            f.size_bytes.to_string(),
115            f.detections.summary.clone(),
116            f.detections.upscaling.to_string(),
117            f.detections.upsampling.to_string(),
118            f.detections.transcoding.to_string(),
119            f.lattice_score
120                .map(|v| format!("{v:.4}"))
121                .unwrap_or_default(),
122            f.cutoff_hz.map(|v| format!("{v:.0}")).unwrap_or_default(),
123            f.cutoff_ratio
124                .map(|v| format!("{v:.3}"))
125                .unwrap_or_default(),
126            f.channels.to_string(),
127            opt_bool(f.fake_stereo),
128            f.phase_correlation
129                .map(|v| format!("{v:.3}"))
130                .unwrap_or_default(),
131            opt_bool(f.phase_inverted),
132            f.clipping.clipped.to_string(),
133            f.clipping.clip_events.to_string(),
134            format!("{:.2}", f.clipping.peak_dbfs),
135            format!("{:.2}", f.clipping.true_peak_dbtp),
136            f.integrated_lufs
137                .map(|v| format!("{v:.1}"))
138                .unwrap_or_default(),
139            opt(f.loudness_peaks.and_then(|p| p.momentary).map(|p| p.lufs)),
140            opt(f
141                .loudness_peaks
142                .and_then(|p| p.momentary)
143                .map(|p| p.start_secs)),
144            opt(f.loudness_peaks.and_then(|p| p.short_term).map(|p| p.lufs)),
145            opt(f
146                .loudness_peaks
147                .and_then(|p| p.short_term)
148                .map(|p| p.start_secs)),
149            f.loudness_range_lu
150                .map(|v| format!("{v:.1}"))
151                .unwrap_or_default(),
152            f.dr_db.map(|v| format!("{v:.1}")).unwrap_or_default(),
153            md5.to_string(),
154            f.bit_depth_evidence
155                .map(|e| match e.method {
156                    crate::analysis::bitdepth::BitDepthMethod::Stored => "Stored",
157                    crate::analysis::bitdepth::BitDepthMethod::NarrowGrid => "NarrowGrid",
158                })
159                .unwrap_or("")
160                .to_string(),
161            opt(f.bit_depth_evidence.map(|e| e.stored_bits)),
162            balance_db,
163            silent_channel.to_string(),
164            hf_side_to_mid_db,
165            hf_reference_side_to_mid_db,
166            hf_narrowed_block_fraction,
167            hf_stereo_narrowed,
168            opt(f.discontinuities.as_ref().map(|d| d.clicks.count)),
169            opt(f.discontinuities.as_ref().map(|d| d.dropouts.count)),
170            discontinuity_locations(f.discontinuities.as_ref().map(|d| &d.clicks)),
171            discontinuity_locations(f.discontinuities.as_ref().map(|d| &d.dropouts)),
172            opt(f.modified_unix),
173            f.dc_offset
174                .as_ref()
175                .map(|dc| dc.max_abs.to_string())
176                .unwrap_or_default(),
177            f.dc_offset
178                .as_ref()
179                .map(|dc| {
180                    dc.channel_means
181                        .iter()
182                        .map(ToString::to_string)
183                        .collect::<Vec<_>>()
184                        .join(";")
185                })
186                .unwrap_or_default(),
187        ];
188        row.extend(phase::values(f.local_phase.as_ref()));
189        row.push(f.file_md5.clone().unwrap_or_default());
190        row.push(f.file_crc32.clone().unwrap_or_default());
191        out.push_str(&row.join(","));
192        out.push('\n');
193    }
194    out
195}

197fn discontinuity_locations(
198    summary: Option<&crate::analysis::discontinuities::EventSummary>,
199) -> String {
200    summary
201        .map(|s| {
202            s.events
203                .iter()
204                .map(|event| {
205                    format!(
206                        "ch{}@{:.6}s/{:.6}s",
207                        event.channel, event.start_secs, event.duration_secs
208                    )
209                })
210                .collect::<Vec<_>>()
211                .join(";")
212        })
213        .unwrap_or_default()
214}

216/// Write the CSV report to `dest`.
217pub fn write_csv(dest: &Path, report: &FolderReport) -> std::io::Result<()> {
218    let mut file = std::fs::File::create(dest)?;
219    file.write_all(build_csv(report).as_bytes())
220}

222/// Build the JSON text for a folder report (pretty-printed, wrapped with a
223/// format marker and version — see `JsonReport`).
224///
225/// # Complete, always
226///
227/// Unlike the CSV, this is a full serialization of every [`crate::FileAnalysis`]
228/// field, in the order the files are given (which is the table's display
229/// order — the frontend hands them over already sorted). **What the user has
230/// chosen to show or hide in the table has no effect here**: a hidden column's
231/// value is still in the JSON, because this file is what a saved analysis is
232/// reloaded from, and a reload that lost whatever happened to be hidden at
233/// save time would be a data-loss bug rather than a formatting choice.
234pub fn build_json(report: &FolderReport) -> serde_json::Result<String> {
235    let wrapped = JsonReport {
236        format: JSON_FORMAT_MARKER.to_string(),
237        version: JSON_FORMAT_VERSION,
238        report: report.clone(),
239    };
240    serde_json::to_string_pretty(&wrapped)
241}
242
243/// Write the JSON report to `dest`.
244pub fn write_json(dest: &Path, report: &FolderReport) -> std::io::Result<()> {
245    let text =
246        build_json(report).map_err(|e| std::io::Error::new(std::io::ErrorKind::InvalidData, e))?;
247    let mut file = std::fs::File::create(dest)?;
248    file.write_all(text.as_bytes())
249}

251/// Parse a previously-saved JSON report back into a [`FolderReport`], so it
252/// can be rendered without re-analyzing any audio. Rejects JSON that doesn't
253/// carry FlacCompagnon's format marker, with a message meant for end users
254/// (someone dropped an unrelated `.json` file).
255///
256/// ```
257/// use flaccompagnon_core::{report, FolderReport};
258///
259/// let empty = FolderReport { root: "/music".into(), files: vec![], has_flac: false };
260///
261/// // build_json / parse_json round-trip the whole report.
262/// let json = report::build_json(&empty).unwrap();
263/// let back = report::parse_json(&json).unwrap();
264/// assert_eq!(back.root, "/music");
265///
266/// // Unrelated JSON is refused with a message meant for the user.
267/// assert!(report::parse_json(r#"{"hello":"world"}"#).is_err());
268/// ```
269pub fn parse_json(text: &str) -> Result<FolderReport, String> {
270    let wrapped: JsonReport = serde_json::from_str(text)
271        .map_err(|e| format!("This doesn't look like a FlacCompagnon JSON report ({e})."))?;
272    if wrapped.format != JSON_FORMAT_MARKER {
273        return Err("This JSON file wasn't exported by FlacCompagnon.".to_string());
274    }
275    if wrapped.version > JSON_FORMAT_VERSION {
276        return Err(
277            "This report was saved by a newer version of FlacCompagnon — please update the app."
278                .to_string(),
279        );
280    }
281    if wrapped.report.files.iter().any(|file| {
282        !matches!(
283            file.detections.summary.as_str(),
284            "Clean" | "Flagged" | "Not analyzed"
285        )
286    }) {
287        return Err(
288            "This report uses an unsupported detection status. Reanalyze the audio files.".into(),
289        );
290    }
291    Ok(wrapped.report)
292}

294fn csv_escape(s: &str) -> String {
295    if s.contains(',') || s.contains('"') || s.contains('\n') {
296        format!("\"{}\"", s.replace('"', "\"\""))
297    } else {
298        s.to_string()
299    }
300}

302fn opt<T: std::fmt::Display>(v: Option<T>) -> String {
303    v.map(|x| x.to_string()).unwrap_or_default()
304}

306fn opt_bool(v: Option<bool>) -> String {
307    v.map(|x| x.to_string()).unwrap_or_default()
