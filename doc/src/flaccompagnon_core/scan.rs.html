8/// Audio file extensions FlacCompagnon will attempt to *analyze* — which is
9/// not the same as the ones it can currently decode.
10///
11/// `.opus` is listed even though there is no Opus decoder yet (see
12/// [`crate::decode`]). Leaving it out was worse, not safer: `.ogg` and `.oga`
13/// are here, an Ogg stream can just as well carry Opus, so the same audio was
14/// accepted under one name and silently refused at the drop under another. A
15/// listed file that fails with "Opus decoding is not supported yet" tells the
16/// user what is going on; a file that vanishes on drop tells them the app is
17/// broken. The day a decoder is wired in, this entry needs no change.
18pub const SUPPORTED_EXTENSIONS: &[&str] = &[
19    "flac", "wav", "wave", "aif", "aiff", "aifc", "alac", "m4a", "mp4", "caf", "ogg", "oga",
20    "opus", "mp3", "aac", "dsf", "dff",
21];
22
23/// Folder names used for generated spectrograms. Files inside one are skipped
24/// so a second scan doesn't try to analyze the app's own PNG output — and, more
25/// to the point, so re-scanning a folder never grows the file list.
26///
27/// Two names, because only the first is written any more: `spectres` was the
28/// original (French) name, and libraries scanned before the rename still hold
29/// those folders. Dropping it from this list would not corrupt anything, but
30/// it would silently change the behaviour of every folder a long-time user
31/// already has — the kind of regression nobody thinks to test for.
32const GENERATED_DIRS: [&str; 2] = ["spectrograms", "spectres"];
33
34/// Returns `true` if `path` has an extension FlacCompagnon knows how to decode.
35///
36/// The check is on the extension only and is case-insensitive; the *real*
37/// container is identified later from the file's magic bytes (which is how a
38/// WAV renamed to `.flac` gets flagged).
39///
40/// ```
41/// use std::path::Path;
42/// use flaccompagnon_core::is_supported_audio;
43///
44/// assert!(is_supported_audio(Path::new("song.flac")));
45/// assert!(is_supported_audio(Path::new("song.FLAC")));   // case-insensitive
46/// assert!(is_supported_audio(Path::new("album.dsf")));   // DSD
47/// assert!(!is_supported_audio(Path::new("cover.jpg")));
48/// assert!(!is_supported_audio(Path::new("README")));     // no extension
49/// ```
50pub fn is_supported_audio(path: &Path) -> bool {
51    path.extension()
52        .and_then(|e| e.to_str())
53        .map(|e| SUPPORTED_EXTENSIONS.contains(&e.to_ascii_lowercase().as_str()))
54        .unwrap_or(false)
55}

57/// List every supported audio file under `root`, sorted, skipping any file that
58/// lives inside a generated spectrogram folder (see [`GENERATED_DIRS`]).
59pub fn list_audio_files(root: &Path, recursive: bool) -> Vec<PathBuf> {
60    let depth = if recursive { usize::MAX } else { 1 };
61    let mut paths: Vec<PathBuf> = walkdir::WalkDir::new(root)
62        .max_depth(depth)
63        .into_iter()
64        // A directory we cannot read (permissions, a broken symlink) is
65        // skipped rather than failing the scan: one unreadable folder must
66        // not cost the user the rest of their library.
67        .filter_map(Result::ok)
68        .map(|e| e.into_path())
69        .filter(|p| p.is_file() && is_supported_audio(p))
70        .filter(|p| {
71            !p.components()
72                .any(|c| GENERATED_DIRS.iter().any(|d| c.as_os_str() == *d))
73        })
74        .collect();
75    paths.sort();
76    paths
77}

79/// Analyze every supported audio file under `root`.
80pub fn analyze_folder(root: &Path, opts: &ScanOptions) -> Result<FolderReport, AnalysisError> {
81    let paths = list_audio_files(root, opts.recursive);
82    let files: Vec<FileAnalysis> = paths.iter().map(|p| analyze_file(p, opts)).collect();
83    let has_flac = files.iter().any(|f| f.flac_md5.is_some());
84
85    Ok(FolderReport {
86        root: root.to_string_lossy().to_string(),
87        files,
88        has_flac,
89    })
90}
