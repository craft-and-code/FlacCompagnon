8/// Audio file extensions FlacCompagnon will attempt to *analyze* — which is
9/// not the same as the ones it can currently decode.
10///
11/// `.opus` is listed even though there is no Opus decoder yet (see
12/// [`crate::decode`]). Leaving it out was worse, not safer: `.ogg` and `.oga`
13/// are here, an Ogg stream can just as well carry Opus, so the same audio was
14/// accepted under one name and silently refused at the drop under another. A
15/// listed file that fails with "Opus decoding is not supported yet" tells the
16/// user what is going on; a file that vanishes on drop tells them the app is
17/// broken. The day a decoder is wired in, this entry needs no change.
18pub const SUPPORTED_EXTENSIONS: &[&str] = &[
19    "flac", "wav", "wave", "aif", "aiff", "aifc", "alac", "m4a", "mp4", "caf", "ogg", "oga",
20    "opus", "mp3", "aac", "dsf", "dff",
21];
22
23/// Folder name used for generated spectrograms. Files inside one are skipped
24/// so a second scan doesn't try to analyze the app's own PNG output — and, more
25/// to the point, so re-scanning a folder never grows the file list.
26const GENERATED_DIR: &str = "spectres";
27
28/// Returns `true` if `path` has an extension FlacCompagnon knows how to decode.
29///
30/// The check is on the extension only and is case-insensitive; the *real*
31/// container is identified later from the file's magic bytes (which is how a
32/// WAV renamed to `.flac` gets flagged).
33///
34/// ```
35/// use std::path::Path;
36/// use flaccompagnon_core::is_supported_audio;
37///
38/// assert!(is_supported_audio(Path::new("song.flac")));
39/// assert!(is_supported_audio(Path::new("song.FLAC")));   // case-insensitive
40/// assert!(is_supported_audio(Path::new("album.dsf")));   // DSD
41/// assert!(!is_supported_audio(Path::new("cover.jpg")));
42/// assert!(!is_supported_audio(Path::new("README")));     // no extension
43/// ```
44pub fn is_supported_audio(path: &Path) -> bool {
45    path.extension()
46        .and_then(|e| e.to_str())
47        .map(|e| SUPPORTED_EXTENSIONS.contains(&e.to_ascii_lowercase().as_str()))
48        .unwrap_or(false)
49}

51/// List every supported audio file under `root`, sorted, skipping any file that
52/// lives inside a generated `spectres` folder.
53pub fn list_audio_files(root: &Path, recursive: bool) -> Vec<PathBuf> {
54    let depth = if recursive { usize::MAX } else { 1 };
55    let mut paths: Vec<PathBuf> = walkdir::WalkDir::new(root)
56        .max_depth(depth)
57        .into_iter()
58        // A directory we cannot read (permissions, a broken symlink) is
59        // skipped rather than failing the scan: one unreadable folder must
60        // not cost the user the rest of their library.
61        .filter_map(Result::ok)
62        .map(|e| e.into_path())
63        .filter(|p| p.is_file() && is_supported_audio(p))
64        .filter(|p| !p.components().any(|c| c.as_os_str() == GENERATED_DIR))
65        .collect();
66    paths.sort();
67    paths
68}

70/// Analyze every supported audio file under `root`.
71pub fn analyze_folder(root: &Path, opts: &ScanOptions) -> Result<FolderReport, AnalysisError> {
72    let paths = list_audio_files(root, opts.recursive);
73    let files: Vec<FileAnalysis> = paths.iter().map(|p| analyze_file(p, opts)).collect();
74    let has_flac = files.iter().any(|f| f.flac_md5.is_some());
75
76    Ok(FolderReport {
77        root: root.to_string_lossy().to_string(),
78        files,
79        has_flac,
80    })
81}

88    fn touch(dir: &Path, rel: &str) {
89        let p = dir.join(rel);
90        if let Some(parent) = p.parent() {
91            fs::create_dir_all(parent).expect("mkdir");
92        }
93        fs::write(&p, b"").expect("write");
94    }

96    #[test]
97    fn every_supported_extension_is_recognized() {
98        for ext in SUPPORTED_EXTENSIONS {
99            let name = format!("track.{ext}");
100            assert!(is_supported_audio(Path::new(&name)), "{ext}");
101            // Uppercase too: file systems are case-insensitive in practice.
102            let upper = format!("track.{}", ext.to_uppercase());
103            assert!(is_supported_audio(Path::new(&upper)), "{upper}");
104        }
105        assert!(!is_supported_audio(Path::new("cover.jpg")));
106        assert!(!is_supported_audio(Path::new("notes.txt")));
107        assert!(!is_supported_audio(Path::new("README")));
108        // An extension that merely *contains* a supported one is not one.
109        assert!(!is_supported_audio(Path::new("track.flacx")));
110    }

112    /// The symptom: a `.opus` file dropped on the app vanished without a
113    /// word, while the *same* Opus audio named `.ogg` was accepted (and then
114    /// failed at decode with a real message). Whether a file is worth trying
115    /// must not depend on which of two names its container was given.
116    #[test]
117    fn opus_is_accepted_under_both_its_container_names() {
118        for name in ["track.opus", "track.ogg", "track.oga", "TRACK.OPUS"] {
119            assert!(is_supported_audio(Path::new(name)), "{name}");
120        }
121    }

123    /// Generated spectrograms live next to the audio; a rescan must not pick
124    /// up anything from there, at any depth.
125    #[test]
126    fn generated_spectres_folders_are_skipped() {
127        let dir = tempfile::tempdir().expect("tempdir");
128        let root = dir.path();
129        touch(root, "a.flac");
130        touch(root, "spectres/a.flac");
131        touch(root, "album/b.flac");
132        touch(root, "album/spectres/b.flac");
133
134        let found = list_audio_files(root, true);
135        assert_eq!(found.len(), 2, "found: {found:?}");
136        assert!(found.iter().all(|p| !p
137            .components()
138            .any(|c| c.as_os_str() == GENERATED_DIR)));
139    }

141    #[test]
142    fn non_recursive_stays_in_the_top_folder() {
143        let dir = tempfile::tempdir().expect("tempdir");
144        let root = dir.path();

146        touch(root, "album/b.flac");

152    /// The list is sorted, because it is what the report's row order comes
153    /// from before the user reorders anything.
154    #[test]
155    fn results_are_sorted() {
156        let dir = tempfile::tempdir().expect("tempdir");
157        let root = dir.path();
158        for name in ["c.flac", "a.flac", "b.flac"] {
159            touch(root, name);
160        }
161        let found = list_audio_files(root, false);
162        let mut sorted = found.clone();
163        sorted.sort();
164        assert_eq!(found, sorted);
165    }
