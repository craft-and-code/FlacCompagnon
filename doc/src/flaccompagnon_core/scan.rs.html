8/// Audio file extensions FlacCompagnon will attempt to analyze.
9pub const SUPPORTED_EXTENSIONS: &[&str] = &[
10    "flac", "wav", "wave", "aif", "aiff", "aifc", "alac", "m4a", "mp4", "caf", "ogg", "oga",
11    "mp3", "aac", "dsf", "dff",
12];
13
14/// Folder name used for generated spectrograms. Files inside one are skipped
15/// so a second scan doesn't try to analyze the app's own PNG output — and, more
16/// to the point, so re-scanning a folder never grows the file list.
17const GENERATED_DIR: &str = "spectres";
18
19/// Returns `true` if `path` has an extension FlacCompagnon knows how to decode.
20///
21/// The check is on the extension only and is case-insensitive; the *real*
22/// container is identified later from the file's magic bytes (which is how a
23/// WAV renamed to `.flac` gets flagged).
24///
25/// ```
26/// use std::path::Path;
27/// use flaccompagnon_core::is_supported_audio;
28///
29/// assert!(is_supported_audio(Path::new("song.flac")));
30/// assert!(is_supported_audio(Path::new("song.FLAC")));   // case-insensitive
31/// assert!(is_supported_audio(Path::new("album.dsf")));   // DSD
32/// assert!(!is_supported_audio(Path::new("cover.jpg")));
33/// assert!(!is_supported_audio(Path::new("README")));     // no extension
34/// ```
35pub fn is_supported_audio(path: &Path) -> bool {
36    path.extension()
37        .and_then(|e| e.to_str())
38        .map(|e| SUPPORTED_EXTENSIONS.contains(&e.to_ascii_lowercase().as_str()))
39        .unwrap_or(false)
40}
41
42/// List every supported audio file under `root`, sorted, skipping any file that
43/// lives inside a generated `spectres` folder.
44pub fn list_audio_files(root: &Path, recursive: bool) -> Vec<PathBuf> {
45    let depth = if recursive { usize::MAX } else { 1 };
46    let mut paths: Vec<PathBuf> = walkdir::WalkDir::new(root)
47        .max_depth(depth)
48        .into_iter()
49        // A directory we cannot read (permissions, a broken symlink) is
50        // skipped rather than failing the scan: one unreadable folder must
51        // not cost the user the rest of their library.
52        .filter_map(Result::ok)
53        .map(|e| e.into_path())
54        .filter(|p| p.is_file() && is_supported_audio(p))
55        .filter(|p| !p.components().any(|c| c.as_os_str() == GENERATED_DIR))
56        .collect();
57    paths.sort();
58    paths
59}
60
61/// Analyze every supported audio file under `root`.
62pub fn analyze_folder(root: &Path, opts: &ScanOptions) -> Result<FolderReport, AnalysisError> {
63    let paths = list_audio_files(root, opts.recursive);
64    let files: Vec<FileAnalysis> = paths.iter().map(|p| analyze_file(p, opts)).collect();
65    let has_flac = files.iter().any(|f| f.flac_md5.is_some());
66
67    Ok(FolderReport {
68        root: root.to_string_lossy().to_string(),
69        files,
70        has_flac,
71    })
72}

74#[cfg(test)]
75mod tests {

77    use std::fs;

79    fn touch(dir: &Path, rel: &str) {
80        let p = dir.join(rel);
81        if let Some(parent) = p.parent() {
82            fs::create_dir_all(parent).expect("mkdir");
83        }
84        fs::write(&p, b"").expect("write");
85    }

87    #[test]
88    fn every_supported_extension_is_recognized() {
89        for ext in SUPPORTED_EXTENSIONS {
90            let name = format!("track.{ext}");
91            assert!(is_supported_audio(Path::new(&name)), "{ext}");
92            // Uppercase too: file systems are case-insensitive in practice.
93            let upper = format!("track.{}", ext.to_uppercase());
94            assert!(is_supported_audio(Path::new(&upper)), "{upper}");
95        }
96        assert!(!is_supported_audio(Path::new("cover.jpg")));
97        assert!(!is_supported_audio(Path::new("notes.txt")));
98        assert!(!is_supported_audio(Path::new("README")));
99        // An extension that merely *contains* a supported one is not one.
100        assert!(!is_supported_audio(Path::new("track.flacx")));
101    }

103    /// Generated spectrograms live next to the audio; a rescan must not pick
104    /// up anything from there, at any depth.
105    #[test]
106    fn generated_spectres_folders_are_skipped() {
107        let dir = tempfile::tempdir().expect("tempdir");
108        let root = dir.path();
109        touch(root, "a.flac");
110        touch(root, "spectres/a.flac");
111        touch(root, "album/b.flac");
112        touch(root, "album/spectres/b.flac");
113
114        let found = list_audio_files(root, true);
115        assert_eq!(found.len(), 2, "found: {found:?}");
116        assert!(found.iter().all(|p| !p
117            .components()
118            .any(|c| c.as_os_str() == GENERATED_DIR)));
119    }

121    #[test]
122    fn non_recursive_stays_in_the_top_folder() {
123        let dir = tempfile::tempdir().expect("tempdir");
124        let root = dir.path();
125        touch(root, "a.flac");
126        touch(root, "album/b.flac");
127
128        assert_eq!(list_audio_files(root, false).len(), 1);
129        assert_eq!(list_audio_files(root, true).len(), 2);
130    }

132    /// The list is sorted, because it is what the report's row order comes
133    /// from before the user reorders anything.
134    #[test]
135    fn results_are_sorted() {
136        let dir = tempfile::tempdir().expect("tempdir");
137        let root = dir.path();
138        for name in ["c.flac", "a.flac", "b.flac"] {
139            touch(root, name);
140        }
141        let found = list_audio_files(root, false);
142        let mut sorted = found.clone();
143        sorted.sort();
144        assert_eq!(found, sorted);
145    }

147    /// A folder that doesn't exist is an empty list, not a panic — the path
148    /// can come from a stale saved report.
149    #[test]
150    fn a_missing_root_yields_nothing() {
151        let missing = Path::new("/definitely/not/a/real/folder/anywhere");
152        assert!(list_audio_files(missing, true).is_empty());
153    }
