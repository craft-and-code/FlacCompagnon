1//! Whole-file fingerprints: MD5 and CRC32 over the bytes on disk.
2//!
3//! # What these are, and what they are not
4//!
5//! These hash the **file**, tags and cover art included — not the audio. Two
6//! files carrying bit-identical audio but a different `ARTIST` tag get
7//! different fingerprints here, and that is the point: this is the identity
8//! of the file as an object, the value that `.sfv` checksums, download
9//! verification and duplicate hunts all work with.
10//!
11//! The audio-level counterpart already exists elsewhere and answers a
12//! different question: [`crate::decode::FlacMd5Status`] compares the decoded
13//! samples against the MD5 a FLAC stores in its own header, which is
14//! unchanged by retagging and identical across a FLAC and the WAV decoded
15//! from it. Neither replaces the other, and confusing the two is easy enough
16//! that the field names say `file_` out loud.
17//!
18//! # Why both, and why one pass
19//!
20//! CRC32 is what the `.sfv` files shipped with lossless releases use, and
21//! what most download tools report; MD5 is what everything else uses. They
22//! are wanted by different people for the same file, and computing them
23//! separately would read the whole file twice for no reason — a 90 MB track
24//! is not free to read, and a folder holds hundreds of them. One read feeds
25//! both.
26//!
27//! CRC32 is a *checksum*, not a cryptographic hash: it detects accidental
28//! corruption, and collisions can be constructed deliberately. It is here
29//! because release conventions use it, not as evidence of anything.
30
31use std::fs::File;
32use std::io::Read;
33use std::path::Path;
34
35use md5::{Digest, Md5};
36
37/// How much is read per iteration. Large enough that the syscall overhead
38/// disappears against the hashing, small enough to stay out of the way of the
39/// analysis buffers running alongside it.
40const CHUNK: usize = 1 << 20; // 1 MiB
41
42/// A file's two fingerprints, as lowercase hex.
43///
44/// Strings rather than the raw values because every consumer — the table, the
45/// CSV, the JSON report — wants the text, and formatting it once here keeps
46/// the three from disagreeing about padding or case.
47#[derive(Debug, Clone, PartialEq, Eq)]
48pub struct FileDigest {
49    /// MD5 of the file's bytes, 32 lowercase hex characters.
50    pub md5: String,
51    /// CRC32 (IEEE, the `.sfv` variant) of the file's bytes, 8 lowercase hex
52    /// characters, zero-padded — an unpadded CRC is the classic way two
53    /// tools disagree about the same file.
54    pub crc32: String,
55}
56
57/// Read `path` once and return both fingerprints.
58///
59/// Errors are the caller's to interpret: an unreadable file still deserves a
60/// row in the table, just without these two columns filled in.
61pub fn file_digest(path: &Path) -> std::io::Result<FileDigest> {
62    let mut file = File::open(path)?;
63    let mut md5 = Md5::new();
64    let mut crc = crc32fast::Hasher::new();
65    let mut buf = vec![0u8; CHUNK];
66    loop {
67        let n = file.read(&mut buf)?;
68        if n == 0 {
69            break;
70        }
71        // `get` rather than indexing: `read` promises `n <= buf.len()`, but
72        // this crate does not take a promise as a reason to allow a panic on
73        // a path that touches user files.
74        let Some(chunk) = buf.get(..n) else { break };
75        md5.update(chunk);
76        crc.update(chunk);
77    }
78    Ok(FileDigest {
79        md5: format!("{:x}", md5.finalize()),
80        crc32: format!("{:08x}", crc.finalize()),
81    })
82}

84#[cfg(test)]
85mod tests {
86    use super::*;
87
88    /// Ground truth from the specifications, not from this implementation:
89    /// RFC 1321 gives MD5("abc"), and CRC-32/ISO-HDLC's check value for
90    /// "123456789" is the one every CRC catalogue lists.
91    #[test]
92    fn known_vectors_from_the_specifications() {
93        let dir = tempfile::tempdir().expect("tempdir");
94
95        let abc = dir.path().join("abc.bin");
96        std::fs::write(&abc, b"abc").expect("write");
97        assert_eq!(
98            file_digest(&abc).expect("digest").md5,
99            "900150983cd24fb0d6963f7d28e17f72"
100        );
101
102        let check = dir.path().join("check.bin");
103        std::fs::write(&check, b"123456789").expect("write");
104        assert_eq!(file_digest(&check).expect("digest").crc32, "cbf43926");
105    }
106
107    /// An empty file has fingerprints, and they are the well-known ones — a
108    /// zero-length read must not be mistaken for a failure.
109    #[test]
110    fn an_empty_file_hashes_to_the_known_empty_values() {
111        let dir = tempfile::tempdir().expect("tempdir");
112        let empty = dir.path().join("empty.bin");
113        std::fs::write(&empty, b"").expect("write");
114        let d = file_digest(&empty).expect("digest");
115        assert_eq!(d.md5, "d41d8cd98f00b204e9800998ecf8427e");
116        assert_eq!(d.crc32, "00000000");
117    }
118
119    /// The chunked loop must produce the same answer as one-shot hashing,
120    /// which only a file larger than [`CHUNK`] can show.
121    #[test]
122    fn chunking_does_not_change_the_result() {
123        let dir = tempfile::tempdir().expect("tempdir");
124        let big = dir.path().join("big.bin");
125        // Deterministic, non-repeating content: a repeating pattern would
126        // hide a chunk-boundary mistake that a changing one exposes.
127        let mut data = Vec::with_capacity(CHUNK * 2 + 1234);
128        let mut x = 0x1234_5678u32;
129        while data.len() < CHUNK * 2 + 1234 {
130            x ^= x << 13;
131            x ^= x >> 17;
132            x ^= x << 5;
133            data.extend_from_slice(&x.to_le_bytes());
134        }
135        std::fs::write(&big, &data).expect("write");
136
137        let mut md5 = Md5::new();
138        md5.update(&data);
139        let mut crc = crc32fast::Hasher::new();
140        crc.update(&data);
141
142        let d = file_digest(&big).expect("digest");
143        assert_eq!(d.md5, format!("{:x}", md5.finalize()));
144        assert_eq!(d.crc32, format!("{:08x}", crc.finalize()));
145    }
146
147    /// Both fields are fixed-width hex — a CRC printed without padding is how
148    /// a checksum silently stops matching the `.sfv` it is compared against.
149    #[test]
150    fn the_hex_is_padded_to_a_fixed_width() {
151        let dir = tempfile::tempdir().expect("tempdir");
152        // Content chosen by search below is unnecessary: any file proves the
153        // widths, and a leading-zero CRC is covered by the empty-file test.
154        let p = dir.path().join("any.bin");
155        std::fs::write(&p, b"whatever").expect("write");
156        let d = file_digest(&p).expect("digest");
157        assert_eq!(d.md5.len(), 32, "{}", d.md5);
158        assert_eq!(d.crc32.len(), 8, "{}", d.crc32);
159        assert!(d.md5.chars().all(|c| c.is_ascii_hexdigit() && !c.is_uppercase()));
160        assert!(d.crc32.chars().all(|c| c.is_ascii_hexdigit() && !c.is_uppercase()));
161    }
162
163    #[test]
164    fn a_missing_file_is_an_error_not_a_fingerprint() {
165        let dir = tempfile::tempdir().expect("tempdir");
166        assert!(file_digest(&dir.path().join("absent.flac")).is_err());
167    }
