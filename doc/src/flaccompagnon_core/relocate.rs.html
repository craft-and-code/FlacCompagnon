1//! Finding files that moved: matching a listing's stale paths against a folder
2//! the user points at.
3//!
4//! # Why by file name, and why a suffix score
5//!
6//! A saved report records absolute paths. Move the library, restore it from a
7//! backup, plug the drive in on another machine, and every path in the report
8//! is wrong while every *file* is fine. The information that survives all of
9//! those is the file name and the tail of its folder structure.
10//!
11//! Matching on the name alone is not enough: a discography holds a dozen
12//! `01 Intro.flac`. So candidates are ranked by how many trailing path
13//! components they share with the old path — `Rammstein/Reise/01 Intro.flac`
14//! beats `Various/Hits/01 Intro.flac` when relocating the former, and no
15//! shared parent at all still matches when there is only one candidate.
16//!
17//! This is deliberately per-file rather than a single "old prefix → new
18//! prefix" rewrite. A listing assembled from several folders — the mixtape
19//! case — has no common prefix to rewrite, but each of its files can still be
20//! found on its own.
21
22use std::collections::HashMap;
23use std::path::{Component, Path, PathBuf};
24
25use serde::{Deserialize, Serialize};
26
27/// One file's move: where the listing thought it was, and where it is now.
28#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
29pub struct Relocation {
30    /// The stale path, exactly as the listing holds it.
31    pub from: String,
32    /// The path found under the folder the user chose.
33    pub to: String,
34}
35
36/// Match each of `missing` against the files under `root`.
37///
38/// Only unambiguous, same-name matches are returned; a file with no candidate
39/// is simply absent from the result, and the caller reports how many were
40/// found rather than failing the whole operation. That matters for a listing
41/// spanning several folders: pointing at one of them should fix that folder's
42/// files and leave the rest alone.
43///
44/// `candidates` is every path under the chosen folder — taking it as an
45/// argument rather than walking the disk here keeps this crate free of the
46/// directory traversal (and the traversal's error handling), and lets the
47/// unit tests work on paths that never existed.
48pub fn match_moved_files(missing: &[String], candidates: &[PathBuf]) -> Vec<Relocation> {
49    // Indexed by file name: the one part of a path a move cannot change.
50    let mut by_name: HashMap<String, Vec<&PathBuf>> = HashMap::new();
51    for path in candidates {
52        if let Some(name) = path.file_name().and_then(|n| n.to_str()) {
53            by_name.entry(name.to_string()).or_default().push(path);
54        }
55    }
56
57    let mut out = Vec::new();
58    for from in missing {
59        let old = Path::new(from);
60        let Some(name) = old.file_name().and_then(|n| n.to_str()) else {
61            continue;
62        };
63        let Some(options) = by_name.get(name) else {
64            continue;
65        };
66        // `max_by_key` returns the *last* maximum, so ties resolve to whichever
67        // candidate the walk found last — arbitrary but stable, and a tie here
68        // means two files that are equally plausible by every signal a path
69        // carries.
70        let Some(best) = options.iter().max_by_key(|c| shared_suffix(old, c)) else {
71            continue;
72        };
73        if let Some(to) = best.to_str() {
74            // A candidate identical to the stale path is not a move. This can
75            // only happen if the file came back between the presence check and
76            // the folder being chosen, and reporting it as relocated would be
77            // a lie in the toast.
78            if to != from {
79                out.push(Relocation {
80                    from: from.clone(),
81                    to: to.to_string(),
82                });
83            }
84        }
85    }
86    out
87}

89/// How many trailing path components `a` and `b` have in common, file name
90/// included.
91///
92/// Comparison is case-insensitive: the same library read from a case-sensitive
93/// filesystem and a case-preserving one differs in exactly this way, and a
94/// score that flipped between them would make the ranking depend on which
95/// machine the report was saved on.
96fn shared_suffix(a: &Path, b: &Path) -> usize {
97    let parts = |p: &Path| -> Vec<String> {
98        p.components()
99            .filter_map(|c| match c {
100                Component::Normal(s) => s.to_str().map(|s| s.to_lowercase()),
101                _ => None,
102            })
103            .collect()
104    };
105    let (a, b) = (parts(a), parts(b));
106    a.iter()
107        .rev()
108        .zip(b.iter().rev())
109        .take_while(|(x, y)| x == y)
110        .count()
111}
