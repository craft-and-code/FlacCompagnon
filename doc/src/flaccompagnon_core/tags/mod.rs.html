1//! Reading and writing metadata tags (title, artist, album, cover art, …).
2//!
3//! Unlike the rest of this crate, this module **does** modify the files it
4//! touches — [`write_tags`] is the one place in FlacCompagnon that writes to
5//! the audio files being analyzed rather than a separate report. It exists to
6//! back the tag-editing panel: batch reads across a selection to pre-fill the
7//! form, and a single batch write to apply the user's edits once they hit
8//! "Enregistrer".
9//!
10//! Backed by [`lofty`](https://docs.rs/lofty), which gives a single API across
11//! FLAC (Vorbis comments), MP3/AIFF/WAV (ID3v2), MP4/M4A (iTunes atoms) and
12//! Ogg/Opus (Vorbis comments). DSD (`.dsf`/`.dff`) isn't a format lofty knows
13//! how to tag; [`read_tags`]/[`write_tags`] return [`TagError::Parse`] for it,
14//! which the caller surfaces as a per-file "not supported" message rather than
15//! failing a whole batch.
16//!
17//! ## Batch-edit semantics
18//!
19//! The tag panel edits several files at once, and a field the user never
20//! touched must not clobber files that had a *different* value than the one
21//! shown (the "valeurs multiples" badge case). [`FieldEdit`] encodes this as
22//! a three-way choice — `Unset` (leave alone), `Clear` (remove the tag) or
23//! `Set` (write this value) — rather than the usual `Option<String>`, which
24//! can't distinguish "leave alone" from "set to empty".
25//!
26//! The extended-tags pop-in edits [`TagSet::extra`] the same way, just keyed
27//! by raw tag name instead of a fixed struct field: [`TagEdits::extra`] is a
28//! sparse `Vec<(String, FieldEdit)>` rather than a `HashMap`, since a batch
29//! write only ever touches a handful of keys and a `Vec` round-trips through
30//! `serde`/Tauri without the string-key caveats a `HashMap` would need.
31//!
32//! ## Module layout
33//!
34//! This module owns the text fields ([`TagSet`], [`FieldEdit`], [`TagEdits`])
35//! and the top-level [`read_tags`]/[`write_tags`] orchestration. Everything
36//! specific to the embedded picture — [`CoverArt`], [`CoverEdit`], MIME
37//! sniffing, mapping a role name to lofty's `PictureType` — lives in
38//! [`cover`], which `read_tags`/`write_tags` call into rather than handling
39//! inline. The extended-tags pop-in's "+" picker — its curated key list and
40//! the resolution against a file's tag type — lives in [`addable`]. Both are
41//! re-exported here so existing callers (`core::tags::CoverArt`,
42//! `core::tags::AddableTag`, etc.) don't need to know the split exists.
43//!
44//! What's left — [`TagSet`]/[`FieldEdit`]/[`TagEdits`] and [`read_tags`]/
45//! [`write_tags`] — is kept as one file rather than split further: it's one
46//! read/write round trip over one set of fields, and the "Batch-edit
47//! semantics" explanation above applies to all of it at once. Splitting the
48//! types from the functions that fill and consume them (or `read_tags` from
49//! `write_tags`) would scatter that one explanation across files that only
50//! make sense read together, for a line count a few dozen over the target —
51//! [`cover`] and [`addable`] were split out because they're each a genuinely
52//! separate concern *within* this module; there isn't a third one left to
53//! carve out here.

68/// Errors that can occur while reading or writing tags.
69#[derive(Debug, thiserror::Error)]
70pub enum TagError {
71    /// The file's format is unsupported for tagging, or its tags could not
72    /// be parsed (path, reason).
73    #[error("{0}: unreadable or unsupported format for tags ({1})")]
74    Parse(String, String),
75    /// The tags could not be written back to the file (path, reason).
76    #[error("{0}: could not write tags ({1})")]
77    Write(String, String),
78    /// The replacement cover image data was invalid (path, reason).
79    #[error("{0}: invalid cover image data ({1})")]
80    Cover(String, String),
81}

83/// The tags of a single file, as read from disk.
84///
85/// Fields absent from the file are `None`. `extra` carries every other
86/// textual tag item lofty recognized but that isn't one of the common fields
87/// above (ISRC, BPM, ReplayGain, custom `TXXX`/user frames, …), keyed by its
88/// format-specific tag name — this backs the "tags étendus" panel.
89#[derive(Debug, Clone, Default, Serialize, Deserialize, PartialEq)]
90pub struct TagSet {
91    /// Track title.
92    pub title: Option<String>,
93    /// Track artist.
94    pub artist: Option<String>,
95    /// Album name.
96    pub album: Option<String>,
97    /// Album artist (as distinct from the track artist, for compilations).
98    pub album_artist: Option<String>,
99    /// Composer.
100    pub composer: Option<String>,
101    /// Release year, as free text (formats vary: `"2019"`, `"2019-04-05"`, …).
102    pub year: Option<String>,
103    /// Track number.
104    pub track: Option<String>,
105    /// Total number of tracks on the disc/release.
106    pub track_total: Option<String>,
107    /// Disc number, for multi-disc releases.
108    pub disc: Option<String>,
109    /// Total number of discs in the release.
110    pub disc_total: Option<String>,
111    /// Genre.
112    pub genre: Option<String>,
113    /// Free-text comment field.
114    pub comment: Option<String>,
115    /// `true` when the file is tagged as part of a compilation.
116    pub compilation: bool,
117    /// Every other textual tag item, keyed by its format-specific tag name.
118    pub extra: Vec<(String, String)>,
119    /// Embedded cover art, when the file has one.
120    pub cover: Option<CoverArt>,
121    /// The MusicBrainz Release ID, if this file already carries one (e.g.
122    /// tagged previously by Picard, or by a ripper that writes it). Lets the
123    /// "Search online" button skip straight to that exact release instead of
124    /// a fuzzy text search — still surfaced under its raw tag name in
125    /// `extra` too, so it's visible in the extended-tags viewer.
126    pub musicbrainz_release_id: Option<String>,
127    /// The tool (and often its version) that produced this file, when it left
128    /// one behind — FLAC's Vorbis comment vendor string, an MP3's ID3v2
129    /// `TSSE` frame, and so on (`ItemKey::EncoderSoftware`). Read-only: this
130    /// is diagnostic information about the file's own history, not something
131    /// the tag panel offers to edit, so it has no `TagEdits` counterpart.
132    #[serde(default)]
133    pub encoder: Option<String>,
134}

136/// A single field's edit instruction for a batch write.
137///
138/// `Unset` is the default so a [`TagEdits`] built from "only the fields the
139/// user actually typed into" naturally leaves everything else untouched.
140#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Default)]
141pub enum FieldEdit {
142    /// Leave this field untouched.
143    #[default]
144    Unset,
145    /// Remove this field's tag entirely.
146    Clear,
147    /// Write this value.
148    Set(String),
149}

151/// A batch of edits to apply to every file in a selection. See the module
152/// docs for why fields aren't plain `Option<String>`.
153#[derive(Debug, Clone, Serialize, Deserialize, Default)]
154pub struct TagEdits {
155    /// Edit instruction for [`TagSet::title`].
156    pub title: FieldEdit,
157    /// Edit instruction for [`TagSet::artist`].
158    pub artist: FieldEdit,
159    /// Edit instruction for [`TagSet::album`].
160    pub album: FieldEdit,
161    /// Edit instruction for [`TagSet::album_artist`].
162    pub album_artist: FieldEdit,
163    /// Edit instruction for [`TagSet::composer`].
164    pub composer: FieldEdit,
165    /// Edit instruction for [`TagSet::year`].
166    pub year: FieldEdit,
167    /// Edit instruction for [`TagSet::track`].
168    pub track: FieldEdit,
169    /// Edit instruction for [`TagSet::track_total`].
170    pub track_total: FieldEdit,
171    /// Edit instruction for [`TagSet::disc`].
172    pub disc: FieldEdit,
173    /// Edit instruction for [`TagSet::disc_total`].
174    pub disc_total: FieldEdit,
175    /// Edit instruction for [`TagSet::genre`].
176    pub genre: FieldEdit,
177    /// Edit instruction for [`TagSet::comment`].
178    pub comment: FieldEdit,
179    /// `None` leaves the compilation flag untouched.
180    pub compilation: Option<bool>,
181    /// Edit instruction for the embedded cover art.
182    pub cover: CoverEdit,
183    /// Sparse add/edit/remove instructions for extended tags, keyed by the
184    /// same raw format-specific tag name [`TagSet::extra`] pairs use (e.g.
185    /// `"BPM"` in a FLAC's Vorbis comments, `"TBPM"` in an MP3's ID3v2
186    /// frames) — only the keys the extended-tags pop-in actually touched are
187    /// present, same sparse convention as the named fields above and for the
188    /// same reason: an untouched key must not clobber a file whose value
189    /// differed from what the panel showed.
190    pub extra: Vec<(String, FieldEdit)>,
191}

193/// Every [`ItemKey`] already surfaced as a named [`TagSet`] field — excluded
194/// from `TagSet::extra` so the extended-tags panel doesn't duplicate them.
195const KNOWN_KEYS: &[ItemKey] = &[
196    ItemKey::TrackTitle,
197    ItemKey::TrackArtist,
198    ItemKey::AlbumTitle,
199    ItemKey::Genre,
200    ItemKey::Comment,
201    ItemKey::AlbumArtist,
202    ItemKey::Composer,
203    ItemKey::Year,
204    ItemKey::RecordingDate,
205    ItemKey::TrackNumber,
206    ItemKey::TrackTotal,
207    ItemKey::DiscNumber,
208    ItemKey::DiscTotal,
209    ItemKey::FlagCompilation,
210    ItemKey::EncoderSoftware,
211];

213fn parse_err(path: &Path, e: impl std::fmt::Display) -> TagError {
214    TagError::Parse(path.display().to_string(), e.to_string())
215}

217/// Read every tag FlacCompagnon knows how to show from `path`'s primary tag
218/// (the tag type a player would actually look at — see
219/// [`lofty::file::FileType::primary_tag_type`]).
220pub fn read_tags(path: &Path) -> Result<TagSet, TagError> {
221    let tagged = read_from_path(path).map_err(|e| parse_err(path, e))?;
222    let mut out = TagSet::default();
223
224    let Some(tag) = tagged.primary_tag() else {
225        return Ok(out);
226    };
227
228    out.title = tag.title().map(|c| c.into_owned());
229    out.artist = tag.artist().map(|c| c.into_owned());
230    out.album = tag.album().map(|c| c.into_owned());
231    out.genre = tag.genre().map(|c| c.into_owned());
232    out.comment = tag.comment().map(|c| c.into_owned());
233    out.album_artist = tag.get_string(ItemKey::AlbumArtist).map(str::to_string);
234    out.composer = tag.get_string(ItemKey::Composer).map(str::to_string);
235    out.year = tag
236        .get_string(ItemKey::Year)
237        .or_else(|| tag.get_string(ItemKey::RecordingDate))
238        .map(str::to_string);
239    out.track = tag.get_string(ItemKey::TrackNumber).map(str::to_string);
240    out.track_total = tag.get_string(ItemKey::TrackTotal).map(str::to_string);
241    out.disc = tag.get_string(ItemKey::DiscNumber).map(str::to_string);
242    out.disc_total = tag.get_string(ItemKey::DiscTotal).map(str::to_string);
243    out.compilation = tag.get_string(ItemKey::FlagCompilation) == Some("1");
244    out.musicbrainz_release_id = tag
245        .get_string(ItemKey::MusicBrainzReleaseId)
246        .map(str::to_string)
247        .filter(|s| !s.is_empty());
248    out.encoder = tag
249        .get_string(ItemKey::EncoderSoftware)
250        .map(str::to_string)
251        .filter(|s| !s.is_empty());
252
253    for item in tag.items() {
254        if KNOWN_KEYS.contains(&item.key()) {
255            continue;
256        }
257        if let Some(text) = item.value().text() {
258            if text.is_empty() {
259                continue;
260            }
261            let label = item
262                .key()
263                .map_key(tag.tag_type())
264                .unwrap_or("?")
265                .to_string();
266            out.extra.push((label, text.to_string()));
267        }
268    }
269
270    out.cover = cover::extract(tag);
271
272    Ok(out)
273}

275fn apply_field(tag: &mut Tag, key: ItemKey, edit: &FieldEdit) {
276    match edit {
277        FieldEdit::Unset => {}
278        FieldEdit::Clear => tag.remove_key(key),
279        FieldEdit::Set(value) => {
280            tag.insert_text(key, value.clone());
281        }
282    }
283}

285/// Apply `edits` to `path`, writing straight to the audio file.
286///
287/// Reads the file's current primary tag first (so untouched fields, and any
288/// tag the panel doesn't know about, survive the round trip), applies only
289/// the fields present in `edits`, and saves. A file with no tag yet gets a
290/// fresh one of the format's primary [`lofty::tag::TagType`].
291pub fn write_tags(path: &Path, edits: &TagEdits) -> Result<(), TagError> {
292    let tagged = read_from_path(path).map_err(|e| parse_err(path, e))?;
293    let tag_type = tagged.primary_tag_type();
294    let mut tag = tagged
295        .primary_tag()
296        .cloned()
297        .unwrap_or_else(|| Tag::new(tag_type));
298
299    apply_field(&mut tag, ItemKey::TrackTitle, &edits.title);
300    apply_field(&mut tag, ItemKey::TrackArtist, &edits.artist);
301    apply_field(&mut tag, ItemKey::AlbumTitle, &edits.album);
302    apply_field(&mut tag, ItemKey::AlbumArtist, &edits.album_artist);
303    apply_field(&mut tag, ItemKey::Composer, &edits.composer);
304    // `ItemKey::Year` has no mapping at all for ID3v2 (only `RecordingDate`
305    // does — see lofty's `ID3V2_MAP`), so writing MP3/WAV/AIFF/AAC tags with
306    // `Year` silently drops the field. `RecordingDate` round-trips on every
307    // format lofty supports (ID3v2 TDRC, Vorbis DATE, MP4 ©day, RIFF ICRD),
308    // so the "year" field is always written there; `read_tags` still checks
309    // the legacy `Year` key first for files tagged by other tools.
310    apply_field(&mut tag, ItemKey::RecordingDate, &edits.year);
311    apply_field(&mut tag, ItemKey::TrackNumber, &edits.track);
312    apply_field(&mut tag, ItemKey::TrackTotal, &edits.track_total);
313    apply_field(&mut tag, ItemKey::DiscNumber, &edits.disc);
314    apply_field(&mut tag, ItemKey::DiscTotal, &edits.disc_total);
315    apply_field(&mut tag, ItemKey::Genre, &edits.genre);
316    apply_field(&mut tag, ItemKey::Comment, &edits.comment);
317
318    if let Some(compilation) = edits.compilation {
319        tag.insert_text(
320            ItemKey::FlagCompilation,
321            if compilation { "1" } else { "0" }.to_string(),
322        );
323    }
324
325    for (key, edit) in &edits.extra {
326        // A key this file's tag type has no mapping for (a Vorbis-only field
327        // on an MP3, say) is silently skipped rather than erroring the whole
328        // batch — the same mismatch already accepted for `Year` above, and
329        // for the same reason: the panel can't know a mixed-format selection
330        // supports every field on every file.
331        if let Some(item_key) = ItemKey::from_key(tag_type, key) {
332            apply_field(&mut tag, item_key, edit);
333        }
334    }
335
336    cover::apply_edit(&mut tag, &edits.cover, &path.display().to_string())?;
337
338    tag.save_to_path(path, WriteOptions::default())
339        .map_err(|e| TagError::Write(path.display().to_string(), e.to_string()))
340}
