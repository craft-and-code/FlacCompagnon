56/// One entry in the extended-tags pop-in's "+" picker: the raw
57/// format-specific key to write (matches `TagSet::extra`'s keys) paired
58/// with a human label.
59#[derive(Debug, Clone, Serialize)]
60pub struct AddableTag {
61    /// Raw format-specific tag name, as `write_tags`'s `edits.extra` expects.
62    pub key: String,
63    /// Human-readable label for the picker.
64    pub label: String,
65}
66
67/// `ADDABLE_KEYS` resolved against `path`'s tag type and filtered to the
68/// ones that format actually supports (`ItemKey::map_key` returns `None`
69/// for a variant with no mapping in that scheme) — so the picker only ever
70/// offers a tag that can actually be written to the file(s) at hand. The
71/// batch always applies one edit to the whole selection, so one
72/// representative file's format is enough to decide what's offered.
73pub fn addable_tags(path: &Path) -> Result<Vec<AddableTag>, TagError> {
74    let tagged = read_from_path(path)
75        .map_err(|e| TagError::Parse(path.display().to_string(), e.to_string()))?;
76    let tag_type = tagged.primary_tag_type();
77    Ok(ADDABLE_KEYS
78        .iter()
79        .filter_map(|(key, label)| {
80            key.map_key(tag_type).map(|raw| AddableTag {
81                key: raw.to_string(),
82                label: (*label).to_string(),
83            })
84        })
85        .collect())
86}

88#[cfg(test)]
89mod tests {
90    use super::*;
91    use lofty::config::WriteOptions;
92    use lofty::tag::{Tag, TagExt as _, TagType};
93
94    /// A minimal, silent WAV — same fixture style as the rest of the crate's
95    /// tests (`hound`-generated, no real audio content needed).
96    fn synth_wav() -> tempfile::TempPath {
97        let file = tempfile::Builder::new().suffix(".wav").tempfile().unwrap();
98        let path = file.into_temp_path();
99        let spec = hound::WavSpec {
100            channels: 2,
101            sample_rate: 44_100,
102            bits_per_sample: 16,
103            sample_format: hound::SampleFormat::Int,
104        };
105        let mut writer = hound::WavWriter::create(&path, spec).unwrap();
106        for _ in 0..1000 {
107            writer.write_sample(0i16).unwrap();
108            writer.write_sample(0i16).unwrap();
109        }
110        writer.finalize().unwrap();
111        // Give the file a tag so `read_from_path` has a `primary_tag_type`
112        // to report rather than falling back to the container's default.
113        let tag = Tag::new(TagType::Id3v2);
114        tag.save_to_path(&path, WriteOptions::default()).unwrap();
115        path
116    }
117
118    /// Every key `addable_tags` offers must resolve back to an `ItemKey` for
119    /// the same file's tag type — the exact round trip `write_tags` relies on
120    /// when the picker's choice comes back as an edit.
121    #[test]
122    fn addable_tags_round_trip_for_their_format() {
123        let path = synth_wav();
124        let tags = addable_tags(&path).unwrap();
125        assert!(!tags.is_empty());
126
127        let tagged = read_from_path(&path).unwrap();
128        let tag_type = tagged.primary_tag_type();
129        for t in &tags {
130            assert!(
131                ItemKey::from_key(tag_type, &t.key).is_some(),
132                "{} did not round-trip",
133                t.key
134            );
135        }
136    }
