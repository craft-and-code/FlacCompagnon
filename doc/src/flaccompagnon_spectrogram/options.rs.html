5/// Maximum plotted pixels, bounding allocations independently of file headers.
6pub const MAX_PIXELS: usize = 16_777_216;

8/// Plot presets shared by hosts; custom dimensions remain available separately.
9#[derive(Debug, Clone, Copy, Default)]
10pub enum PlotSize {
11    /// 1200×627, rounded from the original 900:470 aspect ratio.
12    #[default]
13    Standard,
14    /// Original detailed plot, 1800×940.
15    Large,
16}
17
18impl PlotSize {
19    /// Plotted width and height, excluding the surrounding frame.
20    pub fn dimensions(self) -> (u32, u32) {
21        match self {
22            Self::Standard => (1200, 627),
23            Self::Large => (1800, 940),
24        }
25    }
26}

28/// The actual decoded PCM format and exact number of complete source frames.
29#[derive(Debug, Clone, Copy)]
30pub struct SignalInfo {
31    /// Frames per second, without resampling.
32    pub sample_rate: u32,
33    /// Interleaved samples per frame, from 1 through 32.
34    pub channels: usize,
35    /// Exact source frame count; a mismatched stream is rejected.
36    pub frames: u64,
37}

39/// Frequency mapping of plotted rows.
40#[derive(Debug, Clone, Copy, Default, PartialEq)]
41pub enum FrequencyScale {
42    /// Equal Hz intervals occupy equal vertical distances.
43    #[default]
44    Linear,
45    /// Equal frequency ratios occupy equal distances; minimum must be positive.
46    Logarithmic,
47}
48
49/// Window applied before each Fourier transform.
50#[derive(Debug, Clone, Copy, Default)]
51pub enum Window {
52    /// Periodic Hann; the general-purpose default.
53    #[default]
54    Hann,
55    /// Four-term minimum-sidelobe Blackman-Harris; broader main lobe.
56    BlackmanHarris,
57    /// No taper; suitable for exact bin-centered synthetic references.
58    Rectangular,
59}
60
61/// Combine spectra after transforming channels, avoiding mono cancellation.
62#[derive(Debug, Clone, Copy, Default)]
63pub enum Channels {
64    /// Average channel powers, preserving identical mono/stereo levels.
65    #[default]
66    MeanPower,
67    /// Retain the strongest channel at each frequency.
68    MaximumPower,
69    /// Analyze a single zero-based channel.
70    Single(usize),
71}
72
73/// Reduce spectral observations into a plotted pixel.
74#[derive(Debug, Clone, Copy, Default)]
75pub enum Aggregation {
76    /// Preserve the strongest contributing bin/window, including brief events.
77    #[default]
78    Peak,
79    /// Average linear power over contributing bins/windows, before decibels.
80    MeanPower,
81}

83/// Spectral computation parameters. Dimensions describe the plot, not margins.
84#[derive(Debug, Clone)]
85pub struct AnalysisOptions {
86    /// Plotted width; default 1200 pixels.
87    pub width: u32,
88    /// Plotted height; default round(1200 × 470 / 900) = 627 pixels.
89    pub height: u32,
90    /// FFT length; `None` keeps approximately the 8192/44.1 kHz duration.
91    pub fft_size: Option<usize>,
92    /// Center-to-center hop; `None` uses one eighth of the FFT length.
93    pub hop_size: Option<usize>,
94    /// Taper before the transform.
95    pub window: Window,
96    /// Channel selection/combination.
97    pub channels: Channels,
98    /// Pixel reduction convention.
99    pub aggregation: Aggregation,
100    /// Vertical axis mapping.
101    pub frequency_scale: FrequencyScale,
102    /// Lower plotted frequency in Hz.
103    pub min_frequency: f64,
104    /// Upper plotted frequency; `None` uses decoded Nyquist.
105    pub max_frequency: Option<f64>,
106}
107
108impl Default for AnalysisOptions {
109    fn default() -> Self {
110        let (width, height) = PlotSize::Standard.dimensions();
111        Self {
112            width,
113            height,
114            fft_size: None,
115            hop_size: None,
116            window: Window::Hann,
117            channels: Channels::MeanPower,
118            aggregation: Aggregation::Peak,
119            frequency_scale: FrequencyScale::Linear,
120            min_frequency: 0.0,
121            max_frequency: None,
122        }
123    }
124}

126impl AnalysisOptions {
127    pub(crate) fn validate(&self, signal: SignalInfo) -> Result<(usize, usize, f64), Error> {
128        let invalid = |message: &str| Error::Configuration(message.into());
129        if signal.sample_rate == 0 || !(1..=32).contains(&signal.channels) || signal.frames == 0 {
130            return Err(invalid(
131                "nonempty PCM with a positive rate and 1–32 channels required",
132            ));
133        }
134        let pixels = u64::from(self.width) * u64::from(self.height);
135        if self.width == 0 || self.height == 0 || pixels > MAX_PIXELS as u64 {
136            return Err(invalid(
137                "plot dimensions exceed the pixel limit or are zero",
138            ));
139        }
140        let exponent = (signal.sample_rate as f64 * 8192.0 / 44100.0)
141            .log2()
142            .round() as u32;
143        let automatic = (1_usize << exponent.min(16)).clamp(256, 65536);
144        let fft = self.fft_size.unwrap_or(automatic);
145        let hop = self.hop_size.unwrap_or(fft / 8);
146        if !(16..=65536).contains(&fft) || !fft.is_power_of_two() || hop == 0 || hop > fft {
147            return Err(invalid(
148                "FFT must be a power of two from 16 to 65536; hop must be 1..=FFT",
149            ));
150        }
151        if matches!(self.channels, Channels::Single(channel) if channel >= signal.channels) {
152            return Err(invalid("selected channel is absent"));
153        }
154        let max = self
155            .max_frequency
156            .unwrap_or(f64::from(signal.sample_rate) / 2.0);
157        if !self.min_frequency.is_finite()
158            || !max.is_finite()
159            || self.min_frequency < 0.0
160            || max <= self.min_frequency
161            || max > f64::from(signal.sample_rate) / 2.0
162            || (self.frequency_scale == FrequencyScale::Logarithmic && self.min_frequency <= 0.0)
163        {
164            return Err(invalid(
165                "frequency bounds must be ordered and within decoded Nyquist",
166            ));
167        }
168        Ok((fft, hop, max))
169    }
170}
