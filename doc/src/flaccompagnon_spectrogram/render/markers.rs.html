1//! Explicit physical annotations, validated against the measured axes.
2use super::{canvas::Canvas, text, MarkerAxis, RenderOptions};
3use crate::{Error, Spectrogram};
4use fontdue::Font;
5pub(super) fn draw(
6    image: &mut Canvas,
7    font: &Font,
8    data: &Spectrogram,
9    options: &RenderOptions,
10) -> Result<(), Error> {
11    let [left, top, ..] = options.margins;
12    let (width, height) = data.dimensions();
13    let right = left + width;
14    let bottom = top + height;
15    let size = options.font_size;
16    let scale = size / 15.0;
17    let write = |image: &mut Canvas, label: &str, x: f32, y: f32, px: f32, color, limit| {
18        text::draw(image, font, label, (x, y), px, color, limit)
19    };
20    for marker in &options.markers {
21        let fraction = match marker.axis {
22            MarkerAxis::Frequency => data.fraction(marker.value),
23            MarkerAxis::Time => marker.value / data.duration(),
24        };
25        if !fraction.is_finite() || !(0.0..=1.0).contains(&fraction) {
26            return Err(Error::Configuration(
27                "marker lies outside the plotted physical axis".into(),
28            ));
29        }
30        let (x, y) = match marker.axis {
31            MarkerAxis::Frequency => {
32                let y = bottom
33                    .saturating_sub((fraction * f64::from(height)) as u32)
34                    .min(bottom - 1);
35                image.horizontal(left, y, width, marker.color, marker.dashed);
36                (left + 12, y.saturating_sub((30.0 * scale) as u32).max(top))
37            }
38            MarkerAxis::Time => {
39                let x = (left + (fraction * f64::from(width)) as u32).min(right - 1);
40                image.vertical(x, top, height, marker.color, marker.dashed);
41                (x + 8, top + 8)
42            }
43        };
44        write(
45            image,
46            &marker.label,
47            x as f32,
48            y as f32,
49            size * 0.85,
50            marker.color,
51            right.saturating_sub(x) as f32,
52        );
53    }
54    Ok(())
55}
