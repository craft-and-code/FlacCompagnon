13/// Text supplied by the host, with empty strings hiding individual labels.
14#[derive(Debug, Clone, Default)]
15pub struct ImageLabels {
16    /// Generating application, drawn at the top right.
17    pub generator: String,
18    /// Track title, drawn at the bottom left.
19    pub title: String,
20    /// Artist or supplementary title information.
21    pub artist: String,
22    /// Actual sample rate/depth/channels, drawn at the bottom right.
23    pub audio_description: String,
24}

26/// Which physical axis carries a marker value.
27#[derive(Debug, Clone, Copy)]
28pub enum MarkerAxis {
29    /// A horizontal marker at a frequency in Hz.
30    Frequency,
31    /// A vertical marker at a time in seconds.
32    Time,
33}
34
35/// Optional measured annotation; a marker is not an authenticity verdict.
36#[derive(Debug, Clone)]
37pub struct Marker {
38    /// Axis and physical unit of `value`.
39    pub axis: MarkerAxis,
40    /// Frequency in Hz or time in seconds, within the plotted view.
41    pub value: f64,
42    /// Optional caption.
43    pub label: String,
44    /// Opaque sRGB line/text color.
45    pub color: Color,
46    /// Use a dashed rather than solid line.
47    pub dashed: bool,
48}

50/// Frame colors, independent of the spectral palette.

74/// Appearance options; changing these does not repeat decoding or FFT work.
75#[derive(Debug, Clone)]
76pub struct RenderOptions {
77    /// Continuous color map, monochrome green by default.
78    pub palette: Palette,
79    /// Lower displayed dBFS/bin; quieter values use the first palette color.
80    pub min_db: f64,
81    /// Upper displayed dBFS/bin; louder values use the last palette color.
82    pub max_db: f64,
83    /// Colors of the surrounding frame.
84    pub theme: Theme,
85    /// Left, top, right and bottom margins in pixels.
86    pub margins: [u32; 4],
87    /// Base text size in pixels; headings use proportional sizes.
88    pub font_size: f32,
89    /// Optional custom font bytes; `None` uses embedded OFL Noto Sans.
90    pub font: Option<Arc<[u8]>>,
91    /// Draw frequency/time ticks and units.
92    pub show_axes: bool,
93    /// Draw the level color bar and its dB labels.
94    pub show_legend: bool,
95    /// Draw a subtle grid over the measured image.
96    pub show_grid: bool,
97    /// Optional frequency/time annotations.
98    pub markers: Vec<Marker>,
99}
100
101impl Default for RenderOptions {
102    fn default() -> Self {
103        Self {
104            palette: Palette::default(),

107            theme: Theme::default(),

110            font: None,
111            show_axes: true,
112            show_legend: true,
113            show_grid: false,
114            markers: Vec::new(),

119/// Encode calibrated measurements as a lossless, tagged sRGB PNG in memory.
120/// No filesystem or external process is used, including for text rendering.
121pub fn render_png(
122    data: &Spectrogram,
123    labels: &ImageLabels,
124    options: &RenderOptions,
125) -> Result<Vec<u8>, Error> {
126    options.palette.validate()?;
127    if !options.min_db.is_finite()
128        || !options.max_db.is_finite()
129        || options.min_db >= options.max_db
130        || !options.font_size.is_finite()
131        || !(6.0..=96.0).contains(&options.font_size)
132        || options.markers.len() > 256
133        || options.margins.iter().any(|&n| n > 4096)
134        || options
135            .font
136            .as_ref()
137            .is_some_and(|font| font.len() > 16 * 1024 * 1024)
138    {
139        return Err(Error::Configuration(
140            "invalid rendering bounds, font size or marker count".into(),
141        ));
142    }
143    let (plot_width, plot_height) = data.dimensions();
144    let [left, top, right, bottom] = options.margins;
145    let width = plot_width
146        .checked_add(left)
147        .and_then(|n| n.checked_add(right));
148    let height = plot_height
149        .checked_add(top)
150        .and_then(|n| n.checked_add(bottom));
151    let (width, height) = width
152        .zip(height)
153        .filter(|&(w, h)| u64::from(w) * u64::from(h) <= crate::options::MAX_PIXELS as u64)
154        .ok_or_else(|| Error::Configuration("final image exceeds the pixel limit".into()))?;
155    let font = text::font(options.font.as_deref())?;
156    let mut image = Canvas::new(width, height, options.theme.background)?;
157    for row in 0..plot_height {
158        for column in 0..plot_width {
159            let db = data
160                .level_db(column, row)
161                .ok_or_else(|| Error::Render("inconsistent spectral raster".into()))?;
162            let color = options
163                .palette
164                .color((db - options.min_db) / (options.max_db - options.min_db));
165            image.pixel(left + column, top + plot_height - 1 - row, color);
166        }
167    }
168    axes::draw(&mut image, &font, data, options);
169    labels::draw(&mut image, &font, data, labels, options);
170    markers::draw(&mut image, &font, data, options)?;
171    let mut bytes = Vec::new();
172    {
173        let mut encoder = png::Encoder::new(&mut bytes, width, height);
174        encoder.set_color(png::ColorType::Rgb);
175        encoder.set_depth(png::BitDepth::Eight);
176        encoder.set_source_srgb(png::SrgbRenderingIntent::Perceptual);
177        let render_error = |e: png::EncodingError| Error::Render(e.to_string());
178        encoder
179            .add_itxt_chunk("Software".into(), text::clean(&labels.generator))
180            .map_err(render_error)?;
181        encoder
182            .add_itxt_chunk("Title".into(), text::clean(&labels.title))
183            .map_err(render_error)?;
184        if options.font.is_none() {
185            // Keep the bundled font's full redistribution notice in every
186            // standalone client binary as well as the self-contained PNG.
187            encoder
188                .add_ztxt_chunk(
189                    "FontLicense".into(),
190                    include_str!("../../assets/OFL.txt").into(),
191                )
192                .map_err(render_error)?;
193        }
194        encoder.add_text_chunk("Spectrogram".into(), format!(
195            "FFT={}; hop={}; window={:?}; channels={:?}; pooling={:?}; dBFS/bin={}..{}; sample_rate={}; frames={}",
196            data.fft_size(), data.hop_size(), data.options.window, data.options.channels,
197            data.options.aggregation, options.min_db, options.max_db,
198            data.signal().sample_rate, data.signal().frames)).map_err(render_error)?;
199        let mut writer = encoder.write_header().map_err(render_error)?;
200        writer
201            .write_image_data(&image.pixels)
202            .map_err(render_error)?;
203        writer.finish().map_err(render_error)?;
204    }
205    Ok(bytes)
206}
