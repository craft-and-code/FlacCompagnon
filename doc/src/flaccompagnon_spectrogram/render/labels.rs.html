1//! Host-provided image headings and footer labels.
2use super::{canvas::Canvas, text, ImageLabels, RenderOptions};
3use crate::Spectrogram;
4use fontdue::Font;
5pub(super) fn draw(
6    image: &mut Canvas,
7    font: &Font,
8    data: &Spectrogram,
9    labels: &ImageLabels,
10    options: &RenderOptions,
11) {
12    let [left, top, _, bottom_margin] = options.margins;
13    let (width, height) = data.dimensions();
14    let right = left + width;
15    let bottom = top + height;
16    let size = options.font_size;
17    let scale = size / 15.0;
18    let theme = &options.theme;
19    let write = |image: &mut Canvas, label: &str, x: f32, y: f32, px: f32, color, limit| {
20        text::draw(image, font, label, (x, y), px, color, limit)
21    };
22    if top as f32 >= size * 4.0 {
23        write(
24            image,
25            "S P E C T R O G R A M",
26            left as f32,
27            16.0 * scale,
28            size * 0.8,
29            theme.muted,
30            width as f32 * 0.45,
31        );
32        write(
33            image,
34            "Frequency × time",
35            left as f32,
36            42.0 * scale,
37            size * 1.6,
38            theme.text,
39            width as f32 * 0.45,
40        );
41        let label = text::clean(&labels.generator);
42        let label_width = text::width(font, &label, size * 1.9).min(width as f32 * 0.5);
43        write(
44            image,
45            &label,
46            right as f32 - label_width,
47            18.0 * scale,
48            size * 1.9,
49            theme.text,
50            width as f32 * 0.5,
51        );
52    }
53    if bottom_margin as f32 >= size * 7.0 {
54        let divider = bottom + (68.0 * scale) as u32;
55        image.horizontal(left, divider, width, theme.border, false);
56        let description = text::clean(&labels.audio_description);
57        let description_width = text::width(font, &description, size).min(width as f32 * 0.45);
58        let title_width = (width as f32 - description_width - size * 2.0).max(0.0);
59        write(
60            image,
61            &labels.title,
62            left as f32,
63            divider as f32 + 14.0 * scale,
64            size * 1.4,
65            theme.text,
66            title_width,
67        );
68        write(
69            image,
70            &description,
71            right as f32 - description_width,
72            divider as f32 + 18.0 * scale,
73            size,
74            theme.text,
75            description_width + 1.0,
76        );
77        write(
78            image,
79            &labels.artist,
80            left as f32,
81            divider as f32 + 48.0 * scale,
82            size * 0.85,
83            theme.muted,
84            title_width,
85        );
86        let detail = format!(
87            "{:?} {} · {:?} / pixel",
88            data.options.window,
89            data.fft_size(),
90            data.options.aggregation
91        );
92        let detail_width = text::width(font, &detail, size * 0.7).min(width as f32 * 0.45);
93        write(
94            image,
95            &detail,
96            right as f32 - detail_width,
97            divider as f32 + 50.0 * scale,
98            size * 0.7,
99            theme.muted,
100            detail_width + 1.0,
101        );
102    }
103}
