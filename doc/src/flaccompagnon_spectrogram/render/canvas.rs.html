1//! Clipped RGB drawing primitives shared by text, axes and annotations.
2
3use crate::{Color, Error};
4
5pub(super) struct Canvas {
6    pub width: u32,
7    pub height: u32,
8    pub pixels: Vec<u8>,
9}
10
11impl Canvas {
12    pub fn new(width: u32, height: u32, color: Color) -> Result<Self, Error> {
13        let length = width as usize * height as usize * 3;
14        let mut pixels = Vec::new();
15        pixels
16            .try_reserve_exact(length)
17            .map_err(|e| Error::Render(format!("image allocation failed: {e}")))?;
18        for _ in 0..width as usize * height as usize {
19            pixels.extend_from_slice(&color);
20        }
21        Ok(Self {
22            width,
23            height,
24            pixels,
25        })
26    }
27    pub fn pixel(&mut self, x: u32, y: u32, color: Color) {
28        if x < self.width && y < self.height {
29            let index = (y as usize * self.width as usize + x as usize) * 3;
30            if let Some(pixel) = self.pixels.get_mut(index..index + 3) {
31                pixel.copy_from_slice(&color);
32            }
33        }
34    }
35    pub fn blend(&mut self, x: i32, y: i32, color: Color, alpha: u8) {
36        if x < 0 || y < 0 || x as u32 >= self.width || y as u32 >= self.height {
37            return;
38        }
39        let index = (y as usize * self.width as usize + x as usize) * 3;
40        if let Some(pixel) = self.pixels.get_mut(index..index + 3) {
41            for (value, ink) in pixel.iter_mut().zip(color) {
42                *value = ((u32::from(*value) * (255 - u32::from(alpha))
43                    + u32::from(ink) * u32::from(alpha)
44                    + 127)
45                    / 255) as u8;
46            }
47        }
48    }
49    pub fn horizontal(&mut self, x: u32, y: u32, width: u32, color: Color, dashed: bool) {
50        for dx in 0..width.min(self.width.saturating_sub(x)) {
51            if !dashed || dx % 14 < 7 {
52                self.pixel(x + dx, y, color);
53            }
54        }
55    }
56    pub fn vertical(&mut self, x: u32, y: u32, height: u32, color: Color, dashed: bool) {
57        for dy in 0..height.min(self.height.saturating_sub(y)) {
58            if !dashed || dy % 14 < 7 {
59                self.pixel(x, y + dy, color);
60            }
61        }
62    }
63}
