9/// Measured linear powers and the exact axes used to produce them.
10#[derive(Debug)]
11pub struct Spectrogram {
12    pub(crate) info: SignalInfo,
13    pub(crate) options: AnalysisOptions,
14    pub(crate) fft_size: usize,
15    pub(crate) hop_size: usize,
16    pub(crate) max_frequency: f64,
17    pub(crate) power: Vec<f32>,
18}
19
20impl Spectrogram {
21    /// Native decoded source properties.
22    pub fn signal(&self) -> SignalInfo {
23        self.info
24    }
25    /// Actual transform length, including an automatically selected length.
26    pub fn fft_size(&self) -> usize {
27        self.fft_size
28    }
29    /// Actual distance between FFT centers in source frames.
30    pub fn hop_size(&self) -> usize {
31        self.hop_size
32    }
33    /// Plotted dimensions, excluding labels and margins.
34    pub fn dimensions(&self) -> (u32, u32) {
35        (self.options.width, self.options.height)
36    }
37    /// Duration calculated from decoded complete frames.
38    pub fn duration(&self) -> f64 {
39        self.info.frames as f64 / f64::from(self.info.sample_rate)
40    }
41    /// Peak-amplitude-referenced dBFS/bin at a pixel, with row zero at the bottom.
42    /// Silence is negative infinity; coordinates outside the plot return `None`.
43    pub fn level_db(&self, column: u32, row: u32) -> Option<f64> {
44        if column >= self.options.width || row >= self.options.height {
45            return None;
46        }
47        self.power
48            .get(row as usize * self.options.width as usize + column as usize)
49            .map(|&value| 10.0 * f64::from(value).log10())
50    }
51    pub(crate) fn fraction(&self, frequency: f64) -> f64 {
52        let min = self.options.min_frequency;
53        match self.options.frequency_scale {
54            FrequencyScale::Linear => (frequency - min) / (self.max_frequency - min),
55            FrequencyScale::Logarithmic => (frequency / min).ln() / (self.max_frequency / min).ln(),
56        }
57    }
58}

60/// Stream interleaved PCM into fixed-size spectral buckets, retaining one window.
61pub struct SpectrogramBuilder {
62    result: Spectrogram,
63    received: u64,
64    next_center: u64,
65    last_center: Option<u64>,
66    ring: Vec<f32>,
67    window: Vec<f64>,
68    window_sum: f64,
69    fft: Arc<dyn Fft<f64>>,
70    buffer: Vec<Complex<f64>>,
71    scratch: Vec<Complex<f64>>,
72    bins: Vec<f64>,
73    rows: Vec<f64>,
74    row_counts: Vec<u32>,
75    bin_rows: Vec<Option<usize>>,
76    valid_bins: (usize, usize),
77    counts: Vec<u32>,
78}
79
80impl SpectrogramBuilder {
81    /// Validate all dimensions before allocating; source length must be exact.
82    pub fn new(info: SignalInfo, options: AnalysisOptions) -> Result<Self, Error> {
83        let (fft_size, hop_size, max_frequency) = options.validate(info)?;
84        let height = options.height as usize;
85        let pixels = options.width as usize * height;
86        let mean = matches!(options.aggregation, Aggregation::MeanPower);
87        let mut planner = FftPlanner::<f64>::new();
88        let fft = planner.plan_fft_forward(fft_size);
89        let window = transform::window(options.window, fft_size);
90        let window_sum = window.iter().sum();
91        let result = Spectrogram {
92            info,
93            options,
94            fft_size,
95            hop_size,
96            max_frequency,
97            power: zeroed(pixels)?,
98        };
99        let bin_rows: Vec<Option<usize>> = (0..=fft_size / 2)
100            .map(|bin| {
101                let hz = bin as f64 * f64::from(info.sample_rate) / fft_size as f64;
102                if hz < result.options.min_frequency || hz > max_frequency {
103                    None
104                } else {
105                    Some(((result.fraction(hz) * height as f64) as usize).min(height - 1))
106                }
107            })
108            .collect();
109        let first_bin = bin_rows.iter().position(Option::is_some).ok_or_else(|| {
110            Error::Configuration(
111                "frequency range contains no FFT bin; use a longer transform".into(),
112            )
113        })?;
114        let last_bin = bin_rows
115            .iter()
116            .rposition(Option::is_some)
117            .unwrap_or(first_bin);
118        Ok(Self {
119            ring: zeroed(fft_size * info.channels)?,
120            scratch: zeroed(fft.get_inplace_scratch_len())?,
121            buffer: zeroed(fft_size)?,
122            bins: zeroed(fft_size / 2 + 1)?,
123            rows: zeroed(height)?,
124            row_counts: zeroed(height)?,
125            counts: zeroed(if mean { pixels } else { 0 })?,
126            window,
127            window_sum,
128            fft,
129            result,
130            bin_rows,
131            valid_bins: (first_bin, last_bin),
132            received: 0,
133            next_center: 0,
134            last_center: None,
135        })
136    }
137
138    /// Consume complete, finite interleaved frames without retaining the packet.
139    pub fn push_interleaved(&mut self, samples: &[f32]) -> Result<(), Error> {
140        self.push_interleaved_cancellable(samples, &|| false)
141    }
142
143    /// Consume a packet, checking cancellation between transform windows.
144    pub fn push_interleaved_cancellable(
145        &mut self,
146        samples: &[f32],
147        cancelled: &dyn Fn() -> bool,
148    ) -> Result<(), Error> {
149        let channels = self.result.info.channels;
150        if !samples.len().is_multiple_of(channels) || samples.iter().any(|v| !v.is_finite()) {
151            return Err(Error::Audio(
152                "non-finite sample or incomplete interleaved frame".into(),
153            ));
154        }
155        let frames = (samples.len() / channels) as u64;
156        if frames > self.result.info.frames.saturating_sub(self.received) {
157            return Err(Error::Audio(
158                "stream exceeds the declared frame count".into(),
159            ));
160        }
161        for frame in samples.chunks_exact(channels) {
162            let start = (self.received % self.result.fft_size as u64) as usize * channels;
163            self.ring[start..start + channels].copy_from_slice(frame);
164            self.received += 1;
165            if self.received
166                >= self
167                    .next_center
168                    .saturating_add(self.result.fft_size as u64 / 2)
169            {
170                if cancelled() {
171                    return Err(Error::Cancelled);
172                }
173                self.transform(self.next_center)?;
174                self.next_center = self.next_center.saturating_add(self.result.hop_size as u64);
175            }
176        }
177        Ok(())
178    }
179
180    /// Complete right-boundary windows and reject a missing audio tail.
181    pub fn finish(self) -> Result<Spectrogram, Error> {
182        self.finish_cancellable(&|| false)
183    }
184
185    /// Complete the spectrogram, allowing cancellation during boundary windows.
186    pub fn finish_cancellable(
187        mut self,
188        cancelled: &dyn Fn() -> bool,
189    ) -> Result<Spectrogram, Error> {
190        if self.received != self.result.info.frames {
191            return Err(Error::Audio(
192                "incomplete stream: decoded frame count changed".into(),
193            ));
194        }
195        while self.next_center < self.received {
196            if cancelled() {
197                return Err(Error::Cancelled);
198            }
199            self.transform(self.next_center)?;
200            self.next_center = self.next_center.saturating_add(self.result.hop_size as u64);
201        }
202        if self.last_center != Some(self.received - 1) {
203            if cancelled() {
204                return Err(Error::Cancelled);
205            }
206            self.transform(self.received - 1)?;
207        }
208        if matches!(self.result.options.aggregation, Aggregation::MeanPower) {
209            for (power, count) in self.result.power.iter_mut().zip(&self.counts) {
210                if *count > 0 {
211                    *power /= *count as f32;
212                }
213            }
214        }
215        Ok(self.result)
216    }
217}

219fn zeroed<T: Default + Clone>(length: usize) -> Result<Vec<T>, Error> {
220    let mut result = Vec::new();
221    result
222        .try_reserve_exact(length)
223        .map_err(|e| Error::Configuration(format!("allocation failed: {e}")))?;
224    result.resize(length, T::default());
225    Ok(result)
226}
